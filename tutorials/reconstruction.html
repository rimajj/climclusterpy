Tutorial 4 — Reconstruct the original grid

Capability 4: reconstruct the original grid from cluster-grid output. This tutorial runs the full round trip on synthetic data: cluster the cells, emit a synthetic “model output” NetCDF on the artificial cluster grid together with the two mapping CSVs, then reconstruct the output back onto every original cell and verify each cell recovered exactly its cluster’s value.

The round trip

After clustering (Tutorial 1 — Euclidean clustering), the example writes an artificial-grid output NetCDF laid out as one latitude row with one longitude column per cluster, plus:

  • cluster_id_mapping_{K}.csv — every original cell → its cluster;

  • cluster_compact_map_{K}.csv — each cluster → its lat_index and longitude span on the artificial grid.

reconstruct_original_grid() consumes both, with the grid description, and writes the reconstructed grid. Without a create_clusters() result to hand it, name the three yourself:

from climclusterpy import reconstruct_original_grid

recon_path = reconstruct_original_grid(
    clustered_nc,
    registry=registry,
    id_mapping_csv=id_csv,
    compact_map_csv=compact_csv,    # where each cluster sits on the reduced grid
    output_dir=out_dir,
).path

Verify

Every original valid cell must carry its cluster’s synthetic value, and invalid (ocean) cells must stay NaN. The example checks each cell explicitly and then calls validate_reconstruction_grid():

from climclusterpy.grid import validate_reconstruction_grid

assert validate_reconstruction_grid(recon_path, registry, var_name="vegc")

The complete script

examples/example_4_reconstruct.py
#!/usr/bin/env python3
"""Capability 4 — reconstruct the full original grid from cluster-grid output.

End-to-end synthetic round trip:

1. build synthetic climate+soil and load a registry describing the original grid;
2. cluster the valid cells (Euclidean);
3. emit an artificial-grid "model output" NetCDF plus the two ID/row mapping CSVs
   that :func:`climclusterpy.reconstruct_original_grid` consumes (in production
   :func:`climclusterpy.create_clusters` writes them next to the reduced grid);
4. reconstruct the output back onto every original cell and verify each cell
   recovered exactly its cluster's value.

    python examples/example_4_reconstruct.py
"""

from __future__ import annotations

import os
import tempfile

import numpy as np
import pandas as pd
import xarray as xr
from _synthetic import feature_matrix, load_synthetic

from climclusterpy import cluster_matrix, reconstruct_original_grid
from climclusterpy.grid import validate_reconstruction_grid


def build_artificial_grid_output(labels, registry, out_dir, n_time=3):
    """Write a synthetic cluster-grid output NetCDF + the id/row mapping CSVs.

    The artificial grid is laid out as a single latitude row with one longitude
    column per cluster (``lat_index=0``, cluster ``k`` -> lon column ``k``). Each
    cluster gets a distinct synthetic value ``10*cluster + t`` at time ``t``.

    Returns ``(clustered_nc, id_mapping_csv, compact_map_csv, expected_value_fn)``.
    """
    labels = np.asarray(labels)
    orig_ids = np.asarray(registry.get_valid_original_ids())
    unique = sorted(set(labels.tolist()))
    col_of = {cl: k for k, cl in enumerate(unique)}
    n_clusters = len(unique)

    # id mapping: one row per original (valid) cell -> its cluster.
    id_mapping = pd.DataFrame(
        {
            "original_cell_id": orig_ids,
            "new_cell_id": np.arange(len(orig_ids)),
            "cluster_index": labels,
            "cluster_label": labels,
        }
    )
    id_mapping_csv = os.path.join(out_dir, f"cluster_id_mapping_{n_clusters}.csv")
    id_mapping.to_csv(id_mapping_csv, index=False)

    # compact row map: cluster -> (lat_index, lon start/end) on the artificial grid.
    compact = pd.DataFrame(
        {
            "cluster_label": unique,
            "lat_index": [0] * n_clusters,
            "cluster_index_start": [col_of[cl] for cl in unique],
            "cluster_index_end": [col_of[cl] for cl in unique],
        }
    )
    compact_map_csv = os.path.join(out_dir, f"cluster_compact_map_{n_clusters}.csv")
    compact.to_csv(compact_map_csv, index=False)

    # synthetic per-cluster model output on the (time, lat=1, lon=n_clusters) grid.
    time = np.arange(n_time)
    values = np.empty((n_time, 1, n_clusters), dtype="float32")
    for cl in unique:
        values[:, 0, col_of[cl]] = 10.0 * cl + time
    clustered = xr.Dataset(
        {"vegc": (("time", "lat", "lon"), values)},
        coords={"time": time, "lat": [0.0], "lon": np.arange(n_clusters)},
    )
    clustered_nc = os.path.join(out_dir, "vegc_clustered.nc")
    clustered.to_netcdf(clustered_nc)

    def expected(cluster_label, t):
        return 10.0 * cluster_label + t

    return clustered_nc, id_mapping_csv, compact_map_csv, expected


def main() -> None:
    with tempfile.TemporaryDirectory(prefix="climclu_ex4_") as tmp:
        _paths, datasets, registry = load_synthetic(tmp, n_cells=64)
        X, _meta = feature_matrix(datasets, registry)

        result = cluster_matrix(X, n_clusters=6, locally_weighted=False, standardize=True)
        labels = result.labels
        print(f"Clustered {len(labels)} cells into {len(set(labels.tolist()))} clusters")

        clustered_nc, id_csv, compact_csv, expected = build_artificial_grid_output(
            labels, registry, tmp
        )

        # Without a create_clusters result, name the grid and both mapping files yourself.
        recon_path = reconstruct_original_grid(
            clustered_nc,
            registry=registry,
            id_mapping_csv=id_csv,
            compact_map_csv=compact_csv,
            output_dir=os.path.join(tmp, "reconstructed"),
        ).path
        print("Reconstructed ->", recon_path)

        # Verify: every original valid cell recovered exactly its cluster's value.
        recon = xr.open_dataset(recon_path)["vegc"]
        mask = registry.get_valid_mask()
        positions = np.argwhere(mask)  # row-major, aligned with labels order
        n_time = recon.sizes["time"]
        checked = 0
        for k, (i, j) in enumerate(positions):
            cl = int(labels[k])
            got = recon.values[:, i, j]
            want = np.array([expected(cl, t) for t in range(n_time)], dtype="float32")
            assert np.allclose(got, want), f"cell ({i},{j}) cluster {cl}: {got} != {want}"
            checked += 1
        recon.close()

        # Invalid cells stay NaN; the grid matches the original.
        assert validate_reconstruction_grid(recon_path, registry, var_name="vegc")
        print(f"OK: example 4 (reconstruction round trip) verified {checked} cells.")


if __name__ == "__main__":
    main()
