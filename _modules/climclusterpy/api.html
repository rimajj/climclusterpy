Source code for climclusterpy.api

"""The public API — the whole package in two function calls.

**Tier 1 — the headline functions.** Between them they cover the reason this
package exists: shrink a grid, run your vegetation model on the small one, get the
full grid back.

* :func:`create_clusters` — climate (and optionally soil + model-output) input in,
  model-ready cluster NetCDF plus the mapping CSVs out. One call does feature
  construction, supervision, metric learning, clustering, representative-cell
  selection, and NetCDF writing.
* :func:`reconstruct_original_grid` — model output from the reduced grid in, a
  full-resolution field out, every original cell carrying its cluster's value.

**Tier 2 — building blocks**, for when you want to drive the steps yourself:

* :func:`cluster_matrix` — cluster a feature matrix you built yourself, Euclidean
  or LMNN.
* :func:`compute_supervision` — just the LMNN label signal (Köppen–Geiger,
  ecoregions, or model-output traits).

Everything here is importable from ``climclusterpy`` itself; import it from there.

Heavy/optional dependencies (``torch`` for LMNN, ``geopandas``
for ecoregions, ``pyet`` for SPEI) are imported **inside** the functions that use
them, so ``import climclusterpy`` stays lightweight and the extras are only
required when actually used.

Examples
--------
Reduce a grid to 2000 clusters using a metric learned from model-output traits,
then project a simulation result back onto the full grid::

    from climclusterpy import create_clusters, reconstruct_original_grid

    clusters = create_clusters(
        "/data/climate_input",
        n_clusters=2000,
        locally_weighted=True,
        supervision_data="/data/model_output/trait_grids",
        supervision=["sla", "wooddens", "biomass", "height"],
        output_dir="./clusters_2000",
    )
    # ... run the vegetation model on clusters.netcdf_files ...
    full = reconstruct_original_grid("vegc_2000_clusters.nc", clusters=clusters)

Or, with no supervision data and no training, plain Euclidean clustering:

>>> import numpy as np
>>> from climclusterpy import cluster_matrix
>>> X = np.random.default_rng(0).normal(size=(120, 8))
>>> cluster_matrix(X, n_clusters=5).labels.shape
(120,)
"""

class ClusterResult:
    """Result of :func:`cluster_matrix`.

    Attributes
    ----------
    labels : numpy.ndarray
        Cluster label per cell, shape ``(n_cells,)``, 1-indexed (``1..n_clusters``)
        to match SciPy ``fcluster``.
    n_clusters : int
        Number of clusters requested.
    mode : str
        ``"euclidean"`` or ``"lmnn"``.
    linkage : numpy.ndarray or None
        SciPy linkage matrix ``Z`` of shape ``(n_cells - 1, 4)`` (both modes are
        agglomerative, so this is always populated).
    metadata : pandas.DataFrame or None
        Cell metadata (``cell_id``/``latitude``/``longitude`` where available)
        carried through from a DataFrame input, aligned row-for-row with ``labels``.
    diagnostics : dict
        Mode-specific diagnostics (e.g. LMNN training info); empty for Euclidean.
    distance_matrix : numpy.ndarray or None
        The dense ``(n_cells, n_cells)`` **learned** distance matrix, populated
        only for a per-cell LMNN mode called with
        ``return_distance_matrix=True``. This is what medoid selection needs: a
        medoid is the cluster member minimising total distance *under the learned
        metric*, so it cannot be recovered from ``labels`` and ``linkage`` alone.
        ``None`` otherwise -- it is memory-heavy (``n_cells²`` float64).
    model : object or None
        The **trained metric network** itself (a ``torch.nn.Module``), populated
        only for a per-cell LMNN mode called with ``return_model=True``. It is
        what makes "did these two trainings produce the same metric?" an
        answerable question from outside the library: two learned distance
        matrices can be compared, but a difference between them cannot be
        attributed without the weights. Returned rather than written, so the
        caller decides on persistence (``torch.save(res.model.state_dict(), …)``).
        ``None`` otherwise.
    """

    labels: np.ndarray
    n_clusters: int
    mode: str
    linkage: np.ndarray | None = None
    metadata: pd.DataFrame | None = None
    diagnostics: dict = field(default_factory=dict)
    distance_matrix: np.ndarray | None = None
    model: object | None = None

class SupervisionResult:
    """Result of :func:`compute_supervision` — the label signal LMNN trains on.

    Attributes
    ----------
    labels : pandas.DataFrame
        Canonical per-cell label table with columns
        ``cell_id, latitude, longitude, cluster_label``.
    kind : str
        ``"koppen_geiger"``, ``"ecoregions"`` or ``"output"`` -- or ``"given"`` for labels
        handed to ``create_clusters(supervision=)`` as a table, CSV or array.
    n_classes : int
        Number of distinct supervision classes.
    labels_csv : str or None
        Path to the on-disk labels CSV the generator wrote.
    """

    labels: pd.DataFrame
    kind: str
    n_classes: int
    labels_csv: str | None = None

    @property
    def y(self) -> np.ndarray:
        """The raw ``cluster_label`` values (KG codes / ECO_IDs / 1..K), shape ``(n_cells,)``."""
        return self.labels["cluster_label"].to_numpy()

    @property
    def encoded_y(self) -> np.ndarray:
        """``y`` factorized to contiguous integers ``0..n_classes-1``.

        The torch LMNN trainers (local/enhanced) require contiguous integer classes,
        and KG/ecoregion codes are non-contiguous, so this is the form to feed them.
        """
        return pd.factorize(self.labels["cluster_label"])[0]

class CreateClustersResult:
    """Everything :func:`create_clusters` produced, and what to do with it.

    Attributes
    ----------
    netcdf_files : dict
        ``{variable: path}`` — the cluster NetCDFs on the reduced grid. These are
        the files the vegetation model runs on.
    labels : pandas.DataFrame
        Per-cell assignment: ``cell_id, latitude, longitude, cluster_label``, one
        row per valid cell of the *original* grid.
    labels_csv : str
        Path the label table was written to.
    id_mapping_csv : str or None
        ``cluster_id_mapping_{K}.csv`` — links each reduced-grid cell back to the
        original cells it stands for. Required for reconstruction. ``None`` for
        ``representative="distributed"``, whose map is ``distributed_mapping_csv``.
    compact_map_csv : str or None
        ``cluster_compact_map_{K}.csv`` — the row map needed for correct longitude
        slicing when the reduced grid is large relative to the original one.
        ``None`` if the grid writer did not produce it.
    registry : DataRegistry
        Registry for the original grid, carried so reconstruction needs no
        re-loading.
    n_clusters : int
        Number of clusters actually produced.
    mode : str
        ``"euclidean"`` or ``"lmnn"``.
    representative : str
        How each cluster's cell(s) were chosen: ``"medoid"``, ``"centroid"``,
        ``"random"`` or ``"distributed"``.
    placement : str
        Where each cell was written: ``"native"`` (at the coordinates of the cell whose
        climate it carries) or ``"cluster_center"`` (the pre-0.3.0 layout for
        medoid/centroid, at the cluster's centre latitude).
    output_dir : str
        Directory holding all of the above.
    cells_per_cluster : int
        Real cells written per cluster: 1, except ``cells_per_cluster`` for
        ``representative="distributed"``.
    patches_per_cell : int
        What to set as the vegetation model's patch count (LPJmL's ``npatch``) for
        this grid: the ``npatch`` budget divided by ``cells_per_cluster``.
    selected_cells : pandas.DataFrame
        Which original cells stand for each cluster: ``cluster_label, cell_id,
        latitude, longitude, slot`` — one row per written cell (``slot`` counts
        within the cluster, so a small cluster's repeated cells are visible).
    distributed_mapping_csv : str or None
        ``cluster_distributed_mapping_{K}.csv`` — where each distributed slot sits on
        the written grid. Only for ``representative="distributed"``.
    supervision : SupervisionResult or None
        The label signal the metric was trained on (LMNN mode only).
    feature_columns : list of str
        Feature columns that survived filtering and were clustered on.
    diagnostics : dict
        Mode-specific diagnostics (LMNN training info, feature counts, timings).
    learned_metric_dir : str or None
        Where the distance metric this run trained was saved (``<output_dir>/learned_metric``);
        :func:`load_metric` reads it back, and ``create_clusters(learned_metric=...)`` applies
        it to a new run without training again. ``None`` when no metric was trained here.
    """

    netcdf_files: dict[str, str]
    labels: pd.DataFrame
    labels_csv: str
    id_mapping_csv: str | None
    compact_map_csv: str | None
    registry: object
    n_clusters: int
    mode: str
    representative: str
    output_dir: str
    supervision: SupervisionResult | None = None
    feature_columns: list[str] = field(default_factory=list)
    diagnostics: dict = field(default_factory=dict)
    cells_per_cluster: int = 1
    patches_per_cell: int | None = None
    selected_cells: pd.DataFrame | None = None
    distributed_mapping_csv: str | None = None
    placement: str = "native"
    learned_metric_dir: str | None = None

    @property
    def cluster_files(self) -> list[str]:
        """The written cluster NetCDF paths (without the mapping-CSV entries)."""
        return [p for k, p in sorted(self.netcdf_files.items()) if not k.startswith("__")]

    @property
    def slot_sources(self) -> pd.DataFrame:
        """Where every written cell sits, and where its data comes from: one row per slot.

        Columns ``slot, cluster_label, cell_id, source_latitude, source_longitude,
        grid_latitude, grid_longitude, at_source``. ``source_*`` is the original cell whose
        climate and soil the slot carries; ``grid_*`` is where the slot sits in the written
        reduced-grid files; ``at_source`` is False where the two differ.

        **Why it matters.** Anything that finds a cell's data *by coordinate* -- a NetCDF
        input of the model, a lookup of your own -- reads the place at ``grid_*``, which is
        right only where ``at_source`` is True. With ``representative="distributed"`` that is
        every slot except the repeats of a region with fewer cells than draws (each repeat sits
        at the next free longitude; 2 317 of 16 850 slots at 674 regions x 25 on the global
        half-degree grid). With ``"medoid"``/``"centroid"``/``"random"`` the cells are packed
        into rows, so the latitude is the source's but the longitude generally is not.
        :func:`subset_inputs` writes each slot's *source* values at its grid position, which is
        the safe way to bring any other input onto the reduced grid.

        Raises :class:`ValueError` for a ``placement="cluster_center"`` grid, which records no
        per-slot position.
        """
        return _slot_sources(self)

    def __repr__(self) -> str:
        return (
            f"CreateClustersResult(n_clusters={self.n_clusters}, n_cells={len(self.labels)}, "
            f"mode={self.mode!r}, representative={self.representative!r}, "
            f"placement={self.placement!r}, "
            f"cells_per_cluster={self.cells_per_cluster}, patches_per_cell={self.patches_per_cell}, "
            f"{len(self.cluster_files)} NetCDF files in {self.output_dir!r})"
        )

def _extract_feature_matrix(
    features, feature_columns: list[str] | None
) -> tuple[np.ndarray, pd.DataFrame | None]:
    """Coerce ``features`` to a 2D float matrix, returning ``(X, metadata_df)``.

    Accepts a NumPy array (returned as-is, no metadata) or a pandas DataFrame
    (metadata columns are split off and non-metadata numeric columns become ``X``).
    """
    if isinstance(features, pd.DataFrame):
        if feature_columns is not None:
            missing = [c for c in feature_columns if c not in features.columns]
            if missing:
                raise ValueError(f"feature_columns not found in DataFrame: {missing}")
            feat_cols = list(feature_columns)
        else:
            feat_cols = [
                c
                for c in features.columns
                if c not in _METADATA_COLUMNS and pd.api.types.is_numeric_dtype(features[c])
            ]
            if not feat_cols:
                raise ValueError(
                    "No feature columns found in DataFrame after dropping metadata "
                    f"columns {sorted(_METADATA_COLUMNS)}. Pass feature_columns explicitly."
                )
        meta_cols = [c for c in features.columns if c in _METADATA_COLUMNS]
        metadata = features[meta_cols].copy() if meta_cols else None
        X = features[feat_cols].to_numpy(dtype=float)
        return X, metadata

    X = np.asarray(features, dtype=float)
    if X.ndim != 2:
        raise TypeError(
            "features must be a 2D array (n_cells, n_features) or a pandas DataFrame; "
            f"got array with ndim={X.ndim}. For gridded xarray input use "
            "climclusterpy.clustering.compute_clusters instead."
        )
    return X, None


def _supervision_labels(supervision, n_cells: int) -> np.ndarray:
    """Extract a 1D label array from a SupervisionResult or array-like, checking length."""
    if isinstance(supervision, SupervisionResult):
        y = supervision.y
    else:
        y = np.asarray(supervision)
    y = np.asarray(y).reshape(-1)
    if y.shape[0] != n_cells:
        raise ValueError(
            f"supervision length ({y.shape[0]}) must match the number of cells "
            f"({n_cells}). Supervision labels must be aligned row-for-row with features."
        )
    return y


def _default_supervision_granularity(n_cells: int) -> int:
    """Number of supervision classes to use when the caller does not say.

    Roughly one class per :data:`_CELLS_PER_SUPERVISION_CLASS` cells, clamped to
    ``[30, 674]``: enough classes to describe real ecological variation, few enough
    that each still holds the neighbours LMNN needs to learn from.
    """
    target = round(n_cells / _CELLS_PER_SUPERVISION_CLASS)
    return int(max(_MIN_SUPERVISION_CLASSES, min(_MAX_SUPERVISION_CLASSES, target)))


def _write_cluster_labels_csv(
    registry, labels: np.ndarray, n_clusters: int, output_dir: str
) -> tuple[pd.DataFrame, str]:
    """Write the canonical per-cell label table and return it with its path."""
    lats, lons = registry.get_valid_lat_lon()
    frame = pd.DataFrame(
        {
            "cell_id": registry.get_valid_original_ids(),
            "latitude": lats,
            "longitude": lons,
            "cluster_label": np.asarray(labels).reshape(-1),
        }
    )
    os.makedirs(output_dir, exist_ok=True)
    path = os.path.join(output_dir, f"cluster_labels_with_coords_nr_of_clusters_{n_clusters}.csv")
    frame.to_csv(path, index=False)
    return frame, path


def _supervision_for_lmnn(
    *,
    source: str,
    traits,
    supervision_data,
    ecoregions_shapefile: str | None,
    ecoregion_id_column: str,
    ecoregion_min_cells: int,
    include_dominant_type: bool,
    supervision_granularity: int | None,
    features_df: pd.DataFrame,
    climate_df: pd.DataFrame,
    registry,
    linkage_method: str,
    output_dir: str,
) -> SupervisionResult:
    """Produce the LMNN label signal: from model-output traits, climate zones or ecoregions.

    Trait labels come from clustering the model's own output traits — the strongest signal,
    because it groups cells by the very quantity the reduced grid must reproduce.
    Köppen–Geiger classes are derived from the climate input itself, and ecoregions from a
    polygon file, so neither needs model output.
    """
    n_cells = len(features_df)
    k_sup = supervision_granularity or _default_supervision_granularity(n_cells)

    if source == "given":
        return _given_supervision(traits)
    if source == "koppen_geiger":
        return compute_supervision(
            "koppen_geiger",
            features_df=climate_df,
            output_dir=os.path.join(output_dir, "supervision"),
        )
    if source == "ecoregions":
        return compute_supervision(
            "ecoregions",
            registry=registry,
            ecoregions_shapefile=ecoregions_shapefile,
            ecoregion_id_column=ecoregion_id_column,
            ecoregion_min_cells=ecoregion_min_cells,
            output_dir=os.path.join(output_dir, "supervision"),
        )

    from .data.registry import require_valid_cell_alignment
    from .features.trait_ingest import DOMINANT_TYPE_COLUMN, build_trait_features_df

    traits_df = build_trait_features_df(
        supervision_data,
        registry,
        traits=traits,
        include_dominant_type=include_dominant_type,
    )
    traits_df = require_valid_cell_alignment(traits_df, registry, "The trait supervision table")

    categorical = DOMINANT_TYPE_COLUMN if DOMINANT_TYPE_COLUMN in traits_df.columns else None
    return compute_supervision(
        "output",
        features_df=traits_df,
        cell_ids=traits_df["cell_id"].to_numpy(),
        latitudes=traits_df["latitude"].to_numpy(),
        longitudes=traits_df["longitude"].to_numpy(),
        continuous_cols=list(traits),
        categorical_col=categorical,
        n_clusters=k_sup,
        linkage_method=linkage_method,
        output_dir=os.path.join(output_dir, "supervision"),
    )


def _aligned_supervision_labels(
    supervision: SupervisionResult, features_df: pd.DataFrame
) -> np.ndarray:
    """Return supervision labels in the exact row order of ``features_df``.

    Length-only checks are not enough here: a label vector of the right size but the
    wrong order trains the metric on mislabelled neighbours and produces clusters
    that look plausible and are wrong. Matching is therefore by ``cell_id``.
    """
    labels = supervision.labels
    if "cell_id" not in labels.columns:
        y = supervision.y
        if len(y) != len(features_df):
            raise ValueError(
                f"Supervision has {len(y)} labels for {len(features_df)} cells and no 'cell_id' "
                "column to align on."
            )
        return y

    merged = features_df[["cell_id"]].merge(
        labels[["cell_id", "cluster_label"]], on="cell_id", how="left", validate="one_to_one"
    )
    if merged["cluster_label"].isna().any():
        n_missing = int(merged["cluster_label"].isna().sum())
        raise ValueError(
            f"Supervision covers only {len(merged) - n_missing} of {len(merged)} clustered cells "
            f"({n_missing} without a label). Supervision and climate input must describe the "
            "same grid."
        )
    return merged["cluster_label"].to_numpy()

_PLACEMENTS = ("native", "cluster_center")
_DEFAULT_MIN_GRID_SPACING = 0.1

#: Supervision that needs no model output; any other ``supervision`` value names traits.
_REGION_SUPERVISION = ("koppen_geiger", "ecoregions")
_DEFAULT_SUPERVISION_TRAITS = ("sla", "wooddens", "height", "biomass")


def _resolve_placement(placement, representative: str, min_grid_spacing, function: str):
    """Check ``placement`` and return the lattice spacing it uses (``None`` for native)."""
    if placement not in _PLACEMENTS:
        raise ValueError(f"placement must be 'native' or 'cluster_center'; got {placement!r}.")
    if min_grid_spacing is not None and not min_grid_spacing > 0:
        raise ValueError(
            f"min_grid_spacing must be a positive number of degrees; got {min_grid_spacing!r}."
        )
    if placement == "cluster_center":
        if representative in _SAMPLED_CELL_CHOICES:
            raise ValueError(
                "placement='cluster_center' applies to the medoid and centroid cell choices "
                f"only; representative={representative!r} always writes each drawn cell at its "
                "own position."
            )
        return _DEFAULT_MIN_GRID_SPACING if min_grid_spacing is None else min_grid_spacing
    if min_grid_spacing is not None:
        warnings.warn(
            f"{function}(): min_grid_spacing has no effect with placement='native', which puts "
            "every cell on the source grid's own latitude rows; it only sets the lattice of "
            "placement='cluster_center'.",
            UserWarning,
            stacklevel=3,
        )
    return None


def _is_given_labels(supervision) -> bool:
    """Labels handed in ready-made: a SupervisionResult, a table, an array or a ``.csv`` path."""
    if isinstance(supervision, (SupervisionResult, pd.DataFrame, np.ndarray)):
        return True
    return isinstance(supervision, (str, os.PathLike)) and str(supervision).endswith(".csv")


def _given_supervision(supervision) -> SupervisionResult:
    """Wrap labels handed to ``create_clusters(supervision=)`` as a SupervisionResult."""
    if isinstance(supervision, SupervisionResult):
        return supervision
    labels_csv = None
    if isinstance(supervision, (str, os.PathLike)):
        labels_csv = str(supervision)
        if not os.path.exists(labels_csv):
            raise FileNotFoundError(f"supervision labels CSV not found: {labels_csv}")
        supervision = pd.read_csv(labels_csv)
    if isinstance(supervision, pd.DataFrame):
        if "cluster_label" not in supervision.columns:
            raise ValueError(
                "A supervision table needs a 'cluster_label' column (and a 'cell_id' column to "
                f"be matched to the grid by id); got columns {list(supervision.columns)}."
            )
        table = supervision
    else:
        table = pd.DataFrame({"cluster_label": np.asarray(supervision).reshape(-1)})
    return SupervisionResult(
        labels=table,
        kind="given",
        n_classes=int(table["cluster_label"].nunique()),
        labels_csv=labels_csv,
    )


def _resolve_supervision(supervision, supervision_data, ecoregions_shapefile):
    """Turn ``create_clusters(supervision=...)`` into ``(source, traits)``.

    ``source`` is ``"koppen_geiger"``, ``"ecoregions"``, ``"traits"`` or ``"given"``; the second
    value is the tuple of trait names for ``"traits"``, the labels themselves for ``"given"``,
    and ``None`` otherwise.
    """
    if _is_given_labels(supervision):
        if supervision_data is not None or ecoregions_shapefile is not None:
            raise ValueError(
                "supervision= already holds the labels, so supervision_data and "
                "ecoregions_shapefile would be ignored. Drop them."
            )
        return "given", supervision
    if supervision is None:
        if supervision_data is not None:
            source, traits = "traits", _DEFAULT_SUPERVISION_TRAITS
        else:
            source, traits = "koppen_geiger", None
    elif isinstance(supervision, str):
        if supervision in _REGION_SUPERVISION:
            source, traits = supervision, None
        else:
            source, traits = "traits", (supervision,)
    else:
        traits = tuple(supervision)
        if not traits or not all(isinstance(t, str) for t in traits):
            raise ValueError(
                "supervision must be 'koppen_geiger', 'ecoregions', a trait name or a sequence "
                f"of trait names; got {supervision!r}."
            )
        clash = [t for t in traits if t in _REGION_SUPERVISION]
        if clash:
            raise ValueError(
                f"supervision={supervision!r} mixes {clash[0]!r} with model-output traits. "
                f"Pass supervision={clash[0]!r} on its own, or only trait names."
            )
        source = "traits"

    if source == "traits" and supervision_data is None:
        raise ValueError(
            f"supervision={supervision!r} names model-output trait(s), which are read from "
            "supervision_data=: pass the model output (a directory of per-trait NetCDFs or a "
            "per-cell CSV). supervision='koppen_geiger' or 'ecoregions' need no model output."
        )
    if source != "traits" and supervision_data is not None:
        raise ValueError(
            f"supervision_data is only read for trait supervision; supervision={source!r} would "
            "ignore it. Drop it, or name the trait(s) to learn from in supervision=."
        )
    if source == "ecoregions" and ecoregions_shapefile is None:
        raise ValueError(
            "supervision='ecoregions' needs ecoregions_shapefile=<path to the polygon .shp>."
        )
    if source != "ecoregions" and ecoregions_shapefile is not None:
        raise ValueError("ecoregions_shapefile is only used with supervision='ecoregions'.")
    return source, traits


def _require_n_clusters(n_clusters, n_cells: int | None = None, name: str = "n_clusters") -> int:
    """``n_clusters`` as an int, refusing anything that cannot be a number of clusters.

    A float, a string or a bool is a caller's mistake that scipy would otherwise round,
    truncate or compare into a different number of clusters than the one recorded.
    """
    import numbers

    if isinstance(n_clusters, bool) or not isinstance(n_clusters, numbers.Integral):
        raise TypeError(
            f"{name} must be a whole number; got {n_clusters!r} ({type(n_clusters).__name__})."
        )
    k = int(n_clusters)
    if k < 1:
        raise ValueError(f"{name} must be at least 1; got {k}.")
    if n_cells is not None and k > n_cells:
        raise ValueError(
            f"{name} ({k}) exceeds the number of valid grid cells to cluster ({n_cells}). The reduced grid "
            "cannot have more cells than the original."
        )
    return k


def _cluster_with_learned_metric(
    X, n_clusters, metric, *, linkage_method, feature_names=None, keep_distances=False
):
    """``(labels, linkage, distances_or_None)`` for ``X`` under an already trained metric.

    The same bridge from metric to dendrogram that training uses
    (:func:`~climclusterpy.clustering.lmnn.pipeline.run_lmnn_clustering`), so a metric applied
    here to the features it was trained on gives the labels it was deployed with.
    """
    from scipy.cluster.hierarchy import fcluster, linkage
    from scipy.spatial.distance import squareform

    metric.check_features(feature_columns=feature_names, n_features=X.shape[1])
    D = None
    if metric.kind == "global":
        Z = linkage(metric.transformed(X), method=linkage_method, metric="euclidean")
    else:
        D = metric.distance_matrix(X)
        Z = linkage(squareform(D, checks=False), method=linkage_method)
    labels = np.asarray(fcluster(Z, n_clusters, criterion="maxclust"))
    return labels, Z, (D if keep_distances else None)

def _clusters_for_fraction(fraction: float, n_cells: int) -> int:
    """The number of clusters that is ``fraction`` of ``n_cells``, and say which it is."""
    k = max(1, round(fraction * n_cells))
    logger.info(
        "fraction=%g of %d valid cells: n_clusters=%d (%.3f %% of the cells)",
        fraction,
        n_cells,
        k,
        100.0 * k / n_cells,
    )
    return k


#: The column names a multi-K labels table uses, one per cut: ``cluster_k674`` and so on.
_CUT_COLUMN = re.compile(r"^cluster_k(\d+)$")


def _choose_partition_column(frame, column, fraction, n_cells: int, source: str):
    """``frame`` with the chosen cut in ``cluster_label``, and a description of the choice.

    A labels table holds either one partition (``cluster_label``) or several cuts of one
    dendrogram (``cluster_k7``, ``cluster_k21``, ...). ``column`` names the cut; ``fraction``
    takes the cut whose region count is nearest ``fraction * n_cells``.
    """
    if column is not None and fraction is not None:
        raise ValueError("Pass column= or fraction= to pick the partition's column, not both.")
    if column is not None:
        if column not in frame.columns:
            raise ValueError(f"{source} has no column {column!r}; it has {list(frame.columns)}.")
        chosen, why = column, f"column {column!r}"
    elif fraction is not None:
        _require_fraction("fraction", fraction)
        cuts = [c for c in frame.columns if _CUT_COLUMN.match(str(c)) or c == "cluster_label"]
        if not cuts:
            raise ValueError(
                f"fraction= picks one of a table's cluster_k<N> columns (or cluster_label), and "
                f"{source} has none; it has {list(frame.columns)}."
            )
        wanted = fraction * n_cells
        counts = {c: int(frame[c].nunique(dropna=True)) for c in cuts}
        # Nearest region count; on a tie the finer cut, which is the more faithful one.
        chosen = min(cuts, key=lambda c: (abs(counts[c] - wanted), -counts[c]))
        why = (
            f"column {chosen!r}, {counts[chosen]} regions = "
            f"{100.0 * counts[chosen] / n_cells:.3f} % of {n_cells} cells, the nearest of "
            f"{sorted(counts.values())} to fraction={fraction:g}"
        )
        logger.info("Partition from %s: %s", source, why)
    else:
        if "cluster_label" in frame.columns:
            return frame, None
        cuts = [c for c in frame.columns if _CUT_COLUMN.match(str(c))]
        if cuts:
            raise ValueError(
                f"{source} holds several cuts ({cuts}) and no 'cluster_label' column. Pick one "
                "with column='<name>', or with fraction=<share of cells> for the cut nearest "
                "that many regions."
            )
        raise ValueError(
            f"{source} has no 'cluster_label' column (it has {list(frame.columns)}). That is the "
            "column name every climclusterpy partition table uses, including the ones "
            "compute_supervision writes. Name another column with column='<name>'."
        )
    keep = [c for c in frame.columns if c != "cluster_label" or c == chosen]
    picked = frame[keep].rename(columns={chosen: "cluster_label"})
    return picked, why


def _require_fraction(name: str, value, *, allow_zero: bool = False) -> None:
    import numbers

    number = isinstance(value, numbers.Real) and not isinstance(value, bool)
    lo_ok = number and (value >= 0 if allow_zero else value > 0)
    if not (lo_ok and value <= 1):
        bound = "[0, 1]" if allow_zero else "(0, 1]"
        raise ValueError(f"{name} must be a fraction in {bound}; got {value!r}.")

def check_inputs(climate_input, *, n_clusters: int | None = None, climate_variables=None):
    """Check a climate + soil input before a long run, and report what is wrong with it.

    Runs everything :func:`create_clusters` would check, plus a full scan for missing
    values, without building features or clustering. Use it before submitting a job::

        report = check_inputs("/data/climate_input", n_clusters=2000)
        print(report)          # grid, time axes, units, and every problem found
        report.ok              # False if create_clusters would refuse or go wrong

    What it checks:

    * the input resolves -- every required file present (see :func:`create_clusters`);
    * every climate file is on exactly the ``soil_code`` grid;
    * the climate files share one time axis (same number of steps, all daily or all monthly);
    * units, judged from the values: temperature in °C or K; precipitation in mm per time
      step, not a flux in kg m-2 s-1; radiation in W m-2; humidity in %;
    * no missing values on land cells (cells with a valid ``soil_code``) in any input;
    * ``n_clusters``, if given, is a whole number between 1 and the number of land cells;
    * ``climate_variables``, if given, names known variables the input provides.

    Parameters
    ----------
    climate_input : DataPaths, GenericInputs, dict or str
        As for :func:`create_clusters`.
    n_clusters : int, optional
        The number of clusters you intend to ask for.
    climate_variables : sequence of str, optional
        As for :func:`create_clusters`.

    Returns
    -------
    InputCheck
        ``.ok``, ``.problems`` and ``.warnings`` (lists of plain sentences), and ``.facts``
        (grid size, resolution, land cells, time steps, units). ``print()`` it for a summary.
        It raises only for an argument of the wrong type; every problem with the *data* is
        reported in ``.problems``.

    Notes
    -----
    It reads every input in full once, which for a large daily dataset takes a while --
    but far less than a failed run.
    """
    from .data.checks import InputCheck, check_loaded_inputs, full_missing_scan

    try:
        datasets, registry = _load_input(climate_input)
    except (FileNotFoundError, ValueError, OSError, KeyError) as e:
        return InputCheck(problems=[str(e)])

    report = check_loaded_inputs(datasets, registry, static_names=_static_names(climate_input))
    if report.facts.get("grid", {}).get("n_land_cells"):
        report.problems += full_missing_scan(datasets, registry)
    if n_clusters is not None:
        try:
            _require_n_clusters(n_clusters, report.facts["grid"]["n_land_cells"])
        except ValueError as e:
            report.problems.append(str(e))
    if climate_variables is not None:
        try:
            _resolve_climate_variables(climate_variables, datasets, climate_input)
        except ValueError as e:
            report.problems.append(str(e))
    return report




def _load_input(climate_input):
    """Resolve and load ``climate_input``, dispatching to the generic path for a GenericInputs.

    Returns ``(datasets, registry)``, exactly as :func:`~climclusterpy.data.loader.load_climate_and_soil_data`
    does for a :class:`~climclusterpy.data.loader.DataPaths`/dict/directory input -- the two
    paths are interchangeable from here on.
    """
    from .data.generic_input import GenericInputs, load_generic_inputs
    from .data.loader import load_climate_and_soil_data, resolve_data_paths

    if isinstance(climate_input, GenericInputs):
        return load_generic_inputs(climate_input)
    paths = resolve_data_paths(climate_input)
    return load_climate_and_soil_data(paths)


def _static_names(climate_input) -> frozenset[str]:
    """The non-climate field names a GenericInputs spec declares, else empty.

    ``check_loaded_inputs``/``_check_before_run`` already treat ``{"soil_code",
    "soil_depth"}`` as static unconditionally -- this adds whatever a generic input's own
    ``static`` fields are called, so they aren't flagged for lacking a time dimension.
    """
    from .data.generic_input import GenericInputs

    if isinstance(climate_input, GenericInputs):
        return frozenset(climate_input.static)
    return frozenset()


def _generic_feature_defaults(
    climate_input,
) -> tuple[dict[str, str], list[str] | None, frozenset[str]]:
    """``(variable_aggregations, static_variables, categorical_vars)`` a GenericInputs implies.

    Defaults (``{}``, ``None``, ``{"soil_code"}``) for anything that isn't a GenericInputs,
    so the LPJmL path's behavior (including ``summarize_soil_features``'s own
    ``["soil_depth", "soil_code"]`` default) is unaffected.
    """
    from .data.generic_input import ClimateField, GenericInputs

    if not isinstance(climate_input, GenericInputs):
        return {}, None, frozenset({"soil_code"})
    aggregations = {
        name: f.aggregation
        for name, f in climate_input.climate.items()
        if isinstance(f, ClimateField) and f.aggregation
    }
    static_variables = list(climate_input.static)
    categorical = frozenset(
        name for name, f in climate_input.static.items() if f.kind == "categorical"
    )
    return aggregations, static_variables, categorical


#: The climate variables ``climate_variables=`` may name.
_KNOWN_CLIMATE_VARIABLES = ("tas", "pr", "rsds", "rlds", "hurs", "spei")


def _resolve_climate_variables(climate_variables, datasets, climate_input=None) -> list[str]:
    """The climate variables to summarize: the caller's list, checked, or the defaults present."""
    from .data.generic_input import GenericInputs

    if isinstance(climate_input, GenericInputs):
        # A generic input names its own climate fields -- there is no fixed known set to
        # check against, just the fields the spec actually declared. A field whose name
        # (or data) matches a known climate variable (temperature, say, named "temperature"
        # rather than "tas") dispatches under that variable's logical short code instead of
        # its own key, so it keeps that variable's unit handling -- see
        # resolve_variable_dispatch_names().
        from .features.climate import resolve_variable_dispatch_names

        climate_datasets = {k: datasets[k] for k in climate_input.climate if k in datasets}
        dispatch_names = resolve_variable_dispatch_names(climate_datasets)
        known = sorted(set(dispatch_names.values()))
        if climate_variables is None:
            return known
        names = (
            [climate_variables] if isinstance(climate_variables, str) else list(climate_variables)
        )
        translated = [dispatch_names.get(n, n) for n in names]
        unknown = [n for n, t in zip(names, translated) if t not in known]
        if unknown:
            raise ValueError(f"Unknown climate_variables {unknown}; this input provides {known}.")
        return translated

    if climate_variables is None:
        # Humidity is an optional input: summarize it when it is there, say nothing when not.
        return [v for v in _DEFAULT_CLIMATE_VARIABLES if v != "hurs" or "humid" in datasets]
    names = [climate_variables] if isinstance(climate_variables, str) else list(climate_variables)
    unknown = [v for v in names if v not in _KNOWN_CLIMATE_VARIABLES]
    if unknown:
        raise ValueError(
            f"Unknown climate_variables {unknown}; the known ones are "
            f"{list(_KNOWN_CLIMATE_VARIABLES)} (tas = temperature, pr = precipitation, rsds/rlds "
            "= short/long-wave radiation, hurs = humidity, spei = drought index)."
        )
    if "hurs" in names and "humid" not in datasets:
        raise ValueError(
            "climate_variables asks for 'hurs' (humidity), but the input has no humid file."
        )
    return names


def _resolve_cell_budget(
    representative: str, npatch: int, cells_per_cluster: int | None, n_repeat: int
) -> tuple[int, int]:
    """``(cells_per_cluster, patches_per_cell)`` for a cell choice, or a ValueError saying why not.

    ``npatch`` is the patch budget per cluster. A single-cell choice spends it all in one cell;
    ``"distributed"`` splits it evenly over ``cells_per_cluster`` cells, so every choice costs
    the vegetation model the same number of patches per cluster and the results are comparable.
    """
    _require_n_clusters(npatch, name="npatch (patches per cluster)")
    _require_n_clusters(n_repeat, name="n_repeat (copies of each chosen cell)")
    if cells_per_cluster is not None:
        _require_n_clusters(cells_per_cluster, name="cells_per_cluster")
    if representative != "distributed":
        if cells_per_cluster is not None:
            raise ValueError(
                f"cells_per_cluster only applies to representative='distributed'; got "
                f"representative={representative!r}, which writes one cell per cluster."
            )
        return 1, int(npatch)
    if cells_per_cluster is None:
        raise ValueError(
            "representative='distributed' needs cells_per_cluster: how many member cells of "
            "each cluster to simulate. Each then runs npatch // cells_per_cluster patches, e.g. "
            "npatch=25, cells_per_cluster=25 -> 25 cells of 1 patch each."
        )
    if int(cells_per_cluster) < 1:
        raise ValueError(f"cells_per_cluster must be >= 1; got {cells_per_cluster}.")
    if n_repeat != 1:
        raise ValueError(
            "n_repeat applies to single-cell choices (it copies one cell's climate into several "
            "columns); representative='distributed' already writes several different cells. "
            "Leave n_repeat=1."
        )
    if npatch % cells_per_cluster:
        raise ValueError(
            f"npatch={npatch} patches per cluster cannot be split evenly over "
            f"cells_per_cluster={cells_per_cluster} cells; choose values where cells_per_cluster "
            f"divides npatch (e.g. {cells_per_cluster} x {max(1, npatch // cells_per_cluster)} = "
            f"{cells_per_cluster * max(1, npatch // cells_per_cluster)})."
        )
    return int(cells_per_cluster), int(npatch) // int(cells_per_cluster)


def _load_extraction_inputs(datasets, registry, labels_csv: str, *, preload_climate: bool):
    """The label table and climate arrays every timeseries extractor reads."""
    import gc

    from .timeseries.representative import load_and_clean_cluster_labels

    df_labels = load_and_clean_cluster_labels(labels_csv, registry)

    # Any dataset with a time dimension is a climate input to carry through to the writer --
    # the fixed LPJmL set, "humid", or a custom name from a non-LPJmL input alike. soil_code
    # and soil_depth are static (no time dimension) and are never part of this dict.
    climate = {
        name: ds
        for name, ds in datasets.items()
        if name not in ("soil_code", "soil_depth")
        and any("time" in ds[v].dims for v in ds.data_vars)
    }
    if preload_climate:
        # The extractor fans out over threads; dask-backed arrays are not reliably
        # thread-safe to read, so materialise once (as float32 to halve the cost).
        for name in list(climate):
            climate[name] = climate[name].load()
            for var in list(climate[name].data_vars):
                if climate[name][var].dtype == np.float64:
                    climate[name][var] = climate[name][var].astype(np.float32)
            gc.collect()
    return df_labels, climate

def _source_spacing(registry) -> float:
    """The original grid's cell size in degrees, read off its latitude axis.

    Taken from the axis's two ends, ``(last - first) / (n - 1)``, after checking that every
    step agrees with it. Before 0.3.0 it was the smallest gap between neighbours, which float32
    storage shrinks: 0.070297° instead of 0.0703125° on the 9 km German grid, enough to put
    every cell off its own lattice and stop the build.
    """
    grid = registry.get_original_grid()
    axis, what = np.sort(np.asarray(grid.latitudes, dtype=float)), "latitude"
    if axis.size < 2:
        axis, what = np.sort(np.asarray(grid.longitudes, dtype=float)), "longitude"
    if axis.size < 2:
        raise ValueError("Cannot infer the grid spacing of a one-cell grid.")
    step = (axis[-1] - axis[0]) / (axis.size - 1)
    steps = np.diff(axis)
    tolerance = 10.0**-5 + 8.0 * float(np.finfo(np.float32).eps) * float(np.max(np.abs(axis)))
    if np.any(np.abs(steps - step) > tolerance):
        raise ValueError(
            f"The input's {what} axis is not evenly spaced (steps from {steps.min():.6f}° to "
            f"{steps.max():.6f}°), so the reduced grid cannot be put on the input's own "
            "lattice. With representative='medoid' or 'centroid', placement='cluster_center' "
            "searches for a lattice instead."
        )
    return float(step)


def _stamp_cell_selection(paths, attrs: dict) -> None:
    """Write the cell-choice and patch arithmetic into each NetCDF's global attributes."""
    import netCDF4 as nc4

    for path in paths:
        with nc4.Dataset(path, "a") as ncf:
            for key, value in attrs.items():
                ncf.setncattr(key, value)


def _selected_cells_table(timeseries, labels_frame: pd.DataFrame) -> pd.DataFrame:
    """``cluster_label, cell_id, latitude, longitude, slot`` for every written cell."""
    dim = "cell" if "cell" in timeseries.dims else "cluster"
    table = pd.DataFrame(
        {
            "cluster_label": np.asarray(timeseries["cluster_label"].values).astype(int),
            "latitude": np.asarray(timeseries["latitude"].values, dtype=float),
            "longitude": np.asarray(timeseries["longitude"].values, dtype=float),
        }
    )
    if "cell_id" in timeseries and dim == "cell":
        table["cell_id"] = np.asarray(timeseries["cell_id"].values).astype(int)
        # The distributed extractor writes longitudes in [-180, 180]; report the source's.
        src = labels_frame.set_index("cell_id")
        table["latitude"] = src.loc[table["cell_id"], "latitude"].to_numpy()
        table["longitude"] = src.loc[table["cell_id"], "longitude"].to_numpy()
    else:
        key = labels_frame[["cell_id", "latitude", "longitude"]].copy()
        for c in ("latitude", "longitude"):
            key[c] = key[c].round(5)
            table[c] = table[c].round(5)
        table = table.merge(key, on=["latitude", "longitude"], how="left", validate="one_to_one")
        table["latitude"] = np.asarray(timeseries["latitude"].values, dtype=float)
        table["longitude"] = np.asarray(timeseries["longitude"].values, dtype=float)
    table["slot"] = table.groupby("cluster_label").cumcount()
    return table[["cluster_label", "cell_id", "latitude", "longitude", "slot"]]


def _require_complete_series(timeseries) -> None:
    """Refuse to write a model input whose chosen cells have gaps in their climate series.

    The feature summaries skip missing values, so a land cell missing some time steps
    clusters without complaint -- but if it is the cell chosen to stand for its cluster, the
    gap is copied into the model's forcing file, where the model cannot skip it.
    """
    gaps = []
    for name in timeseries.data_vars:
        if "time" not in timeseries[name].dims:
            continue
        da = timeseries[name]
        missing = da.isnull().any("time")
        n = int(missing.sum())
        if n:
            gaps.append(f"{name} ({n} of {missing.size})")
    if gaps:
        raise ValueError(
            "The cells chosen to stand for their clusters have missing values (NaN) in their "
            f"climate series: {', '.join(gaps)} chosen cells. The model cannot run on a "
            "forcing file with gaps. Fill the gaps in the input, or leave the affected cells "
            "out of the land mask; check_inputs(<input>) lists every land cell with a gap."
        )


def _resolve_write_fields(write_fields, datasets) -> frozenset[str] | None:
    """``write_fields`` checked against the fields the input actually has; ``None`` = all."""
    if write_fields is None:
        return None
    if isinstance(write_fields, str):
        write_fields = (write_fields,)
    fields = frozenset(str(f) for f in write_fields)
    unknown = sorted(fields - set(datasets))
    if unknown or not fields:
        raise ValueError(
            f"write_fields names {unknown or 'nothing'}; the input provides "
            f"{sorted(datasets)}. Pass a non-empty subset of those, or None for all of them."
        )
    return fields


def _only_fields(timeseries, fields, datasets):
    """``timeseries`` without the input fields that are not to be written.

    Only variables named after an input field are candidates; the bookkeeping the writers need
    (labels, coordinates, cell ids) is never dropped.
    """
    # The extractors always make soil_code/soil_depth columns, all-missing when the input has no
    # such field; written out they become a soil map of fill values (soil_code: -2147483648 at
    # every land cell) that a model reads without complaint. Never write a field nobody supplied.
    absent = [v for v in _SOIL_FIELDS if v in timeseries.data_vars and v not in datasets]
    if fields is None:
        return timeseries.drop_vars(absent)
    return timeseries.drop_vars(
        absent + [v for v in timeseries.data_vars if v in datasets and v not in fields]
    )


#: The static fields the extractors carry themselves (LPJmL's soil map and depth). Any other
#: static field of a GenericInputs is cut to the finished grid by _write_static_fields.
_SOIL_FIELDS = ("soil_code", "soil_depth")


def _extra_static_fields(climate_input, datasets, write_fields) -> list[str]:
    """A generic input's own static fields (beyond the soil pair) that are to be written."""
    fields = _resolve_write_fields(write_fields, datasets)
    return [
        name
        for name in sorted(_static_names(climate_input))
        if name not in _SOIL_FIELDS and (fields is None or name in fields)
    ]


def _check_static_placement(static_fields, placement: str) -> None:
    """Refuse, before any work, a layout that could not carry ``static_fields``."""
    if static_fields and placement == "cluster_center":
        raise ValueError(
            f"placement='cluster_center' cannot write the static field(s) {static_fields}: that "
            "layout records no source cell per grid position. Use the default "
            "placement='native', or leave them out of the output with write_fields= (they are "
            "still used for clustering)."
        )


def _write_static_fields(result, datasets, names, *, overwrite: bool) -> dict[str, str]:
    """Cut each named static field to the finished reduced grid, beside its climate files.

    Each grid position gets the value of the cell whose climate it carries, on the same lattice
    as the climate NetCDFs (:func:`~climclusterpy.io.subset.subset_inputs` does the cutting). The
    file is named like the climate files, with the field's name in front.
    """
    from .io.subset import _lattice, _subset_netcdf

    if not names:
        return {}
    # "<field><suffix>.nc": the suffix of any written field's file, e.g. "_4_clusters_…nc".
    field, path = next(
        (k, p)
        for k, p in result.netcdf_files.items()
        if not k.startswith("__") and p.endswith(".nc") and os.path.basename(p).startswith(k)
    )
    suffix = os.path.basename(path)[len(field) :]
    slots, lattice = result.slot_sources, _lattice(result)
    written = {}
    with tempfile.TemporaryDirectory() as tmp:
        for name in names:
            dst = os.path.join(result.output_dir, f"{name}{suffix}")
            if os.path.exists(dst) and not overwrite:
                raise FileExistsError(f"{dst} exists; pass overwrite=True to replace it.")
            src = os.path.join(tmp, f"{name}.nc")
            datasets[name].to_netcdf(src)
            _subset_netcdf(src, dst, slots, lattice, n_jobs=1)
            written[name] = dst
    return written


def _finalize_reduced_grid(
    *,
    datasets,
    registry,
    features_df: pd.DataFrame,
    feature_columns: list,
    labels: np.ndarray,
    labels_frame: pd.DataFrame,
    labels_csv: str,
    n_clusters: int,
    representative_cells: pd.DataFrame | None,
    resolved_representative: str,
    mode: str,
    supervision: SupervisionResult | None,
    diagnostics: dict,
    output_dir: str,
    n_repeat: int,
    npatch: int,
    min_grid_spacing: float,
    overwrite: bool,
    n_jobs: int,
    preload_climate: bool,
    cells_per_cluster: int = 1,
    patches_per_cell: int | None = None,
    seed: int | None = None,
    placement: str = "native",
    write_fields=None,
    learned_metric_dir: str | None = None,
    static_fields: list[str] | tuple[str, ...] = (),
) -> CreateClustersResult:
    """The tail every reduced-grid route shares: timeseries, id mapping, NetCDFs, result.

    Factored out when :func:`create_regions` arrived, so the two public doors cannot drift. The
    round trip depends on the id mapping being created *before* the NetCDF write — the grid writer
    puts the compact map next to it and both CSVs are needed to reconstruct — and that ordering is
    the kind of thing a second copy loses within one release.

    The two drawn cell choices branch here, so both doors get them:

    * ``"random"`` draws one member per cluster and builds with
      :func:`~climclusterpy.grid.native.build_reduced_grid`, so the drawn cell is placed at its
      own latitude. Medoid and centroid do the same under ``placement="native"`` (the default
      since 0.3.0); ``"cluster_center"`` keeps the old centre-latitude lattice.
    * ``"distributed"`` draws ``cells_per_cluster`` members per cluster and writes each at its
      own position on the source grid (:func:`~climclusterpy.grid.distributed.write_distributed_grid`).
    """
    from .grid.build import create_cluster_id_mapping, create_cluster_nc_files

    patches_per_cell = int(npatch) if patches_per_cell is None else int(patches_per_cell)
    diagnostics["representative"] = resolved_representative
    diagnostics["placement"] = placement
    diagnostics["cells_per_cluster"] = int(cells_per_cluster)
    diagnostics["patches_per_cell"] = patches_per_cell
    if resolved_representative in _SAMPLED_CELL_CHOICES:
        diagnostics["cell_selection_seed"] = seed
    stamp = {
        "cell_selection": resolved_representative,
        "cell_placement": placement,
        "npatch_per_cluster": int(npatch),
        "cells_per_cluster": int(cells_per_cluster),
        "patches_per_cell": patches_per_cell,
    }
    if seed is not None:
        stamp["cell_selection_seed"] = int(seed)

    id_mapping_csv: str | None
    compact_map_csv: str | None
    distributed_mapping_csv: str | None = None
    fields = _resolve_write_fields(write_fields, datasets)
    if fields is not None:
        diagnostics["write_fields"] = sorted(fields)

    if resolved_representative == "distributed":
        from .grid.distributed import write_distributed_grid
        from .timeseries.distributed import extract_distributed_cells

        # Never preloaded here: the extractor reads only the drawn cells, straight from the
        # input files where it can, and loading the whole field first would undo exactly that.
        # preload_climate still applies to a field that cannot be read that way.
        df_labels, climate = _load_extraction_inputs(
            datasets, registry, labels_csv, preload_climate=False
        )
        if "soil_code" not in datasets:
            raise ValueError(
                "representative='distributed' needs a 'soil_code' field (LPJmL's own soil "
                "map, or a static field named 'soil_code' in a generic input) -- it is an "
                "LPJmL-specific placement option. Use the default representative='auto' for "
                "a non-LPJmL input."
            )
        soil_code = datasets["soil_code"]
        soil_depth = datasets.get("soil_depth")
        if fields is not None:
            # A field nobody asked for is not read at all: on a daily global forcing the
            # climate read is nearly all of the cost of a distributed grid.
            climate = {name: ds for name, ds in climate.items() if name in fields}
        timeseries = extract_distributed_cells(
            sorted(df_labels["cluster_label"].unique()),
            df_labels,
            climate,
            soil_code.to_dataframe().reset_index(),
            soil_depth.to_dataframe().reset_index() if soil_depth is not None else None,
            soil_code,
            soil_depth,
            npatch=int(cells_per_cluster),
            n_jobs=n_jobs,
            seed=seed,
            preload=preload_climate,
        )
        _require_complete_series(timeseries)
        timeseries = _only_fields(timeseries, fields, datasets)
        netcdf_files = write_distributed_grid(
            timeseries,
            registry,
            output_dir,
            n_clusters=n_clusters,
            cluster_method=mode,
            extra_attrs=stamp,
            overwrite=overwrite,
        )
        distributed_mapping_csv = netcdf_files["__distributed_mapping__"]
        id_mapping_csv = compact_map_csv = None
    elif resolved_representative == "random":
        from .grid.native import build_reduced_grid

        timeseries = _extract_cluster_timeseries(
            datasets,
            registry,
            labels_csv,
            features_df=features_df,
            representative_cells=None,
            n_jobs=n_jobs,
            preload_climate=preload_climate,
            selection_method="random",
            seed=seed,
        )
        _require_complete_series(timeseries)
        timeseries = _only_fields(timeseries, fields, datasets)
        netcdf_files = build_reduced_grid(
            timeseries,
            output_dir,
            source_spacing=_source_spacing(registry),
            labels_csv=labels_csv,
            n_repeat=n_repeat,
            npatch=patches_per_cell,
            timeseries_type="random",
            cluster_method=mode,
            overwrite=overwrite,
        )
        id_mapping_csv = netcdf_files["__id_mapping__"]
        compact_map_csv = os.path.join(output_dir, f"cluster_compact_map_{n_clusters}.csv")
    elif placement == "native":
        from .grid.native import build_reduced_grid

        timeseries = _extract_cluster_timeseries(
            datasets,
            registry,
            labels_csv,
            features_df=features_df,
            representative_cells=representative_cells,
            n_jobs=n_jobs,
            preload_climate=preload_climate,
        )
        _require_complete_series(timeseries)
        timeseries = _only_fields(timeseries, fields, datasets)
        netcdf_files = build_reduced_grid(
            timeseries,
            output_dir,
            source_spacing=_source_spacing(registry),
            labels_csv=labels_csv,
            n_repeat=n_repeat,
            npatch=npatch,
            timeseries_type="representative",
            aggregation_type="monthly",
            cluster_method=mode,
            overwrite=overwrite,
        )
        id_mapping_csv = netcdf_files["__id_mapping__"]
        compact_map_csv = os.path.join(output_dir, f"cluster_compact_map_{n_clusters}.csv")
    else:
        timeseries = _extract_cluster_timeseries(
            datasets,
            registry,
            labels_csv,
            features_df=features_df,
            representative_cells=representative_cells,
            n_jobs=n_jobs,
            preload_climate=preload_climate,
        )
        _require_complete_series(timeseries)
        timeseries = _only_fields(timeseries, fields, datasets)

        # Create the id mapping first so its path is known, and so the grid writer puts
        # the compact map next to it — both CSVs are needed to reconstruct.
        _, id_mapping_csv = create_cluster_id_mapping(
            n_clusters=n_clusters,
            n_repeat=n_repeat,
            labels_csv=labels_csv,
            output_dir=output_dir,
            cluster_labels_order=timeseries["cluster_label"].values.tolist(),
        )

        netcdf_files = create_cluster_nc_files(
            timeseries_dataset=timeseries,
            output_dir=output_dir,
            n_repeat=n_repeat,
            npatch=npatch,
            aggregation_type="monthly",
            timeseries_type="representative",
            parallel=False,
            labels_csv=labels_csv,
            id_mapping_file=id_mapping_csv,
            cluster_method=mode,
            min_grid_spacing=min_grid_spacing,
            overwrite=overwrite,
        )
        compact_map_csv = os.path.join(output_dir, f"cluster_compact_map_{n_clusters}.csv")

    if compact_map_csv is not None and not os.path.exists(compact_map_csv):
        compact_map_csv = None

    _stamp_cell_selection(
        [p for k, p in netcdf_files.items() if not k.startswith("__") and p.endswith(".nc")],
        stamp,
    )
    selected_cells = _selected_cells_table(timeseries, labels_frame)
    if distributed_mapping_csv is not None:
        diagnostics["n_repeated_slots"] = int(
            selected_cells.duplicated(["cluster_label", "cell_id"]).sum()
        )

    result = CreateClustersResult(
        netcdf_files=netcdf_files,
        labels=labels_frame,
        labels_csv=labels_csv,
        id_mapping_csv=id_mapping_csv,
        compact_map_csv=compact_map_csv,
        registry=registry,
        n_clusters=int(pd.Series(labels).nunique()),
        mode=mode,
        representative=resolved_representative,
        output_dir=output_dir,
        supervision=supervision,
        feature_columns=feature_columns,
        diagnostics=diagnostics,
        cells_per_cluster=int(cells_per_cluster),
        patches_per_cell=patches_per_cell,
        selected_cells=selected_cells,
        distributed_mapping_csv=distributed_mapping_csv,
        placement=placement,
        learned_metric_dir=learned_metric_dir,
    )
    static_files = _write_static_fields(result, datasets, list(static_fields), overwrite=overwrite)
    if static_files:
        _stamp_cell_selection(list(static_files.values()), stamp)
        result.netcdf_files.update(static_files)
    _save_run_record(result)
    return result

def _slot_sources(result: CreateClustersResult) -> pd.DataFrame:
    columns = [
        "slot",
        "cluster_label",
        "cell_id",
        "source_latitude",
        "source_longitude",
        "grid_latitude",
        "grid_longitude",
    ]
    if result.distributed_mapping_csv is not None:
        m = pd.read_csv(result.distributed_mapping_csv)
        table = pd.DataFrame(
            {
                "slot": m["cell_index"],
                "cluster_label": m["cluster_label"],
                "cell_id": m["cell_id"],
                "source_latitude": m["original_latitude"],
                "source_longitude": m["original_longitude"],
                "grid_latitude": m["grid_latitude"],
                "grid_longitude": m["grid_longitude"],
            }
        )
    else:
        placement = result.netcdf_files.get("__placement__")
        if placement is None or not os.path.exists(placement):
            raise ValueError(
                "This reduced grid records no per-slot position: it was built with "
                "placement='cluster_center', the layout before 0.3.0. Rebuild it with the "
                "default placement='native' to get slot_sources."
            )
        from .io.subset import _lattice

        _, lon_axis = _lattice(result)
        placed = pd.read_csv(placement)
        source_ids = (
            result.selected_cells.drop_duplicates("cluster_label").set_index("cluster_label")[
                "cell_id"
            ]
            if result.selected_cells is not None
            else None
        )
        rows = []
        for r in placed.itertuples(index=False):
            for j in range(int(r.n_repeat)):
                rows.append(
                    (
                        int(r.cluster_label),
                        None if source_ids is None else int(source_ids.loc[r.cluster_label]),
                        float(r.source_lat),
                        float(r.source_lon),
                        float(r.grid_lat),
                        float(lon_axis[int(r.column_start) + j]),
                    )
                )
        table = pd.DataFrame(rows, columns=columns[1:])
        table.insert(0, "slot", np.arange(len(table)))
    lon_s = table["source_longitude"].where(
        table["source_longitude"] < 180, table["source_longitude"] - 360
    )
    lon_g = table["grid_longitude"].where(
        table["grid_longitude"] < 180, table["grid_longitude"] - 360
    )
    table["at_source"] = np.isclose(table["source_latitude"], table["grid_latitude"], atol=1e-4) & (
        np.isclose(lon_s, lon_g, atol=1e-4)
    )
    return table[[*columns, "at_source"]].reset_index(drop=True)


def _json_safe(value):
    """``value`` with numpy scalars unwrapped and anything JSON cannot hold dropped."""
    if isinstance(value, dict):
        out = {}
        for k, v in value.items():
            v = _json_safe(v)
            if v is not _DROP:
                out[str(k)] = v
        return out
    if isinstance(value, (list, tuple)):
        items = [_json_safe(v) for v in value]
        return [v for v in items if v is not _DROP]
    if isinstance(value, np.generic):
        return value.item()
    if isinstance(value, np.ndarray):
        return value.tolist() if value.size <= 1000 else _DROP
    if value is None or isinstance(value, (str, int, float, bool)):
        return value
    return _DROP

def _save_run_record(result: CreateClustersResult) -> None:
    """Write what :func:`load_clusters` needs to rebuild ``result`` in a later session.

    Paths are stored relative to the output folder, so the folder can be moved or copied to
    another machine as a whole.
    """
    import datetime
    import json

    from . import __version__

    out = os.path.abspath(result.output_dir)

    def rel(path):
        return None if path is None else os.path.relpath(os.path.abspath(path), out)

    result.registry.save(os.path.join(out, ORIGINAL_GRID_FILE))
    selected_csv = None
    if result.selected_cells is not None:
        selected_csv = os.path.join(out, f"selected_cells_{result.n_clusters}.csv")
        result.selected_cells.to_csv(selected_csv, index=False)
    record = {
        "format": "climclusterpy create_clusters run",
        "climclusterpy_version": __version__,
        "written": datetime.datetime.now().isoformat(timespec="seconds"),
        "n_clusters": result.n_clusters,
        "mode": result.mode,
        "representative": result.representative,
        "placement": result.placement,
        "cells_per_cluster": result.cells_per_cluster,
        "patches_per_cell": result.patches_per_cell,
        "netcdf_files": {k: rel(v) for k, v in result.netcdf_files.items()},
        "labels_csv": rel(result.labels_csv),
        "id_mapping_csv": rel(result.id_mapping_csv),
        "compact_map_csv": rel(result.compact_map_csv),
        "distributed_mapping_csv": rel(result.distributed_mapping_csv),
        "learned_metric_dir": rel(result.learned_metric_dir),
        "selected_cells_csv": rel(selected_csv),
        "original_grid": ORIGINAL_GRID_FILE,
        "feature_columns": list(result.feature_columns),
        "supervision": None
        if result.supervision is None
        else {
            "kind": result.supervision.kind,
            "n_classes": result.supervision.n_classes,
            "labels_csv": rel(result.supervision.labels_csv),
        },
        "diagnostics": _json_safe(result.diagnostics),
    }
    with open(os.path.join(out, RUN_RECORD), "w") as fh:
        json.dump(record, fh, indent=2)

def load_clusters(output_dir: str) -> CreateClustersResult:
    """Reload the result of an earlier :func:`create_clusters` (or :func:`create_regions`) run.

    The vegetation-model run between clustering and reconstruction usually takes days, so
    this is how a later Python session gets back everything
    :func:`reconstruct_original_grid` needs -- the mapping files, the per-cell labels and the
    original-grid description -- from the run's output folder alone::

        clusters = load_clusters("./clusters_2000")
        full = reconstruct_original_grid("vegc.nc", clusters=clusters)

    ``reconstruct_original_grid(..., clusters="./clusters_2000")`` does the same in one step.

    Parameters
    ----------
    output_dir : str
        The ``output_dir`` the run wrote to. It holds ``climclusterpy_run.json`` (written by
        every run since version 0.3.0) and ``original_grid.nc``. The folder may have been moved:
        paths are resolved relative to it.

    Returns
    -------
    CreateClustersResult
        As the original call returned it, except that ``supervision`` carries only the
        label table (``labels`` is re-read from its CSV when it still exists) and
        ``diagnostics`` only what could be stored as JSON.

    Raises
    ------
    FileNotFoundError
        If the folder, its run record, or a file the record names is missing.
    """
    import json

    from .data.registry import DataRegistry

    out = os.path.abspath(output_dir)
    record_path = os.path.join(out, RUN_RECORD)
    if not os.path.exists(record_path):
        if not os.path.isdir(out):
            raise FileNotFoundError(f"No such folder: {out}")
        raise FileNotFoundError(
            f"{out} has no {RUN_RECORD}. It is written by create_clusters/create_regions since "
            "climclusterpy 0.3.0; for an older run, pass the pieces to reconstruct_original_grid "
            "directly: registry=DataRegistry.from_soil_code(<the soil_code input>), "
            "id_mapping_csv= and compact_map_csv= (both in that folder)."
        )
    with open(record_path) as fh:
        record = json.load(fh)

    def path(rel, required=True):
        if rel is None:
            return None

    """Turn a climate grid into a model-ready cluster grid, in one call.

    Runs the whole reduction: load the input, summarize it into ecological features,
    optionally learn a distance metric from supervision, cluster the cells, pick a
    representative cell per cluster, and write the reduced-grid NetCDFs plus the
    mapping CSVs that :func:`reconstruct_original_grid` needs to undo it.

    Parameters
    ----------
    climate_input : DataPaths, GenericInputs, dict or str
        The climate and soil input: a
        :class:`~climclusterpy.data.loader.DataPaths`, a mapping of its field names
        to paths, or a directory holding one NetCDF per input named after the field
        (``temperature.nc``, ``precipitation.nc``, ``soil_code.nc``, …). For a model other
        than LPJmL, a :class:`~climclusterpy.GenericInputs`: any named climate and static
        fields and any land mask. Its static fields are used for clustering and written to
        the reduced grid under their own names.
    n_clusters : int, optional
        Number of clusters (the paper's ``K``) — i.e. how many cells the reduced
        grid will have. Give this or ``fraction``, not both.
    fraction : float, optional
        The number of clusters as a share of the input's valid (land) cells, in ``(0, 1]``:
        ``fraction=0.01`` on the 67 420-cell half-degree land grid is ``n_clusters=674``
        (rounded to the nearest whole number, at least 1). It counts clusters, not the cells
        the model then runs: with ``representative="distributed"`` each cluster is run on
        ``cells_per_cluster`` cells. The number used is logged and kept in ``diagnostics``.
    locally_weighted : bool, optional
        ``False`` (default) clusters under plain Euclidean distance: no supervision,
        no training, no optional dependencies. ``True`` learns a locally-weighted
        LMNN metric first, so ecologically meaningful directions in feature space
        count for more. Needs the ``[lmnn]`` extra. The trained metric is saved to
        ``<output_dir>/learned_metric`` (see :attr:`CreateClustersResult.learned_metric_dir`).
    learned_metric : LearnedMetric or str, optional
        Cluster under a metric trained earlier instead of training one: a :class:`LearnedMetric`
        from :func:`load_metric`, or a path it accepts (the ``learned_metric`` folder of an
        earlier run, or a ``model.pth``). Nothing is trained, so ``supervision``,
        ``supervision_data``, ``lmnn_k``, ``rank`` and ``lmnn_kwargs`` are refused, and
        ``lmnn_mode`` and the training use of ``seed`` are ignored. The input's features must be exactly the
        ones the metric was trained on -- same number and, when the metric records them, the
        same names in the same order -- or it raises; a metric that records no names (a bare
        ``model.pth``) is checked on the number only, with a warning.

        **What is guaranteed.** Same features, same metric, same ``linkage_method`` and
        ``n_clusters``: the labels are the ones the metric was deployed with, provided the
        distances are computed on the same kind of device (a GPU and a CPU differ in the last
        bits of float32 arithmetic, which can flip a near-tied merge). Checked on the published
        global metric, see the changelog. **What is not.** The package standardises each
        feature against the input it is given, so the features of another climate or another
        region are on that input's own scale, not the training one's; applying a metric there
        is legitimate but no longer a reproduction. To transfer a metric to a new climate on
        the training scale, build the feature matrix yourself and use
        :func:`cluster_matrix` with ``learned_metric=``.
    supervision : str or sequence of str, optional
        ``locally_weighted=True`` only: what the learned distance is taught to reproduce.

        * ``"koppen_geiger"`` — Köppen–Geiger climate classes computed from the climate
          input itself; no extra data.
        * ``"ecoregions"`` — the polygons of ``ecoregions_shapefile`` (needs the
          ``[ecoregions]`` extra); no model output.
        * a model-output trait such as ``"biomass"``, or several, such as ``("sla",
          "wooddens", "height", "biomass")`` — clustered from ``supervision_data``, which is
          then required. Common aliases are accepted, so ``"biomass"`` matches a ``vegc``
          file.

        * labels you already have — a :class:`SupervisionResult` (e.g. from
          :func:`compute_supervision`), a table or ``.csv`` file with ``cell_id`` and
          ``cluster_label`` columns (matched to the grid by ``cell_id``), or an array of
          labels in the order of the grid's land cells. Nothing is recomputed, so cached
          labels (such as a Köppen–Geiger map that must not be re-derived) are used as they
          are.

        Default ``None``: those four traits when ``supervision_data`` is given,
        Köppen–Geiger otherwise.
    supervision_data : str, optional
        Trait ``supervision`` only, and required there: the vegetation-model output to
        learn from, as either a directory of per-trait NetCDFs or a prepared per-cell CSV
        (auto-detected).
    include_dominant_type : bool, optional
        Trait supervision: also use the categorical ``dominant_type`` trait when the input
        has it (default ``True``).
    supervision_granularity : int, optional
        How many supervision classes to form from the traits. Defaults to roughly
        one class per 316 cells, clamped to ``[30, 674]`` — 213 classes for the
        global 67,420-cell grid.
    ecoregions_shapefile : str, optional
        ``supervision="ecoregions"`` only, and required there: the polygon shapefile. A cell
        gets the polygon its centre falls in, or the nearest one.
    ecoregion_id_column : str, optional
        ``supervision="ecoregions"`` only: the attribute holding the region id (default
        ``"ECO_ID"``).
    ecoregion_min_cells : int, optional
        ``supervision="ecoregions"`` only: an ecoregion with fewer cells than this on the grid
        is merged into the nearest larger one, preferring the same biome and realm within
        1,500 km (see :func:`climclusterpy.supervision.merge_small_ecoregions`), because
        metric learning needs several same-class neighbours for every cell. Default 10, which
        turns the 815 Ecoregions2017 regions on the global half-degree land grid into 623
        classes. ``0`` keeps one class per ecoregion.
    lmnn_mode : str, optional
        Which distance is learned (see :mod:`climclusterpy.clustering.lmnn`):

        * ``"lowrank"`` (default) — a low-rank Mahalanobis metric for every cell, which
          also captures how features interact. The configuration of the published
          results.
        * ``"global"`` — one linear metric for the whole grid.
        * ``"local"`` and ``"enhanced"`` — a **diagonal** metric for every cell (one
          weight per feature, no interactions); ``"enhanced"`` adds class prototypes and
          oversampling for supervision classes of very uneven size.
    lmnn_k : int or None, optional
        Number of target neighbours per cell during metric learning (default 5).
        Not accepted by ``lmnn_mode="lowrank"``, which derives ``k`` per anchor.
    rank : int or None, optional
        ``lmnn_mode="lowrank"`` and ``"global"``: the rank of the learned metric, i.e.
        how many directions in feature space it can stretch. Default ``None``: with
        ``"lowrank"`` 16, or the number of features if that is smaller -- the setting of
        the published global application (16 on 25 features); with ``"global"`` the full
        rank, one direction per feature. Raises with ``"local"`` and ``"enhanced"``,
        whose metric is diagonal and has no rank.
    seed : int or None, optional
        Seed for every random number generator metric learning draws from, **and** for
        the cell draw of ``representative="random"`` / ``"distributed"``. Default ``0``,
        so two identical calls give identical clusters and cells, at any ``n_jobs``.
        ``None`` draws afresh on every call. It changes nothing where nothing is random:
        Euclidean clustering with a computed representative, or a ``learned_metric``.
    lmnn_kwargs : dict, optional
        Extra keyword arguments forwarded to the LMNN trainer. An unknown key
        raises :class:`TypeError` instead of being silently dropped.
    linkage_method : str, optional
        SciPy linkage method for the agglomerative cut (default ``"complete"``).
    supervision_linkage : str, optional
        Linkage method for the **supervision** dendrogram -- the one that turns
        continuous traits into ``supervision_granularity`` classes. Default
        ``"complete"``, as in the published results. ``"average"`` (the default before
        0.4.0) chains badly on these traits: one giant class persists even at very high
        granularity, so it gives weaker supervision. Distinct from ``linkage_method``,
        which cuts the cells under the learned metric.
    representative : str, optional
        Which real cell(s) stand for each cluster:

        * ``"medoid"`` — the member with the smallest total distance to the others,
          in the learned metric;
        * ``"centroid"`` — the member nearest the cluster mean in feature space;
        * ``"auto"`` (default) — medoid when a learned metric is available, centroid
          otherwise;
        * ``"random"`` — one member drawn at random; no representative is computed.
          The drawn cell keeps its own soil and is placed at its own latitude;
        * ``"distributed"`` — ``cells_per_cluster`` members drawn at random, each
          written at its own position on the source grid and each run with
          ``npatch // cells_per_cluster`` patches. No representative is computed,
          and the cluster's result is the mean (or median) of its cells, taken by
          :func:`reconstruct_original_grid`. A cluster smaller than
          ``cells_per_cluster`` has every member repeated (plus a random remainder).

        Every choice costs the model the same ``npatch`` patches per cluster, so
        their results are comparable. See the "Choosing the cells" how-to.
    placement : {"native", "cluster_center"}, optional
        Where a ``"medoid"`` or ``"centroid"`` cell is written on the reduced grid
        (``"random"`` and ``"distributed"`` always write each drawn cell at its own
        position):

        * ``"native"`` (default) — at the latitude of the cell whose climate it carries, on
          the source grid's own latitude spacing, so the model computes day length for the
          latitude that climate came from.
        * ``"cluster_center"`` — at the latitude of the member nearest the cluster's median
          absolute latitude, on a lattice searched with ``min_grid_spacing``. This was the
          only layout before 0.3.0; keep it to reproduce an earlier grid. That member is
          usually not the one whose climate is written there: across 26 real cluster sets,
          17–100 % of clusters ended up away from their own climate's latitude, by up to
          107°, some in the other hemisphere — up to 9.5 hours of day length.
    output_dir : str, optional
        Where the NetCDFs, label table, and mapping CSVs are written.
    n_repeat : int, optional
        Copies of the one chosen cell written per cluster (default 1). Not used with
        ``representative="distributed"``.
    npatch : int, optional
        The patch budget per cluster (default 250). With one cell per cluster that
        cell runs all of it; with ``"distributed"`` it is split evenly, and
        :attr:`CreateClustersResult.patches_per_cell` says what to set as the
        vegetation model's patch count. Recorded in every NetCDF's attributes.
    cells_per_cluster : int, optional
        ``representative="distributed"`` only, and required there: how many member
        cells per cluster to simulate. Must divide ``npatch``.
    climate_variables : sequence of str, optional
        Climate variables to summarize. Defaults to ``("tas", "pr", "rsds", "rlds",
        "hurs")``. Add ``"spei"`` for drought diagnostics — that needs the
        ``[features]`` extra (``pyet``).
    correlation_threshold, pca_variance_threshold, pca_contrib_threshold : float, optional
        Feature-filtering thresholds: drop features correlated above
        ``correlation_threshold``, keep components explaining
        ``pca_variance_threshold`` of variance, and drop features contributing less
        than ``pca_contrib_threshold``.
    min_grid_spacing : float, optional
        ``placement="cluster_center"`` only: minimum latitude spacing of the reduced grid,
        in degrees (default 0.1). Larger values give a coarser, much less memory-hungry
        output grid. Ignored, with a warning, under ``placement="native"``, whose spacing
        is the source grid's.
    n_jobs : int, optional
        Parallel workers for timeseries extraction (``-1`` = all cores).
    preload_climate : bool, optional
        Load the climate arrays into memory before the threaded extraction
        (default ``True``). Faster and thread-safe, but holds the input in RAM —
        set ``False`` for a very large grid on a small machine.
    overwrite : bool, optional
        Recompute and overwrite existing outputs in ``output_dir`` (default
        ``True``). ``False`` reuses NetCDFs that are already there.
    write_fields : sequence of str, optional
        Write only these input fields for the reduced grid (default ``None``: all of them),
        named as the input names them (``"soil_code"``, ``"temperature"``, ...). The label
        table and the mapping CSVs are always written, so the result still reconstructs.
        This is for a model that reads its inputs from the original files through the mapping
        (see :func:`subset_inputs`) and needs only the grid: ``write_fields=("soil_code",)``
        with ``representative="distributed"`` reads and writes no climate at all, which on
        a daily global forcing is the difference between an hour and seconds. With the other
        cell choices the climate is still read to pick and check the cells; only the writing
        is skipped. An unknown name raises :class:`ValueError`.

    Returns
    -------
    CreateClustersResult
        The written NetCDFs, the per-cell labels, the mapping CSVs, and the registry
        — everything :func:`reconstruct_original_grid` needs.

    Raises
    ------
    ValueError
        On an unknown ``representative``/``lmnn_mode``, an input that cannot be
        resolved, or supervision that does not cover the grid.
    ImportError
        For ``locally_weighted=True`` without the ``[lmnn]`` extra.

    Notes
    -----
    Cells are tracked by ``cell_id`` throughout, and the trait/supervision tables
    are checked cell-by-cell against the grid rather than by row count — a
    same-length but differently-ordered input is an error, not a silent
    mis-assignment.

    Examples
    --------
    >>> clusters = create_clusters("/data/climate", n_clusters=2000)  # doctest: +SKIP
    >>> sorted(clusters.netcdf_files)                                 # doctest: +SKIP
    ['__id_mapping__', 'long_wave_radiation', 'precipitation', ...]
    """
    if representative not in {"auto", "medoid", "centroid", *_SAMPLED_CELL_CHOICES}:
        raise ValueError(
            "representative must be 'auto', 'medoid', 'centroid', 'random' or "
            f"'distributed'; got {representative!r}."
        )
    if (n_clusters is None) == (fraction is None):
        raise TypeError(
            "create_clusters() needs exactly one of n_clusters= (a number of clusters) and "
            f"fraction= (a share of the valid cells); got n_clusters={n_clusters!r}, "
            f"fraction={fraction!r}."
        )
    if fraction is None:
        n_clusters = _require_n_clusters(n_clusters)
    else:
        _require_fraction("fraction", fraction)
    _require_fraction("correlation_threshold", correlation_threshold)
    _require_fraction("pca_variance_threshold", pca_variance_threshold)
    _require_fraction("pca_contrib_threshold", pca_contrib_threshold, allow_zero=True)
    min_grid_spacing = _resolve_placement(
        placement, representative, min_grid_spacing, "create_clusters"
    )
    cells_per_cluster, patches_per_cell = _resolve_cell_budget(
        representative, npatch, cells_per_cluster, n_repeat
    )
    draws_cells = representative in _SAMPLED_CELL_CHOICES

    from .data.registry import require_valid_cell_alignment
    from .features.assemble import build_feature_table

    # Metric-learning arguments are meaningless without a metric to learn. Reject
    # them rather than accept-and-ignore: a caller who passes rank= and gets no
    # learned metric has been told the wrong thing about their own run. seed= is the
    # exception: it has a default, and where nothing is random it changes nothing.
    if learned_metric is not None:
        training_only = [
            name
            for name, value in (
                ("supervision", supervision),
                ("supervision_data", supervision_data),
                ("supervision_granularity", supervision_granularity),
                ("ecoregions_shapefile", ecoregions_shapefile),
                ("lmnn_k", lmnn_k),
                ("rank", rank),
                ("lmnn_kwargs", lmnn_kwargs),
            )
            if value is not None
        ]
        if training_only:
            raise ValueError(
                f"{', '.join(training_only)} configure training a metric; learned_metric= uses "
                "one that is already trained, so they would have no effect. Drop them."
            )
        if not isinstance(learned_metric, LearnedMetric):
            learned_metric = load_metric(learned_metric)
    elif not locally_weighted:
        ignored = [
            name
            for name, value in (
                ("rank", rank),
                ("lmnn_kwargs", lmnn_kwargs),
                ("supervision", supervision),
                ("supervision_data", supervision_data),
                ("ecoregions_shapefile", ecoregions_shapefile),
            )
            if value is not None
        ]
        if ignored:
            raise ValueError(
                f"{', '.join(ignored)} require locally_weighted=True; Euclidean "
                "clustering trains no metric, so they would have no effect. Either "
                "set locally_weighted=True or drop them."
            )
    else:
        supervision_source, supervision_trait_names = _resolve_supervision(
            supervision, supervision_data, ecoregions_shapefile
        )

    os.makedirs(output_dir, exist_ok=True)
    datasets, registry = _load_input(climate_input)
    if fraction is not None:
        n_clusters = _clusters_for_fraction(fraction, len(registry.get_valid_original_ids()))
    _check_before_run(datasets, registry, n_clusters, climate_input)
    _resolve_write_fields(write_fields, datasets)  # fail before the expensive part
    static_fields = _extra_static_fields(climate_input, datasets, write_fields)
    _check_static_placement(static_fields, placement)

    _agg_defaults, _static_defaults, _categorical_defaults = _generic_feature_defaults(
        climate_input
    )
    features_df, feature_columns, climate_df = build_feature_table(
        datasets,
        registry,
        climate_variables=_resolve_climate_variables(climate_variables, datasets, climate_input),
        correlation_threshold=correlation_threshold,
        pca_variance_threshold=pca_variance_threshold,
        pca_contrib_threshold=pca_contrib_threshold,
        variable_aggregations=_agg_defaults or None,
        static_variables=_static_defaults,
        categorical_vars=_categorical_defaults,
    )
    features_df = require_valid_cell_alignment(features_df, registry, "The feature table")
    X = features_df[feature_columns].to_numpy(dtype=float)

    n_cells = X.shape[0]
    if n_clusters > n_cells:
        raise ValueError(
            f"n_clusters ({n_clusters}) exceeds the number of valid grid cells ({n_cells}). "
            "The reduced grid cannot have more cells than the original."
        )

    diagnostics: dict = {
        "n_cells": n_cells,
        "n_features": len(feature_columns),
    }
    if fraction is not None:
        diagnostics["fraction"] = float(fraction)
    supervision_result: SupervisionResult | None = None
    distance_matrix = None

    learned_metric_dir = None
    if learned_metric is not None:
        if learned_metric.feature_names is None:
            warnings.warn(
                "The learned metric records no feature names, so only their number "
                f"({learned_metric.n_features}) could be checked against this input's "
                f"{len(feature_columns)} features, not their identity or order.",
                UserWarning,
                stacklevel=2,
            )
        want_medoid = representative in {"auto", "medoid"} and learned_metric.kind != "global"
        labels, _Z, distance_matrix = _cluster_with_learned_metric(
            X,
            n_clusters,
            learned_metric,
            linkage_method=linkage_method,
            feature_names=feature_columns,
            keep_distances=want_medoid,
        )
        diagnostics["learned_metric"] = _metric_summary(learned_metric)
        mode = "lmnn"
    elif locally_weighted:
        supervision_result = _supervision_for_lmnn(
            source=supervision_source,
            traits=supervision_trait_names,
            supervision_data=supervision_data,
            ecoregions_shapefile=ecoregions_shapefile,
            ecoregion_id_column=ecoregion_id_column,
            ecoregion_min_cells=ecoregion_min_cells,
            include_dominant_type=include_dominant_type,
            supervision_granularity=supervision_granularity,
            features_df=features_df,
            climate_df=climate_df,
            registry=registry,
            linkage_method=supervision_linkage,
            output_dir=output_dir,
        )
        y = _aligned_supervision_labels(supervision_result, features_df)
        diagnostics["supervision_kind"] = supervision_result.kind
        diagnostics["n_supervision_classes"] = supervision_result.n_classes

        from .clustering.lmnn.pipeline import run_lmnn_clustering

        # Medoids need the learned distance matrix; 'global' mode never builds one,
        # so asking for it there would raise. Fall through to centroid instead.
        want_medoid = representative in {"auto", "medoid"} and lmnn_mode != "global"
        extra = _check_lmnn_kwargs(
            lmnn_kwargs,
            {
                "mode": lmnn_mode,
                "linkage_method": linkage_method,
                "k": lmnn_k,
                "rank": rank,
                "seed": seed,
                "return_distance_matrix": want_medoid,
                "return_model": lmnn_mode != "global",
            },
        )
        lmnn_result = run_lmnn_clustering(
            X,
            y,
            n_clusters,
            mode=lmnn_mode,
            linkage_method=linkage_method,
            k=lmnn_k,
            rank=rank,
            seed=seed,
            return_distance_matrix=want_medoid,
            return_model=lmnn_mode != "global",
            **extra,
        )
        labels = np.asarray(lmnn_result.labels)
        distance_matrix = lmnn_result.distance_matrix
        diagnostics["lmnn"] = lmnn_result.diagnostics
        learned_metric_dir = _save_trained_metric(
            lmnn_result,
            feature_columns,
            output_dir,
            {
                "trained_by": "create_clusters",
                "lmnn_mode": lmnn_mode,
                "seed": seed,
                "supervision_kind": supervision_result.kind,
                "n_supervision_classes": supervision_result.n_classes,
            },
        )
        diagnostics["supervision_linkage"] = supervision_linkage
        diagnostics["seed"] = seed
        mode = "lmnn"
    else:
        from scipy.cluster.hierarchy import fcluster

        from .clustering.hierarchical import compute_dendrogram

        Z = compute_dendrogram(X, method=linkage_method, metric="euclidean")
        labels = np.asarray(fcluster(Z, n_clusters, criterion="maxclust"))
        mode = "euclidean"

    labels_frame, labels_csv = _write_cluster_labels_csv(registry, labels, n_clusters, output_dir)

    # Medoids need a distance matrix; without one (Euclidean mode, or an LMNN mode
    # that did not return it) fall back to nearest-to-centroid in feature space.
    if representative == "medoid" and distance_matrix is None:
        raise ValueError(
            "representative='medoid' needs a learned distance matrix, which only a "
            "per-cell lmnn_mode ('local', 'enhanced', 'lowrank') with "
            "locally_weighted=True produces. Euclidean clustering and "
            "lmnn_mode='global' have no such matrix — use representative='centroid' "
            "(or 'auto', which falls back automatically)."
        )
    use_medoid = distance_matrix is not None and representative in {"auto", "medoid"}

    representative_cells = None
    if use_medoid:
        from .clustering.medoid import select_medoids

        medoids = select_medoids(distance_matrix, labels)
        representative_cells = pd.DataFrame(
            {
                "cluster_label": list(medoids),
                "latitude": [features_df["latitude"].iloc[i] for i in medoids.values()],
                "longitude": [features_df["longitude"].iloc[i] for i in medoids.values()],
            }
        )
    if draws_cells:
        representative_cells = None
        resolved_representative = representative
    else:
        resolved_representative = "medoid" if use_medoid else "centroid"

    return _finalize_reduced_grid(
        datasets=datasets,
        registry=registry,
        features_df=features_df,
        feature_columns=feature_columns,
        labels=labels,
        labels_frame=labels_frame,
        labels_csv=labels_csv,
        n_clusters=n_clusters,
        representative_cells=representative_cells,
        resolved_representative=resolved_representative,
        mode=mode,
        supervision=supervision_result,
        diagnostics=diagnostics,
        output_dir=output_dir,
        n_repeat=n_repeat,
        npatch=npatch,
        min_grid_spacing=min_grid_spacing,
        overwrite=overwrite,
        n_jobs=n_jobs,
        preload_climate=preload_climate,
        cells_per_cluster=cells_per_cluster,
        patches_per_cell=patches_per_cell,

        seed=seed if draws_cells else None,
        placement=placement,
        learned_metric_dir=learned_metric_dir,
        static_fields=static_fields,
    )




def _region_labels(
    partition,
    features_df: pd.DataFrame,
    *,
    climate_df,
    registry,
    ecoregions_shapefile: str | None,
    ecoregion_id_column: str,
    cold_threshold: float,
    output_dir: str,
    ecoregion_min_cells: int = 0,
    column: str | None = None,
    fraction: float | None = None,
) -> tuple[np.ndarray, SupervisionResult | None, str]:
    """Coerce whatever the caller passed as ``partition`` into labels in ``features_df`` row order.

    Returns ``(labels, supervision_or_None, description)``. Four input forms, in precedence order:
    a :class:`SupervisionResult`, one of the two literals ``"koppen_geiger"`` / ``"ecoregions"``
    (computed here), a path to a labels CSV, a DataFrame, or a bare array.

    Every form that carries a ``cell_id`` is aligned **by** ``cell_id`` rather than by row order.
    That is not defensiveness: :func:`~climclusterpy.supervision.koppen_geiger.assign_cells_to_koppen_geiger`
    drops cells it cannot classify, so a Köppen–Geiger table is routinely *shorter* than the grid,
    and a length check would pass on a table that is silently misaligned. A bare array is the one
    form with nothing to align on, so it is length-checked and documented as positional.
    """
    if (column is not None or fraction is not None) and not (
        isinstance(partition, pd.DataFrame)
        or (isinstance(partition, str) and partition not in {"koppen_geiger", "ecoregions"})
    ):
        raise ValueError(
            "column= and fraction= pick a column of a labels table (a CSV path or a DataFrame); "
            f"they have no effect on partition={_short_repr(partition)}."
        )
    if isinstance(partition, SupervisionResult):
        return (
            _aligned_supervision_labels(partition, features_df),
            partition,
            f"{partition.kind} ({partition.n_classes} classes, supplied)",
        )

    if isinstance(partition, str) and partition in {"koppen_geiger", "ecoregions"}:
        if partition == "koppen_geiger":
            supervision = compute_supervision(
                "koppen_geiger",
                features_df=climate_df,
                cold_threshold=cold_threshold,
                output_dir=os.path.join(output_dir, "partition"),
            )
        else:
            if not ecoregions_shapefile:
                raise ValueError(
                    "partition='ecoregions' needs ecoregions_shapefile=<path to "
                    "Ecoregions2017.shp or equivalent>. The polygons are external data; the "
                    "package does not ship them."
                )
            supervision = compute_supervision(
                "ecoregions",
                registry=registry,
                ecoregions_shapefile=ecoregions_shapefile,
                ecoregion_id_column=ecoregion_id_column,
                ecoregion_min_cells=ecoregion_min_cells,
                output_dir=os.path.join(output_dir, "partition"),
            )
        return (
            _aligned_supervision_labels(supervision, features_df),
            supervision,
            f"{supervision.kind} ({supervision.n_classes} classes, computed here)",
        )

    if isinstance(partition, str):
        if not os.path.exists(partition):
            raise ValueError(
                f"partition={partition!r} is neither 'koppen_geiger', 'ecoregions', nor an "
                "existing labels CSV."
            )
        frame = pd.read_csv(partition)
        source = f"labels CSV {os.path.basename(partition)}"
    elif isinstance(partition, pd.DataFrame):
        frame, source = partition, "labels DataFrame"
    else:
        y = np.asarray(partition)
        if y.ndim != 1:
            raise ValueError(f"partition array must be 1-D; got shape {y.shape}.")
        if len(y) != len(features_df):
            raise ValueError(
                f"partition has {len(y)} labels for {len(features_df)} valid grid cells. A bare "
                "array is matched POSITIONALLY and there is nothing to align on — pass a "
                "DataFrame or CSV with a 'cell_id' column if the orders may differ."
            )
        return y, None, f"array of {len(y)} labels (positional)"

    frame, chosen = _choose_partition_column(frame, column, fraction, len(features_df), source)
    if chosen is not None:
        source = f"{source}, {chosen}"
    if "cell_id" not in frame.columns:
        if len(frame) != len(features_df):
            raise ValueError(
                f"{source} has {len(frame)} rows for {len(features_df)} valid grid cells and no "
                "'cell_id' column to align on."
            )
        return frame["cluster_label"].to_numpy(), None, source

    merged = features_df[["cell_id"]].merge(
        frame[["cell_id", "cluster_label"]], on="cell_id", how="left", validate="one_to_one"
    )
    if merged["cluster_label"].isna().any():
        n_missing = int(merged["cluster_label"].isna().sum())
        raise ValueError(
            f"{source} covers only {len(merged) - n_missing} of {len(merged)} valid grid cells "
            f"({n_missing} without a label). Every cell must belong to a region, or the reduced "
            "grid silently describes a smaller grid than the one you will reconstruct onto."
        )
    return merged["cluster_label"].to_numpy(), None, source


def _short_repr(value) -> str:
    text = repr(value) if isinstance(value, str) else type(value).__name__
    return text if len(text) <= 60 else text[:57] + "..."


def _feature_space_medoids(X: np.ndarray, labels: np.ndarray) -> dict[int, int]:
    """``{label: row index}`` — each region's medoid under the plain Euclidean feature distance.

    The same rule :func:`~climclusterpy.clustering.medoid.select_medoids` applies, and literally
    that function: the distance block is built per region and handed to it, so the definition of a
    medoid lives in one place. What differs is that no global ``n x n`` matrix is ever formed —
    on a 67 420-cell grid that matrix is 18 GB, which is the memory wall this route exists to
    avoid. Peak memory here is the largest single region squared.
    """
    from scipy.spatial.distance import pdist, squareform

    from .clustering.medoid import select_medoids

    medoids: dict[int, int] = {}
    for label in np.unique(labels):
        members = np.flatnonzero(labels == label)
        if members.size == 1:
            medoids[int(label)] = int(members[0])
            continue
        block = squareform(pdist(X[members]))
        inner = select_medoids(block, np.zeros(members.size, dtype=int))[0]
        medoids[int(label)] = int(members[inner])
    return medoids

def create_regions(
    climate_input,
    partition,
    *,
    column: str | None = None,
    fraction: float | None = None,
    ecoregions_shapefile: str | None = None,
    ecoregion_id_column: str = "ECO_ID",
    ecoregion_min_cells: int = 0,
    cold_threshold: float = 0.0,
    representative: str = "centroid",
    placement: str = "native",
    output_dir: str = "./climregion_output",
    n_repeat: int = 1,
    npatch: int = 250,
    cells_per_cluster: int | None = None,
    seed: int | None = 0,
    climate_variables=None,
    correlation_threshold: float = 0.9,
    pca_variance_threshold: float = 0.95,
    pca_contrib_threshold: float = 0.01,
    min_grid_spacing: float | None = None,
    n_jobs: int = -1,
    preload_climate: bool = True,
    overwrite: bool = True,
    write_fields=None,
) -> CreateClustersResult:
    """Build a reduced grid from an EXISTING map of regions — no distance learned, no clustering.

    This is the cheapest route the package offers, and the one to reach for when there is no
    historic run of the model to supervise a distance with and no appetite for the clustering step
    either. An existing categorical map — a Köppen–Geiger climate classification, a published
    ecoregion map, a management-unit or country layer of your own — *is* the partition. Every cell
    keeps the class the map gives it, one representative cell is chosen per class, and the result
    is the same :class:`CreateClustersResult` :func:`create_clusters` returns, so
    :func:`reconstruct_original_grid` consumes it unchanged.

    **The number of regions comes from the map; it is counted, not requested.** To get, say,
    1 % of the cells from a saved table that holds several cuts of one dendrogram (columns
    ``cluster_k7`` … ``cluster_k21320``), pass ``fraction=0.01`` and the cut nearest that many
    regions is used and reported, or name the cut with ``column="cluster_k674"``.

    What it does **not** do is as important as what it does. No metric is trained, so there is no
    seed, no supervision signal, no rank and no training time. No dendrogram is built, so there is
    no :math:`O(n^2)` distance matrix and no memory ceiling from one. On the global half-degree
    land grid that is the difference between an 18 GB array and none at all.

    Parameters
    ----------
    climate_input : DataPaths, GenericInputs, dict or str
        As :func:`create_clusters`: a ``DataPaths``, a ``GenericInputs``, a mapping of field
        names to paths, or a directory of ``<field>.nc`` files.
    partition : SupervisionResult, str, pandas.DataFrame or array-like
        The map. One of:

        * a :class:`SupervisionResult` — what :func:`compute_supervision` returns;
        * ``"koppen_geiger"`` — derive the 30 Köppen–Geiger classes from ``climate_input`` itself
          and use them. Nothing external is needed, and because the classification is a function
          of climate alone it can be recomputed under a future climate;
        * ``"ecoregions"`` — rasterise ``ecoregions_shapefile`` onto the grid and use the polygons;
        * a path to a labels CSV with ``cell_id`` and ``cluster_label`` columns -- or with
          another label column, or several (``cluster_k7``, ``cluster_k21``, ...), picked with
          ``column`` or ``fraction``;
        * a :class:`pandas.DataFrame` with the same columns;
        * a 1-D array of labels, one per valid grid cell, matched **positionally**.

        There is deliberately no ``n_clusters`` parameter. The number of regions is a property of
        the map, it is *counted* rather than requested, and a map whose classes do not all occur
        on your grid simply has fewer. Compare :attr:`CreateClustersResult.n_clusters` against what
        you expected rather than asserting it up front.
    column : str, optional
        A labels table only (CSV path or DataFrame): the column holding the region labels, when
        it is not ``cluster_label`` -- e.g. ``"cluster_k674"`` in a table with one column per
        cut. Give this or ``fraction``, not both.
    fraction : float, optional
        A labels table only: use the cut whose number of regions is nearest ``fraction`` times
        the number of valid cells, among the table's ``cluster_k<N>`` columns (and
        ``cluster_label``). ``fraction=0.01`` on the 67 420-cell half-degree land grid picks
        ``cluster_k674``. The chosen column and its share are logged and recorded in
        ``diagnostics["partition_source"]``.
    ecoregions_shapefile : str, optional
        ``partition="ecoregions"`` only, and required there: path to the ecoregion polygons
        (e.g. the WWF/RESOLVE ``Ecoregions2017.shp``; the package does not ship them). Needs
        the ``[ecoregions]`` extra. Each land cell gets the polygon its centre falls in or on,
        and a cell in no polygon gets the nearest one. A labels CSV already present in
        ``<output_dir>/partition`` is reused only when it was made from the same grid,
        shapefile and settings. Ignored for every other ``partition``.
    ecoregion_id_column : str, optional
        ``partition="ecoregions"`` only: the attribute column of the shapefile that holds the
        region id (default ``"ECO_ID"``). Its values become the region labels, so they must
        be numbers that convert to integers.
    ecoregion_min_cells : int, optional
        ``partition="ecoregions"`` only: merge every ecoregion with fewer cells than this into
        the nearest larger one, as ``create_clusters`` does for its training classes (see
        :func:`climclusterpy.supervision.merge_small_ecoregions`). Default ``0``: the map is
        used as it is, one region per ecoregion.
    cold_threshold : float, optional
        ``partition="koppen_geiger"`` only: the coldest-month mean temperature, in °C, that
        separates temperate (C) from continental (D) climates (default 0.0; -3.0 is the
        other common convention). A cell is temperate when its coldest month is warmer
        than this. Ignored for every other ``partition``.
    representative : str, optional
        How each region's one cell is chosen. ``"centroid"`` (the default) takes the member nearest
        its region's centroid in feature space — :math:`O(nd)`, no distance matrix, and the choice
        that keeps this route cheap end to end. ``"medoid"`` takes the member minimising the sum of
        distances to the other members of its region, which is the rule :func:`create_clusters`
        uses for a learned metric; here the distance is the plain Euclidean one on the same
        features, computed one region at a time, so peak memory is the largest region squared.
        ``"random"`` and ``"distributed"`` draw cells instead of computing a representative, as
        in :func:`create_clusters`: one random member, or ``cells_per_cluster`` random members
        each run with ``npatch // cells_per_cluster`` patches. They are sampling designs, not
        representatives — the point is that nothing about the region has to be summarised to
        pick them — so with these two **no feature table is built** (unless the partition is
        ``"koppen_geiger"``, which is computed from climate). On a daily global forcing that is
        the difference between minutes and hours.
    placement : {"native", "cluster_center"}, optional
        As :func:`create_clusters`: where a ``"centroid"`` or ``"medoid"`` cell is written.
        ``"native"`` (default) puts it at the latitude of the cell whose climate it carries;
        ``"cluster_center"`` is the layout before 0.3.0, kept to reproduce earlier grids.
    output_dir : str, optional
        Where the NetCDFs, label table and mapping CSVs are written (default
        ``"./climregion_output"``; created if missing). A partition computed here
        (``"koppen_geiger"`` or ``"ecoregions"``) is written to its ``partition``
        subfolder.
    n_repeat : int, optional
        As :func:`create_clusters`: copies of the one chosen cell written per region
        (default 1). Must be 1 with ``representative="distributed"``.
    npatch, cells_per_cluster, seed : int, optional
        As :func:`create_clusters`: the patch budget per region (default 250), the member
        cells per region for ``"distributed"`` (required there, must divide ``npatch``), and
        the seed of the cell draw of ``"random"``/``"distributed"`` (default 0, so the same
        call draws the same cells; ``None`` draws afresh). Nothing else here is random, so
        with ``"centroid"`` or ``"medoid"`` it changes nothing.
    climate_variables : sequence of str, optional
        Climate variables to summarize into features, from ``"tas"``, ``"pr"``, ``"rsds"``,
        ``"rlds"``, ``"hurs"`` and ``"spei"``. Defaults to ``("tas", "pr", "rsds", "rlds",
        "hurs")``, leaving out ``"hurs"`` when the input has no humidity file. An unknown
        name, or ``"hurs"`` without a humidity file, raises :class:`ValueError`, as in
        :func:`create_clusters`. Not used when no feature table is built (see
        ``representative``).
    correlation_threshold, pca_variance_threshold, pca_contrib_threshold : float, optional
        Feature-filtering thresholds, as in :func:`create_clusters` (defaults 0.9, 0.95 and
        0.01): drop features correlated above ``correlation_threshold``, keep components
        explaining ``pca_variance_threshold`` of variance, and drop features contributing
        less than ``pca_contrib_threshold``. Each must be a fraction in ``(0, 1]``
        (``[0, 1]`` for ``pca_contrib_threshold``).
    min_grid_spacing : float, optional
        As :func:`create_clusters`: ``placement="cluster_center"`` only, the minimum latitude
        spacing of the reduced grid, in degrees (default 0.1); must be positive. Ignored,
        with a warning, under ``placement="native"``.
    n_jobs : int, optional
        Parallel workers for timeseries extraction (``-1``, the default, uses all cores).
    preload_climate : bool, optional
        As :func:`create_clusters`: load the climate arrays into memory before the threaded
        extraction (default ``True``); set ``False`` for a very large grid on a small
        machine.
    overwrite : bool, optional
        As :func:`create_clusters`: recompute and overwrite existing outputs in
        ``output_dir`` (default ``True``). ``False`` reuses NetCDFs that are already there.
    write_fields : sequence of str, optional
        As :func:`create_clusters`: write only these input fields (default all of them).
        ``write_fields=("soil_code",)`` with ``representative="distributed"`` builds the grid
        and the mapping without reading any climate.

    Returns
    -------
    CreateClustersResult
        With ``mode="regions"``, ``n_clusters`` set to the number of regions actually found, and
        ``supervision`` carrying the map when one was computed or supplied.

    Raises
    ------
    ValueError
        On an unknown ``representative``; an invalid ``npatch`` / ``cells_per_cluster`` /
        ``n_repeat`` combination; ``partition="ecoregions"`` without ``ecoregions_shapefile``; a string ``partition``
        that is neither a known name nor an existing file; a labels table without a
        ``cluster_label`` column, or one that leaves any valid cell unlabelled; a bare array
        of the wrong length; or more regions than valid cells.
    ImportError
        For ``partition="ecoregions"`` without the ``[ecoregions]`` extra.

    Notes
    -----
    **This is not a cheaper version of supervising a learned distance with the same map.** Passing
    a Köppen–Geiger map to :func:`create_clusters` as supervision trains a distance to agree with
    those zones and then clusters the grid into however many clusters you ask for; the zones
    themselves are never the partition. Here they are. The two answer different questions and, on
    the global grid, give materially different accuracy.

    **The regions of a published map are not balanced.** A clustering produces clusters sized by
    the algorithm; a map produces whatever the cartographers drew. In the WWF/RESOLVE Ecoregions
    2017 layer on a half-degree land grid, 31 of the 815 regions present are a single grid cell —
    which those cells then reconstruct exactly, for no reason connected to the method — while
    ECO_ID 0, "rock and ice", is one region of 2 112 cells stretching from Patagonia to northern
    Greenland. Read the class-size distribution in ``diagnostics`` before reading the score.

    Examples
    --------
    >>> from climclusterpy import create_regions, reconstruct_original_grid
    >>> regions = create_regions("climate/", "koppen_geiger", output_dir="reduced_grid/")
    >>> regions.n_clusters                                        # doctest: +SKIP
    30
    >>> # run the vegetation model on reduced_grid/, then:
    >>> full = reconstruct_original_grid("model_out/", clusters=regions)   # doctest: +SKIP
    """
    if representative not in {"centroid", "medoid", *_SAMPLED_CELL_CHOICES}:
        raise ValueError(
            "representative must be 'centroid', 'medoid', 'random' or 'distributed'; got "
            f"{representative!r}. There is no 'auto' here because there is no learned metric to "
            "fall back from — the choice is always available and is always the caller's."
        )
    cells_per_cluster, patches_per_cell = _resolve_cell_budget(
        representative, npatch, cells_per_cluster, n_repeat
    )
    _require_fraction("correlation_threshold", correlation_threshold)
    _require_fraction("pca_variance_threshold", pca_variance_threshold)
    _require_fraction("pca_contrib_threshold", pca_contrib_threshold, allow_zero=True)
    min_grid_spacing = _resolve_placement(
        placement, representative, min_grid_spacing, "create_regions"
    )

    from .data.registry import require_valid_cell_alignment
    from .features.assemble import build_feature_table

    os.makedirs(output_dir, exist_ok=True)
    datasets, registry = _load_input(climate_input)
    _check_before_run(datasets, registry, None, climate_input)
    _resolve_write_fields(write_fields, datasets)  # fail before the expensive part
    static_fields = _extra_static_fields(climate_input, datasets, write_fields)
    _check_static_placement(static_fields, placement)

    # A drawn cell choice compares nothing, so it needs no summary of the climate. The feature
    # table is the expensive step on a daily global forcing, so it is skipped unless the
    # partition itself is computed from climate (Köppen–Geiger); the partition is then aligned
    # on the registry's own cell ids, which is all the other partition forms need.
    needs_features = representative not in _SAMPLED_CELL_CHOICES or (
        isinstance(partition, str) and partition == "koppen_geiger"
    )
    if needs_features:
        _agg_defaults, _static_defaults, _categorical_defaults = _generic_feature_defaults(
            climate_input
        )
        features_df, feature_columns, climate_df = build_feature_table(
            datasets,
            registry,
            climate_variables=_resolve_climate_variables(
                climate_variables, datasets, climate_input
            ),
            correlation_threshold=correlation_threshold,
            pca_variance_threshold=pca_variance_threshold,
            pca_contrib_threshold=pca_contrib_threshold,
            variable_aggregations=_agg_defaults or None,
            static_variables=_static_defaults,
            categorical_vars=_categorical_defaults,
        )
        features_df = require_valid_cell_alignment(features_df, registry, "The feature table")
    else:
        lats, lons = registry.get_valid_lat_lon()
        features_df = pd.DataFrame(
            {"cell_id": registry.get_valid_original_ids(), "latitude": lats, "longitude": lons}
        )
        feature_columns, climate_df = [], None
    X = features_df[feature_columns].to_numpy(dtype=float)

    labels, supervision, partition_source = _region_labels(
        partition,
        features_df,
        climate_df=climate_df,
        registry=registry,
        ecoregions_shapefile=ecoregions_shapefile,
        ecoregion_id_column=ecoregion_id_column,
        ecoregion_min_cells=ecoregion_min_cells,
        cold_threshold=cold_threshold,
        output_dir=output_dir,
        column=column,
        fraction=fraction,
    )
    labels = np.asarray(labels)
    region_ids, region_sizes = np.unique(labels, return_counts=True)
    n_regions = int(region_ids.size)
    if n_regions < 1:
        raise ValueError(f"{partition_source} produced no regions at all.")
    if n_regions > X.shape[0]:
        raise ValueError(
            f"{partition_source} has {n_regions} regions for {X.shape[0]} valid grid cells. The "
            "reduced grid cannot have more cells than the original."
        )

    diagnostics: dict = {
        "n_cells": int(X.shape[0]),
        "n_features": len(feature_columns),
        "partition_source": partition_source,
        "n_regions": n_regions,
        # The distribution, not just the count. A published map is unbalanced by nature and the
        # size of its smallest and largest regions changes what a reconstruction score means.
        "region_size_min": int(region_sizes.min()),
        "region_size_median": float(np.median(region_sizes)),
        "region_size_max": int(region_sizes.max()),
        "n_single_cell_regions": int((region_sizes == 1).sum()),
        "representative": representative,
        "feature_table_built": bool(needs_features),
    }

    labels_frame, labels_csv = _write_cluster_labels_csv(registry, labels, n_regions, output_dir)

    representative_cells = None
    if representative == "medoid":
        medoids = _feature_space_medoids(X, labels)
        representative_cells = pd.DataFrame(
            {
                "cluster_label": list(medoids),
                "latitude": [features_df["latitude"].iloc[i] for i in medoids.values()],
                "longitude": [features_df["longitude"].iloc[i] for i in medoids.values()],
            }
        )

    return _finalize_reduced_grid(
        datasets=datasets,
        registry=registry,
        features_df=features_df,
        feature_columns=feature_columns,
        labels=labels,
        labels_frame=labels_frame,
        labels_csv=labels_csv,
        n_clusters=n_regions,
        representative_cells=representative_cells,
        resolved_representative=representative,
        mode="regions",
        supervision=supervision,
        diagnostics=diagnostics,
        output_dir=output_dir,
        n_repeat=n_repeat,
        npatch=npatch,
        min_grid_spacing=min_grid_spacing,
        overwrite=overwrite,
        n_jobs=n_jobs,
        preload_climate=preload_climate,
        cells_per_cluster=cells_per_cluster,
        patches_per_cell=patches_per_cell,
        write_fields=write_fields,
        seed=seed if representative in _SAMPLED_CELL_CHOICES else None,
        placement=placement,
        static_fields=static_fields,
    )

def draw_cells(
    labels: pd.DataFrame,
    representative: str,
    *,
    cells_per_cluster: int | None = None,
    seed: int | None = 0,
) -> pd.DataFrame:
    """Which cells a drawn cell choice would simulate — the draw alone, with no grid built.

    The same draw :func:`create_clusters` and :func:`create_regions` make for
    ``representative="random"`` or ``"distributed"``: for the same label table (same rows, same
    order) and the same ``seed`` it returns the same cells as their ``selected_cells``. Use it
    to inspect a draw before paying for a build, to draw several seeds cheaply, or to emulate a
    choice on model output you already have for every cell.

    Parameters
    ----------
    labels : pandas.DataFrame
        One row per cell with ``cell_id``, ``cluster_label``, ``latitude`` and ``longitude`` —
        the ``cluster_labels_<K>.csv`` a build writes. Rows with a missing or negative
        ``cluster_label`` are dropped, as the builds drop them.
    representative : {"random", "distributed"}
        ``"random"``: one member per cluster. ``"distributed"``: ``cells_per_cluster`` members
        per cluster; a cluster with fewer members has every member repeated as often as fits,
        plus a random remainder.
    cells_per_cluster : int, optional
        ``"distributed"`` only, and required there.
    seed : int, optional
        Seeds the draw (default 0, the default of :func:`create_clusters` too). ``None``
        draws afresh on every call.

    Returns
    -------
    pandas.DataFrame
        ``cluster_label, cell_id, latitude, longitude, slot, is_repeated``, clusters in
        ascending label order. ``slot`` numbers a cluster's drawn cells from 0; ``is_repeated``
        marks a cell drawn from a cluster smaller than ``cells_per_cluster``.
    """
    from .timeseries.distributed import draw_distributed_cells
    from .timeseries.representative import draw_random_cells

    if representative not in _SAMPLED_CELL_CHOICES:
        raise ValueError(
            f"draw_cells draws cells for representative='random' or 'distributed'; got "
            f"{representative!r}. 'medoid' and 'centroid' are computed, not drawn."
        )
    missing = {"cell_id", "cluster_label", "latitude", "longitude"} - set(labels.columns)
    if missing:
        raise ValueError(f"labels is missing column(s) {sorted(missing)}")
    frame = labels.copy()
    frame["cluster_label"] = pd.to_numeric(frame["cluster_label"], errors="coerce")
    frame = frame.dropna(subset=["cluster_label"])
    frame["cluster_label"] = frame["cluster_label"].astype(int)
    frame = frame[frame["cluster_label"] >= 0]
    clusters = sorted(frame["cluster_label"].unique())

    if representative == "distributed":
        if cells_per_cluster is None or int(cells_per_cluster) < 1:
            raise ValueError(
                "representative='distributed' needs cells_per_cluster >= 1: how many member "
                "cells of each cluster to draw."
            )
        drawn = draw_distributed_cells(clusters, frame, int(cells_per_cluster), seed)
        drawn = drawn.rename(columns={"cell_index_in_cluster": "slot"})
    else:
        if cells_per_cluster not in (None, 1):
            raise ValueError(
                f"cells_per_cluster only applies to representative='distributed'; got "
                f"{cells_per_cluster!r} with 'random', which draws one cell per cluster."
            )
        drawn = draw_random_cells(clusters, frame, seed)
        drawn["slot"] = 0
        drawn["is_repeated"] = False
    drawn["cluster_label"] = drawn["cluster_label"].astype(int)
    drawn["cell_id"] = drawn["cell_id"].astype(int)
    cols = ["cluster_label", "cell_id", "latitude", "longitude", "slot", "is_repeated"]
    return drawn[cols].reset_index(drop=True)

def reconstruct_original_grid(
    model_output,
    clusters: CreateClustersResult | None = None,
    *,
    registry=None,
    id_mapping_csv: str | None = None,
    compact_map_csv: str | None = None,
    output_dir: str | None = None,
    kind: str | None = None,
    distributed_mapping_csv: str | None = None,
    labels_csv: str | None = None,
    aggregate: str = "mean",
) -> ReconstructResult:
    """Project model output from the reduced grid back onto the full original grid.

    Every original cell inherits the value of the cluster it belongs to, so a
    simulation run on ``n_clusters`` cells becomes a full-resolution field. Cells
    are placed by ID through the mapping CSVs, never by coordinate matching.

    Parameters
    ----------
    model_output : str or sequence of str
        The NetCDF(s) the vegetation model produced on the reduced grid — a single
        file, several files, or a directory of them.
    clusters : CreateClustersResult or str, optional
        The result of :func:`create_clusters`, or the ``output_dir`` it wrote to (which is
        read with :func:`load_clusters` -- the way back in a later session). Supplying it is
        the one-liner path: the registry and both mapping CSVs come from it.
    registry : DataRegistry, optional
        Original-grid description. Required when ``clusters`` is not given; get it with
        :meth:`DataRegistry.load` (``original_grid.nc`` in the run's folder) or
        :meth:`DataRegistry.from_soil_code` (the soil-code input that was clustered).
    id_mapping_csv, compact_map_csv : str, optional
        The ``cluster_id_mapping_{K}.csv`` and ``cluster_compact_map_{K}.csv``
        written when the reduced grid was built. Required when ``clusters`` is not
        given.
    output_dir : str, optional
        Where to write the reconstructed NetCDFs. Defaults to a
        ``reconstructed/`` directory beside the input.
    kind : {"output", "distributed"}, optional
        ``"output"`` for a grid with one simulated cell per cluster -- model output, or
        the reduced-grid NetCDFs this package wrote -- and ``"distributed"`` for a
        ``representative="distributed"`` grid. Default: ``"distributed"`` when
        ``clusters`` was built that way (or only ``distributed_mapping_csv`` is given),
        otherwise ``"output"``.
    distributed_mapping_csv, labels_csv : str, optional
        Needed for ``kind="distributed"``; taken from ``clusters`` when given.
    aggregate : {"mean", "median"}, optional
        ``kind="distributed"`` only: how a cluster's simulated cells are combined
        into the value every member receives. ``"mean"`` (default) matches what the
        vegetation model does across the patches inside one cell, so 25 cells × 1
        patch compares directly with 1 cell × 25 patches. ``"median"`` reproduces the
        reconstruction the method was originally published with.

    Returns
    -------
    ReconstructResult
        ``.files`` maps each input to its reconstructed output; ``.path`` is a
        shortcut when there is exactly one.

    Raises
    ------
    ValueError
        If neither ``clusters`` nor the explicit registry/mapping arguments are
        given, no NetCDF is found at ``model_output``, or the inputs do not belong
        together: the two mapping files describe different grids, the model output
        was produced on another reduced grid (values where this grid has no cluster,
        or cluster cells missing from it), or the registry is not the grid that was
        clustered. Nothing is written in that case.
    FileNotFoundError
        If a mapping file or the model output is missing.

    Examples
    --------
    >>> full = reconstruct_original_grid(  # doctest: +SKIP
    ...     "vegc_2000_clusters.nc", clusters=clusters
    ... )
    >>> full.path                                                   # doctest: +SKIP
    './clusters_2000/reconstructed/vegc_2000_clusters_reconstructed.nc'
    """
    if kind not in (None, "output", "distributed"):
        raise ValueError(f"Unknown reconstruction kind: {kind!r}. Use 'output' or 'distributed'.")
    if isinstance(clusters, (str, os.PathLike)):
        clusters = load_clusters(str(clusters))
    if clusters is not None:
        registry = registry if registry is not None else clusters.registry
        id_mapping_csv = id_mapping_csv or clusters.id_mapping_csv
        compact_map_csv = compact_map_csv or clusters.compact_map_csv
        labels_csv = labels_csv or clusters.labels_csv
        distributed_mapping_csv = distributed_mapping_csv or clusters.distributed_mapping_csv
        if kind is None:
            kind = "distributed" if clusters.representative == "distributed" else "output"
    if kind is None:
        kind = "distributed" if distributed_mapping_csv is not None else "output"

    if kind == "distributed":
        if registry is None or distributed_mapping_csv is None or labels_csv is None:
            raise ValueError(
                "Reconstructing a distributed grid needs clusters=<CreateClustersResult> or "
                "registry=, distributed_mapping_csv= and labels_csv=."
            )
    else:
        if aggregate != "mean":
            raise ValueError(
                f"aggregate={aggregate!r} only applies to a distributed grid (several cells per "
                f"cluster); kind={kind!r} has one cell per cluster, so there is nothing to combine."
            )
        if registry is None or id_mapping_csv is None or compact_map_csv is None:
            raise ValueError(
                "reconstruct_original_grid needs either clusters=<CreateClustersResult> or all "
                "of registry=, id_mapping_csv= and compact_map_csv=."
            )

    from .data.loader import list_netcdf_files

    if isinstance(model_output, (str, os.PathLike)):
        path = str(model_output)
        inputs = list_netcdf_files(path) if os.path.isdir(path) else [path]
    else:
        inputs = [str(p) for p in model_output]

    if not inputs:
        raise ValueError(f"No NetCDF files found at {model_output!r} to reconstruct.")

    if output_dir is None:
        output_dir = os.path.join(os.path.dirname(os.path.abspath(inputs[0])), "reconstructed")
    os.makedirs(output_dir, exist_ok=True)

    files: dict[str, str] = {}
    for source in inputs:
        stem = os.path.splitext(os.path.basename(source))[0]
        files[source] = _reconstruct_file(
            source,
            registry,
            id_mapping_csv=id_mapping_csv,
            compact_map_csv=compact_map_csv,
            output_path=os.path.join(output_dir, f"{stem}_reconstructed.nc"),
            kind=kind,
            distributed_mapping_csv=distributed_mapping_csv,
            labels_csv=labels_csv,
            aggregate=aggregate,
        )
    return ReconstructResult(files=files, output_dir=output_dir, kind=kind)

def cluster_matrix(
    features,
    n_clusters: int,
    *,
    locally_weighted: bool | None = None,
    supervision=None,
    linkage_method: str = "complete",
    metric: str = "euclidean",
    standardize: bool = False,
    feature_columns: list[str] | None = None,
    lmnn_mode: str = "lowrank",
    rank: int | None = None,
    seed: int | None = 0,
    return_distance_matrix: bool = False,
    return_model: bool = False,
    lmnn_kwargs: dict | None = None,
    learned_metric=None,
) -> ClusterResult:
    """Cluster grid cells from a feature matrix, under Euclidean or a learned distance.

    Parameters
    ----------
    features : numpy.ndarray or pandas.DataFrame
        Feature matrix ``(n_cells, n_features)``. If a DataFrame, metadata columns
        (``cell_id``/``latitude``/``longitude``/…) are split off automatically and
        carried into :attr:`ClusterResult.metadata`; pass ``feature_columns`` to
        select features explicitly.
    n_clusters : int
        Number of clusters.
    locally_weighted : bool, optional
        ``False`` (default) for plain agglomerative clustering under ``metric``;
        ``True`` to learn an LMNN distance from ``supervision`` first (needs the
        ``[lmnn]`` extra). The same switch as in :func:`create_clusters`.
    supervision : SupervisionResult or array-like, optional
        Required for ``locally_weighted=True``: the per-cell supervision labels (from
        :func:`compute_supervision` or your own array), aligned row-for-row with
        ``features``. Ignored for Euclidean mode.
    linkage_method : str, optional
        SciPy linkage method (default ``"complete"``).
    metric : str, optional
        Distance metric for the Euclidean-mode linkage (default ``"euclidean"``).
    standardize : bool, optional
        Z-score each feature (zero mean, unit variance) before clustering. The
        production pipeline standardizes; pass already-standardized features and
        leave this ``False`` (the default) to avoid double-scaling.
    feature_columns : list of str, optional
        Explicit feature columns when ``features`` is a DataFrame.
    lmnn_mode : str, optional
        Which distance is learned when ``locally_weighted=True``, as in
        :func:`create_clusters`: ``"lowrank"`` (default; a low-rank Mahalanobis metric
        for every cell, the configuration of the published results), ``"global"`` (one
        linear metric for all cells), or ``"local"`` / ``"enhanced"`` (a **diagonal**
        metric for every cell: feature rescaling only, no interactions).
    rank : int or None, optional
        ``lmnn_mode="lowrank"`` and ``"global"``: the rank of the learned metric.
        Default ``None``: with ``"lowrank"`` 16, or the number of features if that is
        smaller; with ``"global"`` the full rank, one direction per feature. Raises with
        ``"local"`` and ``"enhanced"``, whose metric is diagonal and has no rank.
    seed : int or None, optional
        Seed for every random number generator metric learning draws from -- weight
        initialisation, batch order, dropout masks and the train/validation split.
        Default ``0``, so two calls on identical inputs give identical clusters. The
        training itself is **stochastic**: with ``seed=None`` the per-cell modes differ
        from call to call (measured at up to 0.086 reconstruction skill apart).
        Euclidean clustering is deterministic and ignores the seed.
    return_distance_matrix : bool, optional
        Populate :attr:`ClusterResult.distance_matrix` with the dense learned
        ``(n_cells, n_cells)`` matrix (per-cell LMNN modes only; default False).
        Needed to select **medoid** representatives, which is the paper's
        representative rule whenever a learned metric exists. Memory-heavy:
        ``n_cells²`` float64, so ~36 GB at 67k cells.
    return_model : bool, optional
        Populate :attr:`ClusterResult.model` with the trained metric network
        (per-cell LMNN modes only; default False). Ask for it when you need to
        compare two trainings, re-use one metric on new cells, or persist the
        weights alongside a run record: a learned distance matrix shows *that*
        two trainings differ, only the weights say *why*. Cheap -- the model is
        already in memory and this hands back a reference, not a copy.
    lmnn_kwargs : dict, optional
        Extra keyword arguments forwarded to the LMNN trainer. An unknown key
        raises :class:`TypeError` instead of being silently dropped.
    learned_metric : LearnedMetric or str, optional
        Cluster under a metric trained earlier instead of training one: a :class:`LearnedMetric`
        from :func:`load_metric`, or a path it accepts. ``features`` must be exactly the feature
        matrix the metric was trained on -- the same columns, in the same order; when
        ``features`` is a DataFrame and the metric records its column names, they are checked.
        Nothing is trained, so ``supervision``, a ``lmnn_mode`` other than the default,
        ``rank`` and ``lmnn_kwargs`` are refused, and ``seed`` is ignored. Applied to the features it was trained on,
        with the same ``linkage_method``, it reproduces the deployed labels at every
        ``n_clusters`` -- when the distances are computed on the same kind of device (GPU or
        CPU) as the deployed ones; see :meth:`LearnedMetric.distance_matrix`.

    Returns
    -------
    ClusterResult

    Raises
    ------
    ValueError
        If ``locally_weighted=True`` without ``supervision``, or ``rank`` is given to a
        mode that has no rank.
    TypeError
        If ``lmnn_kwargs`` holds a key the selected trainer does not accept.
    """
    mode = "lmnn" if locally_weighted else "euclidean"

    X, metadata = _extract_feature_matrix(features, feature_columns)
    if learned_metric is not None:
        return _cluster_matrix_learned(
            features,
            X,
            metadata,
            n_clusters,
            learned_metric,
            locally_weighted=locally_weighted,
            linkage_method=linkage_method,
            feature_columns=feature_columns,
            return_distance_matrix=return_distance_matrix,
            return_model=return_model,
            standardize=standardize,
            trainer_args={
                "supervision": supervision,
                "lmnn_mode": None if lmnn_mode == "lowrank" else lmnn_mode,
                "rank": rank,
                "lmnn_kwargs": lmnn_kwargs,
            },
        )

    if standardize:
        from sklearn.preprocessing import StandardScaler

        X = StandardScaler().fit_transform(X)

    if X.ndim != 2:
        raise ValueError(
            f"features must be a 2-D (cells x features) table; got shape {X.shape}. A single "
            "feature for every cell is X.reshape(-1, 1)."
        )
    if not np.all(np.isfinite(X)):
        rows = np.flatnonzero(~np.isfinite(X).all(axis=1))
        raise ValueError(
            f"features contains NaN or infinite values in {rows.size} of {X.shape[0]} cells "
            f"(rows {rows[:10].tolist()}{' ...' if rows.size > 10 else ''}). Impute or drop "
            "those cells before clustering."
        )
    n_clusters = _require_n_clusters(n_clusters, X.shape[0])

    if mode == "euclidean":
        if return_distance_matrix:
            raise ValueError(
                "return_distance_matrix=True applies to locally_weighted=True only: it returns "
                "the *learned* distance matrix. Euclidean distances need no model, so "
                "compute them directly (e.g. scipy.spatial.distance.pdist)."
            )
        if return_model:
            raise ValueError(
                "return_model=True applies to locally_weighted=True only. Euclidean clustering "
                "trains nothing, so there is no model to hand back; the metric is "
                f"fully described by metric={metric!r}."
            )
        from scipy.cluster.hierarchy import fcluster

        from .clustering.hierarchical import compute_dendrogram

        Z = compute_dendrogram(X, method=linkage_method, metric=metric)
        labels = fcluster(Z, n_clusters, criterion="maxclust")
        return ClusterResult(
            labels=np.asarray(labels),
            n_clusters=n_clusters,
            mode="euclidean",
            linkage=Z,
            metadata=metadata,
        )

    if mode == "lmnn":
        if supervision is None:
            raise ValueError(
                "locally_weighted=True requires supervision labels. Provide a SupervisionResult "
                "from compute_supervision(...) or an aligned label array via the "
                "`supervision` argument."
            )
        y = _supervision_labels(supervision, X.shape[0])
        from .clustering.lmnn.pipeline import run_lmnn_clustering

        extra = _check_lmnn_kwargs(
            lmnn_kwargs,
            {
                "mode": lmnn_mode,
                "linkage_method": linkage_method,
                "rank": rank,
                "seed": seed,
                "return_distance_matrix": return_distance_matrix,
                "return_model": return_model,
            },
        )
        res = run_lmnn_clustering(
            X,
            y,
            n_clusters,
            mode=lmnn_mode,
            linkage_method=linkage_method,
            rank=rank,
            seed=seed,
            return_distance_matrix=return_distance_matrix,
            return_model=return_model,
            **extra,
        )
        return ClusterResult(
            labels=np.asarray(res.labels),
            n_clusters=n_clusters,
            mode="lmnn",
            linkage=res.linkage,
            metadata=metadata,
            diagnostics=res.diagnostics,
            distance_matrix=res.distance_matrix,
            model=res.model,
        )

    raise AssertionError(f"unreachable: mode={mode!r}")




def _cluster_matrix_learned(
    features,
    X,
    metadata,
    n_clusters,
    learned_metric,
    *,
    locally_weighted,
    linkage_method,
    feature_columns,
    return_distance_matrix,
    return_model,
    standardize,
    trainer_args,
) -> ClusterResult:
    """:func:`cluster_matrix` with ``learned_metric=``: apply it, train nothing."""
    given = [name for name, value in trainer_args.items() if value is not None]
    if given:
        raise ValueError(
            f"{', '.join(given)} configure training a metric; learned_metric= uses one that is "
            "already trained, so they would have no effect. Drop them."
        )
    if locally_weighted is False:
        raise ValueError(
            "locally_weighted=False asks for plain Euclidean clustering, which contradicts "
            "learned_metric=. Leave locally_weighted unset."
        )
    if standardize:
        raise ValueError(
            "standardize=True rescales the features against this matrix's own statistics, "
            "which are not the ones the learned metric was trained on. Pass the features "
            "exactly as the metric was trained on them."
        )
    if not isinstance(learned_metric, LearnedMetric):
        learned_metric = load_metric(learned_metric)
    if not np.all(np.isfinite(X)):
        raise ValueError("features contains NaN or infinite values; impute or drop those cells.")
    n_clusters = _require_n_clusters(n_clusters, X.shape[0])
    names = None
    if isinstance(features, pd.DataFrame):
        names = (
            list(feature_columns)
            if feature_columns is not None
            else [
                c
                for c in features.columns
                if c not in _METADATA_COLUMNS and pd.api.types.is_numeric_dtype(features[c])
            ]
        )
    labels, Z, D = _cluster_with_learned_metric(
        X,
        n_clusters,
        learned_metric,
        linkage_method=linkage_method,
        feature_names=names,
        keep_distances=return_distance_matrix,
    )
    if return_distance_matrix and D is None:
        raise ValueError(
            "return_distance_matrix=True needs a per-cell metric; a 'global' metric clusters "
            "the transformed features directly. Use learned_metric.transformed(X)."
        )
    return ClusterResult(
        labels=labels,
        n_clusters=n_clusters,
        mode="lmnn",
        linkage=Z,
        metadata=metadata,
        diagnostics={"learned_metric": _metric_summary(learned_metric)},
        distance_matrix=D,
        model=learned_metric.model if return_model else None,
    )


def _check_lmnn_kwargs(lmnn_kwargs: dict | None, named: dict) -> dict:
    """Reject ``lmnn_kwargs`` keys that collide with a first-class parameter.

    Passing e.g. ``lmnn_kwargs={"rank": 8}`` alongside the named ``rank`` would
    otherwise end in a bare ``TypeError: got multiple values`` from the forwarding
    call. Name the collision and say which argument to use instead.
    """
    lmnn_kwargs = dict(lmnn_kwargs or {})
    collisions = sorted(set(lmnn_kwargs) & set(named))
    if collisions:
        raise TypeError(
            "lmnn_kwargs must not repeat a first-class parameter: "
            f"{', '.join(collisions)}. Pass "
            f"{', '.join(f'{c}=...' for c in collisions)} directly instead — as a "
            "named argument it is validated against the mode you chose, which is "
            "how a rank given to a mode without one is caught rather than silently ignored."
        )
    return lmnn_kwargs


def _read_labels_csv(csv_path: str) -> pd.DataFrame:

def compute_supervision(
    kind: str,
    *,
    features_df: pd.DataFrame | None = None,
    registry=None,
    ecoregions_shapefile: str | None = None,
    ecoregion_id_column: str = "ECO_ID",
    ecoregion_min_cells: int = 10,
    cell_ids=None,
    latitudes=None,
    longitudes=None,
    continuous_cols: list[str] | None = None,
    categorical_col: str | None = None,
    n_clusters: int | None = None,
    linkage_method: str = "complete",
    cold_threshold: float = 0.0,
    output_dir: str | None = None,
    force_recompute: bool = False,
) -> SupervisionResult:
    """Build the LMNN supervision label signal.

    Three sources are supported via ``kind``:

    * ``"koppen_geiger"`` — climate classes derived from monthly climatology
      features. Requires ``features_df`` carrying ``cell_id``, ``latitude``,
      ``longitude`` and the monthly columns ``tas_month_01..12`` (°C) and
      ``pr_month_01..12`` (mm/month). Import-light (no optional extras).
    * ``"ecoregions"`` — WWF Ecoregions2017 polygons. Requires a
      :class:`~climclusterpy.data.registry.DataRegistry` and a shapefile path.
      **Needs the ``[ecoregions]`` extra** (``geopandas``); imported lazily.
    * ``"output"`` — model-output trait clustering. Requires ``features_df`` with
      the trait columns named in ``continuous_cols`` (plus an optional
      ``categorical_col``), the parallel ``cell_ids``/``latitudes``/``longitudes``
      arrays, and ``n_clusters``. Import-light.

    Parameters
    ----------
    kind : str
        ``"koppen_geiger"``, ``"ecoregions"`` or ``"output"``.
    features_df : pandas.DataFrame, optional
        One row per grid cell. Required for ``"koppen_geiger"``: columns ``cell_id``,
        ``latitude``, ``longitude`` (degrees), ``tas_month_01`` … ``tas_month_12`` (monthly
        mean temperature, °C) and ``pr_month_01`` … ``pr_month_12`` (monthly precipitation,
        mm per month); cells that cannot be classified (missing values) are left out of the
        labels. Required for ``"output"``: the columns named in ``continuous_cols`` and, if
        given, ``categorical_col``. Not used for ``"ecoregions"``.
    registry : DataRegistry, optional
        ``"ecoregions"`` only, and required there: the grid whose land cells are assigned to
        ecoregions (e.g. ``clusters.registry``, or one built with
        :meth:`~climclusterpy.data.registry.DataRegistry.from_soil_code`).
    ecoregions_shapefile : str, optional
        ``"ecoregions"`` only, and required there: path to the polygon shapefile. A cell gets
        the polygon its centre falls in or on, and a cell in no polygon gets the nearest
        one. Polygons in another coordinate system are converted to longitude/latitude; a
        file without one is assumed to be in longitude/latitude already.
    ecoregion_id_column : str, optional
        ``"ecoregions"`` only: the attribute column holding the region id (default
        ``"ECO_ID"``). Its values become the class labels, so they must convert to integers.
    ecoregion_min_cells : int, optional
        ``"ecoregions"`` only: an ecoregion with fewer cells than this on the grid is merged
        into the nearest larger one, preferring the same biome and realm within 1,500 km (see
        :func:`climclusterpy.supervision.merge_small_ecoregions`), because metric learning
        needs several same-class neighbours for every cell. Default 10; ``0`` keeps one class
        per ecoregion. The merges are listed in ``ecoregion_merges_min<n>.csv`` beside the
        labels.
    cell_ids, latitudes, longitudes : array-like, optional
        ``"output"`` only, and all three required there: the cell id and coordinates
        (degrees) of each row of ``features_df``, in the same order. They are copied into
        the labels table as given; their length and order are not checked against
        ``features_df``.
    continuous_cols : list of str, optional
        ``"output"`` only, and required there: the continuous trait columns of
        ``features_df`` to cluster on. Each is standardized on its own (zero mean, unit
        variance); a missing value is replaced by its column's mean first.
    categorical_col : str, optional
        ``"output"`` only: a column of integer category codes (e.g. a dominant vegetation
        type) added to the continuous traits as unscaled one-hot indicator columns. Default
        ``None``: no categorical trait.
    n_clusters : int, optional
        ``"output"`` only, and required there: how many classes to cut the trait dendrogram
        into. It is an upper bound; the dendrogram can yield fewer, and the result's
        ``n_classes`` is the number actually obtained.
    linkage_method : str, optional
        ``"output"`` only: the SciPy linkage method of the trait dendrogram, built under the
        Euclidean distance (default ``"complete"``, as in the published results; it was
        ``"average"`` before 0.4.0). This is what ``supervision_linkage`` sets in
        :func:`create_clusters`.
    cold_threshold : float, optional
        ``"koppen_geiger"`` only: the coldest-month mean temperature, in °C, that separates
        temperate (C) from continental (D) climates (default 0.0; -3.0 is the other common
        convention). A cell is temperate when its coldest month is warmer than this.
    output_dir : str, optional
        Directory the underlying generator writes the labels CSV to. If ``None`` a
        temporary directory is created. Earlier results in it are reused **only for the
        same inputs**: for ``"ecoregions"`` the cell assignment is read back when the grid,
        the shapefile (path, size and modification time) and ``ecoregion_id_column`` all
        match the call that wrote it; for ``"output"`` the trait dendrogram is re-cut when
        the trait matrix and ``linkage_method`` match. Anything else is computed afresh.
        (Before 0.3.0 both were reused on the file name alone, so a second call with other
        inputs in the same folder silently got the first call's labels.)
    force_recompute : bool, optional
        ``"ecoregions"`` and ``"output"``: compute afresh even when a matching saved result
        exists (default ``False``).

    Returns
    -------
    SupervisionResult
        The per-cell ``labels`` table (at least ``cell_id``, ``latitude``, ``longitude``
        and ``cluster_label``), the ``kind``, the number of classes ``n_classes`` and the path
        of the written ``labels_csv``.

    Raises
    ------
    ValueError
        On an unknown ``kind`` or missing required inputs.
    ImportError
        For ``kind="ecoregions"`` without the ``[ecoregions]`` extra installed.
    """
    if output_dir is None:
        output_dir = tempfile.mkdtemp(prefix="climclu_supervision_")
    os.makedirs(output_dir, exist_ok=True)

    if kind == "koppen_geiger":
        if features_df is None:
            raise ValueError("kind='koppen_geiger' requires features_df.")
        from .supervision.koppen_geiger import perform_koppen_geiger_assignment

        result = perform_koppen_geiger_assignment(
            features_df=features_df,
            output_dir=output_dir,
            cold_threshold=cold_threshold,
        )
        labels_csv = result["labels_csv"]
        labels = _read_labels_csv(labels_csv)
        return SupervisionResult(
            labels=labels,
            kind="koppen_geiger",
            n_classes=int(result["n_clusters"]),
            labels_csv=labels_csv,
        )

    if kind == "ecoregions":
        if registry is None or ecoregions_shapefile is None:
            raise ValueError(
                "kind='ecoregions' requires both `registry` and `ecoregions_shapefile`."
            )
        try:
            from .supervision.ecoregions import perform_ecoregion_assignment
        except ImportError as exc:  # geopandas/shapely absent
            raise ImportError(
                "Ecoregion supervision needs the [ecoregions] extra "
                "(geopandas, shapely). Install with: pip install 'climclusterpy[ecoregions]'."
            ) from exc

        result = perform_ecoregion_assignment(
            registry=registry,
            ecoregions_shapefile=ecoregions_shapefile,
            output_dir=output_dir,
            ecoregion_id_column=ecoregion_id_column,
            force_recompute=force_recompute,
            min_cells=ecoregion_min_cells,
        )
        labels_csv = result["labels_csv"]
        labels = _read_labels_csv(labels_csv)
        return SupervisionResult(
            labels=labels,
            kind="ecoregions",
            n_classes=int(result["n_clusters"]),
            labels_csv=labels_csv,
        )

    if kind == "output":
        if features_df is None or continuous_cols is None:
            raise ValueError("kind='output' requires features_df and continuous_cols.")
        if cell_ids is None or latitudes is None or longitudes is None:
            raise ValueError("kind='output' requires cell_ids, latitudes and longitudes arrays.")
        if n_clusters is None:
            raise ValueError("kind='output' requires n_clusters.")
        from .supervision.traits import perform_trait_assignment

        labels_csv = perform_trait_assignment(
            features_df=features_df,
            cell_ids=np.asarray(cell_ids),
            latitudes=np.asarray(latitudes),
            longitudes=np.asarray(longitudes),
            n_clusters=n_clusters,
            output_dir=output_dir,
            continuous_cols=continuous_cols,
            categorical_col=categorical_col,
            linkage_method=linkage_method,
            force_recompute=force_recompute,
        )
        labels = _read_labels_csv(labels_csv)
        return SupervisionResult(
            labels=labels,
            kind="output",
            n_classes=int(labels["cluster_label"].nunique()),
            labels_csv=labels_csv,
        )

    raise ValueError(
        f"Unknown supervision kind: {kind!r}. Use 'koppen_geiger', 'ecoregions', or 'output'."
    )
