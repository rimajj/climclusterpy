Source code for climclusterpy.io.lpjml

"""LPJmL's own binary input formats: read a header and a grid, cut a file down to some cells.

LPJmL reads most cell-indexed inputs not by coordinate but by **position**: record ``i`` of every
``.clm`` file is cell ``i`` of the run's grid file. So cutting an input for a reduced grid means
copying, for each cell of the reduced grid in its order, the record of its source cell -- found by
coordinate in the *full* run's grid file, because the two orders are unrelated (the global
``grid.bin`` is ordered differently from the package's grids, and a positional join between them
is silently wrong).

Three layouts are handled, all copied byte for byte (nothing is decoded or rescaled):

* ``clm`` -- a binary with an ``LPJ...`` header (versions 1-4), one record per cell per year
  (order ``cellyear``); the header is kept with only ``firstcell``/``ncell`` changed;
* ``meta`` -- a ``.clm.json`` description beside a data file; the data is cut and the description
  rewritten with the new cell count;
* ``raw`` -- one fixed-width record per cell and nothing else (LPJmL's soil-code files).
"""

from __future__ import annotations

import json
import os
import struct
from pathlib import Path

import numpy as np
import pandas as pd

__all__ = ["cell_index_in_grid", "read_lpjml_grid", "read_lpjml_header"]

#: LPJmL datatype codes (``include/types.h``).
DATATYPES = {0: np.uint8, 1: np.int16, 2: np.int32, 3: np.float32, 4: np.float64}
_DATATYPE_NAMES = {"byte": 0, "short": 1, "int": 2, "float": 3, "double": 4}
_ORDERS = {1: "cellyear", 2: "yearcell", 3: "cellindex", 4: "cellseq"}


def _id_length(head: bytes) -> int:
    n = 0
    while n < len(head) and n < 16 and (65 <= head[n] <= 90 or head[n] == 95):
        n += 1
    if n < 3 or head[:3] != b"LPJ":
        raise ValueError("not an LPJmL binary: it does not start with an LPJ... header id")
    return n


def read_lpjml_header(path) -> tuple[dict, int]:
    """The header of an LPJmL ``.clm`` file, and its size in bytes.

    Returns a dict with ``id, version, byteorder, order, firstyear, nyear, firstcell, ncell,
    nbands, cellsize_lon, scalar, cellsize_lat, datatype, nstep, timestep``; fields a version-1
    to -3 header does not carry hold LPJmL's defaults.
    """
    with open(path, "rb") as fh:
        head = fh.read(96)
    n = _id_length(head)
    for byteorder in ("<", ">"):
        version = struct.unpack(byteorder + "i", head[n : n + 4])[0]
        if 1 <= version <= 4:
            break
    else:
        raise ValueError(f"{path}: unknown LPJmL header version")
    fmt = byteorder + "6i" + ("ff" if version >= 2 else "") + ("fi" if version >= 3 else "")
    fmt += "ii" if version == 4 else ""
    values = struct.unpack(fmt, head[n + 4 : n + 4 + struct.calcsize(fmt)])
    header = dict(
        zip(
            ("order", "firstyear", "nyear", "firstcell", "ncell", "nbands"), values[:6], strict=True
        )
    )
    header.update(
        id=head[:n].decode(),
        version=version,
        byteorder=byteorder,
        cellsize_lon=values[6] if version >= 2 else 0.5,
        # A version-1 grid is int16 hundredths of a degree: LPJmL's own default.
        scalar=values[7] if version >= 2 else (0.01 if head[:n] == b"LPJGRID" else 1.0),
        cellsize_lat=values[8] if version >= 3 else (values[6] if version >= 2 else 0.5),
        datatype=values[9] if version >= 3 else 1,
        nstep=values[10] if version == 4 else 1,
        timestep=values[11] if version == 4 else 1,
    )
    return header, n + 4 + struct.calcsize(fmt)


def _meta(path) -> tuple[dict, Path]:
    with open(path) as fh:
        meta = json.load(fh)
    data = Path(path).parent / meta["filename"]
    return meta, data

def read_lpjml_grid(path) -> pd.DataFrame:
    """The cells of an LPJmL grid file, in the file's order: ``longitude, latitude``.

    Takes a ``.clm`` grid (``LPJGRID`` header) or a ``.clm.json`` description of one.
    """
    path = os.fspath(path)
    if path.endswith(".json"):
        meta, data = _meta(path)
        dtype = DATATYPES[_DATATYPE_NAMES[meta.get("datatype", "short")]]
        byteorder = ">" if meta.get("bigendian") else "<"
        values = np.fromfile(
            data, dtype=np.dtype(dtype).newbyteorder(byteorder), offset=int(meta.get("offset", 0))
        )
        scalar = float(meta.get("scalar", 0.01 if dtype == np.int16 else 1.0))
        ncell = int(meta["ncell"])
    else:
        header, offset = read_lpjml_header(path)
        dtype = DATATYPES[header["datatype"]]
        values = np.fromfile(
            path, dtype=np.dtype(dtype).newbyteorder(header["byteorder"]), offset=offset
        )
        scalar = float(header["scalar"]) if np.dtype(dtype).kind in "iu" else 1.0
        ncell = int(header["ncell"])
    if values.size != 2 * ncell:
        raise ValueError(f"{path}: {values.size} values for {ncell} cells (expected 2 per cell).")
    pairs = values.reshape(ncell, 2).astype(np.float64)
    if np.dtype(dtype).kind in "iu":
        pairs *= scalar
    return pd.DataFrame({"longitude": pairs[:, 0], "latitude": pairs[:, 1]})




def _keys(longitudes, latitudes) -> np.ndarray:
    """Integer keys at 1e-4 degree, longitudes folded into [-180, 180)."""
    lon = np.asarray(longitudes, dtype=np.float64)
    lon = np.where(lon >= 180.0, lon - 360.0, lon)
    lat = np.asarray(latitudes, dtype=np.float64)
    return np.round(lon * 1e4).astype(np.int64) * 10_000_000 + np.round(lat * 1e4).astype(np.int64)

def cell_index_in_grid(grid_file, latitudes, longitudes) -> np.ndarray:
    """Position of each cell ``(latitudes[i], longitudes[i])`` in an LPJmL grid file.

    This is what a positional join needs: record ``k`` of every ``.clm`` input written for that
    grid belongs to the cell at index ``k``. Cells are matched by coordinate (to 1e-4 degree),
    never by the order of any other table, because LPJmL's global grid files and the package's
    grids order their cells differently.

    Raises
    ------
    ValueError
        If a cell is not in the grid file, or the grid file lists a cell twice.
    """
    grid = read_lpjml_grid(grid_file)
    keys = _keys(grid["longitude"], grid["latitude"])
    order = np.argsort(keys, kind="stable")
    sorted_keys = keys[order]
    if np.any(sorted_keys[1:] == sorted_keys[:-1]):
        raise ValueError(f"{grid_file} lists the same cell more than once.")
    wanted = _keys(longitudes, latitudes)
    pos = np.clip(np.searchsorted(sorted_keys, wanted), 0, sorted_keys.size - 1)
    found = sorted_keys[pos] == wanted
    if not found.all():
        missing = np.flatnonzero(~found)
        lat = np.asarray(latitudes)[missing[:5]].tolist()
        lon = np.asarray(longitudes)[missing[:5]].tolist()
        raise ValueError(
            f"{missing.size} cell(s) are not in {grid_file}, e.g. (lat, lon) "
            f"{list(zip(lat, lon, strict=True))}."
        )
    return order[pos]




def _cut_records(src, dst, index, *, offset, n_source, nyear, record_values, header_bytes=b""):
    """Copy, year by year, the records at ``index`` of a cellyear file. Bytes only."""
    size = os.path.getsize(src) - offset
    n_values = nyear * n_source * record_values
    width, rest = divmod(size, n_values)
    if rest or width not in (1, 2, 4, 8):
        raise ValueError(
            f"{src}: {size} data bytes do not divide into {nyear} year(s) x {n_source} cells x "
            f"{record_values} values."
        )
    record = np.dtype(f"V{width * record_values}")
    data = np.memmap(src, dtype=record, mode="r", offset=offset, shape=(nyear, n_source))
    with open(dst, "wb") as out:
        out.write(header_bytes)
        for year in range(nyear):
            np.ascontiguousarray(data[year][index]).tofile(out)
    return width


def _with_cell_count(header_bytes: bytes, id_length: int, byteorder: str, n: int) -> bytes:
    out = bytearray(header_bytes)
    start = id_length + 4 + 3 * 4  # after id, version, order, firstyear, nyear
    out[start : start + 8] = struct.pack(byteorder + "2i", 0, n)  # firstcell, ncell
    return bytes(out)


def cut_clm(src, dst, index: np.ndarray, n_source: int) -> dict:
    """Write ``dst``: the records of ``src`` at ``index``, its header kept, cell count changed."""
    header, offset = read_lpjml_header(src)
    if _ORDERS.get(header["order"]) != "cellyear":
        raise ValueError(
            f"{src}: order {_ORDERS.get(header['order'], header['order'])!r} -- only cellyear "
            "files can be cut."
        )
    if header["firstcell"] != 0 or header["ncell"] != n_source:
        raise ValueError(
            f"{src} covers cells {header['firstcell']}..{header['firstcell'] + header['ncell']} "
            f"but the source grid has {n_source}: it does not belong to that grid."
        )
    with open(src, "rb") as fh:
        header_bytes = fh.read(offset)
    _cut_records(
        src,
        dst,
        index,
        offset=offset,
        n_source=n_source,
        nyear=header["nyear"],
        record_values=header["nbands"] * header["nstep"],
        header_bytes=_with_cell_count(
            header_bytes, len(header["id"]), header["byteorder"], len(index)
        ),
    )
    return {"format": "clm", "header_version": header["version"]}


def cut_meta(src, dst, index: np.ndarray, n_source: int) -> dict:
    """Cut the data file a ``.clm.json`` describes and write the description beside it."""
    meta, data = _meta(src)
    if meta.get("order", "cellyear") != "cellyear":
        raise ValueError(f"{src}: order {meta['order']!r} -- only cellyear files can be cut.")
    if int(meta.get("firstcell", 0)) != 0 or int(meta["ncell"]) != n_source:
        raise ValueError(f"{src} describes {meta['ncell']} cells; the source grid has {n_source}.")
    offset = int(meta.get("offset", 0))
    with open(data, "rb") as fh:
        header_bytes = fh.read(offset)
    if header_bytes[:3] == b"LPJ":
        header, _ = read_lpjml_header(data)
        header_bytes = _with_cell_count(
            header_bytes, len(header["id"]), header["byteorder"], len(index)
        )
    dst = Path(dst)
    data_dst = dst.parent / Path(meta["filename"]).name
    _cut_records(
        data,
        data_dst,
        index,
        offset=offset,
        n_source=n_source,
        nyear=int(meta.get("nyear", 1)),
        record_values=int(meta.get("nbands", 1)) * int(meta.get("nstep", 1)),
        header_bytes=header_bytes,
    )
    meta.update(filename=data_dst.name, ncell=len(index), firstcell=0)
    meta.pop("grid", None)  # described the full grid, which no longer applies
    dst.write_text(json.dumps(meta, indent=2))
    return {"format": "meta", "data_file": str(data_dst)}


def cut_raw(src, dst, index: np.ndarray, n_source: int) -> dict:
    """Cut a header-less file of one fixed-width record per cell."""
    size = os.path.getsize(src)
    width, rest = divmod(size, n_source)
    if rest or width == 0:
        raise ValueError(f"{src}: {size} bytes is not one record per cell of {n_source} cells.")
    _cut_records(src, dst, index, offset=0, n_source=n_source, nyear=1, record_values=width)
    return {"format": "raw", "record_bytes": width}
