"""Reading, writing, and inspecting NetCDF files.

Thin, dependency-light wrappers around :mod:`xarray` used throughout the
package so that engine choice (``netcdf4``) and coordinate precision are
handled in exactly one place.
"""

from __future__ import annotations

from collections.abc import Iterable
from pathlib import Path

import numpy as np
import xarray as xr

from climclusterpy.data.coords import COORD_PRECISION

__all__ = [
    "enforce_precision",
    "inspect_structure",
    "read_nc",
    "write_nc",
]


def _as_paths(paths: str | Iterable[str]) -> list[Path]:
    if isinstance(paths, str):
        return [Path(paths)]
    return [Path(p) for p in paths]

def read_nc(paths: str | Iterable[str], var_map: dict[str, str] | None = None) -> xr.Dataset:
    """Read one or more NetCDF files into a single xarray Dataset.

    Parameters
    ----------
    paths : str or iterable of str
        File path or list of paths.
    var_map : dict, optional
        Mapping used to rename variables (e.g. ``{"tpr": "precip"}``).

    Returns
    -------
    xarray.Dataset
        An in-memory Dataset.
    """
    files = _as_paths(paths)
    datasets: list[xr.Dataset] = []
    for fp in files:
        ds = xr.open_dataset(fp, engine="netcdf4")
        if var_map:
            rename = {k: v for k, v in var_map.items() if k in ds.variables}
            if rename:
                ds = ds.rename(rename)
        datasets.append(ds)
    if not datasets:
        raise FileNotFoundError("No NetCDF files provided")
    if len(datasets) == 1:
        return datasets[0]
    # Prefer a coordinate-aware merge; fall back to a plain merge for inputs whose
    # coordinates do not form a hypercube (combine_by_coords rejects those).
    try:
        out = xr.combine_by_coords(datasets)
    except ValueError:
        out = xr.merge(datasets)
    return out

def write_nc(ds: xr.Dataset, path: str, encoding: dict[str, dict] | None = None) -> None:
    """Write a Dataset to NetCDF4, creating parent directories as needed.

    Parameters
    ----------
    ds : xarray.Dataset
        Dataset to serialise.
    path : str
        Destination file path.
    encoding : dict, optional
        Per-variable encoding dicts passed through to ``to_netcdf``.
    """
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    ds.to_netcdf(path, engine="netcdf4", format="NETCDF4", encoding=encoding or {})

def enforce_precision(
    ds: xr.Dataset, lat_decimals: int = COORD_PRECISION, lon_decimals: int = COORD_PRECISION
) -> xr.Dataset:
    """Round latitude/longitude coordinates to a fixed decimal precision.

    Float coordinates that differ only in the last bits break the ID-based cell
    matching used across the package, so grids are rounded to a common precision
    before they are compared or joined.

    Parameters
    ----------
    ds : xarray.Dataset
        Dataset whose coordinates should be rounded.
    lat_decimals, lon_decimals : int, optional
        Number of decimals to keep. Both default to
        :data:`climclusterpy.data.coords.COORD_PRECISION`.

    Returns
    -------
    xarray.Dataset
        A new Dataset with rounded coordinates (unchanged if none are present).
    """
    out = ds
    for lat_name in ("latitude", "lat", "Latitude", "LAT"):
        if lat_name in out.coords:
            out = out.assign_coords({lat_name: np.round(out[lat_name].values, lat_decimals)})
            break
    for lon_name in ("longitude", "lon", "Longitude", "LON"):
        if lon_name in out.coords:
            out = out.assign_coords({lon_name: np.round(out[lon_name].values, lon_decimals)})
            break
    return out

def inspect_structure(path: str) -> dict[str, object]:
    """Return a lightweight summary of a NetCDF file's structure.

    Parameters
    ----------
    path : str
        NetCDF file to inspect.

    Returns
    -------
    dict
        Keys ``dims``, ``coords``, ``data_vars``, and ``attrs``.
    """
    ds = xr.open_dataset(path, engine="netcdf4")
    try:
        return {
            "dims": dict(ds.dims),
            "coords": list(ds.coords),
            "data_vars": list(ds.data_vars),
            "attrs": dict(ds.attrs),
        }
    finally:
        ds.close()
