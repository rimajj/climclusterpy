"""Convert NetCDF grids to the LPJmL binary CLM input format.

LPJmL reads climate and soil inputs as CLM binaries, produced by the ``cdf2clm``
and ``cdf2soil`` utilities that ship with an LPJmL installation. This module is a
thin, well-behaved wrapper around those binaries: it locates them, runs them, and
fails with an actionable message when they are missing.

The binaries are found in this order:

1. explicit ``cdf2clm_path`` / ``cdf2soil_path`` arguments (a path given that does not
   exist raises ``FileNotFoundError`` rather than falling back to the places below);
2. ``tools_dir``, if given;
3. ``$LPJROOT/bin``;
4. ``PATH`` (including a login shell, so environment modules are picked up).

Both tools take each axis's step from its **first and last coordinate only**, never
checking the interior, and the check that would report a problem is disabled in the LPJmL
source. An unevenly spaced latitude axis therefore gives every cell the wrong row's climate, with no warning and no failure. So
every file handed to either tool is checked for a strictly uniform axis first, and a
non-uniform one **raises** — see :func:`assert_uniform_axes`. That is a correctness guard,
not tidiness; do not downgrade it to a warning.
"""

from __future__ import annotations

import logging
import os
import shlex
import shutil
import subprocess
from pathlib import Path

logger = logging.getLogger(__name__)

#: Largest departure from a constant axis step, in degrees, that still counts as uniform.
#: Set well below the 0.00125 deg lattice snap seen on real reduced grids so that a
#: snapped axis trips the guard too, and well above float32 representation noise on a
#: coordinate of magnitude 90 (about 8e-06 deg).
UNIFORM_AXIS_TOLERANCE_DEG = 1e-4

_LAT_NAMES = ("lat", "latitude", "Lat", "Latitude", "LAT")
_LON_NAMES = ("lon", "longitude", "Lon", "Longitude", "LON")


def _axis_values(ds, names: tuple[str, ...]):
    """Return the first coordinate in ``names`` present on ``ds``, or ``(None, None)``."""
    for name in names:
        if name in ds.variables or name in ds.coords:
            return name, ds[name].values
        continue
    return None, None

def assert_uniform_axes(
    netcdf_file: str,
    tolerance_deg: float = UNIFORM_AXIS_TOLERANCE_DEG,
) -> dict[str, float]:
    """Fail unless every spatial axis of ``netcdf_file`` is evenly spaced.

    ``cdf2clm`` and ``cdf2soil`` infer each axis's step from its endpoints and never look
    at the interior, so an uneven axis is a silently wrong answer rather than an error.
    This is the check that makes handing them a file safe.

    Parameters
    ----------
    netcdf_file : str
        Path to the file about to be converted.
    tolerance_deg : float, optional
        Largest permitted departure from a constant step, in degrees. Defaults to
        :data:`UNIFORM_AXIS_TOLERANCE_DEG`. **Loosening this re-opens the hazard** — a
        caller who raises it is asserting that the interior of their axis does not matter,
        which for a model input it does.

    Returns
    -------
    dict
        The step found for each axis present, keyed by that axis's variable name. An axis
        of fewer than three points has no interior to disagree with and is reported as-is.

    Raises
    ------
    ValueError
        If any axis's step varies by more than ``tolerance_deg``. The message names the
        index where it first happens and both steps, because "the grid is uneven" is not
        actionable and "row 37 steps 0.75 where every other row steps 0.5" is.
    """
    import numpy as np
    import xarray as xr

    steps: dict[str, float] = {}
    with xr.open_dataset(netcdf_file, decode_times=False) as ds:
        for what, names in (("latitude", _LAT_NAMES), ("longitude", _LON_NAMES)):
            axis_name, values = _axis_values(ds, names)
            if axis_name is None:
                # Checking nothing and returning success is the one outcome a guard must
                # never have: the converters need both axes, so say which is missing.
                raise ValueError(
                    f"{netcdf_file}: no {what} axis found, so its spacing cannot be checked. "
                    f"Accepted names: {list(names)}; the file has {sorted(map(str, ds.variables))}."
                )
            values = np.asarray(values, dtype="float64").ravel()
            if values.size < 2:
                steps[axis_name] = float("nan")
                continue
            diffs = np.diff(values)
            step = float(diffs[0])
            steps[axis_name] = step
            if values.size < 3:
                continue
            offenders = np.flatnonzero(np.abs(diffs - step) > tolerance_deg)
            if offenders.size:
                first = int(offenders[0])
                raise ValueError(
                    f"{netcdf_file}: axis '{axis_name}' is not uniformly spaced. "
                    f"Step {float(diffs[first]):.10g} deg between index {first} "
                    f"({values[first]:.10g}) and {first + 1} ({values[first + 1]:.10g}), "
                    f"against {step:.10g} deg at the start of the axis "
                    f"({offenders.size} of {diffs.size} steps disagree by more than "
                    f"{tolerance_deg:g} deg). cdf2clm/cdf2soil would take the step from the "
                    f"endpoints only and give cells the wrong row's data, silently. Rebuild "
                    f"the grid on a regular lattice — climclusterpy.build_reduced_grid does "
                    f"this by construction."
                )
    return steps




class CLMConverter:
    """
    Python interface for converting NetCDF files to CLM format.
    Wraps the external cdf2soil and cdf2clm utilities.
    """

    def __init__(
        self,
        check_tools: bool = True,
        cdf2soil_path: str | None = None,
        cdf2clm_path: str | None = None,
        tools_dir: str | None = None,
        uniform_axis_tolerance_deg: float = UNIFORM_AXIS_TOLERANCE_DEG,
    ) -> None:
        # A tool path the caller names and that does not exist is a mistake to report, not
        # a hint to ignore: falling back would quietly run whatever other copy is found.
        for name, path in (("cdf2soil_path", cdf2soil_path), ("cdf2clm_path", cdf2clm_path)):
            if path and not os.path.isfile(path):
                raise FileNotFoundError(f"{name}={path!r} does not exist or is not a file.")
        self._cdf2soil_path = cdf2soil_path
        self._cdf2clm_path = cdf2clm_path
        self._tools_dir = tools_dir
        self.uniform_axis_tolerance_deg = uniform_axis_tolerance_deg
        self._base_env = self._build_env()
        self._resolved_bins = self._resolve_bins()
        self.tools_available = self._check_tools() if check_tools else True

        # Variable configuration for CLM conversion
        self.climate_vars: dict[str, dict[str, str | None]] = {
            "precipitation": {"var_name": "precipitation", "units": "mm/day"},
            "temperature": {"var_name": "temperature", "units": "celsius"},
            "short_wave_radiation": {"var_name": "short_wave_radiation", "units": "W/m2"},
            "long_wave_radiation": {"var_name": "long_wave_radiation", "units": "W/m2"},
        }
        self.soil_vars: dict[str, dict[str, str | None]] = {
            "soil_code": {"var_name": "soil_code", "units": None},
            "soil_depth": {"var_name": "soil_depth", "units": "m"},
        }

    def _build_env(self) -> dict[str, str]:
        """Return the child-process environment, with the tool directories on PATH."""
        env = os.environ.copy()
        extra_paths = []
        if self._tools_dir:
            extra_paths.append(self._tools_dir)
        lpjroot = env.get("LPJROOT")
        if lpjroot:
            extra_paths.append(os.path.join(lpjroot, "bin"))
        current_path = env.get("PATH", "")
        env["PATH"] = os.pathsep.join([p for p in extra_paths if p] + [current_path])
        return env

    def _resolve_bins(self) -> dict[str, str | None]:
        """Locate cdf2soil/cdf2clm; values are None when a tool cannot be found."""

        def _resolve(name: str, explicit: str | None) -> str | None:
            if explicit:
                return explicit
            if self._tools_dir:
                cand = os.path.join(self._tools_dir, name)
                if os.path.isfile(cand):
                    return cand
            lpjroot = self._base_env.get("LPJROOT")
            if lpjroot:
                cand = os.path.join(lpjroot, "bin", name)
                if os.path.isfile(cand):
                    return cand
            # Last resort: the PATH built in _build_env (tools_dir + $LPJROOT/bin
            # prepended to the inherited PATH).
            return shutil.which(name, path=self._base_env.get("PATH"))

        return {
            "cdf2soil": _resolve("cdf2soil", self._cdf2soil_path),
            "cdf2clm": _resolve("cdf2clm", self._cdf2clm_path),
        }

    def _missing_tool_message(self, name: str) -> str:
        """Where ``name`` was looked for, and how to point the converter at it."""
        lpjroot = self._base_env.get("LPJROOT")
        places = []
        if self._tools_dir:
            places.append(f"tools_dir={self._tools_dir}")
        places.append(
            f"$LPJROOT/bin ({os.path.join(lpjroot, 'bin')})"
            if lpjroot
            else "$LPJROOT/bin ($LPJROOT is not set)"
        )
        places.append("the PATH, including a login shell's")
        return (
            f"The LPJmL conversion tool '{name}' was not found. Looked in: "
            + "; ".join(places)
            + ". It ships with an LPJmL installation (in its bin/ folder): set LPJROOT to "
            "that installation, load its environment module, or pass tools_dir= (the folder "
            f"holding {name})."
        )

    def _check_tools(self) -> bool:
        # If binaries resolved directly, consider available
        missing = [name for name, path in self._resolved_bins.items() if not path]
        if not missing:
            logger.info("✓ CLM conversion tools available (direct paths)")
            return True
        # Try detecting via login shell environment (sources user's profile and modules)
        detected = []
        for name in missing:
            if self._command_exists_login(name):
                detected.append(name)
        still_missing = [n for n in missing if n not in detected]
        if still_missing:
            logger.warning(f"Missing CLM conversion tools: {still_missing}")
            return False
        logger.info("✓ CLM conversion tools available via login shell")
        return True

    def _command_exists_login(self, name: str) -> bool:
        try:
            proc = subprocess.run(
                ["/bin/bash", "-lc", f"command -v {shlex.quote(name)} >/dev/null 2>&1"],
                env=self._base_env,
                check=False,
                capture_output=True,
                text=True,
            )
            return proc.returncode == 0
        except OSError:  # no bash to ask: the tool cannot be found this way
            return False

    def _run_login(self, args: list[str], cwd: str | None = None) -> subprocess.CompletedProcess:
        cmd_str = " ".join(shlex.quote(a) for a in args)
        # Use login shell to ensure ~/.profile (and modules) are loaded
        return subprocess.run(
            ["/bin/bash", "-lc", cmd_str],
            env=self._base_env,
            cwd=cwd,
            check=True,
            capture_output=True,
            text=True,
        )

    def convert_soil_code(
        self,
        netcdf_file: str,
        output_dir: str,
        var_name: str = "soil_code",
        output_prefix: str | None = None,
    ) -> dict[str, str]:
        if not os.path.exists(netcdf_file):
            raise FileNotFoundError(f"Input file not found: {netcdf_file}")
        if not self.tools_available:
            raise RuntimeError(self._missing_tool_message("cdf2soil"))
        assert_uniform_axes(netcdf_file, self.uniform_axis_tolerance_deg)
        os.makedirs(output_dir, exist_ok=True)

        if output_prefix is None:
            output_prefix = Path(netcdf_file).stem
        grid_file = os.path.join(output_dir, f"{output_prefix}.grid.clm")
        soil_file = os.path.join(output_dir, f"{output_prefix}.soil.bin")

        bin_path = self._resolved_bins.get("cdf2soil") or "cdf2soil"
        cmd = [bin_path, "-float", "-var", var_name, netcdf_file, grid_file, soil_file]

        logger.info(f"Converting soil code with: {' '.join(cmd)}")
        try:
            # Prefer login shell to load user's environment if direct path is not resolved
            if self._resolved_bins.get("cdf2soil"):
                subprocess.run(
                    cmd,
                    check=True,
                    capture_output=True,
                    text=True,
                    cwd=output_dir,
                    env=self._base_env,
                )
            else:
                self._run_login(cmd, cwd=output_dir)
        except subprocess.CalledProcessError as e:
            logger.error(f"cdf2soil failed: {e.stderr}")
            # The tool's OWN message, in the exception, not only in a log nobody
            # reads. `CalledProcessError` renders as "returned non-zero exit status
            # 127" and nothing else, so a caller whose LD_LIBRARY_PATH is missing
            # libnetcdf sees an exit code where the loader had written a sentence
            # naming the library.
            detail = (e.stderr or e.stdout or "").strip()
            raise RuntimeError(
                f"cdf2soil conversion failed: {e}"
                + (f"\ncdf2soil said: {detail}" if detail else "")
            ) from e
        return {"grid_file": grid_file, "soil_file": soil_file}

    def convert_climate_file(
        self,
        netcdf_file: str,
        grid_file: str,
        output_dir: str,
        var_name: str | None = None,
        units: str | None = None,
        output_prefix: str | None = None,
    ) -> str:
        if not os.path.exists(netcdf_file):
            raise FileNotFoundError(f"Input file not found: {netcdf_file}")
        if not self.tools_available:
            raise RuntimeError(self._missing_tool_message("cdf2clm"))
        if not os.path.exists(grid_file):
            raise FileNotFoundError(f"Grid file not found: {grid_file}")
        assert_uniform_axes(netcdf_file, self.uniform_axis_tolerance_deg)
        os.makedirs(output_dir, exist_ok=True)

        basename = Path(netcdf_file).stem
        if var_name is None or units is None:
            d_var, d_units = self._detect_variable_info(basename)
            var_name = var_name or d_var
            units = units or d_units
        output_prefix = output_prefix or basename
        clm_file = os.path.join(output_dir, f"{output_prefix}.clm")

        bin_path = self._resolved_bins.get("cdf2clm") or "cdf2clm"
        cmd = [bin_path, "-float"]
        if var_name:
            cmd += ["-var", var_name]
        if units:
            cmd += ["-units", units]
        cmd += ["-o", clm_file, grid_file, netcdf_file]

        logger.info(f"Converting NetCDF to CLM with: {' '.join(cmd)}")
        try:
            if self._resolved_bins.get("cdf2clm"):
                subprocess.run(
                    cmd,
                    check=True,
                    capture_output=True,
                    text=True,
                    cwd=output_dir,
                    env=self._base_env,
                )
            else:
                self._run_login(cmd, cwd=output_dir)
        except subprocess.CalledProcessError as e:
            logger.error(f"cdf2clm failed: {e.stderr}")
            # The tool's OWN message, in the exception, not only in a log nobody
            # reads. `CalledProcessError` renders as "returned non-zero exit status
            # 127" and nothing else, so a caller whose LD_LIBRARY_PATH is missing
            # libnetcdf sees an exit code where the loader had written a sentence
            # naming the library.
            detail = (e.stderr or e.stdout or "").strip()
            raise RuntimeError(
                f"cdf2clm conversion failed: {e}" + (f"\ncdf2clm said: {detail}" if detail else "")
            ) from e
        return clm_file

    def _detect_variable_info(self, filename: str) -> tuple[str | None, str | None]:
        for key, info in self.climate_vars.items():
            if key in filename:
                return info["var_name"], info["units"]
        for key, info in self.soil_vars.items():
            if key in filename:
                return info["var_name"], info["units"]
        logger.warning(f"Could not auto-detect variable info for: {filename}")
        return None, None

    def convert_netcdf_directory(
        self,
        input_dir: str,
        output_dir: str,
        file_pattern: str = "*.nc",
        grid_file: str | None = None,
    ) -> dict[str, str | None]:
        input_path = Path(input_dir)
        nc_files = list(input_path.glob(file_pattern))
        if not nc_files:
            raise FileNotFoundError(
                f"No NetCDF files in {input_dir} match {file_pattern!r}; check the folder and "
                "the pattern."
            )

        results: dict[str, str | None] = {}

        soil_code_file = None
        if grid_file:
            if not os.path.exists(grid_file):
                raise FileNotFoundError(f"grid_file not found: {grid_file}")
            logger.info(f"Using provided grid file: {grid_file}")
        else:
            soil_code_files = [f for f in nc_files if "soil_code" in f.name]
            if not soil_code_files:
                raise RuntimeError("No soil_code file found and no grid_file provided")
            soil_code_file = str(soil_code_files[0])

        # Every file's axes first, so an uneven one stops the run before anything is written
        # and reaches the caller as the ValueError it is, not wrapped as a failed conversion.
        for f in nc_files:
            assert_uniform_axes(str(f), self.uniform_axis_tolerance_deg)

        if soil_code_file is not None:
            soil_result = self.convert_soil_code(soil_code_file, output_dir)
            grid_file = soil_result["grid_file"]
            results[soil_code_file] = grid_file

        for f in nc_files:
            if "soil_code" in f.name:
                continue
            try:
                results[str(f)] = self.convert_climate_file(str(f), grid_file, output_dir)
            except Exception as e:  # re-raised with the file named; any failure is fatal
                raise RuntimeError(
                    f"Failed to convert {f} to CLM: {e}. Every forcing file a run needs must "
                    "convert; fix this file or remove it from the directory."
                ) from e

        logger.info(f"Conversion summary: {len(results)} files converted")
        return results

def convert_netcdf_to_clm(
    netcdf_files: str | list[str],
    output_dir: str,
    grid_file: str | None = None,
    *,
    soil_code_file: str | None = None,
    strict: bool = True,
    uniform_axis_tolerance_deg: float = UNIFORM_AXIS_TOLERANCE_DEG,
    tools_dir: str | None = None,
) -> dict[str, str | None]:
    """Convert NetCDF grids to the CLM binaries LPJmL reads, and return what was written.

    This is the input half of the reduced-grid round trip:
    :func:`~climclusterpy.build_reduced_grid` writes one NetCDF per forcing variable, and
    this turns them into the ``.clm`` files a model run consumes. Coordinates are written
    as floats (both tools are called with ``-float``), which is what keeps them exact —
    the tools' integer path truncates rather than rounds,
    so a coordinate one bit below a lattice point would land a whole step away. Do not drop
    that flag.

    Every input is checked for a uniform axis first (:func:`assert_uniform_axes`), because
    the tools infer each axis's step from its endpoints alone.

    Parameters
    ----------
    netcdf_files : str or list of str
        The files to convert. Unless ``soil_code_file`` or ``grid_file`` is given, exactly
        one must be a soil-code file, recognised by ``"soil_code"`` in its name; it is
        converted first because it produces the grid file every climate conversion needs.
    output_dir : str
        Where the ``.clm`` / ``.bin`` files land. Created if absent.
    grid_file : str, optional
        An existing ``.grid.clm`` to convert against. When given, no soil-code file is
        needed and none is converted; a path that does not exist raises.
    soil_code_file : str, optional
        The soil-code NetCDF that defines the grid, named explicitly instead of found by
        ``"soil_code"`` in a file name -- safer when names are your own. It need not be in
        ``netcdf_files``; if it is, it is converted once, as the grid. Not together with
        ``grid_file``.
    strict : bool, optional
        Default ``True``: a failed conversion raises. With ``False`` the failure is logged
        and that file's entry is ``None``, which is the older behaviour — use it only when
        a partial result is genuinely useful, never before a model run. A run launched
        against a silently missing forcing file is the failure this default prevents. An
        unevenly spaced axis raises ``ValueError`` either way.
    uniform_axis_tolerance_deg : float, optional
        Passed through to :func:`assert_uniform_axes`.
    tools_dir : str, optional
        The folder holding ``cdf2clm`` and ``cdf2soil``. By default they are looked for in
        ``$LPJROOT/bin`` and then on the ``PATH``.

    Returns
    -------
    dict
        Input path to output path. The soil-code entry maps to the ``.grid.clm`` that was
        written from it. Values are ``None`` only when ``strict=False``.

    Raises
    ------
    RuntimeError
        If no soil-code file is present and no ``grid_file`` was given, or — under
        ``strict`` — if any conversion fails.
    ValueError
        If any input's latitude or longitude axis is not uniformly spaced (whatever
        ``strict`` is), or both ``grid_file`` and ``soil_code_file`` are given.
    FileNotFoundError
        If an input, ``grid_file`` or ``soil_code_file`` does not exist.
    """
    files = [netcdf_files] if isinstance(netcdf_files, str) else list(netcdf_files)
    missing = [f for f in files if not os.path.exists(f)]
    if missing:
        raise FileNotFoundError(f"Input file(s) not found: {missing}")
    converter = CLMConverter(
        uniform_axis_tolerance_deg=uniform_axis_tolerance_deg, tools_dir=tools_dir
    )
    results: dict[str, str | None] = {}

    if grid_file and soil_code_file:
        raise ValueError(
            "Pass grid_file= (an existing grid to convert against) or soil_code_file= (the "
            "NetCDF to make the grid from), not both."
        )
    if grid_file:
        if not os.path.exists(grid_file):
            raise FileNotFoundError(f"grid_file not found: {grid_file}")
    else:
        if soil_code_file is not None:
            if not os.path.exists(soil_code_file):
                raise FileNotFoundError(f"soil_code_file not found: {soil_code_file}")
            soil_file = soil_code_file
        else:
            soil = [f for f in files if "soil_code" in os.path.basename(f)]
            if not soil:
                raise RuntimeError(
                    "No soil-code file to make the grid from: pass soil_code_file=, or "
                    "grid_file= to convert against an existing grid."
                )
            if len(soil) > 1:
                raise RuntimeError(
                    f"{len(soil)} files have 'soil_code' in their name ({soil}); name the one "
                    "that defines the grid with soil_code_file=."
                )
            soil_file = soil[0]
        soil_result = converter.convert_soil_code(soil_file, output_dir)
        grid_file = soil_result["grid_file"]
        results[soil_file] = grid_file
        files = [f for f in files if os.path.abspath(f) != os.path.abspath(soil_file)]

    for f in files:
        try:
            out = converter.convert_climate_file(f, grid_file, output_dir)
            results[f] = out
        except ValueError:
            # An unevenly spaced axis (checked before every conversion) is a file the tool
            # would read wrongly without failing, not a failed conversion: it always reaches
            # the caller as the ValueError it is, whatever strict says.
            raise
        except Exception as e:  # re-raised unless the caller asked for partial results
            if strict:
                raise RuntimeError(
                    f"Failed to convert {f} to CLM: {e}. Every forcing file a run needs "
                    f"must convert, so this is fatal by default; pass strict=False to "
                    f"collect partial results instead."
                ) from e
            logger.error(f"Failed to convert {f}: {e}")
            results[f] = None
    return results
