Source code for climclusterpy.io.subset

"""Cut any model input down to the cells of a reduced grid: ``subset_inputs``.

The reduced grid the package writes carries the climate and soil it clustered on. A real model
run opens more inputs than that -- LPJmL with nitrogen on opens wind, wet days, deposition, soil
pH, slopes, hydrotopes and more -- and each must reach the reduced grid with exactly the values of
the full run, only for fewer cells. This module copies them, without decoding, rescaling or
converting anything:

* **NetCDF** inputs are written on the reduced grid's own lattice -- the same latitude and
  longitude axes as the reduced-grid NetCDFs the package wrote -- with every slot holding its
  *source* cell's values. A model that finds NetCDF cells by coordinate therefore reads the right
  climate even for the slots that do not sit at their source position (the repeated cells of the
  distributed choice, every cell of a packed medoid/centroid grid).
* **LPJmL** ``.clm`` / ``.clm.json`` / raw inputs are cut record by record into the reduced grid's
  cell order, which is the order of its LPJmL grid file. Each slot's source cell is found by
  coordinate in the full run's grid file; a positional join between the two grids would be
  silently wrong.
"""

from __future__ import annotations

import json
import logging
import os
from pathlib import Path

import numpy as np
import pandas as pd

from .lpjml import _keys, cell_index_in_grid, cut_clm, cut_meta, cut_raw, read_lpjml_grid

logger = logging.getLogger(__name__)

__all__ = ["subset_inputs"]

_NETCDF_SUFFIXES = (".nc", ".nc4", ".cdf", ".netcdf")
RECORD = "subset_inputs.json"


def _format(path: str) -> str:
    lower = path.lower()
    if lower.endswith(_NETCDF_SUFFIXES):
        return "netcdf"
    if lower.endswith(".json"):
        return "meta"
    with open(path, "rb") as fh:
        head = fh.read(8)
    if head[:3] == b"CDF" or head[:4] == b"\x89HDF":
        return "netcdf"
    if head[:3] == b"LPJ":
        return "clm"
    return "raw"


def _normalised_inputs(inputs) -> dict[str, str]:
    if isinstance(inputs, str | os.PathLike):
        inputs = [inputs]
    if isinstance(inputs, dict):
        named = {str(k): os.fspath(v) for k, v in inputs.items()}
    else:
        named = {}
        for path in inputs:
            name = Path(path).stem  # input.clm.json -> input.clm, wind.nc -> wind
            if name in named:
                raise ValueError(
                    f"Two inputs would both be called {name!r}; pass a mapping of names to paths."
                )
            named[name] = os.fspath(path)
    missing = [p for p in named.values() if not os.path.exists(p)]
    if missing:
        raise FileNotFoundError(f"Input file(s) not found: {missing}")
    return named


# --------------------------------------------------------------------------- NetCDF


def _lattice(clusters) -> tuple[np.ndarray, np.ndarray]:
    """The latitude and longitude axes of the reduced grid's own NetCDFs."""
    import netCDF4

    from ..data.coords import LAT_NAMES, LON_NAMES

    files = [p for p in clusters.cluster_files if str(p).endswith(".nc") and os.path.exists(p)]
    if not files:
        raise ValueError(
            "The reduced grid has no NetCDF to take its lattice from; build it with at least "
            "one field written (write_fields)."
        )
    with netCDF4.Dataset(files[0]) as nc:
        lat = next(n for n in LAT_NAMES if n in nc.variables)
        lon = next(n for n in LON_NAMES if n in nc.variables)
        return np.asarray(nc[lat][:], dtype=float), np.asarray(nc[lon][:], dtype=float)


def _positions(axis: np.ndarray, values: np.ndarray, *, longitude: bool, what: str) -> np.ndarray:
    def norm(v):
        v = np.round(np.asarray(v, dtype=np.float64), 4)
        return np.where(v >= 180.0, v - 360.0, v) if longitude else v

    a, v = norm(axis), norm(values)
    order = np.argsort(a, kind="stable")
    pos = np.clip(np.searchsorted(a[order], v), 0, a.size - 1)
    if not np.array_equal(a[order][pos], v):
        bad = np.asarray(values)[a[order][pos] != v][:5].tolist()
        raise ValueError(f"{what}: coordinates {bad} are not on the axis.")
    return order[pos]


def _subset_netcdf(src: str, dst: str, slots: pd.DataFrame, lattice, n_jobs: int) -> dict:
    import netCDF4

    from ..data.coords import LAT_NAMES, LON_NAMES
    from .cellread import file_source, read_cell_series

    target_lat, target_lon = lattice
    t_lat = _positions(target_lat, slots["grid_latitude"], longitude=False, what="reduced grid")
    t_lon = _positions(target_lon, slots["grid_longitude"], longitude=True, what="reduced grid")

    with netCDF4.Dataset(src) as nc_in, netCDF4.Dataset(dst, "w", format=nc_in.data_model) as out:
        lat_dim = next((d for d in nc_in.dimensions if d in LAT_NAMES), None)
        lon_dim = next((d for d in nc_in.dimensions if d in LON_NAMES), None)
        if lat_dim is None or lon_dim is None:
            raise ValueError(
                f"{src} has no latitude/longitude dimensions ({list(nc_in.dimensions)})."
            )
        s_lat = _positions(nc_in[lat_dim][:], slots["source_latitude"], longitude=False, what=src)
        s_lon = _positions(nc_in[lon_dim][:], slots["source_longitude"], longitude=True, what=src)

        out.setncatts({a: nc_in.getncattr(a) for a in nc_in.ncattrs()})
        out.setncattr(
            "climclusterpy_subset",
            f"cut from {os.path.basename(src)} to the {len(slots)} cells of a reduced grid; "
            "every cell holds the values of its source cell",
        )
        for name, dim in nc_in.dimensions.items():
            size = {lat_dim: target_lat.size, lon_dim: target_lon.size}.get(name, len(dim))
            out.createDimension(name, None if dim.isunlimited() else size)

        spatial = []
        for name, var in nc_in.variables.items():
            dims = var.dimensions
            if name in (lat_dim, lon_dim):
                continue
            if lat_dim in dims and lon_dim in dims:
                spatial.append(name)
            elif lat_dim in dims or lon_dim in dims:
                logger.info("subset_inputs: %s:%s describes the old axis; not copied", src, name)
            else:
                copy = out.createVariable(name, var.dtype, dims, fill_value=_fill_attr(var))
                copy.setncatts(_attrs(var))
                copy.set_auto_maskandscale(False)
                var.set_auto_maskandscale(False)
                copy[...] = var[...]

        # The new axes: the reduced grid's own lattice, under the source's names and attributes.
        for dim, values in ((lat_dim, target_lat), (lon_dim, target_lon)):
            src_var = nc_in.variables[dim]
            coord = out.createVariable(dim, src_var.dtype, (dim,))
            coord.setncatts(_attrs(src_var))
            coord[:] = values

        for name in spatial:
            var = nc_in.variables[name]
            var.set_auto_maskandscale(False)
            fill = _fill_attr(var)
            if fill is None:
                fill = netCDF4.default_fillvals.get(var.dtype.str[1:], np.nan)
            other = [d for d in var.dimensions if d not in (lat_dim, lon_dim)]
            dims = (*other, lat_dim, lon_dim)
            chunks = (*(1 for _ in other), target_lat.size, target_lon.size)
            new = out.createVariable(
                name, var.dtype, dims, fill_value=fill, zlib=True, complevel=4, chunksizes=chunks
            )
            new.setncatts(_attrs(var))
            new.set_auto_maskandscale(False)
            if len(other) == 0:
                grid = np.full((target_lat.size, target_lon.size), fill, dtype=var.dtype)
                grid[t_lat, t_lon] = np.asarray(var[...]).transpose(
                    var.dimensions.index(lat_dim), var.dimensions.index(lon_dim)
                )[s_lat, s_lon]
                new[...] = grid
            elif len(other) == 1:
                lat_vals = np.asarray(nc_in[lat_dim][:])[s_lat]
                lon_vals = np.asarray(nc_in[lon_dim][:])[s_lon]
                packed = s_lat.astype(np.int64) * len(nc_in.dimensions[lon_dim]) + s_lon
                unique, slot_to_unique = np.unique(packed, return_inverse=True)
                first = np.unique(slot_to_unique, return_index=True)[1]
                source = file_source(src, name, lat_vals[first], lon_vals[first], raw=True)
                series, _ = read_cell_series({name: source}, n_jobs=n_jobs)
                values = series[name]
                n_t = values.shape[0]
                step = max(1, min(n_t, int(2e8 // max(1, target_lat.size * target_lon.size))))
                for t0 in range(0, n_t, step):
                    t1 = min(t0 + step, n_t)
                    buf = np.full((t1 - t0, target_lat.size, target_lon.size), fill, var.dtype)
                    buf[:, t_lat, t_lon] = values[t0:t1][:, slot_to_unique]
                    new[t0:t1] = buf
                del values, series
            else:
                raise ValueError(
                    f"{src}:{name} has {len(other)} non-spatial dimensions {other}; only one "
                    "(time, usually) is supported."
                )
    return {"format": "netcdf", "variables": spatial}


def _fill_attr(var):
    for attr in ("_FillValue", "missing_value"):
        if attr in var.ncattrs():
            return np.asarray(var.getncattr(attr)).ravel()[0]
    return None


def _attrs(var) -> dict:
    return {a: var.getncattr(a) for a in var.ncattrs() if a != "_FillValue"}


# --------------------------------------------------------------------------- LPJmL binaries


def _positional_index(slots: pd.DataFrame, source_grid, target_grid) -> np.ndarray:
    """For each cell of ``target_grid``, in its order, the index of its source cell in ``source_grid``."""
    target = read_lpjml_grid(target_grid)
    if len(target) != len(slots):
        raise ValueError(
            f"{target_grid} has {len(target)} cells but the reduced grid has {len(slots)} slots: "
            "it is not this reduced grid's grid file."
        )
    slot_keys = _keys(slots["grid_longitude"], slots["grid_latitude"])
    lookup = pd.Series(np.arange(len(slots)), index=slot_keys)
    if lookup.index.has_duplicates:
        raise ValueError("Two slots of the reduced grid share one position.")
    target_keys = _keys(target["longitude"], target["latitude"])
    missing = ~np.isin(target_keys, slot_keys)
    if missing.any():
        raise ValueError(
            f"{int(missing.sum())} cells of {target_grid} are not slots of this reduced grid."
        )
    order = lookup.loc[target_keys].to_numpy()
    rows = slots.iloc[order]
    return cell_index_in_grid(
        source_grid, rows["source_latitude"].to_numpy(), rows["source_longitude"].to_numpy()
    )

def subset_inputs(
    clusters,
    inputs,
    output_dir: str,
    *,
    source_grid: str | None = None,
    target_grid: str | None = None,
    n_jobs: int = -1,
) -> dict[str, str]:
    """Cut model input files down to the cells of a reduced grid, keeping their format and type.

    For the inputs the reduced grid itself does not carry -- whatever else the model opens per
    cell. Every slot of the reduced grid gets its *source* cell's record: the cell whose climate
    it carries. A slot that repeats a cell (the distributed choice in a region with fewer cells
    than draws) gets that cell's record again.

    Parameters
    ----------
    clusters : CreateClustersResult or str
        The reduced grid: what :func:`~climclusterpy.create_clusters` or
        :func:`~climclusterpy.create_regions` returned, or their ``output_dir``.
    inputs : str, sequence of str, or mapping of name to path
        The files to cut. The format is taken from each file:

        * NetCDF (``.nc`` and the like) -- every variable on the latitude/longitude grid is
          written on the reduced grid's own lattice, each slot holding its source cell's values
          in the variable's own type, attributes and packing. Variables with no spatial axis
          (time, CO2 series, ...) are copied unchanged. At most one non-spatial axis per
          variable.
        * LPJmL ``.clm`` (an ``LPJ...`` header, order ``cellyear``) -- records cut into the
          reduced grid's cell order; the header kept with only the cell count changed.
        * LPJmL ``.clm.json`` -- the data file it names is cut, and the description rewritten
          for the new cell count (its ``grid`` entry, which named the full grid, is dropped).
        * anything else -- a header-less file of one fixed-width record per cell (LPJmL's
          ``soil.bin``-style raw files).

        A mapping names the outputs in the returned dict; a sequence names each by its file
        name without the last extension (``wind.nc`` -> ``wind``, ``ph.clm.json`` ->
        ``ph.clm``).
    output_dir : str
        Where the cut files go, each under its source's file name (created if missing). A
        ``subset_inputs.json`` there records what was cut from where.
    source_grid : str, optional
        LPJmL inputs only, and required for them: the full run's grid file (``.clm`` with an
        ``LPJGRID`` header, or its ``.clm.json``). Record ``i`` of every full-run input belongs
        to cell ``i`` of this file; each slot's source cell is found in it **by coordinate**.
    target_grid : str, optional
        LPJmL inputs only, and required for them: the reduced grid's own LPJmL grid file, as
        :func:`~climclusterpy.convert_netcdf_to_clm` wrote it. Its cell order is the order the
        cut files are written in.
    n_jobs : int, optional
        Worker processes for reading NetCDF series (default ``-1``, all cores).

    Returns
    -------
    dict
        Input name to the path written (for ``.clm.json``, the description).

    Raises
    ------
    ValueError
        If a slot's source cell is not in an input's grid or in ``source_grid``; if
        ``target_grid`` does not hold exactly this reduced grid's cells; if an LPJmL input is
        given without both grid files, does not cover ``source_grid``'s cells, or is not
        ordered ``cellyear``.
    FileNotFoundError
        If an input does not exist.
    """
    if isinstance(clusters, str | os.PathLike):
        from ..api import load_clusters

        clusters = load_clusters(os.fspath(clusters))
    named = _normalised_inputs(inputs)
    formats = {name: _format(path) for name, path in named.items()}
    slots = clusters.slot_sources

    needs_grids = sorted(n for n, f in formats.items() if f != "netcdf")
    index = None
    if needs_grids:
        if source_grid is None or target_grid is None:
            raise ValueError(
                f"{needs_grids} are LPJmL binaries, which LPJmL reads by position: cutting them "
                "needs source_grid= (the full run's grid file, whose order they follow) and "
                "target_grid= (the reduced grid's own grid file, whose order to write)."
            )
        index = _positional_index(slots, source_grid, target_grid)
        n_source = len(read_lpjml_grid(source_grid))

    os.makedirs(output_dir, exist_ok=True)
    lattice = _lattice(clusters) if "netcdf" in formats.values() else None
    written: dict[str, str] = {}
    record: dict = {
        "reduced_grid": os.path.abspath(clusters.output_dir),
        "n_slots": len(slots),
        "n_repeated_slots": int(slots.duplicated("cell_id").sum()),
        "source_grid": None if source_grid is None else os.path.abspath(source_grid),
        "target_grid": None if target_grid is None else os.path.abspath(target_grid),
        "inputs": {},
    }
    for name, src in named.items():
        dst = os.path.join(output_dir, os.path.basename(src))
        if os.path.abspath(dst) == os.path.abspath(src):
            raise ValueError(f"output_dir would overwrite the input {src}.")
        fmt = formats[name]
        if fmt == "netcdf":
            info = _subset_netcdf(src, dst, slots, lattice, n_jobs)
        elif fmt == "clm":
            info = cut_clm(src, dst, index, n_source)
        elif fmt == "meta":
            info = cut_meta(src, dst, index, n_source)
        else:
            info = cut_raw(src, dst, index, n_source)
        written[name] = dst
        record["inputs"][name] = {"source": os.path.abspath(src), "written": dst, **info}
        logger.info("subset_inputs: %s (%s) -> %s", src, fmt, dst)
    with open(os.path.join(output_dir, RECORD), "w") as fh:
        json.dump(record, fh, indent=2, default=str)
    return written
