Source code for climclusterpy._log

"""The package's progress messages: one logger, shown by default, quietened in one call.

Every module logs through ``logging.getLogger(__name__)``, so all messages sit under the
``"climclusterpy"`` logger. A long run is silent without them, so by default they are shown:
INFO and above go to standard output as plain lines, exactly as the package's old ``print``
calls did. :func:`set_verbosity` turns them down or off; a caller who configures logging
themselves (``logging.basicConfig`` or a handler on ``"climclusterpy"``) gets theirs instead.
"""

from __future__ import annotations

import logging
import sys

__all__ = ["joblib_verbosity", "set_verbosity"]

_ROOT = "climclusterpy"

_LEVELS = {
    "debug": logging.DEBUG,
    "info": logging.INFO,
    "warning": logging.WARNING,
    "error": logging.ERROR,
    "quiet": logging.CRITICAL + 1,
}


class _StdoutHandler(logging.StreamHandler):
    """A handler that writes to whatever ``sys.stdout`` is *now*, not at import time.

    So output that is redirected later (a notebook cell, ``contextlib.redirect_stdout``,
    a test capturing it) still receives the messages.
    """

    def __init__(self) -> None:
        super().__init__(sys.stdout)

    @property
    def stream(self):
        return sys.stdout

    @stream.setter
    def stream(self, value):  # StreamHandler.__init__ assigns it; ignore
        pass

    def emit(self, record: logging.LogRecord) -> None:
        # Step aside once the caller has set up logging of their own -- a handler on the
        # root logger (``logging.basicConfig``) or on ours -- so a message is never printed
        # twice. pytest's capture handlers do not count as the caller's set-up.
        if _caller_configured_logging():
            return
        super().emit(record)


def _caller_configured_logging() -> bool:
    ours = logging.getLogger(_ROOT)
    if any(not getattr(h, "_climclusterpy_default", False) for h in ours.handlers):
        return True
    return any(not type(h).__module__.startswith("_pytest") for h in logging.getLogger().handlers)


def _install_default_handler() -> None:
    logger = logging.getLogger(_ROOT)
    if any(getattr(h, "_climclusterpy_default", False) for h in logger.handlers):
        return
    handler = _StdoutHandler()
    handler.setFormatter(logging.Formatter("%(message)s"))
    handler._climclusterpy_default = True  # type: ignore[attr-defined]
    logger.addHandler(handler)
    if logger.level == logging.NOTSET:
        logger.setLevel(logging.INFO)

def set_verbosity(level: str | int = "info") -> None:
    """How much the package reports while it works.

    Parameters
    ----------
    level : {"debug", "info", "warning", "error", "quiet"} or int, optional
        ``"info"`` (the default) shows the progress messages; ``"warning"`` shows only
        warnings and errors; ``"quiet"`` shows nothing. An int is a :mod:`logging` level.
        Python warnings (:mod:`warnings`, e.g. about a suspicious input) are not affected.

    Examples
    --------
    >>> import climclusterpy
    >>> climclusterpy.set_verbosity("warning")  # a long run, without the progress lines
    """
    if isinstance(level, str):
        try:
            value = _LEVELS[level.lower()]
        except KeyError:
            raise ValueError(
                f"Unknown verbosity {level!r}; use one of {sorted(_LEVELS)} or a logging level."
            ) from None
    else:
        value = int(level)
    logging.getLogger(_ROOT).setLevel(value)




def joblib_verbosity() -> int:
    """The ``verbose=`` for joblib: its per-task progress lines only at the debug level."""
    return 10 if logging.getLogger(_ROOT).getEffectiveLevel() <= logging.DEBUG else 0
