Source code for climclusterpy.clustering.lmnn.saved

"""A learned distance metric as a thing you can save, load and apply again without retraining.

Training a per-cell metric on the global grid takes about half an hour on a GPU and, without a
seed, does not reproduce: two trainings on identical inputs give different clusterings. So the
trained metric is an artifact worth keeping, and reusing it -- another number of clusters, another
region of the same grid, the same grid after a code change -- must not mean training again.

:class:`LearnedMetric` wraps the three kinds the package trains:

* ``"lowrank"`` -- a per-cell low-rank Mahalanobis metric (``lmnn_mode="lowrank"``, the published
  configuration), a :class:`~climclusterpy.clustering.lmnn.local.LocalMahalanobisMLP`;
* ``"diagonal"`` -- per-cell feature weights (``lmnn_mode="local"`` or ``"enhanced"``), a
  :class:`~climclusterpy.clustering.lmnn.local.LocalWeightNetwork`;
* ``"global"`` -- one linear transform ``L`` for the whole grid (``lmnn_mode="global"``).

:meth:`LearnedMetric.save` writes a folder with ``metric.json`` (kind, sizes, the feature columns
it expects, in order) and the weights as a plain tensor file, which :func:`load_metric` reads back
without executing any pickled code. :func:`load_metric` also reads the ``model.pth`` files the
package's research runs wrote with ``torch.save(model)``, including those saved before the module
holding the model class was renamed in 0.2.0 -- but those *are* pickles, and loading a pickle
runs code, so load only files you trust.
"""

from __future__ import annotations

import importlib
import json
import os
import sys
import zipfile
from dataclasses import dataclass, field

import numpy as np

#: ``metric.json`` ``format`` value; bump the suffix on an incompatible change.
FORMAT = "climclusterpy-learned-metric/1"
METRIC_JSON = "metric.json"
WEIGHTS_FILE = "weights.pt"
TRANSFORM_FILE = "transform.npy"

#: Module paths that older saved models name for their class, and where the class lives now.
#: Additive only: each entry keeps the artifacts written before that rename loadable.
LEGACY_MODULES = {
    # 0.2.0 split clustering/lmnn_local.py into the clustering/lmnn/ package.
    "climclusterpy.clustering.lmnn_local": "climclusterpy.clustering.lmnn.local",
}

@dataclass
class LearnedMetric:
    """A trained distance metric, ready to apply to a feature matrix.

    Attributes
    ----------
    kind : str
        ``"lowrank"``, ``"diagonal"`` or ``"global"`` (see the module docstring).
    n_features : int
        Width of the feature matrix the metric applies to.
    model : torch.nn.Module or None
        The per-cell network (``"lowrank"``/``"diagonal"``); ``None`` for ``"global"``.
    transform : numpy.ndarray or None
        ``"global"`` only: the linear map ``L``; distances are Euclidean on ``X @ L.T``.
    rank : int or None
        ``"lowrank"`` only: the rank of each cell's metric factor.
    feature_names : tuple of str or None
        The feature columns the metric was trained on, in order, when known. A metric saved by
        :func:`~climclusterpy.api.create_clusters` records them; a bare ``model.pth`` does not.
    source : str or None
        Where it was loaded from.
    provenance : dict
        What the loader found out about the file (format, legacy module names it needed, the
        research ``meta.json`` next to it, ...), for the run record.
    """

    kind: str
    n_features: int
    model: object | None = None
    transform: np.ndarray | None = None
    rank: int | None = None
    feature_names: tuple[str, ...] | None = None
    source: str | None = None
    provenance: dict = field(default_factory=dict)

    def __repr__(self) -> str:
        rank = f", rank={self.rank}" if self.rank is not None else ""
        names = "" if self.feature_names is None else ", feature names recorded"
        return f"LearnedMetric(kind={self.kind!r}, n_features={self.n_features}{rank}{names})"

    def check_features(self, feature_columns=None, n_features: int | None = None) -> None:
        """Refuse a feature matrix the metric was not trained on.

        The width must match. When both the metric and the caller name the columns, the names
        and their order must match too: a learned metric weights *positions*, so the same width
        with the columns in another order is a well-formed, silently wrong distance.
        """
        width = len(feature_columns) if feature_columns is not None else n_features
        if width is not None and int(width) != self.n_features:
            raise ValueError(
                f"The learned metric applies to {self.n_features} features; the feature matrix "
                f"has {width}. A metric is tied to the exact feature set it was trained on."
            )
        if feature_columns is not None and self.feature_names is not None:
            given = [str(c) for c in feature_columns]
            if given != list(self.feature_names):
                missing = [c for c in self.feature_names if c not in given]
                extra = [c for c in given if c not in self.feature_names]
                detail = (
                    f"missing {missing}, unexpected {extra}"
                    if missing or extra
                    else "the same names in another order"
                )
                raise ValueError(
                    "The feature columns do not match the ones the learned metric was trained "
                    f"on ({detail}). Expected, in order: {list(self.feature_names)}."
                )

    def transformed(self, X) -> np.ndarray:
        """``"global"`` only: the features in the metric space, ``X @ L.T``."""
        if self.kind != "global":
            raise ValueError(f"A {self.kind!r} metric has no single linear transform.")
        return np.asarray(X, dtype=float) @ self.transform.T

    def distance_matrix(self, X, *, device: str | None = None) -> np.ndarray:
        """The dense ``(n_cells, n_cells)`` distance matrix of ``X`` under this metric.

        ``device`` is the torch device for the per-cell kinds (default: CUDA when available,
        else CPU). The value matters for reproduction: the same model evaluated on a GPU and on
        a CPU differs in the last bits of float32 arithmetic, which can move a near-tied merge
        in the dendrogram. Evaluate on the device class the reference was computed on.
        """
        X = np.asarray(X)
        self.check_features(n_features=X.shape[1])
        if self.kind == "global":
            from scipy.spatial.distance import pdist, squareform

            return squareform(pdist(self.transformed(X)))
        import torch

        if device is None:
            device = "cuda" if torch.cuda.is_available() else "cpu"
        if self.kind == "lowrank":
            from .local import compute_local_mahalanobis_distance_matrix

            return compute_local_mahalanobis_distance_matrix(
                np.asarray(X, dtype=np.float32), self.model, device=torch.device(device)
            )
        from .local import compute_distance_matrix_local

        return compute_distance_matrix_local(self.model, np.asarray(X, dtype=np.float32), device)

    def save(self, directory: str) -> str:
        """Write the metric to ``directory`` (created if missing); return the directory.

        Writes ``metric.json`` and either ``weights.pt`` (the network's tensors, no pickled
        code) or ``transform.npy``. :func:`load_metric` reads it back exactly.
        """
        os.makedirs(directory, exist_ok=True)
        record = {
            "format": FORMAT,
            "kind": self.kind,
            "n_features": int(self.n_features),
            "rank": None if self.rank is None else int(self.rank),
            "feature_names": None if self.feature_names is None else list(self.feature_names),
            "architecture": _architecture(self),
            "provenance": _json_safe(self.provenance),
        }
        if self.kind == "global":
            np.save(os.path.join(directory, TRANSFORM_FILE), np.asarray(self.transform))
        else:
            import torch

            state = {k: v.detach().cpu() for k, v in self.model.state_dict().items()}
            torch.save(state, os.path.join(directory, WEIGHTS_FILE))
        with open(os.path.join(directory, METRIC_JSON), "w") as fh:
            json.dump(record, fh, indent=2)
        return directory






def from_model(model, *, feature_names=None, provenance=None) -> LearnedMetric:
    """Wrap a trained per-cell network, as :func:`train_lmnn` returns it."""
    from .local import LocalMahalanobisMLP, LocalWeightNetwork

    if isinstance(model, LocalMahalanobisMLP):
        kind, d, rank = "lowrank", int(model.d), int(model.r)
    elif isinstance(model, LocalWeightNetwork):
        kind, d, rank = "diagonal", int(model.n_features), None
    else:
        raise TypeError(
            f"Not a metric network the package trains: {type(model).__module__}."
            f"{type(model).__name__}. Expected LocalMahalanobisMLP or LocalWeightNetwork."
        )
    model.eval()
    return LearnedMetric(
        kind=kind,
        n_features=d,
        model=model,
        rank=rank,
        feature_names=None if feature_names is None else tuple(str(c) for c in feature_names),
        provenance=dict(provenance or {}),
    )


def from_transform(transform, *, feature_names=None, provenance=None) -> LearnedMetric:
    """Wrap a global-mode linear transform ``L`` (shape ``(k, n_features)``)."""
    L = np.asarray(transform, dtype=float)
    if L.ndim != 2:
        raise ValueError(f"A global transform is 2-D (k, n_features); got shape {L.shape}.")
    return LearnedMetric(
        kind="global",
        n_features=int(L.shape[1]),
        transform=L,
        feature_names=None if feature_names is None else tuple(str(c) for c in feature_names),
        provenance=dict(provenance or {}),
    )


def _architecture(metric: LearnedMetric) -> dict:
    if metric.kind == "global":
        return {"transform_shape": list(np.asarray(metric.transform).shape)}
    if metric.kind == "lowrank":
        first = metric.model.net[0]
        return {
            "class": "LocalMahalanobisMLP",
            "d": metric.n_features,
            "r": metric.rank,
            "hidden": int(first.out_features),
        }
    linears = [m for m in metric.model.mlp if m.__class__.__name__ == "Linear"]
    return {
        "class": "LocalWeightNetwork",
        "n_features": metric.n_features,
        "hidden_dims": [int(m.out_features) for m in linears[:-1]],
        "dropout": float(getattr(metric.model, "dropout_p", 0.0)),
        "normalize_weights": bool(getattr(metric.model, "normalize_weights", True)),
    }


def _json_safe(value):
    if isinstance(value, dict):
        return {str(k): _json_safe(v) for k, v in value.items()}
    if isinstance(value, list | tuple):
        return [_json_safe(v) for v in value]
    if isinstance(value, np.generic):
        return value.item()
    if isinstance(value, str | int | float | bool) or value is None:
        return value
    return str(value)


def _build_network(arch: dict):
    from .local import LocalMahalanobisMLP, LocalWeightNetwork

    if arch["class"] == "LocalMahalanobisMLP":
        d, hidden = int(arch["d"]), int(arch["hidden"])
        # The constructor takes a width *factor* and floors d * factor; (hidden + 0.5) / d
        # floors back to exactly `hidden` whatever the float rounding.
        return LocalMahalanobisMLP(d=d, r=int(arch["r"]), hidden_width_factor=(hidden + 0.5) / d)
    if arch["class"] == "LocalWeightNetwork":
        return LocalWeightNetwork(
            n_features=int(arch["n_features"]),
            hidden_dims=[int(h) for h in arch["hidden_dims"]],
            dropout=float(arch["dropout"]),
            normalize_weights=bool(arch["normalize_weights"]),
        )
    raise ValueError(f"Unknown network class in metric.json: {arch['class']!r}.")


def _legacy_modules_named(path: str) -> list[str]:
    """Which renamed module paths a ``torch.save``d pickle names, read without unpickling it."""
    try:
        with zipfile.ZipFile(path) as z:
            entries = [n for n in z.namelist() if n.endswith("data.pkl")]
            blob = z.read(entries[0]) if entries else b""
    except (zipfile.BadZipFile, OSError, KeyError):
        return []
    return sorted(old for old in LEGACY_MODULES if old.encode() in blob)


def _install_legacy_modules() -> None:
    """Let old pickles resolve their class: point each old module path at today's module.

    The alias is the same module object, not a copy, so an unpickled class *is* today's class.
    A real module of the old name, should one ever exist again, is never shadowed.
    """
    for old, new in LEGACY_MODULES.items():
        if old not in sys.modules:
            sys.modules[old] = importlib.import_module(new)

def load_metric(path: str, *, device: str = "cpu") -> LearnedMetric:
    """Load a learned distance metric saved earlier, to apply it without training again.

    Parameters
    ----------
    path : str
        One of:

        * a folder written by :meth:`LearnedMetric.save` -- which is what
          :func:`~climclusterpy.api.create_clusters` writes to ``<output_dir>/learned_metric``
          whenever it trains a metric. Loaded without executing pickled code;
        * a ``model.pth`` written with ``torch.save(model)`` (the research runs' deploy
          folders), or the folder holding one. This **is** a pickle: loading it runs code, so
          load only files you trust. Files saved before 0.2.0 renamed the model's module load
          too. A ``meta.json`` beside it is read into :attr:`LearnedMetric.provenance`.
    device : str, optional
        Where the network's tensors are loaded (default ``"cpu"``); the distance computation
        picks its own device, see :meth:`LearnedMetric.distance_matrix`.

    Returns
    -------
    LearnedMetric

    Raises
    ------
    FileNotFoundError
        If ``path`` holds neither ``metric.json`` nor ``model.pth``.
    ImportError
        For a network metric without ``torch`` (the ``[lmnn]`` extra).
    TypeError
        If a ``model.pth`` holds something other than a metric network the package trains.
    """
    path = os.fspath(path)
    if os.path.isdir(path):
        if os.path.isfile(os.path.join(path, METRIC_JSON)):
            return _load_saved(path, device)
        pth = os.path.join(path, "model.pth")
        if not os.path.isfile(pth):
            raise FileNotFoundError(
                f"{path} holds neither {METRIC_JSON} (a saved LearnedMetric) nor model.pth."
            )
        path = pth
    if not os.path.isfile(path):
        raise FileNotFoundError(f"No saved metric at {path}.")
    if os.path.basename(path) == METRIC_JSON:
        return _load_saved(os.path.dirname(path) or ".", device)
    return _load_pickle(path, device)




def _load_saved(directory: str, device: str) -> LearnedMetric:
    with open(os.path.join(directory, METRIC_JSON)) as fh:
        record = json.load(fh)
    if record.get("format") != FORMAT:
        raise ValueError(
            f"{directory}/{METRIC_JSON} has format {record.get('format')!r}; this version of "
            f"climclusterpy reads {FORMAT!r}."
        )
    names = record.get("feature_names")
    provenance = {"source_format": FORMAT, "when_saved": record.get("provenance") or {}}
    if record["kind"] == "global":
        metric = from_transform(
            np.load(os.path.join(directory, TRANSFORM_FILE)),
            feature_names=names,
            provenance=provenance,
        )
    else:
        try:
            import torch
        except ImportError as e:
            raise ImportError(
                "Loading a per-cell learned metric needs torch: install the [lmnn] extra."
            ) from e
        model = _build_network(record["architecture"])
        state = torch.load(
            os.path.join(directory, WEIGHTS_FILE), map_location=device, weights_only=True
        )
        model.load_state_dict(state)
        metric = from_model(model, feature_names=names, provenance=provenance)
    metric.source = os.path.abspath(directory)
    return metric


def _load_pickle(path: str, device: str) -> LearnedMetric:
    try:
        import torch
    except ImportError as e:
        raise ImportError("Loading a model.pth needs torch: install the [lmnn] extra.") from e
    legacy = _legacy_modules_named(path)
    _install_legacy_modules()
    # weights_only=False because the file is a pickled model object, not a tensor file.
    obj = torch.load(path, map_location=device, weights_only=False)
    provenance: dict = {
        "source_format": "torch.save(model) pickle",
        "class": f"{type(obj).__module__}.{type(obj).__name__}",
        "renamed_modules_needed": legacy,
    }
    meta = os.path.join(os.path.dirname(path), "meta.json")
    if os.path.isfile(meta):
        with open(meta) as fh:
            provenance["meta_json"] = json.load(fh)
    metric = from_model(obj, provenance=provenance)
    metric.source = os.path.abspath(path)
    return metric
