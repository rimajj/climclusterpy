"""
LMNN (Large Margin Nearest Neighbor) metric learning for ecological clustering.

Trains a Mahalanobis distance metric that pulls same-ecoregion cells together
and pushes different-ecoregion cells apart.

Modes:

- ``"global"``: one linear map ``L`` (Mahalanobis matrix ``M = LᵀL``) for all cells,
  trained in PyTorch (:mod:`~climclusterpy.clustering.lmnn.linear`).
- ``"local"``: a network predicts per-cell feature weights (a diagonal metric).
- ``"enhanced"``: the local metric plus class prototypes and oversampling.
- ``"lowrank"``: a network predicts a per-cell low-rank Mahalanobis metric.

Every mode needs PyTorch, i.e. the ``[lmnn]`` extra; it is imported only when a mode is trained.
"""

from __future__ import annotations

import logging
import pickle
from collections.abc import Callable
from pathlib import Path

import numpy as np
import pandas as pd
from sklearn.discriminant_analysis import LinearDiscriminantAnalysis

logger = logging.getLogger(__name__)

class LDAMetricLearner:
    """
    LDA-based metric learner for ecological clustering.

    Fits scikit-learn's ``LinearDiscriminantAnalysis`` with ``solver='eigen'``
    and ``shrinkage='auto'`` (Ledoit-Wolf shrinkage) on a feature matrix *X*
    supervised by either Köppen-Geiger zone labels or ecoregion labels.

    The fitted discriminant axes define a lower-dimensional space (at most
    ``n_classes − 1`` dimensions) in which same-class cells cluster tightly
    and different-class cells are well separated, analogous to what LMNN
    achieves via a Mahalanobis matrix.

    Parameters
    ----------
    supervision : {'kg', 'ecoregion'}
        Which label set was used for supervision.  Stored as metadata and
        reflected in the default cache filename.
    cache_dir : str or Path, optional
        Directory in which to persist the fitted model as
        ``lda_<supervision>_<fingerprint>.pkl``, the fingerprint being that of the
        ``X`` and ``y`` it was fitted on. When that file already exists and
        ``force_recompute=False``, the stored model is loaded instead of
        refitting; a model fitted on other data is never loaded.
    force_recompute : bool
        When *True*, ignore any existing cache and always refit.

    Attributes
    ----------
    lda_ : LinearDiscriminantAnalysis
        The fitted sklearn LDA estimator (set after calling :meth:`fit`).
    scalings_ : np.ndarray, shape (n_features, n_components)
        Discriminant axis loadings (columns = LDA components).
    feature_importance_ : np.ndarray, shape (n_features,)
        Absolute mean loading per original feature across all discriminant
        axes, normalised to sum to 1.  Analogous to the diagonal of the
        Mahalanobis matrix *M* used by global LMNN.
    n_components_ : int
        Number of discriminant components retained (≤ n_classes − 1).
    """

    def __init__(
        self,
        supervision: str = "kg",
        cache_dir: str | None = None,
        force_recompute: bool = False,
    ) -> None:
        if supervision not in ("kg", "ecoregion"):
            raise ValueError(f"supervision must be 'kg' or 'ecoregion', got '{supervision}'")
        self.supervision = supervision
        self.cache_dir = Path(cache_dir) if cache_dir is not None else None
        self.force_recompute = force_recompute

        self.lda_: LinearDiscriminantAnalysis | None = None
        self.scalings_: np.ndarray | None = None
        self._feature_importance_: np.ndarray | None = None
        self.n_components_: int | None = None
        self._input_key: str | None = None

    # ------------------------------------------------------------------
    # Cache helpers
    # ------------------------------------------------------------------
    def _cache_path(self) -> Path | None:
        if self.cache_dir is None or self._input_key is None:
            return None
        return self.cache_dir / f"lda_{self.supervision}_{self._input_key}.pkl"

    def _try_load(self) -> bool:
        """Try to load a cached model.  Returns True on success."""
        p = self._cache_path()
        if p is None or not p.exists() or self.force_recompute:
            return False
        with open(p, "rb") as fh:
            state = pickle.load(fh)
        self.lda_ = state["lda"]
        self.scalings_ = state["scalings"]
        self._feature_importance_ = state["feature_importance"]
        self.n_components_ = state["n_components"]
        logger.info(
            f"[LDAMetric] Loaded cached LDA ({self.supervision}) from {p} "
            f"— {self.n_components_} components"
        )
        return True

    def _save(self) -> None:
        p = self._cache_path()
        if p is None:
            return
        p.parent.mkdir(parents=True, exist_ok=True)
        state = {
            "lda": self.lda_,
            "scalings": self.scalings_,
            "feature_importance": self._feature_importance_,
            "n_components": self.n_components_,
        }
        with open(p, "wb") as fh:
            pickle.dump(state, fh)
        logger.info(f"[LDAMetric] Saved LDA ({self.supervision}) model to {p}")

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------
    def fit(self, X: np.ndarray, y: np.ndarray) -> LDAMetricLearner:
        """
        Fit the LDA metric learner.

        If a model fitted on this same *X* and *y* is cached (and
        ``force_recompute=False``), it is loaded instead of refitting.

        Parameters
        ----------
        X : np.ndarray, shape (n_cells, n_features)
            Standardised feature matrix.
        y : np.ndarray, shape (n_cells,)
            Integer or string class labels.

        Returns
        -------
        self
        """
        from climclusterpy._cache import fingerprint

        self._input_key = fingerprint(np.asarray(X, dtype=float), np.asarray(y).astype(str))
        if self._try_load():
            return self

        n_classes = len(np.unique(y))
        n_components = min(n_classes - 1, X.shape[1])
        logger.info(
            f"[LDAMetric] Fitting LDA ({self.supervision}) — "
            f"{X.shape[0]} samples, {X.shape[1]} features, "
            f"{n_classes} classes → {n_components} discriminant axes"
        )

        lda = LinearDiscriminantAnalysis(
            solver="eigen",
            shrinkage="auto",
            n_components=n_components,
        )
        lda.fit(X, y)

        self.lda_ = lda
        # scalings_ shape: (n_features, n_components)
        self.scalings_ = lda.scalings_[:, :n_components]
        self.n_components_ = n_components

        # Feature importance: absolute mean loading across all axes, normalised
        abs_loadings = np.abs(self.scalings_)  # (n_features, n_components)
        mean_abs = abs_loadings.mean(axis=1)  # (n_features,)
        total = mean_abs.sum()
        self._feature_importance_ = mean_abs / total if total > 0 else mean_abs

        self._save()
        return self

    def transform(self, X: np.ndarray) -> np.ndarray:
        """
        Project *X* into the LDA discriminant space.

        Parameters
        ----------
        X : np.ndarray, shape (n_cells, n_features)

        Returns
        -------
        X_lda : np.ndarray, shape (n_cells, n_components)
        """
        if self.lda_ is None:
            raise RuntimeError("Call fit() before transform().")
        return self.lda_.transform(X)

    @property
    def feature_importance_(self) -> np.ndarray:
        """
        Absolute mean loading per original feature across all discriminant
        axes, normalised to sum to 1.
        """
        if self._feature_importance_ is None:
            raise RuntimeError("Call fit() before accessing feature_importance_.")
        return self._feature_importance_

    def feature_importance_df(self, feature_names: list[str]) -> pd.DataFrame:
        """
        Return a DataFrame with columns ``['feature', 'importance']`` sorted
        by importance descending.  Mirrors the output of
        :func:`compute_feature_importance` for easy comparison.
        """
        imp = self.feature_importance_
        df = pd.DataFrame({"feature": feature_names, "importance": imp})
        df.sort_values("importance", ascending=False, inplace=True)
        df.reset_index(drop=True, inplace=True)
        return df




def _require_torch(mode: str) -> None:
    """Fail with the install command, not a bare ModuleNotFoundError, when torch is absent."""
    import importlib.util

    if importlib.util.find_spec("torch") is None:
        raise ImportError(
            f"lmnn_mode={mode!r} is trained with PyTorch, which is not installed. Install the "
            "metric-learning extra: pip install 'climclusterpy[lmnn]'."
        )


def _reject_unconsumed_kwargs(kwargs: dict, target: Callable, mode: str) -> None:
    """Raise if any extra keyword is not a parameter of the trainer that will run.

    Extra keywords used to be forwarded blindly, so a name the target trainer did
    not accept was either a confusing ``TypeError`` from deep inside the call
    stack or -- worse, for keywords that happened to bind to a *different*
    function's parameter -- silently dropped: a caller asking for a reduced-rank
    metric in a mode that cannot give one used to get a full-rank one and no
    warning. Fail fast instead.
    """
    import inspect

    accepted = {
        name
        for name, p in inspect.signature(target).parameters.items()
        if p.kind in (inspect.Parameter.POSITIONAL_OR_KEYWORD, inspect.Parameter.KEYWORD_ONLY)
    }
    unknown = sorted(set(kwargs) - accepted)
    if unknown:
        raise TypeError(
            f"train_lmnn(mode={mode!r}) got unexpected keyword argument(s): "
            f"{', '.join(unknown)}. The {mode} trainer ({target.__name__}) does not "
            f"accept them, and silently ignoring them would produce a model that "
            f"does not match what you asked for. Accepted extras: "
            f"{', '.join(sorted(accepted - {'X', 'y', 'labels'}))}."
        )

def train_lmnn(
    X: np.ndarray,
    labels: np.ndarray,
    mode: str = "global",
    k: int | None = None,
    max_iter: int = 1000,
    # Options of the per-cell network modes ("local", "enhanced", "lowrank")
    hidden_dims: list[int] | None = None,
    margin: float = 1.0,
    learning_rate: float | None = None,
    batch_size: int = 128,
    max_epochs: int = 100,
    dropout: float = 0.1,
    weight_decay: float = 1e-4,
    patience: int = 10,
    rank: int | None = None,
    seed: int | None = None,
    verbose: bool = False,
    **kwargs,
) -> tuple[np.ndarray, np.ndarray, dict] | tuple[object, Callable, dict]:
    """Train an LMNN metric-learning model in one of four modes.

    Parameters
    ----------
    X : numpy.ndarray
        Feature matrix ``(n_samples, n_features)``; should be standardized.
    labels : numpy.ndarray
        Class labels (e.g. ecoregion IDs), shape ``(n_samples,)``.
    mode : str, optional
        ``"global"`` learns one linear map ``L`` for all cells (classic LMNN);
        ``"local"`` learns feature-conditioned **diagonal** local weights via a
        neural network; ``"enhanced"`` adds prototypes and oversampling to the
        diagonal metric; ``"lowrank"`` learns a per-cell **low-rank Mahalanobis**
        metric that captures cross-feature interaction (the published
        configuration, ``rank=16`` on 25 features). Default ``"global"``.
    k : int or None, optional
        Number of target neighbours (default 5). Not accepted by
        ``mode="lowrank"``, which derives ``k`` per anchor from ``k_min``,
        ``k_max`` and ``beta``.
    max_iter : int, optional
        Global mode only: maximum number of training iterations (default 1000).
    hidden_dims : list of int, optional
        Local/enhanced modes: hidden-layer sizes for the network (default
        ``[64, 32]``).
    margin : float, optional
        Global, local and enhanced modes: impostor separation margin (default 1.0).
    learning_rate : float or None, optional
        Optimiser step size. ``None`` (default) uses the mode's own default:
        1e-2 for global mode (Adam on the entries of ``L``), 1e-3 for the
        network modes.
    batch_size : int, optional
        Network modes only: training batch size (default 128).
    max_epochs : int, optional
        Network modes only: maximum training epochs (default 100).
    dropout : float, optional
        Network modes only: dropout probability (default 0.1).
    weight_decay : float, optional
        Local/enhanced/lowrank only: L2 regularization strength (default 1e-4).
    patience : int, optional
        Network modes only: early-stopping patience (default 10).
    rank : int or None, optional
        Rank of the learned metric. ``mode="global"``: number of rows of ``L``
        (``None`` = full rank, ``n_features``). ``mode="lowrank"``: rank of each
        cell's metric factor (``None`` = 16, the published setting on 25
        features). A rank above ``n_features`` is lowered to ``n_features``.
        The diagonal modes (``"local"``, ``"enhanced"``) have no rank, so
        passing it there raises.
    seed : int or None, optional
        Seed for every RNG the trainer draws from. ``None`` (default) leaves the
        global RNGs untouched, so the network modes are **not reproducible**;
        pass a seed for any run whose numbers you intend to cite. Global mode
        has no random component and is reproducible either way.
    verbose : bool, optional
        Log training progress (default ``False``).
    **kwargs
        Further options of the trainer the mode runs:
        :func:`~climclusterpy.clustering.lmnn.linear.train_global_lmnn`
        (e.g. ``push_weight``, ``n_candidates``, ``refresh_every``),
        :func:`~climclusterpy.clustering.lmnn.local.train_local_lmnn`,
        :func:`~climclusterpy.clustering.lmnn.enhanced.train_enhanced_local_lmnn` or
        :func:`~climclusterpy.clustering.lmnn.lowrank.train_local_mahalanobis_lmnn`.
        A name that trainer does not accept raises ``TypeError``.

    Returns
    -------
    tuple
        For ``mode="global"``: ``(L, M, diagnostics)`` -- the linear map ``L`` of
        shape ``(rank, n_features)``, the Mahalanobis matrix ``M = L.T @ L`` of
        shape ``(n_features, n_features)``, and a diagnostics dict (loss history,
        convergence info, the rank used).
        For the other modes: ``(model, distance_fn, diagnostics)`` -- the trained
        network, a callable computing the pairwise distance matrix
        (``distance_fn(X) -> D``), and a diagnostics dict.

    Raises
    ------
    ImportError
        If PyTorch (the ``[lmnn]`` extra) is not installed.
    ValueError
        For an unknown mode, or ``rank``/``k`` given to a mode that cannot honour it.
    TypeError
        For a keyword argument the selected trainer does not accept.

    Notes
    -----
    Global mode assumes one universal feature weighting across all cells, while
    the per-cell modes let the metric vary with local environmental context
    (e.g. temperature matters more in the tropics).

    Examples
    --------
    Global LMNN::

        L, M, diag = train_lmnn(X, labels, mode="global", rank=10, seed=0)
        X_transformed = transform_features(X, L)

    Local LMNN::

        model, distance_fn, diag = train_lmnn(X, labels, mode="local", seed=0)
        D = distance_fn(X)  # pairwise distances
    """
    # `output_dim` was the global mode's name for what every mode now calls `rank`.
    # An explicit value is refused with the new name, so an old call does not silently
    # train a full-rank metric; `None` asks for nothing and is dropped.
    old_rank = kwargs.pop("output_dim", None)
    if old_rank is not None:
        raise TypeError(
            "output_dim was renamed to rank in climclusterpy 0.4.0 (one name for the metric's "
            f"rank in every mode). Pass rank={old_rank!r} instead."
        )
    # The diagonal modes learn one weight per feature per cell; there is no rank to
    # choose, and accepting one and dropping it would hand back a different metric
    # from the one asked for.
    if mode in ("local", "enhanced") and rank is not None:
        raise ValueError(
            f"rank applies to mode='global' and mode='lowrank' only; mode={mode!r} learns a "
            "diagonal metric, which has no rank."
        )
    # `k` is a fixed target-neighbour count; lowrank picks k per anchor from the
    # adaptive schedule instead, so accepting k there and dropping it would be
    # the same silent-wrong-answer failure.
    if mode == "lowrank" and k is not None:
        raise ValueError(
            "mode='lowrank' chooses target neighbours per anchor from an adaptive "
            "schedule, so a fixed k does not apply. Tune it with k_min, k_max and "
            "beta (defaults 3, 15, 0.005 -- the published setting)."
        )
    if mode not in ("global", "local", "enhanced", "lowrank"):
        raise ValueError(
            f"Unknown mode: '{mode}'. Use 'global', 'local', 'enhanced', or 'lowrank'.\n"
            "  - 'global': Classic LMNN with one linear map for all cells\n"
            "  - 'local': Feature-conditioned LMNN with neural network (DIAGONAL metric)\n"
            "  - 'enhanced': Local LMNN with adaptive k, prototypes, and oversampling\n"
            "  - 'lowrank': Per-cell LOW-RANK Mahalanobis metric (captures feature\n"
            "    interactions; the published configuration, rank=16)"
        )
    if k is None:
        k = 5
    _require_torch(mode)
    # The network modes keep the step size they have always been run with.
    net_learning_rate = 1e-3 if learning_rate is None else learning_rate

    if mode == "global":
        from .linear import train_global_lmnn

        _reject_unconsumed_kwargs(kwargs, train_global_lmnn, mode)
        if learning_rate is not None:
            kwargs["learning_rate"] = learning_rate
        return train_global_lmnn(
            X,
            labels,
            k=k,
            rank=rank,
            margin=margin,
            max_iter=max_iter,
            seed=seed,
            verbose=verbose,
            **kwargs,
        )
    elif mode == "local":
        from .local import compute_distance_matrix_local, train_local_lmnn

        _reject_unconsumed_kwargs(kwargs, train_local_lmnn, mode)
        model, diagnostics = train_local_lmnn(
            X,
            labels,
            k=k,
            margin=margin,
            hidden_dims=hidden_dims or [64, 32],
            dropout=dropout,
            learning_rate=net_learning_rate,
            weight_decay=weight_decay,
            batch_size=batch_size,
            max_epochs=max_epochs,
            patience=patience,
            seed=seed,
            verbose=verbose,
            **kwargs,
        )

        # Create distance function closure
        def distance_fn(X_new: np.ndarray) -> np.ndarray:
            return compute_distance_matrix_local(model, X_new, verbose=verbose)

        return model, distance_fn, diagnostics
    elif mode == "enhanced":
        # Enhanced mode: delegate directly — no parameter renaming needed here.
        # Callers must use the same names as train_enhanced_local_lmnn:
        #   enable_prototypes, tiny_class_threshold, enable_oversampling,
        #   small_class_threshold, per_ecoregion_metrics_path, etc.
        from .enhanced import train_enhanced_local_lmnn

        _reject_unconsumed_kwargs(kwargs, train_enhanced_local_lmnn, mode)
        model, distance_fn, diagnostics = train_enhanced_local_lmnn(
            X,
            labels,
            k=k,
            margin=margin,
            hidden_dims=hidden_dims or [64, 32],
            dropout=dropout,
            learning_rate=net_learning_rate,
            weight_decay=weight_decay,
            batch_size=batch_size,
            max_epochs=max_epochs,
            patience=patience,
            seed=seed,
            verbose=verbose,
            **kwargs,
        )

        return model, distance_fn, diagnostics
    else:  # "lowrank"
        from .lowrank import DEFAULT_RANK, make_distance_fn, train_local_mahalanobis_lmnn

        _reject_unconsumed_kwargs(kwargs, train_local_mahalanobis_lmnn, mode)
        model, diagnostics = train_local_mahalanobis_lmnn(
            X,
            labels,
            rank=DEFAULT_RANK if rank is None else rank,
            dropout_rate=dropout,
            learning_rate=net_learning_rate,
            weight_decay=weight_decay,
            batch_size=batch_size,
            max_epochs=max_epochs,
            patience=patience,
            seed=seed,
            verbose=verbose,
            **kwargs,
        )
        return model, make_distance_fn(model), diagnostics

def compute_feature_importance(
    M: np.ndarray,
    feature_names: list[str],
) -> pd.DataFrame:
    """
    Compute feature importance from Mahalanobis matrix.

    Importance = diagonal of M (weight on each feature's squared distance).
    Higher values indicate features that contribute more to the learned metric.

    Args:
        M: Mahalanobis matrix (n_features, n_features)
        feature_names: List of feature names

    Returns
    -------
        DataFrame with columns ['feature', 'importance'] sorted by importance
    """
    logger = logging.getLogger(__name__)
    logger.debug("[LMNN] Computing feature importance from Mahalanobis matrix")

    # Extract diagonal (self-weights)
    importance = np.diag(M)

    # Create DataFrame
    importance_df = pd.DataFrame(
        {
            "feature": feature_names,
            "importance": importance,
        }
    ).sort_values("importance", ascending=False)

    # Normalize to sum to 1 for interpretability
    importance_df["importance_normalized"] = (
        importance_df["importance"] / importance_df["importance"].sum()
    )

    logger.debug("[LMNN] Top 5 most important features:")
    for _idx, row in importance_df.head(5).iterrows():
        logger.debug(
            f"  {row['feature']}: {row['importance']:.4f} ({row['importance_normalized']:.2%})"
        )

    return importance_df

def transform_features(X: np.ndarray, L: np.ndarray) -> np.ndarray:
    """
    Apply LMNN transformation to features.

    Args:
        X: Feature matrix (n_samples, n_features)
        L: Transformation matrix (rank, n_features)

    Returns
    -------
        X_transformed: Transformed features (n_samples, rank)
    """
    logging.getLogger(__name__).debug(
        f"[LMNN] Transforming {X.shape[0]} samples with learned metric"
    )

    X_transformed = X @ L.T

    logging.getLogger(__name__).debug(f"[LMNN] Output shape: {X_transformed.shape}")

def compute_mahalanobis_distances(
    X: np.ndarray,
    M: np.ndarray,
) -> np.ndarray:
    """
    Compute pairwise Mahalanobis distances.

    D[i, j] = sqrt((x_i - x_j).T @ M @ (x_i - x_j))

    Args:
        X: Feature matrix (n_samples, n_features)
        M: Mahalanobis matrix (n_features, n_features)

    Returns
    -------
        Distance matrix (n_samples, n_samples)
    """
    n_samples = X.shape[0]
    D = np.zeros((n_samples, n_samples))

    logging.getLogger(__name__).debug(
        f"[LMNN] Computing Mahalanobis distances for {n_samples} samples (may be slow)"
    )

    for i in range(n_samples):
        if i % 100 == 0 and i > 0:
            logging.getLogger(__name__).debug(f"  Progress: {i}/{n_samples}")

        diff = X - X[i, :]  # (n_samples, n_features)
        # (diff @ M @ diff.T) gives squared distances
        D[i, :] = np.sqrt(np.sum((diff @ M) * diff, axis=1))

    return D
