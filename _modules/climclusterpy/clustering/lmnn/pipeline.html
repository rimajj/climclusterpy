Source code for climclusterpy.clustering.lmnn.pipeline

"""Path-free LMNN clustering driver (promoted from ``pipelines/run_lmnn_ecoregion_clustering.py``).

The legacy experiment script mixed data loading, hardcoded ``/p`` paths, feature
caching, and plotting in with the actual metric-learning-plus-clustering logic.
This module extracts just that logic into a single reusable function,
:func:`run_lmnn_clustering`, that takes a feature matrix and supervision labels
already in memory and returns cluster labels — no files, no fixed paths.

Heavy dependencies stay lazy: importing this module (and
:mod:`climclusterpy.clustering.lmnn`) does **not** import ``torch`` — that only
happens inside :func:`train_lmnn`, when a metric is actually trained. Every mode
needs ``torch``, provided by the ``[lmnn]`` extra.
"""

from __future__ import annotations

from dataclasses import dataclass, field

import numpy as np

@dataclass
class LmnnClusteringResult:
    """Result of :func:`run_lmnn_clustering`.

    Attributes
    ----------
    labels : numpy.ndarray
        1-indexed cluster label per cell (``1..n_clusters``), shape ``(n_cells,)``,
        from :func:`scipy.cluster.hierarchy.fcluster` with ``criterion="maxclust"``.
    linkage : numpy.ndarray
        The SciPy linkage matrix ``Z`` of shape ``(n_cells - 1, 4)``.
    n_clusters : int
        Requested number of clusters.
    mode : str
        The LMNN mode used (``"global"``, ``"local"``, ``"enhanced"`` or ``"lowrank"``).
    diagnostics : dict
        Training diagnostics returned by :func:`train_lmnn`.
    transform : numpy.ndarray or None
        Global-mode linear transform ``L`` (``X @ L.T`` is the metric space);
        ``None`` for local/enhanced modes.
    mahalanobis : numpy.ndarray or None
        Global-mode Mahalanobis matrix ``M = L.T @ L``; ``None`` otherwise.
    distance_matrix : numpy.ndarray or None
        Local/enhanced dense ``(n_cells, n_cells)`` distance matrix, only kept when
        ``return_distance_matrix=True`` (it is memory-heavy for large grids).
    model : object or None
        The trained per-cell metric network (a ``torch.nn.Module``), only kept when
        ``return_model=True``; ``None`` for global mode, whose learned metric is the
        pair ``transform``/``mahalanobis`` and involves no network.
    """

    labels: np.ndarray
    linkage: np.ndarray
    n_clusters: int
    mode: str
    diagnostics: dict = field(default_factory=dict)
    transform: np.ndarray | None = None
    mahalanobis: np.ndarray | None = None
    distance_matrix: np.ndarray | None = None
    model: object | None = None

def run_lmnn_clustering(
    X: np.ndarray,
    labels: np.ndarray,
    n_clusters: int,
    *,
    mode: str = "global",
    linkage_method: str = "complete",
    rank: int | None = None,
    k: int | None = None,
    seed: int | None = None,
    verbose: bool = False,
    return_distance_matrix: bool = False,
    return_model: bool = False,
    **lmnn_kwargs,
) -> LmnnClusteringResult:
    """Learn an LMNN metric from supervision labels, then hierarchically cluster.

    This is the in-memory, path-free core of the LMNN clustering pipeline. It

    1. trains an LMNN metric on ``X`` supervised by ``labels`` via
       :func:`climclusterpy.clustering.lmnn.train_lmnn`, then
    2. builds a SciPy linkage and cuts it to ``n_clusters``.

    The bridge from metric to linkage depends on the mode:

    * ``"global"`` — learns a linear transform ``L``; clustering runs on the
      transformed features ``X @ L.T`` with ordinary Euclidean linkage (cheap; no
      dense distance matrix is materialized).
    * ``"local"`` / ``"enhanced"`` — learns a per-cell (diagonal) metric realized
      as a dense precomputed distance matrix ``D``; clustering runs on ``D``.
    * ``"lowrank"`` — learns a per-cell **low-rank Mahalanobis** metric
      (``M_i = L_i L_iᵀ``, ``L_i`` of shape ``(d, rank)``), also realized as a
      dense ``D``. Unlike the diagonal modes it captures cross-feature
      interaction; this is the configuration the published results deploy.

    Parameters
    ----------
    X : numpy.ndarray
        Standardized feature matrix, shape ``(n_cells, n_features)``.
    labels : numpy.ndarray
        Supervision labels, shape ``(n_cells,)`` — e.g. Köppen-Geiger codes,
        ecoregion IDs, or output-cluster labels. For ``"local"``/``"enhanced"``
        modes the labels are integer-encoded internally (the torch trainers require
        contiguous integer classes); ``"global"`` accepts arbitrary label values.
    n_clusters : int
        Number of clusters to cut the dendrogram into.
    mode : str, optional
        LMNN mode: ``"global"`` (default), ``"local"``, ``"enhanced"`` or ``"lowrank"``.
    linkage_method : str, optional
        SciPy linkage method (default ``"complete"``, matching the production path).
    rank : int or None, optional
        Rank of the learned metric, for the two modes that have one.
        ``mode="global"``: number of rows of the linear map ``L`` (``None`` = full
        rank). ``mode="lowrank"``: rank ``r`` of each cell's Mahalanobis factor
        (``None`` = 16, the setting the published results use on 25 features).
        A rank above the number of features is lowered to it. The diagonal modes
        (``"local"``, ``"enhanced"``) have no rank and raise if given one.
    k : int or None, optional
        LMNN target-neighbour count (default 5). Not accepted by
        ``mode="lowrank"``, which derives it per anchor.
    seed : int or None, optional
        Seed every RNG the torch-backed modes draw from. ``None`` (default)
        leaves them untouched, so ``local``/``enhanced``/``lowrank`` are **not
        reproducible**; pass a seed for any run whose numbers you will cite.
    verbose : bool, optional
        Forwarded to :func:`train_lmnn`.
    return_distance_matrix : bool, optional
        Keep the dense per-cell distance matrix in the result (default False).
        Required to select medoid representatives downstream.
    return_model : bool, optional
        Keep the trained per-cell metric network in the result (default False).
        Not available for ``mode="global"``, which learns no network.
    **lmnn_kwargs
        Extra keyword arguments forwarded verbatim to :func:`train_lmnn` (e.g.
        ``max_iter`` for global mode, or ``max_epochs``/``margin`` for local modes).
        A name the selected trainer does not accept raises ``TypeError``.

    Returns
    -------
    LmnnClusteringResult
        The cluster labels, linkage, and metric artifacts.

    Raises
    ------
    ValueError
        If ``mode`` is not one of ``"global"``, ``"local"``, ``"enhanced"``,
        ``"lowrank"``, or ``rank`` is given to a diagonal mode.
    ImportError
        If ``torch`` (the ``[lmnn]`` extra) is not installed.
    """
    from scipy.cluster.hierarchy import fcluster
    from scipy.cluster.hierarchy import linkage as _linkage
    from scipy.spatial.distance import squareform

    from .train import train_lmnn, transform_features

    if mode not in ("global", "local", "enhanced", "lowrank"):
        raise ValueError(
            f"Unknown LMNN mode: {mode!r}. Use 'global', 'local', 'enhanced', or 'lowrank'."
        )

    X = np.asarray(X, dtype=float)
    y = np.asarray(labels)
    if X.ndim != 2:
        raise ValueError(f"X must be 2D (n_cells, n_features); got shape {X.shape}.")
    if y.shape[0] != X.shape[0]:
        raise ValueError(f"labels length ({y.shape[0]}) must match X rows ({X.shape[0]}).")

    if return_distance_matrix and mode == "global":
        raise ValueError(
            "return_distance_matrix=True is not available for mode='global', which "
            "learns a linear transform L and clusters the transformed features "
            "without ever materializing a dense distance matrix. Use the returned "
            "`transform`/`mahalanobis`, or a per-cell mode "
            "('local', 'enhanced', 'lowrank')."
        )

    if return_model and mode == "global":
        raise ValueError(
            "return_model=True is not available for mode='global', which learns a "
            "single Mahalanobis matrix rather than a network. Its learned metric is "
            "returned in full as `transform` (L) and `mahalanobis` (M); a per-cell "
            "mode ('local', 'enhanced', 'lowrank') has a model to hand back."
        )

    transform = mahalanobis = distance_matrix = model = None

    if mode == "global":
        # One linear map L for all cells; clustering on X @ L.T with Euclidean
        # distance is clustering under the learned metric, with no n x n matrix.
        L, M, diagnostics = train_lmnn(
            X,
            y,
            mode="global",
            k=k,
            rank=rank,
            seed=seed,
            verbose=verbose,
            **lmnn_kwargs,
        )
        X_metric = transform_features(X, L)
        Z = _linkage(X_metric, method=linkage_method, metric="euclidean")
        transform, mahalanobis = L, M
    else:
        # torch local/enhanced/lowrank → per-cell metric realized as a precomputed
        # matrix. The torch trainers cast labels via LongTensor, so classes must
        # be contiguous integers.
        _, y = np.unique(y, return_inverse=True)
        trained_model, distance_fn, diagnostics = train_lmnn(
            X,
            y,
            mode=mode,
            k=k,
            rank=rank,
            seed=seed,
            verbose=verbose,
            **lmnn_kwargs,
        )
        D = distance_fn(X)
        Z = _linkage(squareform(D, checks=False), method=linkage_method)
        if return_distance_matrix:
            distance_matrix = D
        if return_model:
            model = trained_model

    cluster_labels = fcluster(Z, n_clusters, criterion="maxclust")

    return LmnnClusteringResult(
        labels=cluster_labels,
        linkage=Z,
        n_clusters=n_clusters,
        mode=mode,
        diagnostics=diagnostics,
        transform=transform,
        mahalanobis=mahalanobis,
        distance_matrix=distance_matrix,
        model=model,
    )
