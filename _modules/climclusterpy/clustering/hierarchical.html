Source code for climclusterpy.clustering.hierarchical

"""Agglomerative clustering of grid cells under plain Euclidean distance.

The unsupervised mode: standardize the feature matrix, build a linkage tree, cut it
at *K*. No training, no supervision, no optional dependencies — and for the
high-*K* end of the range it is competitive with the learned metric, which is why it
is the default.

Accepts precomputed shortcuts in priority order — a linkage matrix, a distance
matrix, or a feature matrix — so a linkage cached from an earlier run (or an
LMNN-learned distance) can be cut at many *K* values without recomputing anything.

Input data is expected to be already aligned to the original grid by
:class:`~climclusterpy.data.registry.DataRegistry`; no regridding happens here.
"""

import logging
import os

import numpy as np
import pandas as pd
import xarray as xr
from scipy.cluster.hierarchy import fcluster, linkage
from sklearn.preprocessing import StandardScaler

from climclusterpy import _settings
from climclusterpy._cache import fingerprint, read_fingerprint, write_fingerprint
from climclusterpy._mpi import is_mpi_root
from climclusterpy.data.registry import DataRegistry

logger = logging.getLogger(__name__)


def calculate_temporal_means(
    data: xr.DataArray, time_dim: str = "time", aggregation: str = "seasonal"
) -> xr.DataArray:
    """
    Calculate temporal means from time series data.

    Args:
            data: xarray DataArray with standardized 'latitude' and 'longitude' coordinates
            time_dim: name of time dimension
            aggregation: one of 'monthly', 'seasonal', or 'yearly'
    """
    data = _ensure_datetime_coord(data, time_dim=time_dim)
    data = data.assign_coords(year=data[time_dim].dt.year)

    if aggregation == "monthly":
        data = data.assign_coords(month=data[time_dim].dt.month)
        data = data.groupby(["year", "month"]).mean(dim=time_dim, skipna=True)
        data = data.transpose("latitude", "longitude", "year", "month")

    elif aggregation == "seasonal":
        data = data.assign_coords(season=((data[time_dim].dt.month - 1) // 3 + 1))
        data = data.groupby(["year", "season"]).mean(dim=time_dim, skipna=True)
        data = data.transpose("latitude", "longitude", "year", "season")

    elif aggregation == "yearly":
        data = data.groupby("year").mean(dim=time_dim, skipna=True)
        data = data.transpose("latitude", "longitude", "year")

    else:
        raise ValueError(f"Unknown aggregation type: {aggregation}")

    return data


def _ensure_datetime_coord(data: xr.DataArray, time_dim: str = "time") -> xr.DataArray:
    """Ensure time coordinate is datetime-like; coerce ints to monthly dates.

    A time axis that is already dates (numpy or cftime) is kept. An integer index (as in
    synthetic datasets) is read as consecutive months starting January 2000.
    """
    # If .dt accessor works, it's already datetime-like
    try:
        _ = data[time_dim].dt  # xarray raises AttributeError when the axis is not dates
        return data
    except (AttributeError, TypeError):
        # Build a simple monthly DateRange starting from 2000-01-01
        periods = int(data.sizes.get(time_dim, data[time_dim].size))
        new_time = pd.date_range("2000-01-01", periods=periods, freq="MS")
        return data.assign_coords({time_dim: new_time})

def stack_feature_arrays(*datasets: xr.DataArray) -> np.ndarray:
    """Flatten gridded DataArrays into one ``(n_cells, n_features)`` matrix.

    Each input is expected in ``(latitude, longitude, [time dims])`` order; the
    trailing dimensions of each array become feature columns, and the arrays are
    concatenated along the feature axis.

    Not to be confused with
    :func:`climclusterpy.features.combine_features`, which joins two *tables*
    (climate and soil DataFrames) on cell identity.

    Parameters
    ----------
    *datasets : xarray.DataArray
        Gridded arrays sharing the same latitude/longitude grid.

    Returns
    -------
    numpy.ndarray
        Feature matrix of shape ``(n_lat * n_lon, sum_of_trailing_sizes)``.
    """
    feature_list = []
    for ds in datasets:
        arr = ds.values
        lat_size, lon_size = arr.shape[0], arr.shape[1]
        n_spatial = lat_size * lon_size
        n_time = int(np.prod(arr.shape[2:]))
        arr_reshaped = arr.reshape(n_spatial, n_time)
        feature_list.append(arr_reshaped)

    return np.hstack(feature_list)

def compute_dendrogram(
    X: np.ndarray, method: str = "complete", metric: str = "euclidean"
) -> np.ndarray:
    """Compute hierarchical linkage (dendrogram) for feature matrix X.

    Parameters
    ----------
    - X: 2D array (n_samples, n_features)
    - method: linkage method (default 'complete')
    - metric: distance metric (default 'euclidean')

    Returns
    -------
    - Z: linkage matrix of shape (n_samples-1, 4)
    """
    if X.ndim != 2:
        raise ValueError("X must be 2D (n_samples, n_features)")
    return linkage(X, method=method, metric=metric)




def _require_one_row_per_cell(n_rows: int, registry: DataRegistry, what: str) -> None:
    """Refuse an input whose cell count differs from the registry's valid cells.

    The labels are written next to the registry's cell ids in order, so any other count
    pairs labels with the wrong cells (or fails later with an unrelated pandas error).
    """
    n_cells = len(registry.get_valid_original_ids())
    if n_rows != n_cells:
        raise ValueError(
            f"{what} describes {n_rows} cells, but the grid has {n_cells} valid cells. It must "
            "have one entry per valid cell, in the registry's cell order."
        )


def _refuse_missing_values(X: np.ndarray, column_names: list[str]) -> None:
    """Raise when any cell has a missing feature, naming how many cells and which features.

    Dropping those rows instead would shift every later label onto the wrong cell.
    """
    bad = np.isnan(X)
    if not bad.any():
        return
    n_bad_cells = int(bad.any(axis=1).sum())
    bad_cols = np.flatnonzero(bad.any(axis=0))
    names = list(dict.fromkeys(column_names[i] for i in bad_cols))
    raise ValueError(
        f"{n_bad_cells} of {X.shape[0]} cells have missing (NaN) feature values, in "
        f"{names[:10]}{' ...' if len(names) > 10 else ''}. Every valid cell needs every "
        "feature; fill or mask those cells before clustering."
    )


def _labels_table(registry: DataRegistry, labels, max_lat_diff: float | None) -> pd.DataFrame:
    """The registry's valid cells with their cluster label, optionally split by latitude band."""
    lat_non_nan, lon_non_nan = registry.get_valid_lat_lon()
    orig_ids = registry.get_valid_original_ids()
    df = pd.DataFrame(
        {
            "cell_id": orig_ids,
            "latitude": lat_non_nan,
            "longitude": lon_non_nan,
            "cluster_label": labels,
        }
    )
    if max_lat_diff is not None and max_lat_diff > 0:
        df["subcluster"] = (df["latitude"] // max_lat_diff).astype(int)
        df["cluster_label"] = df["cluster_label"].astype(str) + "_" + df["subcluster"].astype(str)
        df = df.drop(columns=["subcluster"])
    return df


def _write_labels_on_root(df: pd.DataFrame, out_csv: str) -> None:
    """Write the label table from MPI rank 0 only, so parallel ranks do not collide."""
    if is_mpi_root():
        df.to_csv(out_csv, index=False)
        logger.info(f"Wrote cluster labels to {out_csv}")
    else:
        logger.info(f"MPI non-root: Skipping write (handled by rank 0): {out_csv}")

def process_cluster_labels(
    climate_data,
    registry: DataRegistry,
    aggregation: str,
    n_clusters: int,
    out_dir: str,
    max_lat_diff: float | None = None,
    precomputed_features: np.ndarray | None = None,
    precomputed_distance_matrix: np.ndarray | None = None,
    precomputed_linkage: np.ndarray | None = None,
):
    """Compute or load dendrogram, cut into n clusters, and write CSV of labels with coords.

    Args:
            climate_data: dict of climate datasets (used if precomputed_features/distance_matrix is None)
            registry: DataRegistry with valid cell information
            aggregation: temporal aggregation type ('monthly'/'seasonal'/'yearly'); only used
                         on the from-raw-climate path (ignored when a precomputed input is given)
            n_clusters: number of clusters to create
            out_dir: output directory
            max_lat_diff: optional latitude-based subclustering
            precomputed_features: Optional (n_cells, n_features) array.
                                  If provided, skips time series aggregation and uses these features.
            precomputed_distance_matrix: Optional (n_cells, n_cells) distance matrix.
                                         If provided, uses distance matrix directly (takes priority over features).
            precomputed_linkage: Optional linkage matrix Z of shape (n_samples-1, 4).
                                If provided, skips DM + squareform + linkage — just fcluster
                                (takes priority over both of the above).

    Every input must have exactly one row per valid cell of ``registry``, in its order, and
    no missing value: a ``ValueError`` says otherwise, because dropping or misaligning rows
    would put labels on the wrong cells.

    Returns the output CSV path. Only MPI rank 0 writes it.
    """
    os.makedirs(out_dir, exist_ok=True)

    # Fast path: pre-computed linkage Z — no dendrogram computation needed at all
    if precomputed_linkage is not None:
        logger.info(
            f"[Clustering] Using pre-computed linkage Z (shape: {precomputed_linkage.shape})"
        )
        Z = np.asarray(precomputed_linkage)
        if Z.ndim != 2 or Z.shape[1] != 4:
            raise ValueError(
                f"precomputed_linkage must be a scipy linkage matrix of shape (n_cells - 1, 4); "
                f"got shape {Z.shape}."
            )
        # A tree over n cells has n - 1 merges.
        _require_one_row_per_cell(Z.shape[0] + 1, registry, "precomputed_linkage")
        logger.info(f"[Clustering] Cutting dendrogram to {n_clusters} clusters…")
        labels = fcluster(Z, t=n_clusters, criterion="maxclust")
        df = _labels_table(registry, labels, max_lat_diff)
        csv_path = os.path.join(
            out_dir, f"cluster_labels_with_coords_nr_of_clusters_{n_clusters}.csv"
        )
        _write_labels_on_root(df, csv_path)
        logger.info(f"[Clustering] Unique cluster count: {df['cluster_label'].nunique()}")
        return csv_path

    # Different dendrogram filename for different input types
    if precomputed_distance_matrix is not None:
        dendrogram_path = os.path.join(out_dir, "dendrogram_distance_matrix.npy")
    elif precomputed_features is not None:
        dendrogram_path = os.path.join(out_dir, "dendrogram_precomputed.npy")
    else:
        dendrogram_path = os.path.join(out_dir, "dendrogram.npy")

    if precomputed_distance_matrix is not None:
        # Use precomputed distance matrix directly for clustering
        logger.info(
            f"[Clustering] Using precomputed distance matrix (shape: {precomputed_distance_matrix.shape})"
        )
        # Verify it's square and symmetric
        if precomputed_distance_matrix.shape[0] != precomputed_distance_matrix.shape[1]:
            raise ValueError(
                f"Distance matrix must be square, got shape {precomputed_distance_matrix.shape}"
            )
        _require_one_row_per_cell(
            precomputed_distance_matrix.shape[0], registry, "precomputed_distance_matrix"
        )

        # Convert to condensed form for scipy linkage (required for precomputed distances)
        from scipy.spatial.distance import squareform

        logger.info("[Clustering] Converting distance matrix to condensed form for linkage...")
        distance_condensed = squareform(precomputed_distance_matrix, checks=False)
        logger.info(f"[Clustering] Condensed distance vector length: {len(distance_condensed)}")
        linkage_input, linkage_metric = distance_condensed, "precomputed"
    elif precomputed_features is not None:
        # Use precomputed features directly
        logger.info(
            f"[Clustering] Using precomputed features (shape: {precomputed_features.shape})"
        )
        X = np.asarray(precomputed_features, dtype=float)
        if X.ndim != 2:
            raise ValueError(
                f"precomputed_features must be 2-D (n_cells, n_features); got shape {X.shape}."
            )
        _require_one_row_per_cell(X.shape[0], registry, "precomputed_features")
        _refuse_missing_values(X, [f"column {i}" for i in range(X.shape[1])])
        logger.info(f"[Clustering] Scaling features {X.shape}…")
        scaler = StandardScaler()
        Xn = scaler.fit_transform(X)
        linkage_input, linkage_metric = Xn, "euclidean"
    else:
        # Original time-series aggregation logic
        logger.info(f"[Clustering] Aggregating variables ({aggregation})…")

        # MAX_TIME_STEPS used to shorten every series here; say so if it is still set.
        _settings.warn_retired_time_limit()

        precip = climate_data["precipitation"]["precipitation"]
        logger.info(f"  - precipitation dims: {dict(precip.sizes)}")
        precipitation_agg = calculate_temporal_means(precip, aggregation=aggregation)
        logger.info(f"    -> precipitation aggregated dims: {dict(precipitation_agg.sizes)}")

        temp = climate_data["temperature"]["temperature"]
        logger.info(f"  - temperature dims: {dict(temp.sizes)}")
        temperature_agg = calculate_temporal_means(temp, aggregation=aggregation)
        logger.info(f"    -> temperature aggregated dims: {dict(temperature_agg.sizes)}")

        swr = climate_data["short_wave_radiation"]["short_wave_radiation"]
        logger.info(f"  - short_wave_radiation dims: {dict(swr.sizes)}")
        swr_agg = calculate_temporal_means(swr, aggregation=aggregation)
        logger.info(f"    -> short_wave_radiation aggregated dims: {dict(swr_agg.sizes)}")

        lwr = climate_data["long_wave_radiation"]["long_wave_radiation"]
        logger.info(f"  - long_wave_radiation dims: {dict(lwr.sizes)}")
        lwr_agg = calculate_temporal_means(lwr, aggregation=aggregation)
        logger.info(f"    -> long_wave_radiation aggregated dims: {dict(lwr_agg.sizes)}")

        # Humidity is optional: used when the input has it, left out when it does not.
        humid_parts = []
        if climate_data.get("humid") is not None:
            humid = climate_data["humid"]["humid"]
            logger.info(f"  - humid dims: {dict(humid.sizes)}")
            humid_agg = calculate_temporal_means(humid, aggregation=aggregation)
            logger.info(f"    -> humid aggregated dims: {dict(humid_agg.sizes)}")
            humid_parts.append(registry.flatten_on_valid(humid_agg))
        logger.info("[Clustering] Building feature matrix…")
        # Humidity, when present, goes last in the climate feature stack
        climate_parts = [
            registry.flatten_on_valid(precipitation_agg),
            registry.flatten_on_valid(temperature_agg),
            registry.flatten_on_valid(swr_agg),
            registry.flatten_on_valid(lwr_agg),
            *humid_parts,
        ]
        X_soil = registry.flatten_on_valid(climate_data["soil_code"]["soil_code"]).reshape(-1, 1)
        X_soildepth = registry.flatten_on_valid(climate_data["soil_depth"]["soil_depth"]).reshape(
            -1, 1
        )
        blocks = [
            ("precipitation", climate_parts[0]),
            ("temperature", climate_parts[1]),
            ("short_wave_radiation", climate_parts[2]),
            ("long_wave_radiation", climate_parts[3]),
            *(("humid", part) for part in humid_parts),
            ("soil_code", X_soil),
            ("soil_depth", X_soildepth),
        ]
        X = np.hstack([part for _, part in blocks])
        _refuse_missing_values(X, [name for name, part in blocks for _ in range(part.shape[1])])
        logger.info(f"[Clustering] Feature matrix shape: {X.shape}. Scaling…")
        scaler = StandardScaler()
        Xn = scaler.fit_transform(X)
        linkage_input, linkage_metric = Xn, "euclidean"

    # A saved dendrogram is reused only for the input it was linked from; the file name alone
    # used to decide, so a second call with other features in the same folder got the first
    # call's clusters back.
    key = fingerprint(np.asarray(linkage_input, dtype=float), linkage_metric, "complete")
    if os.path.exists(dendrogram_path) and read_fingerprint(dendrogram_path) == key:
        Z = np.load(dendrogram_path)
        logger.info(f"[Clustering] Loaded cached dendrogram: {dendrogram_path}")
    else:
        logger.info("[Clustering] Computing hierarchical linkage (O(N^2)) — can take minutes…")
        Z = linkage(linkage_input, method="complete", metric=linkage_metric)
        np.save(dendrogram_path, Z)
        write_fingerprint(dendrogram_path, key)
        logger.info(f"Saved dendrogram to {dendrogram_path}")

    logger.info(f"[Clustering] Cutting dendrogram to {n_clusters} clusters…")
    labels = fcluster(Z, t=n_clusters, criterion="maxclust")

    df = _labels_table(registry, labels, max_lat_diff)
    # Include aggregation type in filename to avoid overwriting when multiple aggregations are used
    out_csv = os.path.join(out_dir, f"cluster_labels_{aggregation}_{n_clusters}.csv")
    _write_labels_on_root(df, out_csv)
    return out_csv

def perform_hierarchical_clustering(
    datasets: dict,
    registry: DataRegistry,
    n_clusters: int,
    output_dir: str,
    aggregation: str = "monthly",
    precomputed_features: np.ndarray | None = None,
    precomputed_distance_matrix: np.ndarray | None = None,
    precomputed_linkage: np.ndarray | None = None,
) -> dict:
    """Run hierarchical clustering using time series aggregation, precomputed features, OR distance matrix.

    Args:
            datasets: dict-like climate data as returned from loader (keys like 'precipitation', 'temperature', ...)
            registry: DataRegistry instance initialized for the datasets
            n_clusters: desired number of clusters
            output_dir: directory to write cluster label CSV
            aggregation: temporal aggregation ('monthly','seasonal','yearly') used to build
                         features from raw climate on the from-scratch path; IGNORED when a
                         precomputed feature/distance/linkage input is supplied.
            precomputed_features: Optional feature matrix (n_cells, n_features).
                                  If provided, skips time series aggregation and uses these features directly.
                                  This enables LMNN-based or summary-statistic-based clustering.
            precomputed_distance_matrix: Optional pairwise distance matrix (n_cells, n_cells).
                                         If provided, skips feature extraction and uses distance matrix directly.
                                         Takes priority over precomputed_features.
            precomputed_linkage: Optional scipy linkage matrix Z of shape (n_samples-1, 4).
                                If provided, skips DM loading, squareform, and linkage entirely — just fcluster.

    Returns
    -------
            dict with keys: 'n_clusters', 'method', 'labels_csv'
    """
    os.makedirs(output_dir, exist_ok=True)
    # Reuse process_cluster_labels which writes the CSV and returns its path
    labels_csv = process_cluster_labels(
        datasets,
        registry,
        aggregation=aggregation,
        n_clusters=n_clusters,
        out_dir=output_dir,
        precomputed_features=precomputed_features,
        precomputed_distance_matrix=precomputed_distance_matrix,
        precomputed_linkage=precomputed_linkage,
    )
    method_tag = "hierarchical"
    if precomputed_linkage is not None:
        method_tag = "hierarchical_precomputed_linkage"
    elif precomputed_distance_matrix is not None:
        method_tag = "hierarchical_distance_matrix"
    elif precomputed_features is not None:
        method_tag = "hierarchical_precomputed"

    return {"n_clusters": n_clusters, "method": method_tag, "labels_csv": labels_csv}
