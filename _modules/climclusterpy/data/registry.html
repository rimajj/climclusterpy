from __future__ import annotations

"""
Central data registry for grids, mappings, and dataset alignment.

Concepts:
- OriginalGrid: derived from soil code dataset; assigns a stable original_cell_id per lat/lon.
- ArtificialGrid: user-provided grid with new_cell_id; mapping to original cells for repeated cells.
- DataRegistry: singleton-ish instance to hold global grids/mappings and provide aligned I/O helpers.

This module minimizes ad-hoc alignment in other code: consumers fetch aligned data
through the registry, ensuring consistent coordinates and IDs.
"""

import logging
from dataclasses import dataclass

import numpy as np
import pandas as pd
import xarray as xr

from .coords import LAT_NAMES, LON_NAMES

logger = logging.getLogger(__name__)

"""Registry assumes datasets are already standardized by the loader."""


def _find_coord_names(ds: xr.Dataset) -> tuple[str, str]:
    """Find coordinate names, preferring canonical 'latitude' and 'longitude'.

    After standardization, all datasets should have canonical names.
    Falls back to checking variants for backward compatibility.
    """
    # Prefer canonical names (should always work after standardization)
    if "latitude" in ds.coords and "longitude" in ds.coords:
        return "latitude", "longitude"

    # Fallback for unstandardized datasets
    lat = next((n for n in LAT_NAMES if n in ds.coords), None)
    lon = next((n for n in LON_NAMES if n in ds.coords), None)
    if lat is None or lon is None:
        raise ValueError(f"Could not find lat/lon coords. Available: {list(ds.coords)}")
    return lat, lon

class DataRegistry:
    """Central access point for grids and aligned data.

    It records which cells of the original grid are land (from the soil-code map) and gives
    each one a stable ``cell_id``; reconstruction uses it to put values back in place.

    Ways to get one:

    - :func:`~climclusterpy.create_clusters` builds it and puts it on its result
      (``clusters.registry``), and saves it as ``original_grid.nc`` in its output folder;
    - :meth:`DataRegistry.load` reads such a saved file back;
    - :meth:`DataRegistry.from_soil_code` builds it from the soil-code NetCDF of the
      original input -- the same file ``create_clusters`` was given.

    Lower level: ``DataRegistry().init_original_grid_from_soilcode(soil_code_ds)``.
    """

    #: ``format`` attribute of the file :meth:`save` writes, checked by :meth:`load`.
    FILE_FORMAT = "climclusterpy original grid"

    @classmethod
    def from_soil_code(cls, soil_code, var_name: str = "soil_code") -> DataRegistry:
        """Build the original-grid description from a soil-code NetCDF (path or Dataset).

        Cells where the soil code is missing are not part of the grid. This is exactly what
        :func:`~climclusterpy.create_clusters` does with its ``soil_code`` input, so the same
        file gives the same ``cell_id`` for every cell.

        A cell counts as missing when its value is NaN or equals the variable's ``_FillValue``
        / ``missing_value``; when neither is declared, the first of -9999, -999, 9999, -32768
        and -1000000 that occurs anywhere in the data is treated as the fill value. A soil code
        of 0 is not a soil type -- LPJmL skips such a cell -- so it is left out too (with a
        warning; before 0.3.0 it counted as land). The
        remaining cells get consecutive ``cell_id`` values 0, 1, 2, … in row-major
        (latitude, then longitude) order. If the variable has a ``time`` dimension, only its
        first time step is used.

        Parameters
        ----------
        soil_code : str, os.PathLike or xarray.Dataset
            The soil-code input: a path to the NetCDF file, or an already opened Dataset.
            A Dataset has its coordinates standardized (see
            :func:`~climclusterpy.data.coords.standardize_dataset_coordinates`); a file is
            opened the way :func:`~climclusterpy.create_clusters` opens it. In both cases
            the variable is found by exact
            name, then case-insensitively, then under the aliases ``soiltexture``,
            ``soil_tex``, ``soilcode``, then as the only data variable in the file, then by
            a ``long_name`` or ``standard_name`` attribute containing ``var_name`` with
            underscores read as spaces ("soil code" by default).
        var_name : str, optional
            Name of the soil-code variable (default ``"soil_code"``).

        Returns
        -------
        DataRegistry
            A registry holding the original grid, its cell ids and its land mask.

        Raises
        ------
        FileNotFoundError
            If ``soil_code`` is a path that does not exist.
        ValueError
            If the file cannot be read or no soil-code variable can be identified in it.
        """
        from .coords import standardize_dataset_coordinates
        from .loader import _canonicalize_variable, _open_dataset

        if isinstance(soil_code, xr.Dataset):
            opened = standardize_dataset_coordinates(soil_code)
        else:
            opened = _open_dataset(str(soil_code))
        # The same name resolution for a path and an open Dataset, so a file that uses one of
        # the older soil variable names works either way.
        ds = _canonicalize_variable(
            opened, target=var_name, synonyms=("soiltexture", "soil_tex", "soilcode")
        )
        if "time" in ds[var_name].dims:
            ds = ds.isel(time=0, drop=True)
        registry = cls()
        registry.init_original_grid_from_soilcode(ds, var_name=var_name)
        return registry

    def save(self, path: str) -> str:
        """Write the original grid (coordinates, cell ids, land mask) to a small NetCDF.

        The file holds ``original_cell_id`` (int64, -1 for cells outside the grid) and
        ``valid`` (int8, 1 for land cells) on the ``(latitude, longitude)`` grid. It is the
        ``original_grid.nc`` that :func:`~climclusterpy.create_clusters` writes, and
        :meth:`load` reads it back.

        Parameters
        ----------
        path : str or os.PathLike
            File to write. An existing file at that path is overwritten.

        Returns
        -------
        str or os.PathLike
            ``path``, unchanged.

        Raises
        ------
        RuntimeError
            If the registry holds no original grid yet.
        """
        orig = self.get_original_grid()
        ds = xr.Dataset(
            {
                "original_cell_id": (("latitude", "longitude"), orig.cell_ids.astype("int64")),
                "valid": (("latitude", "longitude"), self.get_valid_mask().astype("int8")),
            },
            coords={"latitude": orig.latitudes, "longitude": orig.longitudes},
            attrs={"format": self.FILE_FORMAT, "source": self._original_source or ""},
        )
        ds.to_netcdf(path)
        return path

    @classmethod
    def load(cls, path: str) -> DataRegistry:
        """Read back a file written by :meth:`save` (``original_grid.nc`` in a run's folder).

        Parameters
        ----------
        path : str or os.PathLike
            The NetCDF file written by :meth:`save`.

        Returns
        -------
        DataRegistry
            A registry with the saved grid, cell ids and land mask.

        Raises
        ------
        FileNotFoundError
            If ``path`` does not exist.
        ValueError
            If the file was not written by :meth:`save` (its ``format`` attribute is missing
            or different, or it has no ``original_cell_id`` variable). To build a registry
            from the soil-code input instead, use :meth:`from_soil_code`.
        """
        import os

        if not os.path.exists(path):
            raise FileNotFoundError(f"Original-grid file not found: {path}")
        with xr.open_dataset(path) as raw:
            ds = raw.load()
        if ds.attrs.get("format") != cls.FILE_FORMAT or "original_cell_id" not in ds:
            raise ValueError(
                f"{path} is not an original-grid file written by DataRegistry.save(). To build "
                "one from the soil-code input instead, use DataRegistry.from_soil_code(path)."
            )
        registry = cls()
        registry._original = OriginalGrid(
            latitudes=ds["latitude"].values,
            longitudes=ds["longitude"].values,
            cell_ids=ds["original_cell_id"].values.astype("int64"),
        )
        registry._valid_mask = ds["valid"].values.astype(bool)
        registry._original_source = ds.attrs.get("source") or None
        return registry



    def __init__(self) -> None:
        self._original: OriginalGrid | None = None
        self._artificial: ArtificialGrid | None = None
        # 2D boolean mask on original grid (latitude, longitude)
        self._valid_mask = None
        # Track provenance of original grid for richer diagnostics
        self._original_source: str | None = None

    # --- original grid ---
    def init_original_grid_from_soilcode(
        self, soil_ds: xr.Dataset, var_name: str = "soil_code"
    ) -> OriginalGrid:
        ds = soil_ds
        lat_name, lon_name = _find_coord_names(ds)
        lats = ds[lat_name].values
        lons = ds[lon_name].values

        # derive a valid mask from soil dataset (non-NaN where soil is defined)
        # Allow legacy soil variable names when the canonical var is not present
        used_var = var_name
        if var_name not in ds:
            # common legacy names
            legacy = ("soiltexture", "soil_tex", "soilcode", "soild")
            found = next((v for v in ds.data_vars if v.lower() in legacy), None)
            if found is not None:
                used_var = found
            else:
                raise ValueError(
                    f"Expected variable '{var_name}' in soil dataset to derive valid mask"
                )
        da = ds[used_var]

        # Robust sentinel detection: a soil_code variable may use an integer sentinel
        # (e.g. -9999) to preserve integer dtype, or it may use NaN for missing values.
        # We detect any configured _FillValue or missing_value in encoding/attrs and
        # fall back to scanning for common sentinel values in the data. Cells equal
        # to the sentinel are treated as invalid.
        def _detect_sentinel(dataarray):
            enc = dataarray.encoding
            cand = enc.get("_FillValue", enc.get("missing_value"))
            if cand is None:
                cand = dataarray.attrs.get("_FillValue")
            if cand is None:
                cand = dataarray.attrs.get("missing_value")
            # If still unknown, scan for common sentinels in the array values
            if cand is None:
                vals = dataarray.values
                cand = next(
                    (s for s in (-9999, -999, 9999, -32768, int(-1e6)) if np.any(vals == s)),
                    None,
                )
            return cand

        sentinel = _detect_sentinel(da)

        # Build a boolean valid DataArray: True for real data, False for missing/sentinel
        valid_da = ~da.isnull()
        if sentinel is not None:
            valid_da = valid_da & (da != sentinel)

        # Soil code 0 is not a soil type: LPJmL's soil map has no entry for it and the model
        # skips such a cell ("Invalid soilcode=0, cell skipped"). A grid that kept it would
        # cluster, and write forcing for, cells the model never simulates. Before 0.3.0 a
        # code of 0 counted as land.
        zero_code = valid_da & (da == 0)
        n_zero = int(zero_code.sum())
        if n_zero:
            valid_da = valid_da & ~zero_code
            logger.warning(
                f"{n_zero} cell(s) have soil code 0, which is no soil type (LPJmL skips them); "
                "they are left out of the grid."
            )

        # reduce any extra dims beyond spatial
        reduce_dims = [d for d in valid_da.dims if d not in (lat_name, lon_name)]
        if reduce_dims:
            valid_da = valid_da.any(dim=reduce_dims)
        valid_mask = valid_da.transpose(lat_name, lon_name).values.astype(bool)

        source = soil_ds.encoding.get("source") or soil_ds.attrs.get("source")

        return self._finalize_grid(lats, lons, valid_mask, source)

    def _finalize_grid(
        self,
        lats: np.ndarray,
        lons: np.ndarray,
        valid_mask: np.ndarray,
        source: str | None,
    ) -> OriginalGrid:
        """Assign consecutive ``cell_id``s to a boolean ``(lat, lon)`` valid mask.

        Shared tail of every ``init_original_grid_from_*`` constructor: cells outside
        ``valid_mask`` get ``-1``, the rest get 0, 1, 2, … in row-major (latitude, then
        longitude) order.
        """
        nlat = lats.size
        nlon = lons.size
        ids = np.full((nlat, nlon), -1, dtype=np.int64)  # Initialize with -1 (invalid)

        valid_count = np.sum(valid_mask)
        if valid_count > 0:
            ids[valid_mask] = np.arange(valid_count, dtype=np.int64)

        logger.info(f"{valid_count} valid cells found in original grid.")

        grid = OriginalGrid(latitudes=lats, longitudes=lons, cell_ids=ids)
        self._original = grid
        self._original_source = source

        if ids.shape != valid_mask.shape:
            raise ValueError("Derived valid mask shape does not match original grid shape")
        self._valid_mask = valid_mask
        return grid

    @classmethod
    def from_mask(cls, mask, var_name: str = "mask") -> DataRegistry:
        """Build the original-grid description from an explicit land-mask NetCDF (path or Dataset).

        Unlike :meth:`from_soil_code`, this applies no model-specific convention: a cell is land
        when the mask variable is truthy and not missing (NaN), full stop. Use this when the
        input isn't LPJmL's soil-code map — any other model's own land/sea mask, a shapefile
        rasterized upstream, or a hand-built mask.

        Parameters
        ----------
        mask : str, os.PathLike or xarray.Dataset
            The mask input: a path to a NetCDF file, or an already opened Dataset. A nonzero,
            non-NaN value means land.
        var_name : str, optional
            Name of the mask variable (default ``"mask"``); found by exact name, then
            case-insensitively, then as the only data variable in the file.

        Returns
        -------
        DataRegistry
        """
        from .coords import standardize_dataset_coordinates
        from .loader import _canonicalize_variable, _open_dataset

        if isinstance(mask, xr.Dataset):
            opened = standardize_dataset_coordinates(mask)
        else:
            opened = _open_dataset(str(mask))
        ds = _canonicalize_variable(opened, target=var_name, synonyms=())
        da = ds[var_name]
        if "time" in da.dims:
            da = da.isel(time=0, drop=True)
        lat_name, lon_name = _find_coord_names(ds)
        lats = ds[lat_name].values
        lons = ds[lon_name].values

        valid_da = (~da.isnull()) & (da != 0)
        reduce_dims = [d for d in valid_da.dims if d not in (lat_name, lon_name)]
        if reduce_dims:
            valid_da = valid_da.any(dim=reduce_dims)
        valid_mask = valid_da.transpose(lat_name, lon_name).values.astype(bool)

        registry = cls()
        registry._finalize_grid(lats, lons, valid_mask, source=None)
        return registry

    @classmethod
    def from_complete_data(cls, datasets: dict[str, xr.Dataset]) -> DataRegistry:
        """Build the original-grid description from "every supplied input has data here".

        No mask file at all: a cell is land when every dataset in ``datasets`` has a real
        (non-NaN) value there (its first time step, for a variable that has a ``time``
        dimension). This is the fallback land-mask route for a model with no soil-code-like
        input and no separate mask file.

        Parameters
        ----------
        datasets : dict[str, xarray.Dataset]
            Mapping of input name to its (already opened, coordinate-standardized) Dataset.
            Must contain at least one entry.

        Returns
        -------
        DataRegistry
        """
        if not datasets:
            raise ValueError("from_complete_data needs at least one dataset to derive a mask from")

        valid_mask = None
        lats = lons = None
        lat_name = lon_name = None
        for ds in datasets.values():
            lat_name, lon_name = _find_coord_names(ds)
            lats = ds[lat_name].values
            lons = ds[lon_name].values
            data_vars = [v for v in ds.data_vars if v not in (lat_name, lon_name, "time")]
            if not data_vars:
                continue
            da = ds[data_vars[0]]
            if "time" in da.dims:
                da = da.isel(time=0, drop=True)
            present = ~da.isnull()
            reduce_dims = [d for d in present.dims if d not in (lat_name, lon_name)]
            if reduce_dims:
                present = present.all(dim=reduce_dims)
            present_mask = present.transpose(lat_name, lon_name).values.astype(bool)
            valid_mask = present_mask if valid_mask is None else (valid_mask & present_mask)

        if valid_mask is None:
            raise ValueError("None of the supplied datasets had a usable data variable")

        registry = cls()
        registry._finalize_grid(lats, lons, valid_mask, source=None)
        return registry

    @classmethod
    def from_cell_list(cls, cells, lat_col: str = "lat", lon_col: str = "lon") -> DataRegistry:
        """Build the original-grid description from a plain list of cell coordinates.

        For a model that names its cells rather than running on a dense raster (e.g. a
        LPJ-GUESS-style text gridlist): the grid's latitude/longitude axes are the *unique*
        coordinate values appearing in ``cells``, and only the listed cells are marked valid
        — everywhere else on that axis grid is left out, exactly like any other sparse mask.

        Parameters
        ----------
        cells : str, os.PathLike or pandas.DataFrame
            A CSV path (or any path ``pandas.read_csv`` can open) or an already-loaded
            DataFrame with one row per cell.
        lat_col, lon_col : str, optional
            Column names holding latitude/longitude (default ``"lat"``/``"lon"``).

        Returns
        -------
        DataRegistry
        """
        if isinstance(cells, pd.DataFrame):
            df = cells
        else:
            df = pd.read_csv(cells)
        if lat_col not in df.columns or lon_col not in df.columns:
            raise ValueError(
                f"Cell list is missing column(s): need '{lat_col}' and '{lon_col}', got "
                f"{list(df.columns)}"
            )

        lats = np.sort(df[lat_col].unique())
        lons = np.sort(df[lon_col].unique())
        lat_index = {v: i for i, v in enumerate(lats)}
        lon_index = {v: i for i, v in enumerate(lons)}

        valid_mask = np.zeros((lats.size, lons.size), dtype=bool)
        for lat_val, lon_val in zip(df[lat_col], df[lon_col]):
            valid_mask[lat_index[lat_val], lon_index[lon_val]] = True

        registry = cls()
        registry._finalize_grid(lats, lons, valid_mask, source=None)
        return registry



    def get_original_grid(self) -> OriginalGrid:
        if self._original is None:
            raise RuntimeError(
                "Original grid not initialized. Call init_original_grid_from_soilcode() first."
            )
        return self._original

    # --- artificial grid ---
    def set_artificial_grid(
        self, grid_latitudes: np.ndarray, grid_longitudes: np.ndarray, mapping_df: pd.DataFrame
    ) -> ArtificialGrid:
        # validate mapping contains only original ids present
        orig = self.get_original_grid()
        valid_ids = set(orig.cell_ids.ravel().tolist())
        if not set(mapping_df["original_cell_id"].tolist()).issubset(valid_ids):
            raise ValueError("Mapping contains unknown original_cell_id values")
        if not {"new_cell_id", "original_cell_id"}.issubset(mapping_df.columns):
            raise ValueError("mapping_df must contain columns: new_cell_id, original_cell_id")
        ag = ArtificialGrid(
            latitudes=np.asarray(grid_latitudes),
            longitudes=np.asarray(grid_longitudes),
            mapping_df=mapping_df.copy(),
        )
        self._artificial = ag
        return ag

    def get_artificial_grid(self) -> ArtificialGrid:
        if self._artificial is None:
            raise RuntimeError("Artificial grid not set. Call set_artificial_grid() first.")
        return self._artificial

    def align_to_original(self, ds: xr.Dataset) -> xr.Dataset:
        """
        Map `ds` to the original grid using nearest-neighbor mapping.
        Handles 1D or 2D lat/lon grids with ascending or descending coordinates.
        Emits diagnostics:
        - dataset sources
        - number of cells requiring mapping
        - maximum lat/lon shift with source/target coordinates.
        """
        import warnings

        import numpy as np

        # get original grid
        orig = self.get_original_grid()
        lat_name, lon_name = _find_coord_names(ds)
        tgt_lat = np.asarray(orig.latitudes)
        tgt_lon = np.asarray(orig.longitudes)

        # quick check for exact match
        lat_ok = np.array_equal(ds[lat_name].values, tgt_lat)
        lon_ok = np.array_equal(ds[lon_name].values, tgt_lon)
        if lat_ok and lon_ok:
            out = ds.copy()
            out["original_cell_id"] = ((lat_name, lon_name), orig.cell_ids)
            return out

        # --- build 2D target grid ---
        tgt_lat_grid, tgt_lon_grid = np.meshgrid(tgt_lat, tgt_lon, indexing="ij")

        # --- find nearest source indices ---
        src_lat = ds[lat_name].values
        src_lon = ds[lon_name].values

        def nearest_index_1d(src, tgt):
            """Find nearest indices, handles ascending or descending source arrays."""
            ascending = src[0] < src[-1]
            src_sorted = src if ascending else src[::-1]
            idx = np.searchsorted(src_sorted, tgt)
            idx = np.clip(idx, 1, len(src_sorted) - 1)
            left = src_sorted[idx - 1]
            right = src_sorted[idx]
            idx = np.where(np.abs(tgt - left) <= np.abs(tgt - right), idx - 1, idx)
            if not ascending:
                idx = len(src) - 1 - idx  # reverse indices back
            return idx

        lat_idx_1d = nearest_index_1d(src_lat, tgt_lat)
        lon_idx_1d = nearest_index_1d(src_lon, tgt_lon)

        # --- build full 2D index grids ---
        lat_idx_grid, lon_idx_grid = np.meshgrid(lat_idx_1d, lon_idx_1d, indexing="ij")

        # --- get chosen source coordinates ---
        chosen_lat_grid = src_lat[lat_idx_grid]
        chosen_lon_grid = src_lon[lon_idx_grid]

        # --- compute diffs ---
        lat_diffs = np.abs(chosen_lat_grid - tgt_lat_grid)
        lon_diffs = np.minimum(
            np.abs(chosen_lon_grid - tgt_lon_grid), 360 - np.abs(chosen_lon_grid - tgt_lon_grid)
        )

        # --- mapped dataset ---
        mapped_ds = ds.isel({lat_name: lat_idx_1d, lon_name: lon_idx_1d}).copy()
        mapped_ds = mapped_ds.assign_coords({lat_name: tgt_lat, lon_name: tgt_lon})
        mapped_ds["original_cell_id"] = ((lat_name, lon_name), orig.cell_ids)

        # --- diagnostics ---
        mismatched_cells = int(np.count_nonzero((lat_diffs != 0.0) | (lon_diffs != 0.0)))
        lat_max_idx = int(np.argmax(lat_diffs))
        lon_max_idx = int(np.argmax(lon_diffs))
        lat_max_diff = float(lat_diffs.ravel()[lat_max_idx])
        lon_max_diff = float(lon_diffs.ravel()[lon_max_idx])
        lat_max_tgt = float(tgt_lat_grid.ravel()[lat_max_idx])
        lat_max_src = float(chosen_lat_grid.ravel()[lat_max_idx])
        lon_max_tgt = float(tgt_lon_grid.ravel()[lon_max_idx])
        lon_max_src = float(chosen_lon_grid.ravel()[lon_max_idx])

        # dataset sources
        ds_source = ds.encoding.get("source") or ds.attrs.get("source")
        orig_source = getattr(self, "_original_source", None)

        # Nearest-cell mapping is a regrid only while every target cell falls inside some
        # source cell. Past half a cell it is extrapolation: a grid that lies somewhere else
        # entirely would otherwise be "mapped" onto its own edge cells with only a warning.
        def _step(v):
            v = np.asarray(v, dtype=float)
            return float(np.median(np.abs(np.diff(v)))) if v.size > 1 else float("inf")

        lat_limit = 0.5 * max(_step(src_lat), _step(tgt_lat)) + 1e-6
        lon_limit = 0.5 * max(_step(src_lon), _step(tgt_lon)) + 1e-6
        valid = self._valid_mask if self._valid_mask is not None else np.ones_like(lat_diffs, bool)
        too_far = ((lat_diffs > lat_limit) | (lon_diffs > lon_limit)) & valid
        if too_far.any():
            raise ValueError(
                f"{ds_source or 'A dataset'} is not on a grid that covers the original grid: "
                f"{int(too_far.sum())} land cell(s) would take their value from a cell up to "
                f"{max(lat_max_diff, lon_max_diff):.4g} deg away (latitude {lat_max_tgt:.6g} -> "
                f"{lat_max_src:.6g}, longitude {lon_max_tgt:.6g} -> {lon_max_src:.6g}), more "
                "than half a grid cell. Regrid it onto the soil_code grid first, or check "
                "that it covers the same region."
            )

        # emit single warning
        msg_lines = ["Grid mismatch detected; mapping to original grid (nearest)."]
        if ds_source:
            msg_lines.append(f"  Dataset source: {ds_source}")
        if orig_source:
            msg_lines.append(f"  Original grid source: {orig_source}")
        msg_lines.append(
            f"  Source cells requiring interpolation: {mismatched_cells} of {tgt_lat_grid.size}"
        )
        msg_lines.append(
            f"  Max |lat shift| = {lat_max_diff:.6f} at target {lat_max_tgt:.6f} → chosen src {lat_max_src:.6f}"
        )
        msg_lines.append(
            f"  Max |lon shift| = {lon_max_diff:.6f} at target {lon_max_tgt:.6f} → chosen src {lon_max_src:.6f}"
        )
        msg = "\n".join(msg_lines)
        warnings.warn(msg, UserWarning, stacklevel=2)

        return mapped_ds

    def attach_original_ids(self, ds: xr.Dataset, tol: float = 1e-5) -> xr.Dataset:
        """Attach original_cell_id to a dataset already on the original grid; no interpolation.

        This function is strict by default but accepts small numeric deviations between
        the dataset coordinates and the registered original grid. If coordinates are
        within the given absolute tolerance ``tol`` they will be coerced to the exact
        original grid coordinates to preserve downstream equality checks.

        Args:
            ds: dataset whose coordinates should match the original grid.
            tol: absolute tolerance for numeric coordinate differences (default 1e-5).

        Raises ValueError if ds coordinates do not match the original grid within
        the provided tolerance.
        """
        import warnings

        orig = self.get_original_grid()
        lat_name, lon_name = _find_coord_names(ds)
        ds_lat = ds[lat_name].values
        ds_lon = ds[lon_name].values

        lat_ok = np.array_equal(ds_lat, orig.latitudes)
        lon_ok = np.array_equal(ds_lon, orig.longitudes)
        if lat_ok and lon_ok:
            out = ds.copy()
            out["original_cell_id"] = ((lat_name, lon_name), orig.cell_ids)
            return out

        # Allow small numeric deviations
        lat_close = np.allclose(ds_lat, orig.latitudes, rtol=0.0, atol=tol)
        lon_close = np.allclose(ds_lon, orig.longitudes, rtol=0.0, atol=tol)
        if not (lat_close and lon_close):
            raise ValueError(
                "Dataset grid mismatch; expected dataset already on original grid. No interpolation performed."
            )

        # Coerce coordinates to exact original values to satisfy strict downstream checks
        out = ds.copy()
        out = out.assign_coords({lat_name: orig.latitudes, lon_name: orig.longitudes})
        out["original_cell_id"] = ((lat_name, lon_name), orig.cell_ids)
        warnings.warn(
            f"Dataset coordinates matched original grid within tol={tol}; coordinates were coerced to exact original values.",
            UserWarning,
            stacklevel=2,
        )
        return out

    def flatten_on_valid(self, da: xr.DataArray) -> np.ndarray:
        """Flatten a DataArray (lat, lon, ...) to (n_valid, prod(other_dims)) restricting to valid cells.

        The input must be aligned to the original grid (use align_to_original beforehand).

        Optimized version: extract values first, then use numpy operations (100x faster).
        """
        # Identify lat/lon dimensions
        dims = list(da.dims)
        lat_dim = None
        lon_dim = None
        for d in dims:
            d_lower = d.lower()
            if d_lower in ("lat", "latitude", "y"):
                lat_dim = d
            elif d_lower in ("lon", "longitude", "x"):
                lon_dim = d

        if lat_dim is None or lon_dim is None:
            # Fallback
            lat_dim, lon_dim = _find_coord_names(da.to_dataset(name="_tmp"))

        # Load data in chunks to avoid memory issues, convert to float32
        # Use dask if available for parallel processing
        if hasattr(da, "chunk") and hasattr(da.data, "chunks"):
            # Already chunked with dask - compute with float32
            arr = da.astype(np.float32).values
        else:
            # Load directly as float32
            arr = np.asarray(da.values, dtype=np.float32)

        # Find dimension indices for lat/lon
        lat_idx = dims.index(lat_dim)
        lon_idx = dims.index(lon_dim)

        # Build transpose order: (lat, lon, other...)
        other_indices = [i for i in range(len(dims)) if i not in (lat_idx, lon_idx)]
        transpose_order = [lat_idx, lon_idx, *other_indices]

        # Transpose using numpy (much faster than xarray)
        if transpose_order != list(range(len(dims))):
            arr = np.transpose(arr, transpose_order)

        # Reshape spatial dims to flat
        lat_size, lon_size = arr.shape[0], arr.shape[1]
        rest_size = int(np.prod(arr.shape[2:])) if arr.ndim > 2 else 1
        arr_2d = arr.reshape(lat_size * lon_size, rest_size)

        # Apply mask
        mask_flat = self.get_valid_mask().reshape(lat_size * lon_size)
        return arr_2d[mask_flat]

    def attach_new_cell_id(self, ds_on_artificial: xr.Dataset) -> xr.Dataset:
        """Attach per-point new_cell_id along a point dimension based on mapping_df.

        Note: This is a simplified representation; in practice you may build a point index
        by flattening spatial dims.
        """
        ag = self.get_artificial_grid()
        # build a deterministic mapping array for flattened order
        nlat = ag.latitudes.size
        nlon = ag.longitudes.size
        flat_indices = np.arange(nlat * nlon)
        # map flat index to new_cell_id by joining with mapping (assuming 1-1 for simplicity here)
        # In real use, mapping_df can contain repeats (multiple new cells per original); preserve as-is.
        new_ids = ag.mapping_df["new_cell_id"].values
        if new_ids.size != flat_indices.size:
            # if mismatch, fallback to broadcasting new_cell_id by nearest mapping order (placeholder)
            # For full fidelity, extend mapping to a 2D raster matching the artificial grid.
            new_ids = np.resize(new_ids, flat_indices.size)
        grid_ids = new_ids.reshape(nlat, nlon)
        out = ds_on_artificial.copy()
        out["new_cell_id"] = (("latitude", "longitude"), grid_ids)
        return out






def require_valid_cell_alignment(features_df: pd.DataFrame, registry, what: str) -> pd.DataFrame:
    """Return ``features_df`` sorted into registry valid-cell order, or fail loudly.

    Every downstream step (label CSVs, mapping CSVs, reconstruction) identifies a
    cell by its position in the registry's valid-cell sequence. A table that has
    lost or reordered cells would still *run* while silently attributing values to
    the wrong places, so the alignment is checked elementwise rather than by length.
    """
    if "cell_id" not in features_df.columns:
        raise ValueError(f"{what} is missing a 'cell_id' column; cannot verify cell alignment.")

    expected = np.asarray(registry.get_valid_original_ids())
    ordered = features_df.sort_values("cell_id").reset_index(drop=True)
    actual = ordered["cell_id"].to_numpy()

    if actual.shape != expected.shape or not np.array_equal(actual, expected):
        missing = sorted(set(expected.tolist()) - set(actual.tolist()))
        extra = sorted(set(actual.tolist()) - set(expected.tolist()))
        raise ValueError(
            f"{what} does not cover the grid's valid cells exactly: it has {len(actual)} rows "
            f"for {len(expected)} valid cells"
            + (
                f", missing cell_ids {missing[:5]}{'…' if len(missing) > 5 else ''}"
                if missing
                else ""
            )
            + (f", unexpected cell_ids {extra[:5]}{'…' if len(extra) > 5 else ''}" if extra else "")
            + ". Cell identity must match the climate input's grid."
        )
    return ordered
