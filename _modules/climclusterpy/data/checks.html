Source code for climclusterpy.data.checks

"""Checks on the climate and soil input, run before anything expensive happens.

:func:`~climclusterpy.create_clusters` runs the cheap ones on every call and refuses to
continue on a problem; :func:`~climclusterpy.check_inputs` runs all of them on their own, so a
user can find out what is wrong with an input before submitting a long job.

A *problem* is something that would make the run fail later or give a wrong answer without
saying so (mixed time axes, precipitation given as a flux, missing values inside the land
mask). A *warning* is something worth knowing that does not stop the run.
"""

from __future__ import annotations

from dataclasses import dataclass, field

import numpy as np
import xarray as xr

__all__ = [
    "CLIMATE_INPUTS",
    "InputCheck",
    "check_loaded_inputs",
    "full_missing_scan",
    "missing_value_problems",
]

#: The climate inputs and what each must be given in.
CLIMATE_INPUTS = {
    "temperature": "°C (Kelvin is recognised and converted)",
    "precipitation": "mm per time step (mm/day for daily data, mm/month for monthly data)",
    "short_wave_radiation": "W m-2",
    "long_wave_radiation": "W m-2",
    "humid": "relative humidity, %",
}

#: How many time steps the cheap checks look at to judge an input's units.
_SAMPLE_STEPS = 24

@dataclass
class InputCheck:
    """What :func:`~climclusterpy.check_inputs` found.

    Attributes
    ----------
    problems : list of str
        Each would make :func:`~climclusterpy.create_clusters` fail, or give a wrong result.
    warnings : list of str
        Worth knowing; they do not stop a run.
    facts : dict
        The grid and the inputs as the package reads them: grid size, resolution, number
        of land cells, time steps and step length per variable, units attribute, and the
        sampled value range.
    """

    problems: list[str] = field(default_factory=list)
    warnings: list[str] = field(default_factory=list)
    facts: dict = field(default_factory=dict)

    @property
    def ok(self) -> bool:
        """True when there is no problem (warnings allowed)."""
        return not self.problems

    def raise_if_problems(self) -> None:
        """Raise a single ValueError listing every problem, if there is any."""
        if self.problems:
            listed = "\n".join(f"  - {p}" for p in self.problems)
            raise ValueError(
                f"The input has {len(self.problems)} problem(s):\n{listed}\n"
                "Run climclusterpy.check_inputs(<input>) for the full report."
            )



    def __str__(self) -> str:
        lines = ["Input check: " + ("OK" if self.ok else f"{len(self.problems)} problem(s)")]
        grid = self.facts.get("grid")
        if grid:
            lines.append(
                f"  grid: {grid['n_lat']} x {grid['n_lon']} (latitude x longitude), "
                f"resolution {grid['resolution']}, {grid['n_land_cells']} land cells, "
                f"latitude {grid['lat_range'][0]:g} to {grid['lat_range'][1]:g}, "
                f"longitude {grid['lon_range'][0]:g} to {grid['lon_range'][1]:g}"
            )
        for name, f in self.facts.get("variables", {}).items():
            step = f.get("step")
            lines.append(
                f"  {name}: {f.get('n_time', '-')} time steps"
                + (f" ({step})" if step else "")
                + (f", units {f['units']!r}" if f.get("units") else ", no units attribute")
                + (
                    f", sampled range {f['min']:.4g} to {f['max']:.4g}"
                    if f.get("min") is not None
                    else ""
                )
            )
        lines += [f"  PROBLEM: {p}" for p in self.problems]
        lines += [f"  warning: {w}" for w in self.warnings]
        return "\n".join(lines)




def _step_label(days: float | None, n_time: int) -> str | None:
    if days is None:
        return None
    if days < 2:
        return "daily"
    if 27 <= days <= 32:
        return "monthly"
    if 360 <= days <= 367:
        return "yearly"
    return f"every {days:g} days"


def _data_var(ds: xr.Dataset, name: str) -> xr.DataArray:
    return ds[name] if name in ds.data_vars else ds[next(iter(ds.data_vars))]


def _land_sample(da: xr.DataArray, mask: np.ndarray) -> np.ndarray:
    """Values on land cells over the first few time steps (cheap: reads at most 24 steps)."""
    if "time" in da.dims:
        da = da.isel(time=slice(0, _SAMPLE_STEPS))
    da = da.transpose(..., "latitude", "longitude")
    vals = np.asarray(da.values, dtype="float64")
    return vals[..., mask].ravel()


def _grid_facts(registry) -> dict:
    orig = registry.get_original_grid()
    lats, lons = np.asarray(orig.latitudes, float), np.asarray(orig.longitudes, float)

    def res(v):
        return float(np.median(np.abs(np.diff(v)))) if v.size > 1 else None

    rlat, rlon = res(lats), res(lons)
    if rlat is not None and rlon is not None and np.isclose(rlat, rlon):
        resolution = f"{rlat:g} deg"
    else:
        resolution = f"{rlat} x {rlon} deg"
    return {
        "n_lat": int(lats.size),
        "n_lon": int(lons.size),
        "resolution": resolution,
        "n_land_cells": int(registry.get_valid_mask().sum()),
        "lat_range": (float(lats.min()), float(lats.max())),
        "lon_range": (float(lons.min()), float(lons.max())),
    }


def _units_problems(
    name: str, units: str, sample: np.ndarray, step: str | None
) -> tuple[list, list]:
    """(problems, warnings) about the units of one climate input, judged from a sample."""
    problems, warns = [], []
    finite = sample[np.isfinite(sample)]
    if finite.size == 0:
        return problems, warns
    mean, vmin, vmax = float(finite.mean()), float(finite.min()), float(finite.max())
    u = (units or "").replace(" ", "").lower()
    if name == "temperature":
        if not (-100.0 <= mean <= 60.0 or 150.0 <= mean <= 350.0):
            problems.append(
                f"temperature averages {mean:.4g} over the sampled land cells, which is neither "
                "°C nor K. Check the file and its scale_factor/add_offset attributes."
            )
    elif name == "precipitation":
        flux = "s-1" in u or "/s" in u or "s^-1" in u
        if flux or vmax < 1e-3:
            why = (
                f"its units attribute is {units!r}"
                if flux
                else f"its largest sampled value is {vmax:.3g}"
            )
            per = "86400 (daily data)" if step == "daily" else "the seconds in each month"
            problems.append(
                f"precipitation looks like a flux in kg m-2 s-1 ({why}), but it must be "
                f"mm per time step. Multiply by {per} first."
            )
        if vmin < -1e-6:
            warns.append(
                f"precipitation has negative values (down to {vmin:.4g}); small ones are common "
                "after regridding, large ones point at a wrong file or offset."
            )
    elif name in ("short_wave_radiation", "long_wave_radiation"):
        if vmax > 2000.0 or vmin < -1.0:
            warns.append(
                f"{name} ranges {vmin:.4g} to {vmax:.4g}; W m-2 was expected (0 to about 1400). "
                "Radiation given as an energy sum (J m-2 per day) is 86 400 times larger."
            )
    elif name == "humid":
        if vmax > 100.5 or vmin < -0.5:
            warns.append(
                f"humid ranges {vmin:.4g} to {vmax:.4g}; relative humidity in % was expected."
            )
        elif vmax <= 1.0:
            warns.append(
                f"humid ranges {vmin:.4g} to {vmax:.4g}: that looks like a fraction, but % "
                "(0-100) was expected."
            )
    return problems, warns


def check_loaded_inputs(
    datasets: dict, registry, static_names: frozenset[str] | set[str] = frozenset()
) -> InputCheck:
    """The cheap checks on already-loaded inputs: grid, time axes and units.

    Reads at most 24 time steps per variable, so it costs next to nothing next to the
    feature construction that follows. Missing values are checked separately, on the
    feature table (see :func:`missing_value_problems`), or in full by
    :func:`~climclusterpy.check_inputs`.

    ``static_names`` names the non-climate fields (``{"soil_code", "soil_depth"}`` for the
    LPJmL path, or whatever a generic input's own static fields are called) -- these are
    expected to have no time dimension, so they are skipped here rather than reported as a
    problem; use :func:`full_missing_scan` to check them for missing values instead.
    """
    static_names = set(static_names) | {"soil_code", "soil_depth"}
    report = InputCheck()
    report.facts["grid"] = _grid_facts(registry)
    if report.facts["grid"]["n_land_cells"] == 0:
        source = "soil_code" if "soil_code" in datasets else "the land mask"
        report.problems.append(
            f"{source} has no valid cell: every value is missing or equals its fill value, so "
            "there is no land to cluster."
        )
        return report
    mask = registry.get_valid_mask()

    variables: dict[str, dict] = {}
    from climclusterpy.features.climate import _time_step_days

    for name in datasets:
        if name in static_names or datasets[name] is None:
            continue
        da = _data_var(datasets[name], name)
        facts: dict = {"units": da.attrs.get("units")}
        if "time" not in da.dims:
            report.problems.append(
                f"{name} has no 'time' dimension (dimensions: {list(da.dims)}); climate inputs "
                "must be time series."
            )
            variables[name] = facts
            continue
        facts["n_time"] = int(da.sizes["time"])
        facts["step"] = _step_label(_time_step_days(da), facts["n_time"])
        sample = _land_sample(da, mask)
        finite = sample[np.isfinite(sample)]
        if finite.size:
            facts["min"], facts["max"] = float(finite.min()), float(finite.max())
            if np.isclose(facts["min"], facts["max"]):
                report.warnings.append(
                    f"{name} has the same value ({facts['min']:.4g}) everywhere in the sampled "
                    "time steps; it carries no information to cluster on."
                )
        probs, warns = _units_problems(name, facts["units"], sample, facts["step"])
        report.problems += probs
        report.warnings += warns
        variables[name] = facts
    report.facts["variables"] = variables

    lengths = {n: f["n_time"] for n, f in variables.items() if "n_time" in f}
    steps = {n: f.get("step") for n, f in variables.items() if "n_time" in f}
    if len(set(lengths.values())) > 1 or len({s for s in steps.values() if s}) > 1:
        listed = ", ".join(
            f"{n}: {lengths[n]}" + (f" {steps[n]}" if steps[n] else "") for n in lengths
        )
        report.problems.append(
            f"the climate inputs do not share one time axis ({listed} time steps). They must "
            "all cover the same period at the same step -- all daily or all monthly."
        )
    return report


#: Which input a feature column comes from, by the prefix of its name.
_FEATURE_SOURCES = {
    "tas": "temperature",
    "pr": "precipitation",
    "rsds": "short_wave_radiation",
    "rlds": "long_wave_radiation",
    "hurs": "humid",
    "spei": "the SPEI drought index (from temperature, precipitation and radiation)",
    "soil": "soil_code/soil_depth",
    "kg": "temperature/precipitation (climate-zone diagnostics)",
}


def missing_value_problems(features, feature_columns, *, max_examples: int = 5) -> list[str]:
    """One problem per input whose feature columns hold missing (NaN/inf) values on land cells.

    A missing value anywhere in a land cell's series ends up here: every summary built from
    that series is NaN. The message names the input, counts the cells and gives a few of
    their coordinates, which is what a user needs to find them in the file.
    """
    values = features[feature_columns].to_numpy(dtype=float)
    bad = ~np.isfinite(values)
    if not bad.any():
        return []
    by_source: dict[str, np.ndarray] = {}
    for k, col in enumerate(feature_columns):
        if not bad[:, k].any():
            continue
        prefix = col.split("_", 1)[0]
        source = _FEATURE_SOURCES.get(prefix, f"the input behind feature '{col}'")
        by_source[source] = by_source.get(source, np.zeros(len(features), bool)) | bad[:, k]
    problems = []
    for source, rows in by_source.items():
        idx = np.flatnonzero(rows)
        examples = ", ".join(
            f"({features['latitude'].iloc[i]:g}, {features['longitude'].iloc[i]:g})"
            for i in idx[:max_examples]
        )
        problems.append(
            f"{source} has missing values (NaN) in {idx.size} land cell(s), e.g. at "
            f"(latitude, longitude) {examples}. A land cell is one inside the grid's land "
            "mask; either fill the gaps, or leave those cells out of the mask so they are "
            "not part of the grid."
        )
    return problems


def full_missing_scan(datasets: dict, registry, *, max_examples: int = 5) -> list[str]:
    """Missing values on land cells in every input, reading each one in full.

    This is the expensive half of :func:`~climclusterpy.check_inputs`: it reads every time
    step once (lazily, chunk by chunk, when the input was opened with dask).
    """
    mask = registry.get_valid_mask()
    orig = registry.get_original_grid()
    problems = []
    for name in datasets:
        if datasets[name] is None:
            continue
        da = _data_var(datasets[name], name)
        missing = da.isnull()
        extra = [d for d in missing.dims if d not in ("latitude", "longitude")]
        if extra:
            missing = missing.any(dim=extra)
        missing = np.asarray(missing.transpose("latitude", "longitude").values, dtype=bool)
        holes = missing & mask
        n = int(holes.sum())
        if n:
            ii, jj = np.nonzero(holes)
            examples = ", ".join(
                f"({float(orig.latitudes[i]):g}, {float(orig.longitudes[j]):g})"
                for i, j in list(zip(ii, jj))[:max_examples]
            )
            steps = " at one or more time steps" if extra else ""
            mask_note = (
                "A land cell is one with a valid soil_code; either fill the gaps, or mark "
                "those cells missing in soil_code so they are not part of the grid."
                if "soil_code" in datasets
                else "Either fill the gaps, or leave those cells out of the land mask."
            )
            problems.append(
                f"{name} has missing values (NaN){steps} in {n} land cell(s), e.g. at "
                f"(latitude, longitude) {examples}. {mask_note}"
            )
    return problems
