"""
High-level data loading for climate and soil datasets.

Provides one entry point: load_climate_and_soil_data(paths) that
opens files, validates grid compatibility, regrids soil depth if needed,
initializes the DataRegistry based on the soil_code, aligns all data to the
original grid, and returns the prepared dict along with the registry.
"""

from __future__ import annotations

import logging
import os
from dataclasses import dataclass

import numpy as np
import xarray as xr

from .. import _settings
from .coords import (
    check_grid_compatibility,
    standardize_dataset_coordinates,
)
from .registry import DataRegistry

logger = logging.getLogger(__name__)

# Package-level sentinels for deterministic writes
INT_SENTINEL = -9999
# Choose a FLOAT_SENTINEL that is exactly representable in float32 and float64.
# Use a (negative) power-of-two so the value is exactly representable in binary
# floating point formats. 2**100 (~1.27e30) is within float32/float64 ranges
# and will be preserved exactly when stored as float32 or float64.
FLOAT_SENTINEL = float(-(2**100))


def _canonicalize_variable(ds: xr.Dataset, target: str, synonyms: tuple[str, ...]) -> xr.Dataset:
    """Return a dataset containing a single data variable named exactly `target`.
    Resolution order:
      1) Exact match (case-sensitive), else case-insensitive
      2) First matching synonym (case-insensitive)
      3) If single data var present, rename it to target
      4) Attribute-based heuristic (long_name/standard_name contains target words)
      5) Else raise with clear diagnostics.
    """
    if target in ds.data_vars:
        return ds[[target]]
    # Case-insensitive exact
    for v in ds.data_vars:
        if v.lower() == target.lower():
            return ds[[v]].rename({v: target})
    # Synonyms
    syn_lower = tuple(s.lower() for s in synonyms)
    for v in ds.data_vars:
        if v.lower() in syn_lower:
            return ds[[v]].rename({v: target})
    # Single-var fallback
    dvars = list(ds.data_vars)
    if len(dvars) == 1:
        return ds[[dvars[0]]].rename({dvars[0]: target})
    # Attribute-based
    t_words = target.replace("_", " ").lower()
    for v in dvars:
        attrs = ds[v].attrs or {}
        ln = str(attrs.get("long_name", "")).lower()
        sn = str(attrs.get("standard_name", "")).lower()
        if t_words in ln or t_words in sn:
            return ds[[v]].rename({v: target})
    raise ValueError(f"Unable to canonicalize variable for target '{target}'. Available: {dvars}")

def _open_dataset(path: str) -> xr.Dataset:
    """Open one input NetCDF, naming the file in any error it raises."""
    try:
        return _open_dataset_unchecked(path)
    except FileNotFoundError:
        raise
    except OSError as e:
        raise OSError(f"Could not read input file {path}: {e}") from e
    except (ValueError, KeyError) as e:
        raise ValueError(f"Could not read input file {path}: {e}") from e


def _to_float32_keeping_source(ds: xr.Dataset, path: str) -> None:
    """Cast float64 variables to float32 in place, keeping the record of where each came from.

    ``astype`` drops a variable's encoding, and with it the source file, the stored block layout
    and the on-disk name -- what lets the distributed extractor read the few cells it needs
    straight from the file instead of decompressing the whole grid
    (:mod:`climclusterpy.io.cellread`). The on-disk name is kept under its own key because the
    loader renames variables afterwards.
    """
    for var in list(ds.data_vars):
        if ds[var].dtype == np.float64:
            # Only the provenance survives the cast; the rest of the encoding (dtype, fill,
            # compression) described the float64 variable and would be wrong if written back.
            ds[var] = ds[var].astype(np.float32)
            ds[var].encoding = {"source": path}
        ds[var].encoding.setdefault("source", path)
        ds[var].encoding["ccp_source_variable"] = var


def _open_dataset_unchecked(path: str) -> xr.Dataset:
    logger.info(f"[Loader] Opening dataset: {path}")
    # Opened lazily in dask chunks along time, so a global daily forcing is never loaded
    # whole. The chunk length sets memory use; it can move a time average in its last
    # floating-point digit (the order of summation), never more. A file without a time axis
    # (soil) simply gets one chunk.
    chunk_time = _settings.chunk_time()
    ds = xr.open_dataset(path, chunks={"time": chunk_time})
    logger.info(f"[Loader] Using dask chunks on time: {chunk_time} (enables parallel processing)")

    # Convert all float64 data variables to float32 to halve memory usage
    _to_float32_keeping_source(ds, path)

    # Some soil depth files include crs variable not needed for processing
    if "crs" in ds.variables:
        ds = ds.drop_vars("crs")
    # Standardize coordinates once here
    return standardize_dataset_coordinates(ds)

def load_climate_and_soil_data(
    paths: DataPaths,
) -> tuple[dict[str, xr.Dataset], DataRegistry]:
    """
    Load all required datasets, ensure grids are compatible, and align via DataRegistry.

    Returns a tuple: (climate_data_dict, registry)
    where climate_data_dict has keys: precipitation, temperature, short_wave_radiation,
    long_wave_radiation, humid, soil_code, soil_depth.
    """
    # Open datasets
    precipitation_data = _canonicalize_variable(
        _open_dataset(paths.precipitation),
        target="precipitation",
        synonyms=("tpr", "pr", "precip", "precipitation_flux", "tp"),
    )
    temperature_data = _canonicalize_variable(
        _open_dataset(paths.temperature),
        target="temperature",
        synonyms=("TMean", "tmean", "tas", "t2m", "temp"),
    )
    short_wave_radiation_data = _canonicalize_variable(
        _open_dataset(paths.short_wave_radiation),
        target="short_wave_radiation",
        synonyms=("SWR", "rsds", "SWdown", "swdown"),
    )
    long_wave_radiation_data = _canonicalize_variable(
        _open_dataset(paths.long_wave_radiation),
        target="long_wave_radiation",
        synonyms=("LWR", "rlds", "LWdown", "lwdown"),
    )

    # Humid is optional - only load if path is provided
    if paths.humid:
        humid_data = _canonicalize_variable(
            _open_dataset(paths.humid),
            target="humid",
            synonyms=("hurs", "rh", "humidity", "relative_humidity"),
        )
    else:
        humid_data = None

    soil_code_data = _canonicalize_variable(
        _open_dataset(paths.soil_code),
        target="soil_code",
        synonyms=("soiltexture", "soil_tex", "soilcode"),
    )
    soildepth_data = _canonicalize_variable(
        _open_dataset(paths.soil_depth),
        target="soil_depth",
        synonyms=("soildepth", "soild", "depth", "soil_depth_m"),
    )
    # Ensure soil variables are 2D by dropping time if present
    for label, ds_var in ("soil_code", soil_code_data), ("soil_depth", soildepth_data):
        if "time" in ds_var.dims:
            logger.info(
                f"[Loader] Converting '{label}' to 2D by selecting first time step (dropping 'time' dimension)."
            )
            ds2 = ds_var.isel(time=0)
            if "time" in ds2.coords:
                ds2 = ds2.drop_vars("time")
            if label == "soil_code":
                soil_code_data = ds2
            else:
                soildepth_data = ds2
    logger.info("[Loader] Validating grid compatibility (climate vs soil_code)…")

    # Validate climate vs soil_code grids
    check_grid_compatibility(precipitation_data, soil_code_data, "precipitation", "soil_code")
    check_grid_compatibility(temperature_data, soil_code_data, "temperature", "soil_code")
    check_grid_compatibility(
        short_wave_radiation_data, soil_code_data, "short_wave_radiation", "soil_code"
    )
    check_grid_compatibility(
        long_wave_radiation_data, soil_code_data, "long_wave_radiation", "soil_code"
    )
    if humid_data is not None:
        check_grid_compatibility(humid_data, soil_code_data, "humid", "soil_code")

    # Soil depth: may differ; will be mapped in align_to_original using nearest.
    soildepth_final = soildepth_data

    # Initialize registry from soil_code
    logger.info("[Loader] Initializing DataRegistry from soil_code…")
    registry = DataRegistry()
    registry.init_original_grid_from_soilcode(soil_code_data, var_name="soil_code")
    logger.info("[Loader] Aligning datasets to original grid…")
    precipitation_data = registry.align_to_original(precipitation_data)
    temperature_data = registry.align_to_original(temperature_data)
    short_wave_radiation_data = registry.align_to_original(short_wave_radiation_data)
    long_wave_radiation_data = registry.align_to_original(long_wave_radiation_data)
    if humid_data is not None:
        humid_data = registry.align_to_original(humid_data)
    soil_code_data = registry.align_to_original(soil_code_data)
    soildepth_final = registry.align_to_original(soildepth_final)
    logger.info("[Loader] Alignment complete.")

    climate_data = {
        "precipitation": precipitation_data,
        "temperature": temperature_data,
        "short_wave_radiation": short_wave_radiation_data,
        "long_wave_radiation": long_wave_radiation_data,
        "soil_code": soil_code_data,
        "soil_depth": soildepth_final,
    }

    # Only include humid if it was provided
    if humid_data is not None:
        climate_data["humid"] = humid_data

    return climate_data, registry

def create_test_data(
    paths: DataPaths,
    n_cells: int = 100,
    out_dir: str | None = None,
    pick_last: bool = False,
    use_nan: bool = False,
) -> tuple[dict[str, xr.Dataset], DataRegistry]:
    """Create a small test dataset by selecting ``n_cells`` valid grid cells.

    Loads the full datasets via :func:`load_climate_and_soil_data`, then picks
    cells in row-major order from the soil grid's valid mask (starting top-left;
    ``pick_last=True`` selects the last ``n_cells`` valid cells instead). Latitude
    rows and longitude columns containing no selected cell are dropped (the grid
    is compacted), any remaining non-selected cells are masked to ``NaN`` so
    exactly ``n_cells`` cells stay valid, and a new :class:`DataRegistry` is
    initialised on the reduced grid.

    Parameters
    ----------
    paths : DataPaths
        Paths to the full input datasets.
    n_cells : int, optional
        Number of valid cells to keep (default 100, at least 1).
    out_dir : str, optional
        If given, each reduced dataset is saved as NetCDF inside this folder.
    pick_last : bool, optional
        Select the last ``n_cells`` valid cells instead of the first.
    use_nan : bool, optional
        Mask non-selected cells with ``NaN`` (as opposed to the fill sentinel).

    Returns
    -------
    tuple
        ``(climate_data_small, registry_small)`` — the reduced datasets dict and
        a :class:`DataRegistry` on the reduced grid.
    """
    if n_cells < 1:
        raise ValueError(f"n_cells must be at least 1; got {n_cells}.")
    full_data, _full_registry = load_climate_and_soil_data(paths)

    soil_ds = full_data["soil_code"]

    # The loader names every input's coordinates 'latitude'/'longitude', puts them all on the
    # soil grid and drops any time axis from the soil fields.
    soil_lat_name, soil_lon_name = "latitude", "longitude"
    lats = soil_ds[soil_lat_name].values
    lons = soil_ds[soil_lon_name].values

    # Choose a small rectangular tile covering at least n_cells
    nlat_total, nlon_total = lats.size, lons.size

    # Determine valid cells from the soil code (present and not 0, as the registry decides),
    # reduce any extra dims
    soil_var = "soil_code"
    valid = soil_ds[soil_var].notnull() & (soil_ds[soil_var] != 0)
    reduce_dims = [d for d in valid.dims if d not in (soil_lat_name, soil_lon_name)]
    if reduce_dims:
        valid = valid.any(dim=reduce_dims)
    valid = valid.transpose(soil_lat_name, soil_lon_name)
    valid_np = valid.values.astype(bool)

    # Build a full-grid mask with True on the first/last N valid cells (row-major)
    idx_flat = np.flatnonzero(valid_np.reshape(-1))
    if idx_flat.size == 0:
        raise ValueError(
            f"The soil_code input {paths.soil_code} has no valid cell (every cell is missing "
            "or 0), so no test cells can be picked. Check that it is the right file."
        )
    keep = min(n_cells, idx_flat.size)
    if pick_last:
        chosen = idx_flat[-keep:]
    else:
        chosen = idx_flat[:keep]
    mask_flat = np.zeros((nlat_total * nlon_total,), dtype=bool)
    mask_flat[chosen] = True
    mask = mask_flat.reshape(nlat_total, nlon_total)

    # Compact grid: choose a contiguous rectangular slice that covers all selected cells
    # This preserves the original grid spacing (no irregular gaps in coords). At least one
    # cell is chosen (n_cells >= 1 and the soil map has a valid cell), so both are non-empty.
    row_idx = np.where(mask.any(axis=1))[0]
    col_idx = np.where(mask.any(axis=0))[0]
    row_start, row_end = int(row_idx.min()), int(row_idx.max())
    col_start, col_end = int(col_idx.min()), int(col_idx.max())

    # Create boolean masks (or slices) for selection
    rows_keep = slice(row_start, row_end + 1)
    cols_keep = slice(col_start, col_end + 1)

    lats_comp = lats[rows_keep]
    lons_comp = lons[cols_keep]
    mask_comp = mask[row_start : row_end + 1, col_start : col_end + 1]

    def _apply_mask(ds: xr.Dataset) -> xr.Dataset:
        dlat, dlon = soil_lat_name, soil_lon_name
        # First subset ds to compacted rows/cols (positions align across datasets after alignment)
        # Map compaction indices to this dataset's coordinate positions via equality
        # Since grids were aligned, coordinate values should match and order preserved.
        lat_vals = ds[dlat].values
        lon_vals = ds[dlon].values
        # Determine indices where ds coords equal compacted coords
        # Use np.isin with order-preserving index lookup
        lat_idx = np.nonzero(np.isin(lat_vals, lats_comp))[0]
        lon_idx = np.nonzero(np.isin(lon_vals, lons_comp))[0]
        small = ds.isel({dlat: lat_idx, dlon: lon_idx})

        # Construct a compacted mask DataArray and align to the subset dataset
        mask_da_comp = xr.DataArray(
            mask_comp,
            dims=(soil_lat_name, soil_lon_name),
            coords={soil_lat_name: lats_comp, soil_lon_name: lons_comp},
        )
        mask_da = mask_da_comp.rename({soil_lat_name: dlat, soil_lon_name: dlon})
        mask_da = mask_da.assign_coords({dlat: small[dlat], dlon: small[dlon]})
        out_vars = {}
        for v in small.data_vars:
            da = small[v]
            # Only mask variables that include both spatial dims
            if dlat in da.dims and dlon in da.dims:
                if v == "soil_code":
                    # Use integer fill for soiltexture to preserve integer dtype.
                    # Replace masked-out cells with an integer sentinel (INT_SENTINEL) or leave as NaN
                    sentinel = INT_SENTINEL
                    if use_nan:
                        # keep NaNs for missing cells; do not cast to integer
                        da = da.where(mask_da)
                    else:
                        # Every kept cell is a valid soil code, so no NaN reaches the cast.
                        da = da.where(mask_da, other=sentinel).astype("int32")
                        da.encoding["_FillValue"] = int(sentinel)
                else:
                    da = da.where(mask_da)
            out_vars[v] = da
        # Preserve coords
        small_masked = xr.Dataset(out_vars, coords={dlat: small[dlat], dlon: small[dlon]})
        # Keep any remaining coordinates/attrs
        small_masked = small_masked.assign_coords(
            {k: small.coords[k] for k in small.coords if k not in (dlat, dlon)}
        )
        small_masked.attrs.update(ds.attrs)
        return small_masked

    reduced = {k: _apply_mask(v) for k, v in full_data.items()}

    # The reduced datasets use an integer sentinel for masked-out soil cells
    # (to preserve integer dtype for `soil_code`). When reporting the number
    # of valid cells we must therefore exclude that sentinel (and any
    # configured _FillValue / missing_value) because those are non-NaN
    # integer placeholders, not real valid soil codes.
    da = reduced["soil_code"]["soil_code"]
    # Masked-out cells hold INT_SENTINEL (or NaN with use_nan=True).
    valid_count = int((da.notnull() & (da != INT_SENTINEL)).sum())
    logger.info(f"[Loader] Valid cells in reduced soil_code: {valid_count}")

    # Initialize a new registry on the reduced soil dataset (NaNs mark invalid)
    registry_small = DataRegistry()
    registry_small.init_original_grid_from_soilcode(reduced["soil_code"], var_name="soil_code")
    # Align all reduced datasets to attach original_cell_id for the reduced grid
    for k in list(reduced.keys()):
        reduced[k] = registry_small.align_to_original(reduced[k])

    if out_dir:
        _write_reduced(reduced, out_dir, use_nan=use_nan)

    return reduced, registry_small

def create_test_data_by_latlon_range(
    paths: DataPaths,
    lat_range: tuple[float, float],
    lon_range: tuple[float, float],
    out_dir: str | None = None,
) -> tuple[dict[str, xr.Dataset], DataRegistry]:
    """A test dataset of every grid cell inside a latitude/longitude box.

    Loads the full datasets with :func:`load_climate_and_soil_data`, cuts each to the box
    (using its own coordinate names) and builds a new :class:`DataRegistry` on the cut soil
    grid.

    Parameters
    ----------
    paths : DataPaths
        Paths to the full input datasets.
    lat_range : tuple of float
        Inclusive, in either order; ascending and descending latitude axes both work.
    lon_range : tuple of float
        Inclusive. With ``lon_min > lon_max`` the box wraps across the dateline (longitudes
        ``>= lon_min`` or ``<= lon_max``).
    out_dir : str, optional
        If given, each cut dataset is saved as NetCDF inside this folder.

    Returns
    -------
    tuple
        ``(climate_data_small, registry_small)``.
    """
    full_data, _full_registry = load_climate_and_soil_data(paths)
    # The loader names every input's coordinates 'latitude'/'longitude', puts them all on the
    # soil grid and drops any time axis from the soil fields.
    s_lat, s_lon = "latitude", "longitude"

    lat_min, lat_max = lat_range
    lon_min, lon_max = lon_range
    # Normalize lat order for comparisons
    lat_min_n = min(lat_min, lat_max)
    lat_max_n = max(lat_min, lat_max)

    # Determine indices from soil grid to validate non-empty selection
    soil_ds = full_data["soil_code"]
    s_lat_vals = soil_ds[s_lat].values
    s_lon_vals = soil_ds[s_lon].values

    lat_mask_soil = (s_lat_vals >= lat_min_n) & (s_lat_vals <= lat_max_n)
    if lon_min <= lon_max:
        lon_mask_soil = (s_lon_vals >= lon_min) & (s_lon_vals <= lon_max)
    else:
        # Wrap-around selection (e.g., 350..10 for 0..360 longitudes)
        lon_mask_soil = (s_lon_vals >= lon_min) | (s_lon_vals <= lon_max)

    lat_idx_soil = np.where(lat_mask_soil)[0]
    lon_idx_soil = np.where(lon_mask_soil)[0]
    if lat_idx_soil.size == 0 or lon_idx_soil.size == 0:
        raise ValueError(
            f"No grid cells found within lat [{lat_min},{lat_max}] and lon [{lon_min},{lon_max}]"
        )

    # Every input sits on the soil grid, so the soil grid's selection applies to all of them.
    reduced = {k: v.isel({s_lat: lat_idx_soil, s_lon: lon_idx_soil}) for k, v in full_data.items()}

    # Initialize a new registry on the reduced soil dataset
    registry_small = DataRegistry()
    registry_small.init_original_grid_from_soilcode(reduced["soil_code"], var_name="soil_code")

    # Align all reduced datasets to attach original_cell_id for the reduced grid
    for k in list(reduced.keys()):
        reduced[k] = registry_small.align_to_original(reduced[k])

    if out_dir:
        _write_reduced(reduced, out_dir, use_nan=False)

    return reduced, registry_small




def _write_reduced(reduced: dict[str, xr.Dataset], out_dir: str, use_nan: bool) -> None:
    """Save each reduced dataset as ``<out_dir>/<name>_test.nc``.

    ``original_cell_id`` and ``soil_code`` are written as int32 with :data:`INT_SENTINEL` for
    missing cells (``soil_code`` keeps NaN in memory when ``use_nan`` is set); every other
    variable as float64, with :data:`FLOAT_SENTINEL` for missing cells unless ``use_nan``.
    """
    os.makedirs(out_dir, exist_ok=True)
    comp = {"zlib": True, "complevel": 4, "shuffle": True}
    for key, ds in reduced.items():
        ds_write = ds.copy()
        encoding = {}
        for v in list(ds_write.data_vars):
            da = ds_write[v]
            if v in ("original_cell_id", "soil_code"):
                if v == "original_cell_id":
                    da = da.where(da != -1)  # the registry marks cells off the grid with -1
                if not (use_nan and v == "soil_code"):
                    da = da.fillna(INT_SENTINEL).astype("int32")
                da.attrs["missing_value"] = INT_SENTINEL
                encoding[v] = {**comp, "dtype": "int32", "_FillValue": INT_SENTINEL}
            elif use_nan:
                encoding[v] = {**comp, "dtype": "float64"}
            else:
                da = da.fillna(FLOAT_SENTINEL).astype("float64")
                da.attrs["missing_value"] = FLOAT_SENTINEL
                encoding[v] = {**comp, "dtype": "float64", "_FillValue": FLOAT_SENTINEL}
            ds_write[v] = da

        # Coordinates are stored as float32, without a fill value.
        for names, word in (
            (("lat", "latitude", "y", "YLAT", "LAT"), "lat"),
            (("lon", "longitude", "x", "XLONG", "LON"), "lon"),
        ):
            coord = next((n for n in names if n in ds_write.coords), None)
            if coord is None:
                coord = next((n for n in ds_write.coords if word in n.lower()), None)
            if coord is not None:
                ds_write[coord] = ds_write[coord].astype("float32")
                # Without an explicit None, xarray writes a NaN fill value, which CF
                # conventions forbid on a coordinate.
                encoding[coord] = {"dtype": "float32", "_FillValue": None}

        ds_write.to_netcdf(os.path.join(out_dir, f"{key}_test.nc"), encoding=encoding)


def list_netcdf_files(directory: str) -> list[str]:
    """Return the ``.nc`` files directly inside ``directory``."""
    import glob

    return sorted(glob.glob(os.path.join(directory, "*.nc")))


def resolve_data_paths(climate_input) -> DataPaths:
    """Coerce ``climate_input`` to a :class:`~climclusterpy.data.loader.DataPaths`.

    Accepts a ``DataPaths``, a mapping of its field names to paths, or a directory
    containing one NetCDF per input named after the field (``temperature.nc``,
    ``precipitation.nc``, …).
    """
    from dataclasses import fields as dataclass_fields

    if isinstance(climate_input, DataPaths):
        return climate_input

    if isinstance(climate_input, dict):
        valid = [f.name for f in dataclass_fields(DataPaths)]
        required = [f.name for f in dataclass_fields(DataPaths) if f.default != ""]
        unknown = sorted(set(climate_input) - set(valid))
        missing = [k for k in required if k not in climate_input]
        if unknown or missing:
            raise ValueError(
                "climate_input mapping "
                + (f"has unknown key(s) {unknown}" if unknown else "")
                + (" and " if unknown and missing else "")
                + (f"lacks required key(s) {missing}" if missing else "")
                + f". Valid keys: {valid} (only 'humid' is optional)."
            )
        return DataPaths(**climate_input)

    path = str(climate_input)
    if not os.path.isdir(path):
        raise ValueError(
            f"climate_input {path!r} is not a directory. Pass a DataPaths, a mapping of "
            "input names to file paths, or a directory holding one NetCDF per input "
            "(temperature.nc, precipitation.nc, short_wave_radiation.nc, "
            "long_wave_radiation.nc, soil_code.nc, soil_depth.nc)."
        )

    resolved: dict[str, str] = {}
    missing: list[str] = []
    for f in dataclass_fields(DataPaths):
        candidate = os.path.join(path, f"{f.name}.nc")
        if os.path.exists(candidate):
            resolved[f.name] = candidate
        elif f.default == "":  # optional input (humid)
            continue
        else:
            missing.append(f"{f.name}.nc")
    if missing:
        present = sorted(os.path.basename(p) for p in list_netcdf_files(path))
        raise FileNotFoundError(
            f"climate_input directory {path!r} is missing {missing}. Found: "
            f"{present or 'no .nc files'}. Either rename the files to match, or pass an "
            "explicit DataPaths / mapping."
        )
    return DataPaths(**resolved)
