Source code for climclusterpy.data.generic_input

"""A model-agnostic input description: any named climate and static fields.

:class:`~climclusterpy.data.loader.DataPaths` names six fixed fields because it describes
LPJmL-FIT's own input. A different vegetation model may supply a different set of climate
drivers under its own names, a different set of static (soil, elevation, land-cover, …)
fields, and its own land mask or cell list rather than a soil-code raster.
:class:`GenericInputs` and :func:`load_generic_inputs` are the model-agnostic counterpart:
any number of named climate fields (at least one), any number of named static fields each
declared categorical or continuous, and an explicit land mask -- from a mask file, a plain
cell list, or (failing both) wherever every supplied field has data.

``DataPaths`` and :func:`~climclusterpy.data.loader.load_climate_and_soil_data` are
untouched by this module; the two paths are wired together in :mod:`climclusterpy.api`.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Literal

import xarray as xr

from .registry import DataRegistry

__all__ = ["ClimateField", "GenericInputs", "StaticField", "load_generic_inputs"]

@dataclass
class ClimateField:
    """One named climate input.

    Attributes
    ----------
    path : str
        NetCDF file path.
    variable : str, optional
        The variable name inside the file, when the file holds more than one data
        variable (or its single variable isn't named after the field). Default: resolved
        the way :class:`~climclusterpy.data.loader.DataPaths` resolves a field with no
        synonym list -- exact name, then case-insensitively, then the file's only data
        variable, then by a matching ``long_name``/``standard_name`` attribute.
    aggregation : {"mean", "sum"}, optional
        For a variable the package doesn't already know how to summarize (anything
        outside temperature/precipitation/radiation/humidity): whether its monthly
        climatology should be a mean (e.g. wind speed, vapour pressure deficit) or a sum
        (e.g. a second precipitation-like flux). Default: ``"mean"``.
    """

    path: str
    variable: str | None = None
    aggregation: Literal["mean", "sum"] | None = None

@dataclass
class StaticField:
    """One named static (non-climate) input -- soil, elevation, land cover, ….

    Attributes
    ----------
    path : str
        NetCDF file path.
    kind : {"categorical", "continuous"}
        Categorical fields (an integer class code, e.g. a soil or land-cover class) are
        one-hot encoded; continuous fields (elevation, soil depth, texture fractions) are
        kept as a numeric feature. There is no default -- declare it explicitly.
    variable : str, optional
        As :attr:`ClimateField.variable`.
    """

    path: str
    kind: Literal["categorical", "continuous"]
    variable: str | None = None

@dataclass
class GenericInputs:
    """A model-agnostic input: any named climate and static fields, any land mask.

    Attributes
    ----------
    climate : dict[str, str or ClimateField]
        At least one entry, name -> file path (or a :class:`ClimateField` for a
        non-default variable name or aggregation). Nothing is mandatory beyond "at least
        one" -- unlike :class:`~climclusterpy.data.loader.DataPaths`, there is no fixed
        set of required fields.
    static : dict[str, StaticField], optional
        Any number of named static fields, each declared categorical or continuous.
    land_mask : str, optional
        An explicit land-mask NetCDF (see :meth:`DataRegistry.from_mask`). Takes priority
        over ``grid``.
    grid : str, optional
        A plain list of cell coordinates (CSV path, or a DataFrame) -- a model's own
        gridlist rather than a raster (see :meth:`DataRegistry.from_cell_list`). Used only
        when ``land_mask`` is not given.

    When neither ``land_mask`` nor ``grid`` is given, the land mask is derived from
    "every supplied field has data here" (:meth:`DataRegistry.from_complete_data`).
    """

    climate: dict[str, str | ClimateField]
    static: dict[str, StaticField] = field(default_factory=dict)
    land_mask: str | None = None
    grid: str | None = None




def _as_climate_field(value: str | ClimateField) -> ClimateField:
    return value if isinstance(value, ClimateField) else ClimateField(path=str(value))


def _open_named_variable(path: str, name: str, variable: str | None) -> xr.Dataset:
    from .loader import _canonicalize_variable, _open_dataset

    opened = _open_dataset(path)
    if variable is not None:
        if variable not in opened.data_vars:
            raise ValueError(
                f"{path!r} has no variable {variable!r} for input {name!r}. Available: "
                f"{list(opened.data_vars)}"
            )
        return opened[[variable]].rename({variable: name})
    return _canonicalize_variable(opened, target=name, synonyms=())


def load_generic_inputs(spec: GenericInputs) -> tuple[dict[str, xr.Dataset], DataRegistry]:
    """Open every field in ``spec``, derive the land mask, and align everything to it.

    Returns ``(datasets, registry)`` in the same shape
    :func:`~climclusterpy.data.loader.load_climate_and_soil_data` returns: a dict keyed by
    each field's own name (the ``spec.climate``/``spec.static`` dict keys -- not
    canonicalized into any fixed set), and the :class:`~climclusterpy.data.registry.DataRegistry`
    they were aligned to.
    """
    if not spec.climate:
        raise ValueError("GenericInputs.climate needs at least one entry")

    opened: dict[str, xr.Dataset] = {}
    for name, value in spec.climate.items():
        cfield = _as_climate_field(value)
        opened[name] = _open_named_variable(cfield.path, name, cfield.variable)
    for name, sfield in spec.static.items():
        opened[name] = _open_named_variable(sfield.path, name, sfield.variable)

    if spec.land_mask is not None:
        registry = DataRegistry.from_mask(spec.land_mask)
    elif spec.grid is not None:
        registry = DataRegistry.from_cell_list(spec.grid)
    else:
        registry = DataRegistry.from_complete_data(opened)

    aligned = {name: registry.align_to_original(ds) for name, ds in opened.items()}
    return aligned, registry
