"""Latitude/longitude coordinate handling and grid compatibility checks.

The single place where the package decides what a coordinate is called, how
precisely it is stored, and when two grids count as the same grid. Every module
that touches coordinates imports from here, so cell identity stays consistent
from raw input through clustering to reconstruction.
"""

from __future__ import annotations

from typing import TYPE_CHECKING

import numpy as np
import xarray as xr

__all__ = [
    "COORD_DTYPE",
    "COORD_PRECISION",
    "LAT_NAMES",
    "LON_NAMES",
    "check_grid_compatibility",
    "get_lat_lon_coords",
    "regrid_to_reference",
    "standardize_coordinates",
    "standardize_dataset_coordinates",
]

#: Coordinate names accepted on input, in the order they are probed.
LAT_NAMES = ["latitude", "lat", "Latitude", "LAT"]
LON_NAMES = ["longitude", "lon", "Longitude", "LON"]

# Coordinates are rounded to this precision (and stored in this dtype) everywhere:
# float noise in the last bits would otherwise make identical cells compare unequal.
COORD_PRECISION: int = 5
COORD_DTYPE = np.float32

if TYPE_CHECKING:
    import pandas as pd  # for type hints only


def get_lat_lon_coords(
    dataset: xr.Dataset | xr.DataArray, name: str
) -> tuple[np.ndarray, np.ndarray]:
    """
    Return latitude and longitude coordinate arrays.

    After standardization, all datasets should have 'latitude' and 'longitude' coords.
    Falls back to checking common variants for backward compatibility.

    Raises ValueError if either coordinate cannot be found.
    """
    # Primary: canonical names (should always work after standardization)
    if "latitude" in dataset.coords and "longitude" in dataset.coords:
        return dataset["latitude"].values, dataset["longitude"].values

    # Fallback: check variants for unstandardized datasets
    lat_coord = None
    lon_coord = None

    for lat_name in LAT_NAMES:
        if lat_name in dataset.coords:
            lat_coord = dataset[lat_name].values
            break

    for lon_name in LON_NAMES:
        if lon_name in dataset.coords:
            lon_coord = dataset[lon_name].values
            break

    if lat_coord is None:
        available_coords = list(dataset.coords.keys())
        raise ValueError(
            f"Could not find latitude coordinate in {name}. Available coordinates: {available_coords}"
        )
    if lon_coord is None:
        available_coords = list(dataset.coords.keys())
        raise ValueError(
            f"Could not find longitude coordinate in {name}. Available coordinates: {available_coords}"
        )

    return lat_coord, lon_coord


def check_grid_compatibility(
    data1: xr.Dataset | xr.DataArray,
    data2: xr.Dataset | xr.DataArray,
    name1: str,
    name2: str,
) -> None:
    """Ensure two datasets share identical lat/lon grids; raise ValueError otherwise.

    Expects datasets to have standardized 'latitude' and 'longitude' coordinates.
    """
    lat1, lon1 = get_lat_lon_coords(data1, name1)
    lat2, lon2 = get_lat_lon_coords(data2, name2)

    for axis, a, b in (("latitude", lat1, lat2), ("longitude", lon1, lon2)):
        if not np.array_equal(a, b):
            raise ValueError(
                f"Grid mismatch: {name1} and {name2} are not on the same grid "
                f"({axis} differs). {_describe_axis_difference(axis, a, b, name1, name2)} "
                "Every climate input must be on exactly the soil_code grid: regrid the "
                "offending file first (e.g. with CDO `remapnn` or xarray `.interp`)."
            )


def _describe_axis(values: np.ndarray) -> str:
    v = np.asarray(values, dtype=float)
    if v.size == 0:
        return "0 values"
    step = f", step {float(np.median(np.diff(v))):.6g}" if v.size > 1 else ""
    return f"{v.size} values from {v[0]:.6g} to {v[-1]:.6g}{step}"


def _describe_axis_difference(axis, a, b, name1, name2) -> str:
    """One sentence saying HOW two coordinate axes differ, so the user knows what to fix."""
    text = f"{name1}: {_describe_axis(a)}; {name2}: {_describe_axis(b)}."
    a, b = np.asarray(a, dtype=float), np.asarray(b, dtype=float)
    if a.size == b.size and a.size > 0:
        if np.array_equal(a, b[::-1]):
            text += f" The {axis} axis runs in opposite directions."
        else:
            k = int(np.argmax(a != b))
            text += f" First difference at index {k}: {a[k]:.6g} vs {b[k]:.6g}."
    return text

def regrid_to_reference(
    data_to_regrid: xr.Dataset | xr.DataArray,
    reference_data: xr.Dataset | xr.DataArray,
    method: str = "nearest",
) -> xr.Dataset | xr.DataArray:
    """
    Regrid data_to_regrid to the grid of reference_data using xarray.interp.

    Expects both datasets to have standardized 'latitude' and 'longitude' coordinates.
    Falls back to detecting coordinate names for backward compatibility.
    """
    ref_lat, ref_lon = get_lat_lon_coords(reference_data, "reference")

    # Try canonical names first
    if "latitude" in data_to_regrid.coords and "longitude" in data_to_regrid.coords:
        return data_to_regrid.interp(method=method, latitude=ref_lat, longitude=ref_lon)

    # Fallback: detect coordinate names
    data_lat_name = None
    data_lon_name = None
    for lat_name in LAT_NAMES:
        if lat_name in data_to_regrid.coords:
            data_lat_name = lat_name
            break
    for lon_name in LON_NAMES:
        if lon_name in data_to_regrid.coords:
            data_lon_name = lon_name
            break

    if data_lat_name is None or data_lon_name is None:
        available_coords = list(data_to_regrid.coords.keys())
        raise ValueError(
            f"Could not find lat/lon coordinates in data_to_regrid. Available: {available_coords}"
        )

    interp_kwargs = {data_lat_name: ref_lat, data_lon_name: ref_lon}
    return data_to_regrid.interp(method=method, **interp_kwargs)

def standardize_dataset_coordinates(
    dataset: xr.Dataset,
    lat_coord_name: str | None = None,
    lon_coord_name: str | None = None,
    precision: int = COORD_PRECISION,
    dtype=COORD_DTYPE,
) -> xr.Dataset:
    """Standardize coordinates in an xarray Dataset: rename to canonical names, round, cast dtype.

    - Auto-detect lat/lon coord names if not provided.
    - Rename to canonical 'latitude' and 'longitude'.
    - Round to `precision` decimals and cast to `dtype`.
    - Shift longitudes above 180 degrees down by 360, so a 0–360 grid becomes
      -180–180. The longitude axis is **not** re-sorted afterwards, so such a grid
      keeps its original order (on a half-degree grid: 0 … 180, then -179.5 … -0.5).

    Parameters
    ----------
    dataset : xarray.Dataset
        The dataset whose coordinates are standardized. It is not modified in
        place.
    lat_coord_name : str, optional
        Name of the latitude coordinate in ``dataset``. Default ``None``: take the
        first of ``latitude``, ``lat``, ``Latitude``, ``LAT`` that is present.
    lon_coord_name : str, optional
        Name of the longitude coordinate. Default ``None``: take the first of
        ``longitude``, ``lon``, ``Longitude``, ``LON`` that is present.
    precision : int, optional
        Number of decimal places the coordinate values (in degrees) are rounded to
        (default 5).
    dtype : numpy dtype, optional
        Data type the coordinate values are cast to before rounding (default
        ``numpy.float32``).

    Returns
    -------
    xarray.Dataset
        A copy of ``dataset`` with coordinates named ``latitude`` and ``longitude``
        holding the rounded, cast and longitude-shifted values.

    Raises
    ------
    ValueError
        If a coordinate name is not given and none of the accepted names is
        present.
    KeyError
        If a name passed as ``lat_coord_name`` or ``lon_coord_name`` is not in
        ``dataset``.
    """
    # Detect coordinate names
    if lat_coord_name is None:
        lat_coord_name = next((n for n in LAT_NAMES if n in dataset.coords), None)
        if lat_coord_name is None:
            raise ValueError(
                f"Could not find a latitude coordinate. Accepted names: {LAT_NAMES}. Available "
                f"coordinates: {list(dataset.coords.keys())}. Rename it first, e.g. "
                f"ds.rename({{'<yours>': 'latitude'}})."
            )
    if lon_coord_name is None:
        lon_coord_name = next((n for n in LON_NAMES if n in dataset.coords), None)
        if lon_coord_name is None:
            raise ValueError(
                f"Could not find a longitude coordinate. Accepted names: {LON_NAMES}. Available "
                f"coordinates: {list(dataset.coords.keys())}. Rename it first, e.g. "
                f"ds.rename({{'<yours>': 'longitude'}})."
            )

    # Standardize values
    lat_vals = np.round(np.asarray(dataset[lat_coord_name].values, dtype=dtype), precision)
    lon_vals = np.round(np.asarray(dataset[lon_coord_name].values, dtype=dtype), precision)

    # Normalize longitudes from [0, 360] to [-180, 180] for CLM compatibility
    # CLM tools require longitudes in [-180, 180] range
    lon_vals = np.where(lon_vals > 180, lon_vals - 360, lon_vals)

    # Rename to canonical names if needed
    rename_dict = {}
    if lat_coord_name != "latitude":
        rename_dict[lat_coord_name] = "latitude"
    if lon_coord_name != "longitude":
        rename_dict[lon_coord_name] = "longitude"

    if rename_dict:
        dataset = dataset.rename(rename_dict)

    # Assign standardized coordinate values
    return dataset.assign_coords({"latitude": lat_vals, "longitude": lon_vals})

def standardize_coordinates(
    values: np.ndarray | pd.Series | list,
    name: str | None = None,
    precision: int = COORD_PRECISION,
    dtype=COORD_DTYPE,
):
    """Standardize a 1D sequence of coordinates by rounding and casting dtype.

    Accepts numpy arrays, pandas Series, or lists. Returns a numpy array of dtype,
    or a Series when given one. Unlike :func:`standardize_dataset_coordinates`,
    longitudes are **not** shifted from 0–360 to -180–180 here.

    Parameters
    ----------
    values : numpy.ndarray, pandas.Series or list
        Coordinate values in degrees.
    name : str, optional
        Accepted but not used: it has no effect on the result.
    precision : int, optional
        Number of decimal places the values are rounded to (default 5).
    dtype : numpy dtype, optional
        Data type the values are cast to before rounding (default ``numpy.float32``).

    Returns
    -------
    numpy.ndarray or pandas.Series
        The rounded values in ``dtype``. A Series input gives a Series with the same
        index and name; any other input gives an array.
    """
    import pandas as pd

    is_series = isinstance(values, pd.Series)

    arr = np.asarray(values, dtype=dtype)
    arr = np.round(arr, precision)
    # Preserve pandas Series if input was a Series
    if is_series:
        return pd.Series(arr, index=values.index, name=values.name)
    return arr
