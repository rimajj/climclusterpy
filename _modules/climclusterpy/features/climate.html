Source code for climclusterpy.features.climate

"""Climate features: per-cell summaries of temperature, precipitation and radiation.

Raw time series are a poor clustering input — two cells with identical means can have
completely different seasonality, and the series are far too long to cluster directly.
This module reduces each cell's climate to interpretable summary statistics (annual
means and variability, seasonality, extremes, monthly climatology) that ecologists
recognise, plus the daily→monthly reduction that bioclimatic classifications need.

Water-balance features live in :mod:`climclusterpy.features.hydro`, soil in
:mod:`climclusterpy.features.soil`, and the derived diagnostic families in
:mod:`climclusterpy.features.diagnostics`.
"""

from __future__ import annotations

import itertools
import logging
import warnings

import numpy as np
import pandas as pd
import xarray as xr

from climclusterpy.data.registry import DataRegistry

logger = logging.getLogger(__name__)

__all__ = [
    "DAYS_PER_MONTH",
    "monthly_climatology_from_daily",
    "resolve_variable_dispatch_names",
    "summarize_climate_features",
    "to_celsius",
]

#: Days per month for a non-leap year — monthly totals are derived from monthly
#: means with these weights.
DAYS_PER_MONTH = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]

#: Aliases recognised for each known logical climate variable, matched against a
#: dataset's own dict key and the names of its data variables (e.g. 'temperature' and
#: 'tas' both resolve to the logical name 'tas').
_VAR_ALIASES: dict[str, list[str]] = {
    "tas": ["tas", "temperature", "temp"],
    "pr": ["pr", "precip", "precipitation"],
    "rsds": ["rsds", "short_wave", "shortwave", "rsd"],
    "rlds": ["rlds", "long_wave", "longwave", "rld"],
    "hurs": ["hurs", "humid", "rh", "relative_humidity"],
}


def resolve_variable_dispatch_names(datasets: dict[str, xr.Dataset]) -> dict[str, str]:
    """For each dataset key, the name :func:`summarize_climate_features` dispatches it under.

    A key that matches a known climate variable (by the alias rules in ``_VAR_ALIASES``,
    checked against the key itself and its data variable names) maps to that variable's
    logical short code (``tas``/``pr``/``rsds``/``rlds``/``hurs``), so it keeps that
    variable's special unit handling no matter what the caller named it. Everything else
    maps to its own key, unchanged -- a custom climate variable from a non-LPJmL input.

    Used both by ``summarize_climate_features`` itself and by callers that need to know,
    ahead of time, which name a given dataset key will be requested under (e.g. to resolve
    a ``climate_variables=`` default for a generic input).
    """
    matched_key_for: dict[str, str] = {}  # logical name -> the dataset key that filled it
    for key, ds in datasets.items():
        dvars = [v.lower() for v in ds.data_vars]
        for logical, aliases in _VAR_ALIASES.items():
            if logical in matched_key_for:
                continue
            if logical in str(key).lower() or any(
                a in str(key).lower() or any(a in dv for dv in dvars) for a in aliases
            ):
                matched_key_for[logical] = key
    key_to_logical = {key: logical for logical, key in matched_key_for.items()}
    return {key: key_to_logical.get(key, key) for key in datasets}


# Above this mean value the series cannot be °C for any Earth-like climate, so it
# is Kelvin. Cheaper and more robust than trusting a `units` attribute, which is
# frequently missing or wrong in model output.
_KELVIN_MEAN_THRESHOLD = 200.0

def to_celsius(da: xr.DataArray) -> xr.DataArray:
    """Return the temperature series in °C, converting from Kelvin if needed.

    Parameters
    ----------
    da : xarray.DataArray
        Temperature series in either °C or K.

    Returns
    -------
    xarray.DataArray
        The same series in °C.
    """
    arr = da
    # compute() loads a lazy (dask) mean and is a no-op on an array already in memory.
    mean_val = float(arr.mean().compute())

    if mean_val > _KELVIN_MEAN_THRESHOLD:
        arr = arr - 273.15
    return arr




def _calendar_parts(time_values) -> tuple[np.ndarray, np.ndarray, np.ndarray] | None:
    """``(months, years, days)`` of a time axis, or None when it carries no dates.

    Handles numpy ``datetime64`` (what xarray decodes a standard calendar to) as well as
    ``cftime`` objects (what it decodes a ``noleap``/``360_day`` calendar to).
    """
    values = np.asarray(time_values)
    if values.size == 0:
        return None
    if np.issubdtype(values.dtype, np.datetime64):
        idx = pd.DatetimeIndex(values)
        return idx.month.to_numpy(), idx.year.to_numpy(), idx.day.to_numpy()
    if all(hasattr(values[0], a) for a in ("month", "year", "day")):
        return (
            np.array([t.month for t in values]),
            np.array([t.year for t in values]),
            np.array([t.day for t in values]),
        )
    return None


def _monthly_climatology_by_date(da: xr.DataArray, da_flat: np.ndarray) -> np.ndarray | None:
    """``(n_cells, 12)`` mean of each calendar month, from monthly data with dates; else None.

    Returns None when the time axis carries no dates, or when some calendar month has no
    time step at all (the caller then reports that the climatology is incomplete).
    """
    if "time" not in da.coords or da.sizes.get("time") != da_flat.shape[1]:
        return None
    parts = _calendar_parts(da["time"].values)
    if parts is None:
        return None
    months = parts[0]
    if not all((months == m).any() for m in range(1, 13)):
        return None
    with warnings.catch_warnings():
        warnings.simplefilter("ignore", RuntimeWarning)
        return np.stack(
            [np.nanmean(da_flat[:, months == m], axis=1) for m in range(1, 13)], axis=1
        ).astype(np.float32)


def _time_step_days(da: xr.DataArray) -> float | None:
    """Median spacing of the time axis in days, or None when it carries no dates."""
    if "time" not in da.coords or da.sizes.get("time", 0) < 2:
        return None
    values = np.asarray(da["time"].values)
    try:
        if np.issubdtype(values.dtype, np.datetime64):
            diffs = np.diff(values).astype("timedelta64[s]").astype(float) / 86400.0
        elif hasattr(values[0], "month"):
            diffs = np.array(
                [(b - a).total_seconds() / 86400.0 for a, b in itertools.pairwise(values)]
            )
        else:
            return None
    except (TypeError, AttributeError):
        return None
    return float(np.median(diffs))


def _is_daily(da: xr.DataArray, n_timesteps: int, fallback_threshold: int) -> bool:
    """True for daily data: from the dates when there are any, else from the step count."""
    step = _time_step_days(da)
    if step is not None:
        return step < 2.0
    return n_timesteps > fallback_threshold

def monthly_climatology_from_daily(
    temp_da: xr.DataArray, prec_da: xr.DataArray
) -> tuple[xr.DataArray, xr.DataArray]:
    """Reduce daily temperature and precipitation to 12-month climatologies.

    Parameters
    ----------
    temp_da : xarray.DataArray
        Daily mean temperature with a ``time`` dimension (°C).
    prec_da : xarray.DataArray
        Daily precipitation with a ``time`` dimension (mm/day).

    Returns
    -------
    temp_clim : xarray.DataArray
        Mean temperature per calendar month, dims ``(month, ...)`` with
        ``month`` in 1–12 (°C).
    prec_clim : xarray.DataArray
        Mean precipitation *total* per calendar month, dims ``(month, ...)``
        (mm/month).

    Notes
    -----
    Precipitation needs two steps, not one. Grouping daily values by calendar
    month and summing would add up every year's rain into one figure; resampling
    to monthly totals first and only then averaging across years gives the
    per-month total the classification rules are defined on.
    """
    temp_clim = temp_da.groupby("time.month").mean("time")

    prec_monthly_total = prec_da.resample(time="1ME").sum(dim="time")
    prec_clim = prec_monthly_total.groupby("time.month").mean("time")

    return temp_clim, prec_clim

def summarize_climate_features(
    datasets: dict[str, xr.Dataset],
    registry: DataRegistry,
    variables: list[str] | None = None,
    include_monthly_climatology: bool = True,
    variable_aggregations: dict[str, str] | None = None,
) -> pd.DataFrame:
    """
    Compute ecological summary statistics for climate variables.

    For each grid cell, computes interpretable summaries that capture
    annual means, seasonal patterns, and extremes.

    Args:
        datasets: Dict mapping variable names to xarray Datasets
                  Expected keys: 'tas', 'pr', 'rsds', 'rlds', 'hurs'
        registry: DataRegistry containing valid cell mask and coordinates
        variables: List of variables to summarize (default: all climate vars + SPEI)
        include_monthly_climatology: If True (default), include 12 monthly climatology
            features for temperature (tas_month_01..tas_month_12) and precipitation
            (pr_month_01..pr_month_12). These are needed for Köppen-Geiger classification.
        variable_aggregations: For a variable the package doesn't specially know (anything
            outside tas/pr/rsds/rlds/hurs/spei), whether its monthly climatology should be a
            mean or a sum (default: mean). Ignored for the known variables, which keep their
            own fixed recipe (temperature/radiation/humidity: mean; precipitation: sum).

    Returns
    -------
        DataFrame with columns:
        - cell_id, latitude, longitude
        - tas_mean_annual, tas_month_01..tas_month_12, tas_range, tas_std, tas_seasonality
        - pr_mean_annual, pr_month_01..pr_month_12, pr_cv_monthly, pr_wetdays_freq, pr_max_monthly
        - rsds_mean_annual, rsds_seasonal_amplitude, rsds_std
        - rlds_mean_annual, rlds_seasonal_amplitude, rlds_std
        - hurs_mean_annual, hurs_std, hurs_dry_min
        - spei_3_mean, spei_6_mean, spei_12_mean (drought indices at 3/6/12 month scales)
    """
    # Imported here rather than at module level: `diagnostics` imports DAYS_PER_MONTH
    # from this module (so a top-level import would be circular), and `hydro` pulls in
    # the optional `pyet` PET backend only when SPEI is actually requested.
    from climclusterpy.features.diagnostics import compute_kg_diagnostic_features
    from climclusterpy.features.hydro import summarize_spei_features

    if variables is None:
        variables = ["tas", "pr", "rsds", "rlds", "hurs", "spei"]

    logger.info(f"[FeatureSummary] Summarizing climate features for {len(variables)} variables")

    # Get valid cell coordinates
    valid_lats, valid_lons = registry.get_valid_lat_lon()
    n_cells = len(valid_lats)

    # Initialize output structure
    features = {
        "cell_id": list(range(n_cells)),
        "latitude": valid_lats,
        "longitude": valid_lons,
    }

    # Map each dataset to the name summarize_climate_features dispatches on: the known
    # logical short code (tas/pr/rsds/rlds/hurs) when it matches, else the dataset's own
    # key -- so a custom variable ("wind", "vpd", ...) is still usable in `variables`,
    # and a known one keeps its special unit handling even under a descriptive key like
    # "temperature" rather than "tas".
    dispatch_names = resolve_variable_dispatch_names(datasets)
    found: dict[str, xr.Dataset] = {}
    for key, ds in datasets.items():
        found.setdefault(dispatch_names[key], ds)

    # Process each variable
    for var_name in variables:
        # SPEI needs special handling - computed from multiple variables
        if var_name == "spei":
            # Check if we have the required variables
            if "tas" in found and "pr" in found and "rsds" in found:
                features.update(summarize_spei_features(found, registry))
            else:
                missing = [v for v in ["tas", "pr", "rsds"] if v not in found]
                logger.warning(
                    f"[FeatureSummary] WARNING: Cannot compute SPEI - missing variables: {missing}"
                )
            continue

        if var_name not in found:
            logger.warning(f"[FeatureSummary] WARNING: Variable '{var_name}' not found, skipping")
            continue

        ds = found[var_name]

        # Extract the data variable (assume single data var or same name as key)
        if var_name in ds.data_vars:
            da = ds[var_name]
        else:
            # Fallback: use first data variable
            data_vars = [v for v in ds.data_vars if v not in ["lat", "lon", "time"]]
            if not data_vars:
                logger.warning(
                    f"[FeatureSummary] WARNING: No data variables found for '{var_name}', skipping"
                )
                continue
            da = ds[data_vars[0]]

        # Flatten to valid cells with chunked parallel processing
        # This avoids loading huge arrays into memory and uses float32
        da_flat = registry.flatten_on_valid(da)

        # da_flat is already float32 from registry.flatten_on_valid
        # No need to convert again - just ensure it's a numpy array
        if hasattr(da_flat, "compute"):
            # If it's still a dask array, compute it now
            da_flat = np.asarray(da_flat.compute(), dtype=np.float32)
        elif not isinstance(da_flat, np.ndarray):
            da_flat = np.asarray(da_flat, dtype=np.float32)

        if var_name == "tas":
            # Every temperature threshold downstream (Köppen–Geiger classes, growing-season
            # diagnostics) is in °C. Kelvin is recognised by its magnitude, not by the units
            # attribute, which model output often gets wrong or leaves out.
            if float(np.nanmean(da_flat)) > _KELVIN_MEAN_THRESHOLD:
                logger.info(f"[FeatureSummary] '{var_name}' looks like Kelvin; converting to °C")
                da_flat = da_flat - np.float32(273.15)
            features.update(
                _summarize_temperature(da, da_flat, var_name, include_monthly_climatology)
            )
        elif var_name == "pr":
            features.update(
                _summarize_precipitation(da, da_flat, var_name, include_monthly_climatology)
            )
        elif var_name in ["rsds", "rlds"]:
            features.update(_summarize_radiation(da, da_flat, var_name))
        elif var_name == "hurs":
            features.update(_summarize_humidity(da, da_flat, var_name))
        elif da_flat.ndim == 1:
            # Generic, but static (no time dimension left after flattening): just mean/std.
            logger.info(f"[FeatureSummary] Using generic static summary for '{var_name}'")
            features[f"{var_name}_mean"] = np.nanmean(da_flat)
            features[f"{var_name}_std"] = np.nanstd(da_flat)
        else:
            aggregation = (variable_aggregations or {}).get(var_name, "mean")
            logger.info(
                f"[FeatureSummary] Using generic summary for '{var_name}' (aggregation={aggregation})"
            )
            features.update(
                _summarize_generic(da, da_flat, var_name, aggregation, include_monthly_climatology)
            )

    # ── KG-diagnostic scalar features ────────────────────────────────────────
    # These are computed from the monthly climatology arrays that were just built
    # by _summarize_temperature and _summarize_precipitation.  Reconstruct the
    # (n_cells, 12) arrays directly from the feature dict so we don't need to
    # change the helper signatures.
    tas_month_cols = [f"tas_month_{m:02d}" for m in range(1, 13)]
    pr_month_cols = [f"pr_month_{m:02d}" for m in range(1, 13)]

    has_tas_months = all(c in features for c in tas_month_cols)
    has_pr_months = all(c in features for c in pr_month_cols)

    if has_tas_months and has_pr_months and include_monthly_climatology:
        monthly_T = np.column_stack([features[c] for c in tas_month_cols])  # (n, 12)
        monthly_P = np.column_stack([features[c] for c in pr_month_cols])  # (n, 12)
        lats = np.asarray(valid_lats, dtype=np.float32)
        kg_feats = compute_kg_diagnostic_features(monthly_T, monthly_P, lats)
        features.update(kg_feats)
        logger.info(
            f"[FeatureSummary] Added {len(kg_feats)} KG-diagnostic features: {list(kg_feats)}"
        )
    else:
        logger.info(
            "[FeatureSummary] Skipping KG-diagnostic features "
            f"(has_tas_months={has_tas_months}, has_pr_months={has_pr_months}, "
            f"include_monthly_climatology={include_monthly_climatology})"
        )
    # ─────────────────────────────────────────────────────────────────────────

    df = pd.DataFrame(features)
    logger.info(f"[FeatureSummary] Created {len(df.columns)} climate features for {n_cells} cells")

    return df




def _summarize_generic(
    da: xr.DataArray,
    da_flat: np.ndarray,
    var_name: str,
    aggregation: str = "mean",
    include_monthly_climatology: bool = True,
) -> dict:
    """Per-cell summary for a climate variable the package doesn't specially know.

    Unlike the dedicated tas/pr/rsds/rlds/hurs summarizers, there is no built-in unit
    handling here -- the caller's own units are used as-is. Produces an annual aggregate
    (mean or sum, per ``aggregation``), its std, and -- when a clean monthly climatology can
    be built -- 12 monthly columns using the same aggregation and a seasonality (std across
    those 12 months). Standardization across cells happens later, uniformly, in
    :func:`climclusterpy.features.filtering.standardize_features`.

    Args:
        da: Original xarray DataArray (used for its time coordinate).
        da_flat: Pre-flattened array of shape (n_cells, n_timesteps).
        var_name: Variable name prefix for feature names.
        aggregation: ``"mean"`` or ``"sum"`` -- how daily values roll up into a month, and
            how a month's value represents the year.
        include_monthly_climatology: If True, include the 12 monthly columns.
    """
    agg_fn = np.nansum if aggregation == "sum" else np.nanmean
    annual_key = "sum_annual" if aggregation == "sum" else "mean_annual"

    features: dict = {
        f"{var_name}_{annual_key}": agg_fn(da_flat, axis=1),
        f"{var_name}_std": np.nanstd(da_flat, axis=1),
    }

    n_cells, n_timesteps = da_flat.shape
    monthly = None
    if include_monthly_climatology and n_timesteps >= 12:
        is_daily = _is_daily(da, n_timesteps, fallback_threshold=365)
        if is_daily and "time" in da.dims:
            parts = _calendar_parts(da["time"].values)
            if parts is not None:
                months, _years, days = parts
                is_no_leap_calendar = "noleap" in str(type(da["time"].values[0])).lower()
                keep = (
                    np.ones_like(months, dtype=bool)
                    if is_no_leap_calendar
                    else ~((months == 2) & (days == 29))
                )
                months_kept = months[keep]
                da_flat_kept = da_flat[:, keep]
                monthly = np.full((n_cells, 12), np.nan, dtype=np.float32)
                with warnings.catch_warnings():
                    warnings.simplefilter("ignore", RuntimeWarning)
                    for m in range(12):
                        month_mask = months_kept == (m + 1)
                        if np.any(month_mask):
                            monthly[:, m] = agg_fn(da_flat_kept[:, month_mask], axis=1)
        elif not is_daily:
            # Already monthly: each step is one month's aggregate, so the per-month mean
            # across years is wanted regardless of `aggregation` -- same recipe the
            # temperature/precipitation summarizers use for dated monthly data.
            by_date = _monthly_climatology_by_date(da, da_flat)
            if by_date is not None:
                monthly = by_date
            else:
                n_years = n_timesteps // 12
                if n_years > 0 and n_timesteps % 12 == 0:
                    reshaped = da_flat[:, : n_years * 12].reshape(n_cells, n_years, 12)
                    monthly = np.nanmean(reshaped, axis=1)
                else:
                    warnings.warn(
                        f"'{var_name}': {n_timesteps} monthly time steps are not whole years "
                        f"({n_timesteps % 12} months left over), so its 12 monthly-climatology "
                        "features are left out. Trim the input to whole years to keep them.",
                        UserWarning,
                        stacklevel=2,
                    )

    if monthly is not None:
        features[f"{var_name}_seasonality"] = np.nanstd(monthly, axis=1)
        for m in range(12):
            features[f"{var_name}_month_{m + 1:02d}"] = monthly[:, m]

    return features


def _summarize_temperature(
    da: xr.DataArray,
    da_flat: np.ndarray,
    var_name: str,
    include_monthly_climatology: bool = True,
) -> dict:
    """
    Temperature summaries:
    - Mean annual temperature
    - Annual range (max - min)
    - Annual standard deviation
    - Seasonality (monthly std)
    - Monthly climatology (12 monthly means, optional, for Köppen-Geiger).

    All computations use the pre-flattened valid-cell array for efficiency.

    Args:
        da: Original xarray DataArray (used for time coordinate when resampling daily to monthly)
        da_flat: Pre-flattened array of shape (n_cells, n_timesteps)
        var_name: Variable name prefix for feature names (e.g., 'tas')
        include_monthly_climatology: If True, include 12 monthly mean features
            (tas_month_01 through tas_month_12) needed for Köppen-Geiger classification
    """
    features = {}

    # All computations on da_flat (n_cells × n_timesteps) - much faster than xarray groupby

    # Mean annual
    features[f"{var_name}_mean_annual"] = np.nanmean(da_flat, axis=1)

    # Annual std
    features[f"{var_name}_std"] = np.nanstd(da_flat, axis=1)

    # Annual range
    features[f"{var_name}_range"] = np.nanmax(da_flat, axis=1) - np.nanmin(da_flat, axis=1)

    # Seasonality: compute monthly climatology efficiently from flattened data
    monthly_means = None
    n_cells = da_flat.shape[0]
    n_timesteps = da_flat.shape[1]

    if n_timesteps >= 12:
        # Check if data is daily (more than 365 timesteps per year suggests daily data)
        # Daily data: ~365 * n_years timesteps; Monthly data: 12 * n_years timesteps
        # From the dates when there are any: a count alone mistakes 31+ years of monthly
        # data for daily data.
        is_daily = _is_daily(da, n_timesteps, fallback_threshold=365)

        if is_daily:
            # Daily data: need to resample to monthly using the time coordinate
            logger.info(
                f"[{var_name}] Detected daily data ({n_timesteps} timesteps), resampling to monthly..."
            )

            # Use xarray's time coordinate to group by month
            # da should have a 'time' dimension
            time_coord = da["time"].values if "time" in da.dims else None

            if time_coord is not None:
                parts = _calendar_parts(time_coord)
                if parts is None:
                    raise ValueError(
                        f"'{var_name}' has {n_timesteps} time steps, which looks like daily data, "
                        f"but its time coordinate holds {np.asarray(time_coord).dtype} values, not "
                        "dates, so days cannot be assigned to months. Give the time axis CF "
                        "'units' and 'calendar' attributes (e.g. 'days since 1901-01-01', "
                        "'noleap') so it decodes to dates, or provide monthly data."
                    )
                months, years, days = parts

                if months is not None:
                    # Check if this is a no-leap calendar (365 days per year)
                    # If so, we don't need to filter leap days
                    is_no_leap_calendar = (
                        "noleap" in str(type(time_coord[0])).lower()
                        or "365" in str(type(time_coord[0])).lower()
                    )

                    if not is_no_leap_calendar:
                        # Exclude Feb 29 (leap days) for standard calendars
                        non_leap_mask = ~((months == 2) & (days == 29))
                        da_flat_filtered = da_flat[:, non_leap_mask]
                        months_filtered = months[non_leap_mask]
                        years_filtered = years[non_leap_mask]
                        logger.info(f"[{var_name}] Filtered {np.sum(~non_leap_mask)} leap days")
                    else:
                        # No leap calendar - use all data
                        da_flat_filtered = da_flat
                        months_filtered = months
                        years_filtered = years
                        logger.info(
                            f"[{var_name}] No-leap calendar detected, using all {n_timesteps} days"
                        )

                    # Count unique years
                    unique_years = np.unique(years_filtered)
                    n_years = len(unique_years)

                    if n_years > 0:
                        # Compute monthly climatology by averaging all days of each month across all years
                        monthly_means = np.zeros((n_cells, 12), dtype=np.float32)
                        monthly_counts = np.zeros(12, dtype=np.int32)

                        for month_idx in range(12):
                            month_mask = months_filtered == (month_idx + 1)  # months are 1-indexed
                            if np.any(month_mask):
                                month_data = da_flat_filtered[:, month_mask]
                                monthly_means[:, month_idx] = np.nanmean(month_data, axis=1)
                                monthly_counts[month_idx] = np.sum(month_mask)

                        logger.info(
                            f"[{var_name}] Computed monthly climatology from {n_years} years ({n_timesteps} days)"
                        )
                        logger.info(f"[{var_name}] Days per month: {list(monthly_counts)}")
                    else:
                        logger.warning(
                            f"[{var_name}] WARNING: Not enough data for monthly climatology"
                        )
                        monthly_means = None
            else:
                logger.warning(
                    f"[{var_name}] WARNING: No time coordinate found, cannot compute monthly climatology"
                )
                monthly_means = None
        else:
            # Monthly data: original logic
            n_months_total = n_timesteps
            n_years = n_months_total // 12

            by_month = _monthly_climatology_by_date(da, da_flat)
            if by_month is not None:
                # The dates say which month each step is, so the series may start in any
                # month and need not cover whole years.
                monthly_means = by_month
            elif n_years > 0 and n_months_total % 12 == 0:
                # No dates: assume the series starts in January.
                reshaped = da_flat[:, : n_years * 12].reshape(n_cells, n_years, 12)
                monthly_means = np.nanmean(reshaped, axis=1)  # (n_cells, 12)
                logger.info(
                    f"[{var_name}] Computed monthly climatology from {n_years} years of monthly data"
                )
            else:
                warnings.warn(
                    f"'{var_name}': {n_timesteps} monthly time steps are not whole years "
                    f"({n_months_total % 12} months left over), so its 12 monthly-climatology "
                    "features -- and the Köppen–Geiger diagnostics built from them -- are left "
                    "out. Trim the input to whole years to keep them.",
                    UserWarning,
                    stacklevel=2,
                )

        # Seasonality = std across the 12 monthly means
        if monthly_means is not None:
            features[f"{var_name}_seasonality"] = np.nanstd(monthly_means, axis=1)
        else:
            features[f"{var_name}_seasonality"] = features[f"{var_name}_std"]
    else:
        # Not enough timesteps
        features[f"{var_name}_seasonality"] = features[f"{var_name}_std"]

    # Include monthly climatology features for Köppen-Geiger classification
    if include_monthly_climatology and monthly_means is not None:
        logger.info(f"[{var_name}] Adding 12 monthly climatology features")
        for month_idx in range(12):
            month_num = month_idx + 1  # 1-indexed month
            features[f"{var_name}_month_{month_num:02d}"] = monthly_means[:, month_idx]
    elif include_monthly_climatology and monthly_means is None:
        logger.warning(
            f"[{var_name}] WARNING: include_monthly_climatology=True but monthly_means is None!"
        )
        logger.info(f"[{var_name}]   da_flat.shape = {da_flat.shape}")

    return features


def _summarize_precipitation(
    da: xr.DataArray,
    da_flat: np.ndarray,
    var_name: str,
    include_monthly_climatology: bool = True,
) -> dict:
    """
    Precipitation summaries:
    - Mean annual precipitation
    - Monthly seasonality (coefficient of variation)
    - Wet-day frequency (days >1 mm/month)
    - Maximum monthly precipitation
    - Monthly climatology (12 monthly totals, optional, for Köppen-Geiger).

    All computations use the pre-flattened valid-cell array for efficiency.

    Args:
        da: Original xarray DataArray (not currently used but kept for API consistency)
        da_flat: Pre-flattened array of shape (n_cells, n_timesteps)
        var_name: Variable name prefix for feature names (e.g., 'pr')
        include_monthly_climatology: If True, include 12 monthly sum features
            (pr_month_01 through pr_month_12) needed for Köppen-Geiger classification.
            These are mean monthly totals in mm/month.
    """
    features = {}

    # All vectorized numpy operations on da_flat

    # Mean annual
    mean_vals = np.nanmean(da_flat, axis=1)
    features[f"{var_name}_mean_annual"] = mean_vals

    # CV (coefficient of variation): std / mean
    std_vals = np.nanstd(da_flat, axis=1)
    # Avoid division by zero
    cv = np.zeros_like(mean_vals)
    mask = mean_vals > 0
    cv[mask] = std_vals[mask] / mean_vals[mask]
    features[f"{var_name}_cv_monthly"] = cv

    # Wet-day frequency (proportion of timesteps > threshold)
    threshold = 1.0  # mm
    wet_days = np.nanmean(da_flat > threshold, axis=1)
    features[f"{var_name}_wetdays_freq"] = wet_days

    # Maximum monthly precipitation
    features[f"{var_name}_max_monthly"] = np.nanmax(da_flat, axis=1)

    # Compute monthly climatology for Köppen-Geiger classification
    # For precipitation, we need monthly totals (mm/month)
    monthly_sums = None
    if include_monthly_climatology and da_flat.shape[1] >= 12:
        n_cells = da_flat.shape[0]
        n_timesteps = da_flat.shape[1]

        # Detect if data is daily (more than 2 years of daily data)
        is_daily = _is_daily(da, n_timesteps, fallback_threshold=365 * 2)

        if is_daily:
            # Daily data: need to resample to monthly first, removing leap days
            logger.info(
                f"[{var_name}] Detected daily data ({n_timesteps} timesteps), resampling to monthly"
            )

            # Get time coordinate from original DataArray for leap day detection
            time_coord = da.coords.get("time", None)
            if time_coord is not None:
                # Remove leap days (Feb 29) to get exactly 365 days per year
                time_vals = time_coord.values

                # Identify non-leap-day indices
                if hasattr(time_vals[0], "month"):
                    # cftime or datetime objects
                    non_leap_mask = np.array(
                        [
                            not (getattr(t, "month", 0) == 2 and getattr(t, "day", 0) == 29)
                            for t in time_vals
                        ]
                    )
                else:
                    # pandas timestamps or numpy datetime64
                    import pandas as pd

                    time_pd = pd.to_datetime(time_vals)
                    non_leap_mask = ~((time_pd.month == 2) & (time_pd.day == 29))

                # Apply mask to remove leap days
                da_flat_no_leap = da_flat[:, non_leap_mask]
                n_days_no_leap = da_flat_no_leap.shape[1]
                n_years = n_days_no_leap // 365

                logger.info(
                    f"[{var_name}] After removing leap days: {n_days_no_leap} days, {n_years} complete years"
                )

                if n_years > 0:
                    # Trim to complete years
                    da_flat_trimmed = da_flat_no_leap[:, : n_years * 365]

                    # Reshape to (n_cells, n_years, 365)
                    reshaped_daily = da_flat_trimmed.reshape(n_cells, n_years, 365)

                    # Aggregate daily to monthly using DAYS_PER_MONTH
                    # For precipitation: sum days in each month to get monthly totals
                    monthly_data = np.zeros((n_cells, n_years, 12))
                    day_idx = 0
                    for month_idx, days_in_month in enumerate(DAYS_PER_MONTH):
                        month_slice = reshaped_daily[:, :, day_idx : day_idx + days_in_month]
                        # Sum daily values to get monthly total
                        monthly_data[:, :, month_idx] = np.nansum(month_slice, axis=2)
                        day_idx += days_in_month

                    # Compute mean monthly sums across years
                    monthly_sums = np.nanmean(monthly_data, axis=1)  # (n_cells, 12)
                    logger.info(f"[{var_name}] Computed monthly climatology from daily data")
            else:
                logger.warning(
                    f"[{var_name}] WARNING: Daily data detected but no time coordinate found"
                )
        else:
            # Monthly data: original logic
            n_months_total = n_timesteps
            n_years = n_months_total // 12

            by_month = _monthly_climatology_by_date(da, da_flat)
            if by_month is not None:
                monthly_sums = by_month
            elif n_years > 0 and n_months_total % 12 == 0:
                # No dates: assume the series starts in January.
                # Reshape to (n_cells, n_years, 12)
                reshaped = da_flat[:, : n_years * 12].reshape(n_cells, n_years, 12)
                # For precipitation: compute monthly sum per year, then average across years
                # If data is already monthly, values are monthly totals; take mean across years
                # This gives mean monthly precipitation for each month (mm/month)
                monthly_sums = np.nanmean(reshaped, axis=1)  # (n_cells, 12)
            else:
                warnings.warn(
                    f"'{var_name}': {n_timesteps} monthly time steps are not whole years "
                    f"({n_months_total % 12} months left over), so its 12 monthly-climatology "
                    "features -- and the Köppen–Geiger diagnostics built from them -- are left "
                    "out. Trim the input to whole years to keep them.",
                    UserWarning,
                    stacklevel=2,
                )

        # Add monthly features if we computed them successfully
        if monthly_sums is not None:
            logger.info(f"[{var_name}] Adding 12 monthly climatology features")
            for month_idx in range(12):
                month_num = month_idx + 1  # 1-indexed month
                features[f"{var_name}_month_{month_num:02d}"] = monthly_sums[:, month_idx]
    elif include_monthly_climatology:
        logger.warning(
            f"[{var_name}] WARNING: include_monthly_climatology=True but da_flat.shape[1] < 12"
        )
        logger.info(f"[{var_name}]   da_flat.shape = {da_flat.shape}")

    return features


def _summarize_radiation(da: xr.DataArray, da_flat: np.ndarray, var_name: str) -> dict:
    """
    Radiation summaries (shortwave/longwave):
    - Mean annual radiation
    - Seasonal amplitude (max - min monthly mean)
    - Annual std.
    """
    features = {}

    # Mean annual
    features[f"{var_name}_mean_annual"] = np.nanmean(da_flat, axis=1)

    # Annual std
    features[f"{var_name}_std"] = np.nanstd(da_flat, axis=1)

    # Seasonal amplitude (simplified as range)
    features[f"{var_name}_seasonal_amplitude"] = np.nanmax(da_flat, axis=1) - np.nanmin(
        da_flat, axis=1
    )

    return features


def _summarize_humidity(da: xr.DataArray, da_flat: np.ndarray, var_name: str) -> dict:
    """
    Relative humidity summaries:
    - Mean annual RH
    - Annual std
    - Dry-season minimum (lowest value).
    """
    features = {}

    # Mean annual
    features[f"{var_name}_mean_annual"] = np.nanmean(da_flat, axis=1)

    # Annual std
    features[f"{var_name}_std"] = np.nanstd(da_flat, axis=1)

    # Dry-season minimum
    features[f"{var_name}_dry_min"] = np.nanmin(da_flat, axis=1)

    return features
