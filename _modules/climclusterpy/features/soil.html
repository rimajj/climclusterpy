"""Soil features: one-hot soil class and continuous soil depth.

Soil type is categorical, so it is one-hot encoded rather than treated as an
ordinal number — soil class 5 is not "more" than class 2. The fitted encoder can be
cached so that separate runs over the same grid produce the *same* columns in the
same order, which matters when a learned metric is reused across runs.
"""

from __future__ import annotations

import logging
import os
import pickle
from pathlib import Path

import numpy as np
import pandas as pd
import xarray as xr
from sklearn.preprocessing import OneHotEncoder

from climclusterpy.data.registry import DataRegistry

logger = logging.getLogger(__name__)

__all__ = ["encode_categorical", "encode_soil_code", "summarize_soil_features"]


def encode_categorical(
    values: np.ndarray,
    var_name: str = "soil_code",
    encoder_cache_path: str | None = None,
) -> tuple[pd.DataFrame, OneHotEncoder]:
    """
    One-hot encode an array of integer category codes.

    The encoder is fit on the provided values and optionally persisted to
    *encoder_cache_path* so that downstream pipelines can reload it
    consistently without refitting.  The resulting column names follow the
    pattern ``<var_name>_<value>``.

    Args:
        values: 1-D array of integer category values (may contain NaN).
        var_name: Name of the field, used as the output column prefix
            (default ``"soil_code"``, for backward compatibility).
        encoder_cache_path: Optional filesystem path (str or Path) at which to
            save/load the fitted ``OneHotEncoder`` as a pickle file.  When the
            file already exists the cached encoder is loaded and used to
            transform ``values``; otherwise the encoder is fit and saved.

    Returns
    -------
        ohe_df: DataFrame whose columns are named ``<var_name>_<value>``
            (dtype float32, one column per unique non-NaN code).
        encoder: The fitted ``OneHotEncoder`` instance.
    """
    # Replace NaN with a sentinel integer so the encoder can process the array
    NAN_SENTINEL = -9999
    codes_filled = np.where(np.isnan(values.astype(float)), NAN_SENTINEL, values)
    codes_int = codes_filled.astype(np.int32).reshape(-1, 1)

    # Load or fit encoder
    if encoder_cache_path is not None and Path(encoder_cache_path).exists():
        with open(encoder_cache_path, "rb") as fh:
            encoder = pickle.load(fh)
        logger.info(f"[SoilOHE] Loaded cached {var_name} encoder from {encoder_cache_path}")
    else:
        encoder = OneHotEncoder(
            sparse_output=False,
            handle_unknown="ignore",
            dtype=np.float32,
        )
        encoder.fit(codes_int)
        if encoder_cache_path is not None:
            Path(encoder_cache_path).parent.mkdir(parents=True, exist_ok=True)
            with open(encoder_cache_path, "wb") as fh:
                pickle.dump(encoder, fh)
            logger.info(f"[SoilOHE] Saved {var_name} encoder to {encoder_cache_path}")

    # Transform
    ohe_array = encoder.transform(codes_int)  # (n_cells, n_codes)

    # Build column names: use the actual category values stored in the encoder
    # Filter out the NaN sentinel from column names
    categories = encoder.categories_[0]
    nan_col = f"{var_name}_nan"
    col_names = [f"{var_name}_{int(c)}" if c != NAN_SENTINEL else nan_col for c in categories]

    ohe_df = pd.DataFrame(ohe_array, columns=col_names)

    # Zero-out the sentinel column (cells that had NaN) and drop it
    if nan_col in ohe_df.columns:
        ohe_df.drop(columns=[nan_col], inplace=True)

    n_codes = len(ohe_df.columns)
    logger.info(
        f"[SoilOHE] One-hot encoded {var_name} into {n_codes} binary columns "
        f"({n_codes} unique categories found)"
    )
    return ohe_df, encoder

def encode_soil_code(
    soil_code_values: np.ndarray,
    encoder_cache_path: str | None = None,
) -> tuple[pd.DataFrame, OneHotEncoder]:
    """One-hot encode an array of integer soil codes (see :func:`encode_categorical`)."""
    return encode_categorical(soil_code_values, "soil_code", encoder_cache_path)

def summarize_soil_features(
    datasets: dict[str, xr.Dataset],
    registry: DataRegistry,
    soil_vars: list[str] | None = None,
    encoder_cache_path: str | None = None,
    categorical_vars: frozenset[str] | set[str] = frozenset({"soil_code"}),
) -> pd.DataFrame:
    """
    Extract static (soil, or any other non-climate) features.

    Each field in ``soil_vars`` is either continuous (kept as a numeric feature) or
    categorical (one-hot encoded into binary columns named ``<var_name>_<value>``, so that
    metric-learning algorithms are not misled by the arbitrary ordinal structure of a raw
    integer code). By default only ``soil_code`` is treated as categorical -- everything
    else, including any field outside the original soil_code/soil_depth pair, is continuous
    unless named in ``categorical_vars``.

    Args:
        datasets: Dict mapping variable names to xarray Datasets.
                  Expected keys: ``'soil_depth'``, ``'soil_code'`` (or any other static
                  field name, for a non-LPJmL input).
        registry: DataRegistry containing valid cell mask.
        soil_vars: List of static variables to process
            (default: ``['soil_depth', 'soil_code']``).
        encoder_cache_path: Optional path at which to save/reload the fitted
            ``OneHotEncoder`` for each categorical field. Pass a stable path
            (e.g. inside the feature cache directory) so the encoder is
            reloaded consistently on subsequent runs. When more than one
            categorical field is given, each gets its own cache file (the
            field name is inserted before the file extension).
        categorical_vars: Which of ``soil_vars`` are categorical (one-hot encoded)
            rather than continuous. Default: ``{"soil_code"}``, matching the
            original LPJmL-specific behavior.

    Returns
    -------
        DataFrame with columns:
        - ``cell_id``, ``latitude``, ``longitude``
        - one continuous column per continuous field
        - ``<var_name>_<value>`` (binary, one column per unique category) per categorical field
    """
    if soil_vars is None:
        soil_vars = ["soil_depth", "soil_code"]

    logger.info(f"[FeatureSummary] Summarizing soil features for {len(soil_vars)} variables")

    # Get valid cell coordinates
    valid_lats, valid_lons = registry.get_valid_lat_lon()
    n_cells = len(valid_lats)

    # Initialize output metadata
    base = {
        "cell_id": list(range(n_cells)),
        "latitude": valid_lats,
        "longitude": valid_lons,
    }

    # Collect continuous features and the raw categorical arrays separately
    continuous_features: dict[str, np.ndarray] = {}
    categorical_vals: dict[str, np.ndarray] = {}

    for var_name in soil_vars:
        if var_name not in datasets:
            logger.warning(
                f"[FeatureSummary] WARNING: Soil variable '{var_name}' not found, skipping"
            )
            continue

        ds = datasets[var_name]

        # Extract data variable
        if var_name in ds.data_vars:
            da = ds[var_name]
        else:
            data_vars = [v for v in ds.data_vars if v not in ["lat", "lon", "time"]]
            if not data_vars:
                logger.warning(
                    f"[FeatureSummary] WARNING: No data variables found for '{var_name}', skipping"
                )
                continue
            da = ds[data_vars[0]]

        # Flatten to valid cells
        if "time" in da.dims:
            da_vals = registry.flatten_on_valid(da.isel(time=0))
        else:
            da_vals = registry.flatten_on_valid(da)

        da_vals = np.asarray(da_vals).ravel()

        if var_name in categorical_vars:
            # Defer one-hot encoding until after all vars are loaded
            categorical_vals[var_name] = da_vals
            unique_codes = np.unique(da_vals[~np.isnan(da_vals.astype(float))])
            logger.info(
                f"[FeatureSummary] {var_name}: {len(unique_codes)} unique values "
                f"from {unique_codes.min():.0f} to {unique_codes.max():.0f} "
                f"— will one-hot encode"
            )
        else:
            # soil_depth and any other continuous static variable
            continuous_features[var_name] = da_vals

    # Build base DataFrame with continuous features
    df = pd.DataFrame({**base, **continuous_features})

    # One-hot encode each categorical field and concatenate
    for var_name, vals in categorical_vals.items():
        cache_path = encoder_cache_path
        if cache_path is not None and len(categorical_vals) > 1:
            stem, ext = os.path.splitext(cache_path)
            cache_path = f"{stem}_{var_name}{ext}"
        ohe_df, _encoder = encode_categorical(vals, var_name, encoder_cache_path=cache_path)
        ohe_df.index = df.index
        df = pd.concat([df, ohe_df], axis=1)

    logger.info(f"[FeatureSummary] Created {len(df.columns)} soil features for {n_cells} cells")
    return df
