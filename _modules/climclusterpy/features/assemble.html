Source code for climclusterpy.features.assemble

"""Join the per-variable feature tables into the matrix that gets clustered.

Climate and soil features are computed separately, then merged **on cell identity**
(``cell_id`` plus coordinates) rather than by row position — the two tables are built
from the same registry, but joining on identity means a mismatch surfaces as missing
rows instead of silently pairing the wrong cells.
"""

from __future__ import annotations

import logging

import pandas as pd

logger = logging.getLogger(__name__)

__all__ = ["METADATA_COLUMNS", "build_feature_table", "combine_features"]

#: Columns that identify a cell rather than describe it — never clustering features.
METADATA_COLUMNS = frozenset(
    {"cell_id", "latitude", "longitude", "lat", "lon", "ecoregion_id", "cluster_label"}
)

def combine_features(climate_df: pd.DataFrame, soil_df: pd.DataFrame) -> pd.DataFrame:
    """
    Merge climate and soil features on cell_id.

    Args:
        climate_df: DataFrame from summarize_climate_features()
        soil_df: DataFrame from summarize_soil_features()

    Returns
    -------
        Merged DataFrame with all features
    """
    logger.info("[FeatureSummary] Combining climate and soil features")

    # Merge on cell_id
    merged = climate_df.merge(soil_df, on=["cell_id", "latitude", "longitude"], how="inner")

    logger.info(
        f"[FeatureSummary] Combined features: {len(merged)} cells, {len(merged.columns)} columns"
    )

    return merged




def build_feature_table(
    datasets,
    registry,
    *,
    climate_variables,
    correlation_threshold: float,
    pca_variance_threshold: float,
    pca_contrib_threshold: float,
    variable_aggregations: dict[str, str] | None = None,
    static_variables: list[str] | None = None,
    categorical_vars: frozenset[str] | set[str] = frozenset({"soil_code"}),
) -> tuple[pd.DataFrame, list[str], pd.DataFrame]:
    """Build the standardized, de-correlated, PCA-filtered per-cell feature table.

    Returns ``(features_df, feature_columns, climate_summary_df)``. The unfiltered
    climate summary comes back too, because Köppen–Geiger supervision needs the raw
    monthly climatology columns that filtering may drop.

    ``variable_aggregations``, ``static_variables`` and ``categorical_vars`` only matter
    for an input outside the known LPJmL set -- see
    :func:`~climclusterpy.features.climate.summarize_climate_features` and
    :func:`~climclusterpy.features.soil.summarize_soil_features`.
    """
    from climclusterpy.features.climate import summarize_climate_features
    from climclusterpy.features.filtering import (
        diagnostic_pca_filter,
        remove_correlated_features,
        standardize_features,
    )
    from climclusterpy.features.soil import summarize_soil_features

    climate_df = summarize_climate_features(
        datasets,
        registry,
        variables=list(climate_variables),
        variable_aggregations=variable_aggregations,
    )
    soil_df = summarize_soil_features(
        datasets, registry, soil_vars=static_variables, categorical_vars=categorical_vars
    )
    raw = combine_features(climate_df, soil_df)

    feature_cols = [c for c in raw.columns if c not in METADATA_COLUMNS]
    if not feature_cols:
        raise ValueError(
            "Feature construction produced no feature columns. Check that the climate and "
            "soil inputs contain the expected variables."
        )

    # A missing value anywhere in a land cell's series makes its summaries NaN, which the
    # statistics below cannot take -- say which input and which cells, instead of letting it
    # surface as a scikit-learn error about PCA.
    from climclusterpy.data.checks import missing_value_problems

    problems = missing_value_problems(raw, feature_cols)
    if problems:
        raise ValueError(
            "The input has missing values on land cells:\n"
            + "\n".join(f"  - {p}" for p in problems)
        )

    standardized, _scaler = standardize_features(raw, feature_cols)
    # NOTE: the second return value is the list of features that were *removed*,
    # so the surviving columns have to be derived by difference.
    decorrelated, removed_cols, _corr = remove_correlated_features(
        standardized, feature_cols, threshold=correlation_threshold
    )
    surviving_cols = [c for c in feature_cols if c not in set(removed_cols)]
    filtered, _pca_info = diagnostic_pca_filter(
        decorrelated,
        surviving_cols,
        variance_threshold=pca_variance_threshold,
        contrib_threshold=pca_contrib_threshold,
    )

    final_cols = [c for c in filtered.columns if c not in METADATA_COLUMNS]
    if not final_cols:
        raise ValueError(
            "Feature filtering removed every feature. Loosen correlation_threshold "
            f"(currently {correlation_threshold}) or pca_contrib_threshold "
            f"(currently {pca_contrib_threshold})."
        )
    return filtered, final_cols, climate_df
