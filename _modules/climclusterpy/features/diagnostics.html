"""Ecophysiological and Köppen-diagnostic features.

Two families of derived features that raw statistics miss, both computed from the
monthly climatologies:

* **Köppen diagnostics** (``kg_diag_*``) — the quantities the Köppen–Geiger rules
  are actually defined on (warmest/coldest month, seasonality ratios, aridity
  index), so a clustering can see the same structure a climate classification does.
* **Ecology diagnostics** (``eco_diag_*``) — threshold and ratio transformations
  with independent ecological meaning: vapour-pressure deficit, growing degree
  days, frost-free days, PET and the P/PET ratio, water-deficit months, dry-spell
  length, and precipitation intensity.

Both families are *protected* from correlation and PCA filtering. They may
correlate with raw features numerically while encoding non-redundant
threshold-nonlinear information, so dropping them on a correlation test would throw
away exactly the ecological signal they were added for.
"""

from __future__ import annotations

def compute_kg_diagnostic_features(
    monthly_T: np.ndarray,
    monthly_P: np.ndarray,
    latitudes: np.ndarray,
) -> dict[str, np.ndarray]:
    """
    Compute KG-diagnostic scalar features from 12-month climatological means.

    These features capture the exact quantities used by the Köppen-Geiger
    classification rule (temperature extremes, precipitation seasonality,
    aridity) as continuous scalars.  They are prefixed with ``kg_diag_`` so
    that the filtering pipeline can identify and protect them.

    Parameters
    ----------
    monthly_T : np.ndarray, shape (n_cells, 12)
        Mean temperature (°C) for each of the 12 calendar months.
    monthly_P : np.ndarray, shape (n_cells, 12)
        Mean precipitation (mm/month) for each of the 12 calendar months.
    latitudes : np.ndarray, shape (n_cells,)
        Cell latitudes in decimal degrees (negative = Southern Hemisphere).

    Returns
    -------
    dict mapping ``kg_diag_*`` feature names to 1-D arrays of length n_cells.
    """
    # Month indices: NH summer = Apr(3)–Sep(8), NH winter = Oct(9)–Mar(2)
    # For SH cells the halves are flipped (hemisphere derived from latitude sign).
    NH_SUMMER = np.array([3, 4, 5, 6, 7, 8])  # 0-indexed months Apr–Sep
    NH_WINTER = np.array([9, 10, 11, 0, 1, 2])  # Oct–Mar

    T_cold = monthly_T.min(axis=1)
    T_warm = monthly_T.max(axis=1)
    T_annual = monthly_T.mean(axis=1)

    P_dry_min = monthly_P.min(axis=1)
    P_wet_max = monthly_P.max(axis=1)

    # Hemisphere-aware seasonal precip
    is_southern = latitudes < 0
    # NH conventions first; then flip for SH cells
    P_summer_nh = monthly_P[:, NH_SUMMER].min(axis=1)
    P_winter_nh = monthly_P[:, NH_WINTER].min(axis=1)
    P_summer_sh = monthly_P[:, NH_WINTER].min(axis=1)
    P_winter_sh = monthly_P[:, NH_SUMMER].min(axis=1)

    P_dry_summer = np.where(is_southern, P_summer_sh, P_summer_nh)
    P_dry_winter = np.where(is_southern, P_winter_sh, P_winter_nh)

    # Seasonality ratio (wettest/driest monthly precip); clip to avoid explosion
    # Add small epsilon to driest month to avoid div-by-zero
    P_seasonality_ratio = np.clip(
        P_wet_max / np.maximum(P_dry_min, 0.1),
        1.0,
        100.0,
    )

    # Aridity offset from the KG threshold
    # KG aridity threshold: Pann > 2*Tann + 28 (all-year precipitation; shift for
    # summer/winter precipitation patterns is not applied here — that is handled
    # by the KG module).  Negative values indicate arid conditions.
    aridity_index = monthly_P.sum(axis=1) - (2.0 * T_annual + 28.0)

    return {
        "kg_diag_T_cold": T_cold.astype(np.float32),
        "kg_diag_T_warm": T_warm.astype(np.float32),
        "kg_diag_P_dry_min": P_dry_min.astype(np.float32),
        "kg_diag_P_wet_max": P_wet_max.astype(np.float32),
        "kg_diag_P_dry_summer": P_dry_summer.astype(np.float32),
        "kg_diag_P_dry_winter": P_dry_winter.astype(np.float32),
        "kg_diag_P_seasonality_ratio": P_seasonality_ratio.astype(np.float32),
        "kg_diag_aridity_index": aridity_index.astype(np.float32),
    }

def compute_vpd_features(
    monthly_T: np.ndarray,
    monthly_q: np.ndarray,
) -> dict[str, np.ndarray]:
    """
    Vapour Pressure Deficit (VPD) features from monthly temperature and
    specific humidity.

    VPD = e_s(T) * (1 - RH/100)
    e_s(T) = 0.611 * exp(17.27*T / (T + 237.3))  [kPa]
    RH ≈ 100 * (q * P) / (0.622 * e_s(T))        [% ; P = 101.325 kPa]

    Ecological motivation:
    - VPD is the primary driver of plant water stress and stomatal closure,
      more direct than either temperature or humidity alone
    - Mean VPD captures background atmospheric aridity
    - Max monthly VPD captures peak physiological stress intensity
    - Stress months captures duration of high-VPD conditions
    - Particularly important for differentiating semi-arid systems where
      P/PET alone does not capture atmospheric demand timing

    Note: may correlate with temperature and humidity features but encodes
    a nonlinear multiplicative interaction between them — protected from
    filtering as ecology_diagnostic.

    Parameters
    ----------
    monthly_T : (n_cells, 12)  Mean temperature per month [°C]
    monthly_q : (n_cells,)     Mean annual specific humidity [kg/kg]

    Returns
    -------
    dict with keys: eco_diag_vpd_mean, eco_diag_vpd_max_monthly,
                    eco_diag_vpd_stress_months
    """
    P_ATM = 101.325  # kPa, standard atmospheric pressure

    # Saturation vapour pressure per month [kPa]
    e_s = 0.611 * np.exp(17.27 * monthly_T / (monthly_T + 237.3))

    # Approximate RH per month from annual-mean specific humidity
    # RH_m ≈ 100 * (q * P) / (0.622 * e_s_m)
    q_expanded = monthly_q[:, np.newaxis]  # (n, 1)
    RH_monthly = np.clip(
        100.0 * (q_expanded * P_ATM) / (0.622 * np.maximum(e_s, 1e-6)),
        0.0,
        100.0,
    )

    # VPD per month [kPa]
    vpd_monthly = np.maximum(e_s * (1.0 - RH_monthly / 100.0), 0.0)

    vpd_mean = vpd_monthly.mean(axis=1).astype(np.float32)
    vpd_max_monthly = vpd_monthly.max(axis=1).astype(np.float32)
    vpd_stress_months = (vpd_monthly > 1.0).sum(axis=1).astype(np.float32)

    return {
        "eco_diag_vpd_mean": vpd_mean,
        "eco_diag_vpd_max_monthly": vpd_max_monthly,
        "eco_diag_vpd_stress_months": vpd_stress_months,
    }


def compute_growing_season_features(
    monthly_T: np.ndarray,
) -> dict[str, np.ndarray]:
    """
    Growing season thermal accumulation features from monthly temperatures.

    Works on the monthly means directly, with no daily reconstruction:

    * GDD above a base is the sum over months of ``max(T_month - base, 0)`` times
      the days in that month. This underestimates slightly in months whose mean is
      close to the base, where some days are above it and some below.
    * Frost-free days is the sum over months of the days in that month times
      ``clip(T_month / 5 + 0.5, 0, 1)``: a month at -2.5 °C or colder counts no day,
      one at +2.5 °C or warmer counts every day, linear in between.

    Ecological motivation:
    - GDD captures total thermal energy available for plant growth
      during the growing season — standard input to vegetation models
    - Frost-free days captures growing season length directly
    - Together they distinguish maritime from continental climates
      that have equal mean annual temperature
    - GDD_5 is the temperate standard; GDD_10 is the warm-season/crop standard
    - May correlate with temperature seasonality but captures threshold-
      nonlinear integration — protected as ecology_diagnostic

    Parameters
    ----------
    monthly_T : (n_cells, 12) Mean temperature per month [°C]

    Returns
    -------
    dict with keys: eco_diag_gdd_5, eco_diag_gdd_10, eco_diag_frost_free_days
    """
    days_per_month = np.array(DAYS_PER_MONTH, dtype=np.float32)

    # Approximate GDD from monthly means using the Thom (1966) method:
    # For month m with mean T_m:
    #   GDD_base(m) ≈ max(T_m - base, 0) * days_in_month(m)
    # This is a good approximation when |T_m - base| >> daily_range/2
    # and slightly underestimates in the transition zone.
    gdd_5_monthly = np.maximum(monthly_T - 5.0, 0.0) * days_per_month[np.newaxis, :]
    gdd_10_monthly = np.maximum(monthly_T - 10.0, 0.0) * days_per_month[np.newaxis, :]

    gdd_5 = gdd_5_monthly.sum(axis=1).astype(np.float32)
    gdd_10 = gdd_10_monthly.sum(axis=1).astype(np.float32)

    # Frost-free days: approximate as sum of days in months with T_m >= 0°C
    # For months near 0°C, use linear interpolation:
    # fraction_frost_free ≈ clip(T_m / 5 + 0.5, 0, 1)
    # This gives 0% at T_m = -2.5°C, 100% at T_m = +2.5°C, linear between
    frac_frost_free = np.clip(monthly_T / 5.0 + 0.5, 0.0, 1.0)
    frost_free_days = (frac_frost_free * days_per_month[np.newaxis, :]).sum(axis=1)
    frost_free_days = frost_free_days.astype(np.float32)

    return {
        "eco_diag_gdd_5": gdd_5,
        "eco_diag_gdd_10": gdd_10,
        "eco_diag_frost_free_days": frost_free_days,
    }


def compute_water_balance_features(
    monthly_P: np.ndarray,
    monthly_T: np.ndarray,
    rsds_mean: np.ndarray,
) -> dict[str, np.ndarray]:
    """
    Water balance features from precipitation and potential evapotranspiration.

    PET is computed using Priestley-Taylor approximation from monthly T and
    mean annual shortwave radiation:
        Rn ≈ 0.77 * rsds * 0.0864   [MJ/m²/day]
        PET_monthly ≈ 1.26 * (Δ * Rn) / (Δ + γ) * days_per_month
    where Δ is the slope of the saturation vapour pressure curve and γ
    is the psychrometric constant.

    Ecological motivation:
    - P/PET is the single most important axis of global vegetation distribution,
      separating humid from arid systems more cleanly than either P or T alone
    - Annual PET captures evaporative demand independent of water supply
    - Water deficit months captures duration of physiological drought stress
    - These complement rather than replace the KG aridity index scalar
      (Pann - 2*Tann + 28) which uses a linear temperature approximation —
      P/PET from Priestley-Taylor PET is physically more rigorous
    - Protected as ecology_diagnostic despite possible correlation with
      existing aridity features

    Parameters
    ----------
    monthly_P : (n_cells, 12) Monthly mean precipitation [mm/month]
    monthly_T : (n_cells, 12) Monthly mean temperature [°C]
    rsds_mean : (n_cells,)    Mean annual shortwave radiation [W/m²]

    Returns
    -------
    dict with keys: eco_diag_pet_mean, eco_diag_p_pet_ratio,
                    eco_diag_water_deficit_months
    """
    days_per_month = np.array(DAYS_PER_MONTH, dtype=np.float32)

    # Net radiation (MJ/m²/day) from mean annual shortwave
    Rn_daily = 0.77 * rsds_mean * 0.0864  # (n_cells,)

    # Psychrometric constant γ ≈ 0.066 kPa/°C (at sea level)
    gamma = 0.066

    # Slope of saturation vapour pressure curve Δ(T) [kPa/°C]
    # Δ = 4098 * e_s(T) / (T + 237.3)^2
    e_s = 0.611 * np.exp(17.27 * monthly_T / (monthly_T + 237.3))
    delta = 4098.0 * e_s / (monthly_T + 237.3) ** 2  # (n_cells, 12)

    # Priestley-Taylor PET [mm/day] per month
    # PET_day = 1.26 * (Δ / (Δ + γ)) * (Rn / 2.45)
    # where 2.45 MJ/kg is the latent heat of vaporisation
    Rn_expand = Rn_daily[:, np.newaxis]  # (n_cells, 1)
    pet_daily = 1.26 * (delta / (delta + gamma)) * (Rn_expand / 2.45)
    pet_daily = np.maximum(pet_daily, 0.0)  # (n_cells, 12)

    # Monthly PET [mm/month]
    PET_monthly = pet_daily * days_per_month[np.newaxis, :]  # (n_cells, 12)

    pet_mean = PET_monthly.mean(axis=1).astype(np.float32)

    # P/PET ratio (annual)
    P_annual = monthly_P.sum(axis=1)
    PET_annual = PET_monthly.sum(axis=1)
    p_pet_ratio = np.clip(
        P_annual / np.maximum(PET_annual, 1.0),
        0.01,
        10.0,
    ).astype(np.float32)

    # Water deficit months: months where P < PET
    water_deficit_months = (monthly_P < PET_monthly).sum(axis=1).astype(np.float32)

    return {
        "eco_diag_pet_mean": pet_mean,
        "eco_diag_p_pet_ratio": p_pet_ratio,
        "eco_diag_water_deficit_months": water_deficit_months,
    }


def compute_dry_spell_features(
    monthly_P: np.ndarray,
    pr_wetdays_freq: np.ndarray,
    pr_mean_annual: np.ndarray,
) -> dict[str, np.ndarray]:
    """
    Precipitation event structure features from monthly precipitation totals.

    Uses monthly summaries to approximate dry spell statistics without
    requiring daily data:

    - Dry spell max: the expected longest dry run of a year when every day is wet
      with the annual wet-day fraction ``p`` independently (geometric model),
      ``-ln(365) / ln(1 - p)``, clipped to 1-365 days; ``1 - p`` is clipped to
      [0.01, 0.99]
    - Dry spell mean: ``1 / p`` (with ``p`` at least 0.01)
    - Precipitation intensity on wet days: mean daily P / ``p``
    - Wettest quarter fraction: exact from monthly totals (the only feature here
      that uses ``monthly_P``)

    Ecological motivation:
    - Dry spell length determines how long plants must survive without water
    - Precipitation intensity on wet days captures rainfall erosivity and
      infiltration vs runoff partitioning
    - Wettest quarter fraction distinguishes monsoon systems from year-round
      precipitation — important for tropical savanna vs forest differentiation
    - May correlate with wet-day frequency and CV but captures the temporal
      clustering of precipitation events — protected as ecology_diagnostic

    Parameters
    ----------
    monthly_P :       (n_cells, 12) Monthly mean precipitation [mm/month]
    pr_wetdays_freq : (n_cells,)    Fraction of days that are wet (>1mm)
    pr_mean_annual :  (n_cells,)    Mean annual precipitation [mm/day]

    Returns
    -------
    dict with keys: eco_diag_dry_spell_max, eco_diag_dry_spell_mean,
                    eco_diag_precip_intensity_wet_days,
                    eco_diag_precip_wettest_quarter_frac
    """
    # ── Precipitation intensity on wet days [mm/day] ──────────────────────
    # intensity = mean_daily_P / wet_day_fraction
    wet_frac = np.maximum(pr_wetdays_freq, 0.01)  # avoid div-by-zero
    mean_daily_P = np.maximum(pr_mean_annual, 0.0)
    precip_intensity = (mean_daily_P / wet_frac).astype(np.float32)

    # ── Wettest quarter fraction ──────────────────────────────────────────
    precip_wettest_quarter_frac = _compute_wettest_quarter_fraction(monthly_P)

    # ── Dry spell statistics (monthly approximation) ──────────────────────
    # Model: on a dry day (1 - wet_frac), the probability of a dry spell
    # continuing is (1 - wet_frac). The expected maximum dry spell length
    # for N=365 days drawn from geometric(p=wet_frac) is approximately:
    #   E[max] ≈ -log(N) / log(1 - p)
    # where p = wet-day probability.
    dry_frac = np.clip(1.0 - pr_wetdays_freq, 0.01, 0.99)
    # Mean dry spell length = 1 / p ≈ 1 / wet_frac
    dry_spell_mean = (1.0 / np.maximum(pr_wetdays_freq, 0.01)).astype(np.float32)

    # Expected annual maximum dry spell from geometric distribution
    # E[max] ≈ -ln(365) / ln(1 - p) where p = wet_frac
    log_dry = np.log(np.maximum(dry_frac, 1e-6))
    dry_spell_max = np.clip(
        -np.log(365.0) / np.minimum(log_dry, -1e-6),
        1.0,
        365.0,
    ).astype(np.float32)

    return {
        "eco_diag_dry_spell_max": dry_spell_max,
        "eco_diag_dry_spell_mean": dry_spell_mean,
        "eco_diag_precip_intensity_wet_days": precip_intensity,
        "eco_diag_precip_wettest_quarter_frac": precip_wettest_quarter_frac,
    }


def _compute_wettest_quarter_fraction(monthly_P: np.ndarray) -> np.ndarray:
    """
    Fraction of annual precipitation falling in the wettest 3 consecutive months.
    Uses a rolling 3-month window over the 12-month climatology.

    Parameters
    ----------
    monthly_P : (n_cells, 12) Monthly precipitation [mm/month]

    Returns
    -------
    (n_cells,) fraction in [0, 1]. The annual total is taken as at least 1 mm, so the
    fraction is at least 0.25 only where annual precipitation is at least 1 mm; a cell
    with no precipitation at all gets 0.
    """
    annual_total = np.maximum(monthly_P.sum(axis=1, keepdims=True), 1.0)

    max_quarter = np.zeros(monthly_P.shape[0], dtype=np.float32)
    for start in range(12):
        indices = [start % 12, (start + 1) % 12, (start + 2) % 12]
        quarter_sum = monthly_P[:, indices].sum(axis=1)
        max_quarter = np.maximum(max_quarter, quarter_sum)

    return (max_quarter / annual_total.squeeze()).astype(np.float32)

def compute_all_ecology_diagnostic_features(
    monthly_T: np.ndarray,
    monthly_P: np.ndarray,
    monthly_q: np.ndarray,
    rsds_mean: np.ndarray,
    pr_wetdays_freq: np.ndarray,
    pr_mean_annual: np.ndarray,
) -> dict[str, np.ndarray]:
    """
    Compute all 13 ecology_diagnostic features from cached summary data.

    This is the top-level entry point for augmenting an existing feature set.
    All inputs are available from the existing raw feature CSV — no raw NetCDF
    data or daily time series are required.

    Parameters
    ----------
    monthly_T :       (n_cells, 12) Monthly mean temperature [°C]
    monthly_P :       (n_cells, 12) Monthly mean precipitation [mm/month]
    monthly_q :       (n_cells,)    Annual mean specific humidity [kg/kg]
    rsds_mean :       (n_cells,)    Annual mean shortwave radiation [W/m²]
    pr_wetdays_freq : (n_cells,)    Fraction of days that are wet (>1mm)
    pr_mean_annual :  (n_cells,)    Mean annual precipitation [mm/day]

    Returns
    -------
    dict mapping eco_diag_* feature names to (n_cells,) arrays
    """
    feats: dict[str, np.ndarray] = {}
    feats.update(compute_vpd_features(monthly_T, monthly_q))
    feats.update(compute_growing_season_features(monthly_T))
    feats.update(compute_water_balance_features(monthly_P, monthly_T, rsds_mean))
    feats.update(compute_dry_spell_features(monthly_P, pr_wetdays_freq, pr_mean_annual))

    if set(feats) != set(ECOLOGY_DIAGNOSTIC_FEATURES):
        raise ValueError(
            "The ecology diagnostics computed do not match the declared feature list: "
            f"extra {sorted(set(feats) - set(ECOLOGY_DIAGNOSTIC_FEATURES))}, missing "

        )
