"""
Feature filtering and standardization for ecological clustering.

Provides:
- Standardization (zero mean, unit variance)
- Correlation-based feature removal
- PCA-based diagnostic filtering
"""

from __future__ import annotations

import logging
import pickle
import warnings
from pathlib import Path

import numpy as np
import pandas as pd
from scipy.stats import spearmanr
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler

logger = logging.getLogger(__name__)

# Column-name prefixes that are protected from correlation/PCA filtering by default.
# Any column whose name starts with one of these prefixes will be passed through
# unchanged when protected_cols is not explicitly provided.
# eco_diag_: ecology_diagnostic features — physically non-redundant ecological
#   indices (VPD, GDD, PET, P/PET, dry spells) that encode threshold-nonlinear
#   or ratio transformations with independent ecological meaning.
_AUTO_PROTECTED_PREFIXES = ("tas_month_", "pr_month_", "kg_diag_", "soil_", "eco_diag_")

def standardize_features(
    df: pd.DataFrame,
    feature_cols: list[str],
) -> tuple[pd.DataFrame, StandardScaler]:
    """
    Standardize features to zero mean, unit variance.

    Args:
        df: DataFrame with features
        feature_cols: List of column names to standardize

    Returns
    -------
        df_scaled: DataFrame with standardized features (replaces original columns)
        scaler: Fitted StandardScaler for inverse transform
    """
    logger.info(f"[FeatureFilter] Standardizing {len(feature_cols)} features")

    # Copy dataframe
    df_scaled = df.copy()

    # Extract feature matrix
    X = df[feature_cols].values

    # Fit scaler
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(X)

    # Replace columns
    df_scaled[feature_cols] = X_scaled

    logger.info("[FeatureFilter] Standardization complete")
    logger.info(f"  Mean (should be ~0): {np.mean(X_scaled, axis=0)[:5]}")
    logger.info(f"  Std (should be ~1): {np.std(X_scaled, axis=0)[:5]}")

    return df_scaled, scaler

def remove_correlated_features(
    df: pd.DataFrame,
    feature_cols: list[str],
    threshold: float = 0.9,
    method: str = "spearman",
    protected_cols: list[str] | None = None,
) -> tuple[pd.DataFrame, list[str], pd.DataFrame]:
    """Remove highly correlated features whose ``|rho|`` exceeds ``threshold``.

    When two features are correlated above threshold, keeps the one with
    lower mean absolute correlation to all other features.

    Args:
        df: DataFrame with features
        feature_cols: List of column names to analyze
        threshold: Correlation threshold (default 0.9)
        method: 'spearman' or 'pearson'
        protected_cols: Columns to pass through without analysis.  When
            ``None`` (default), any column in *feature_cols* whose name starts
            with one of the :data:`_AUTO_PROTECTED_PREFIXES` (``tas_month_*``,
            ``pr_month_*``, ``kg_diag_*``, ``soil_*``) is automatically
            protected.

    Returns
    -------
        df_filtered: DataFrame with redundant features removed
        removed_features: List of feature names that were dropped
        corr_matrix: Correlation matrix (before filtering, on analysed cols)
    """
    # Resolve protected cols
    if protected_cols is None:
        protected_cols = [
            c for c in feature_cols if any(c.startswith(p) for p in _AUTO_PROTECTED_PREFIXES)
        ]
    protected_set = set(protected_cols)
    analysed_cols = [c for c in feature_cols if c not in protected_set]

    if protected_cols:
        logger.info(
            f"[FeatureFilter] remove_correlated_features: protecting "
            f"{len(protected_cols)} cols from removal "
            f"({protected_cols[:5]}{'...' if len(protected_cols) > 5 else ''})"
        )

    if not analysed_cols:
        logger.info("[FeatureFilter] No analysable columns — skipping correlation filter")
        identity_corr = pd.DataFrame(
            np.eye(len(feature_cols)), index=feature_cols, columns=feature_cols
        )
        return df.copy(), [], identity_corr

    logger.info(
        f"[FeatureFilter] Removing correlated features (|ρ| > {threshold}) from {len(analysed_cols)} cols"
    )

    # Extract feature matrix (analysed cols only)
    X = df[analysed_cols].values

    # Compute correlation matrix (on analysed cols only)
    if method == "spearman":
        # A column that is the same in every cell (e.g. a uniform soil depth) has no defined
        # correlation; numpy/scipy say so with a RuntimeWarning on every run. It is handled
        # below (its entries stay NaN and it is never chosen for removal), so the warning only
        # tells the user about something that is not wrong.
        with warnings.catch_warnings(), np.errstate(divide="ignore", invalid="ignore"):
            warnings.simplefilter("ignore", RuntimeWarning)
            corr_matrix = spearmanr(X, axis=0, nan_policy="omit")[0]
        # Handle single feature case
        if len(analysed_cols) == 1:
            corr_matrix = np.array([[1.0]])
        elif np.ndim(corr_matrix) != 2:
            # scipy returns one bare NaN instead of a matrix when a column is constant (e.g.
            # an input with the same value everywhere). The same rank correlation, column by
            # column, leaves only that column's entries undefined.
            corr_matrix = pd.DataFrame(X).corr(method="spearman").to_numpy()
    else:
        corr_matrix = np.corrcoef(X.T)

    # Convert to DataFrame for easier handling
    corr_df = pd.DataFrame(corr_matrix, index=analysed_cols, columns=analysed_cols)

    # Find highly correlated pairs
    removed_features: list[str] = []
    keep_features = analysed_cols.copy()

    for i, feat_i in enumerate(analysed_cols):
        if feat_i in removed_features:
            continue

        for j, feat_j in enumerate(analysed_cols[i + 1 :], start=i + 1):
            if feat_j in removed_features:
                continue

            corr_val = abs(corr_matrix[i, j])

            if corr_val > threshold:
                mean_corr_i = np.mean(np.abs(corr_matrix[i, :]))
                mean_corr_j = np.mean(np.abs(corr_matrix[j, :]))

                to_remove = feat_i if mean_corr_i > mean_corr_j else feat_j

                if to_remove not in removed_features:
                    removed_features.append(to_remove)
                    keep_features.remove(to_remove)
                    other = feat_j if to_remove == feat_i else feat_i
                    logger.info(f"  Removing '{to_remove}' (|ρ| = {corr_val:.3f} with '{other}')")

    # Filter dataframe
    df_filtered = df.drop(columns=removed_features)

    logger.info(f"[FeatureFilter] Removed {len(removed_features)} correlated features")
    logger.info(f"[FeatureFilter] Remaining features: {len(keep_features) + len(protected_cols)}")

    return df_filtered, removed_features, corr_df

def diagnostic_pca_filter(
    df: pd.DataFrame,
    feature_cols: list[str],
    variance_threshold: float = 0.95,
    contrib_threshold: float = 0.01,
    protected_cols: list[str] | None = None,
) -> tuple[pd.DataFrame, dict]:
    """Use PCA to identify and remove low-contribution features.

    Does NOT project data — only uses PCA for feature selection.

    Process:

    1. Compute PCA on analysable features (not protected).
    2. Find the minimum ``N`` components explaining at least
       ``variance_threshold`` (e.g. 95%).
    3. For each feature, compute its total absolute contribution across the
       first ``N`` components.
    4. Remove features with contribution below ``contrib_threshold`` (e.g. 1%).

    Args:
        df: DataFrame with features
        feature_cols: List of column names to analyze
        variance_threshold: Cumulative variance to retain (default 0.95)
        contrib_threshold: Minimum feature contribution to keep (default 0.01)
        protected_cols: Columns to pass through without analysis.  When
            ``None`` (default), any column in *feature_cols* whose name starts
            with one of the :data:`_AUTO_PROTECTED_PREFIXES` (``tas_month_*``,
            ``pr_month_*``, ``kg_diag_*``, ``soil_*``) is automatically
            protected.

    Returns
    -------
        df_filtered: DataFrame with low-contribution features removed.
        diagnostics: Dict with the PCA object, component info, contributions, and removed/kept feature lists.
    """
    # Resolve protected cols
    if protected_cols is None:
        protected_cols = [
            c for c in feature_cols if any(c.startswith(p) for p in _AUTO_PROTECTED_PREFIXES)
        ]
    protected_set = set(protected_cols)
    analysed_cols = [c for c in feature_cols if c not in protected_set]

    if protected_cols:
        logger.info(
            f"[FeatureFilter] diagnostic_pca_filter: protecting "
            f"{len(protected_cols)} cols from removal "
            f"({protected_cols[:5]}{'...' if len(protected_cols) > 5 else ''})"
        )

    if not analysed_cols:
        logger.info("[FeatureFilter] No analysable columns — skipping PCA filter")
        return df.copy(), {
            "pca": None,
            "n_components": 0,
            "variance_explained": np.array([]),
            "cumulative_variance": np.array([]),
            "feature_contributions": pd.DataFrame(),
            "removed_features": [],
            "kept_features": feature_cols,
        }

    logger.info(
        f"[FeatureFilter] Running diagnostic PCA (variance threshold = {variance_threshold}) on {len(analysed_cols)} cols"
    )

    # Extract feature matrix (analysed cols only)
    X = df[analysed_cols].values

    # Fit PCA with all components
    pca = PCA(n_components=min(X.shape[0], X.shape[1]))
    pca.fit(X)

    # Find number of components for variance threshold
    cumsum_var = np.cumsum(pca.explained_variance_ratio_)
    n_components = np.argmax(cumsum_var >= variance_threshold) + 1

    logger.info(
        f"[FeatureFilter] {n_components} components explain {cumsum_var[n_components - 1]:.3f} of variance"
    )

    # Compute feature contributions to first N components
    loadings = pca.components_[:n_components, :]  # (n_components, n_features)
    feature_contrib = np.sum(np.abs(loadings), axis=0) / n_components

    # Normalise contributions to sum to 1
    feature_contrib_norm = feature_contrib / np.sum(feature_contrib)

    contrib_df = pd.DataFrame(
        {
            "feature": analysed_cols,
            "contribution": feature_contrib_norm,
        }
    ).sort_values("contribution", ascending=False)

    removed_features = contrib_df[contrib_df["contribution"] < contrib_threshold][
        "feature"
    ].tolist()
    keep_features = contrib_df[contrib_df["contribution"] >= contrib_threshold]["feature"].tolist()

    logger.info(
        f"[FeatureFilter] Removing {len(removed_features)} features with contribution < {contrib_threshold}"
    )
    for feat in removed_features:
        contrib_val = contrib_df.loc[contrib_df["feature"] == feat, "contribution"].values[0]
        logger.info(f"  Removing '{feat}' (contribution = {contrib_val:.4f})")

    df_filtered = df.drop(columns=removed_features)

    diagnostics = {
        "pca": pca,
        "n_components": n_components,
        "variance_explained": pca.explained_variance_ratio_,
        "cumulative_variance": cumsum_var,
        "feature_contributions": contrib_df,
        "removed_features": removed_features,
        "kept_features": keep_features + protected_cols,
    }

    logger.info(f"[FeatureFilter] Remaining features: {len(keep_features) + len(protected_cols)}")

    return df_filtered, diagnostics

def apply_block_pca_compression(
    df: pd.DataFrame,
    tas_monthly_cols: list[str],
    pr_monthly_cols: list[str],
    cache_dir: str,
    variance_threshold: float = 0.99,
    force_recompute: bool = False,
) -> tuple[pd.DataFrame, dict]:
    """Replace raw monthly T/P columns with PCA scores (block PCA).

    Fits (or loads from cache) one :class:`sklearn.decomposition.PCA` per
    block – one for the 12 temperature monthly columns, one for the 12
    precipitation monthly columns – retaining enough components to explain
    *variance_threshold* (default 99 %) of within-block variance.

    The raw monthly columns are dropped from the returned DataFrame and
    replaced by ``tas_month_PC1 … tas_month_PCN`` and
    ``pr_month_PC1 … pr_month_PCM``.

    PCA objects are serialised to
    ``{cache_dir}/block_pca_tas.pkl`` and
    ``{cache_dir}/block_pca_pr.pkl``.

    Args:
        df: DataFrame containing (at least) the monthly columns.
        tas_monthly_cols: Ordered list of the 12 temperature monthly column
            names (e.g. ``['tas_month_01', …, 'tas_month_12']``).
        pr_monthly_cols: Same for precipitation.
        cache_dir: Directory where PCA cache files are stored/read.
        variance_threshold: Fraction of variance to retain per block
            (default 0.99).
        force_recompute: If ``True``, ignore cached PCA objects and refit.

    Returns
    -------
        df_compressed: DataFrame with raw monthly cols replaced by PC cols.
        info: Dict with the fitted PCA objects (pca_tas, pca_pr), the per-block PC counts (n_tas_pcs, n_pr_pcs), and the new PC column-name lists (tas_pc_cols, pr_pc_cols).
    """
    cache_path = Path(cache_dir)
    cache_path.mkdir(parents=True, exist_ok=True)

    tas_cache = cache_path / "block_pca_tas.pkl"
    pr_cache = cache_path / "block_pca_pr.pkl"

    def _fit_or_load(cols: list[str], cache_file: Path, label: str) -> tuple[PCA, int]:
        """Return (fitted_pca, n_components_retained)."""
        if not force_recompute and cache_file.exists():
            with open(cache_file, "rb") as fh:
                pca_obj = pickle.load(fh)
            # Determine how many components cover variance_threshold
            cumvar = np.cumsum(pca_obj.explained_variance_ratio_)
            n_keep = int(np.argmax(cumvar >= variance_threshold)) + 1
            logger.info(
                f"[BlockPCA] Loaded cached PCA for {label} from {cache_file}  "
                f"({n_keep} PCs @ {variance_threshold * 100:.0f}% var)"
            )
            return pca_obj, n_keep

        X = df[cols].values.astype(np.float32)
        pca_obj = PCA(n_components=min(X.shape[0], X.shape[1]))
        pca_obj.fit(X)
        cumvar = np.cumsum(pca_obj.explained_variance_ratio_)
        n_keep = int(np.argmax(cumvar >= variance_threshold)) + 1
        logger.info(
            f"[BlockPCA] Fitted PCA for {label}: "
            f"{n_keep}/{len(cols)} PCs explain "
            f"{cumvar[n_keep - 1] * 100:.1f}% variance"
        )
        with open(cache_file, "wb") as fh:
            pickle.dump(pca_obj, fh, protocol=pickle.HIGHEST_PROTOCOL)
        logger.info(f"[BlockPCA] Saved PCA cache → {cache_file}")
        return pca_obj, n_keep

    # -----------------------------------------------------------------
    # Temperature block
    # -----------------------------------------------------------------
    if tas_monthly_cols and all(c in df.columns for c in tas_monthly_cols):
        pca_tas, n_tas = _fit_or_load(tas_monthly_cols, tas_cache, "tas")
        X_tas = df[tas_monthly_cols].values.astype(np.float32)
        scores_tas = pca_tas.transform(X_tas)[:, :n_tas]  # (n_cells, n_tas)
        # Whiten: divide by sqrt(eigenvalue) so each PC has unit variance (std=1)
        std_tas = np.sqrt(pca_tas.explained_variance_[:n_tas])
        scores_tas = scores_tas / std_tas[np.newaxis, :]
        tas_pc_cols = [f"tas_month_PC{i + 1}" for i in range(n_tas)]
        logger.info(
            f"[BlockPCA] Whitened T scores: std range "
            f"{scores_tas.std(axis=0).min():.3f}–{scores_tas.std(axis=0).max():.3f}"
        )
    else:
        logger.warning("[BlockPCA] WARNING: tas monthly cols missing — skipping T block PCA")
        pca_tas, n_tas, tas_pc_cols, scores_tas = None, 0, [], None

    # -----------------------------------------------------------------
    # Precipitation block
    # -----------------------------------------------------------------
    if pr_monthly_cols and all(c in df.columns for c in pr_monthly_cols):
        pca_pr, n_pr = _fit_or_load(pr_monthly_cols, pr_cache, "pr")
        X_pr = df[pr_monthly_cols].values.astype(np.float32)
        scores_pr = pca_pr.transform(X_pr)[:, :n_pr]  # (n_cells, n_pr)
        # Whiten: divide by sqrt(eigenvalue) so each PC has unit variance (std=1)
        std_pr = np.sqrt(pca_pr.explained_variance_[:n_pr])
        scores_pr = scores_pr / std_pr[np.newaxis, :]
        pr_pc_cols = [f"pr_month_PC{i + 1}" for i in range(n_pr)]
        logger.info(
            f"[BlockPCA] Whitened P scores: std range "
            f"{scores_pr.std(axis=0).min():.3f}–{scores_pr.std(axis=0).max():.3f}"
        )
    else:
        logger.warning("[BlockPCA] WARNING: pr monthly cols missing — skipping P block PCA")
        pca_pr, n_pr, pr_pc_cols, scores_pr = None, 0, [], None

    # -----------------------------------------------------------------
    # Build output DataFrame
    # -----------------------------------------------------------------
    drop_cols = [c for c in tas_monthly_cols if c in df.columns] + [
        c for c in pr_monthly_cols if c in df.columns
    ]
    df_out = df.drop(columns=drop_cols)

    if scores_tas is not None:
        for k, col in enumerate(tas_pc_cols):
            df_out[col] = scores_tas[:, k]

    if scores_pr is not None:
        for k, col in enumerate(pr_pc_cols):
            df_out[col] = scores_pr[:, k]

    logger.info(
        f"[BlockPCA] Replaced {len(drop_cols)} raw monthly cols with "
        f"{len(tas_pc_cols)} T-PCs + {len(pr_pc_cols)} P-PCs"
    )

    info = {
        "pca_tas": pca_tas,
        "pca_pr": pca_pr,
        "n_tas_pcs": n_tas,
        "n_pr_pcs": n_pr,
        "tas_pc_cols": tas_pc_cols,
        "pr_pc_cols": pr_pc_cols,
    }
    return df_out, info
