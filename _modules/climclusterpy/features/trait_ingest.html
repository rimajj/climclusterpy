Source code for climclusterpy.features.trait_ingest

"""Read vegetation-model trait output into a per-cell table.

Trait supervision (:mod:`climclusterpy.supervision.traits`) needs one row per grid
cell with one column per trait. Model output rarely arrives that way, so this
module accepts the two forms it actually comes in and produces that table:

* **a directory of trait NetCDFs** — one file per trait (``sla_2000_2019.nc``,
  ``height_2000_2019.nc``, …), gridded and usually with a time axis. Continuous
  traits are averaged over time; a categorical ``dominant_type`` is reduced by
  temporal mode (the type that prevails, not a meaningless average).
* **a prepared per-cell CSV** — already one row per cell, e.g. written by an
  earlier run or by the user's own aggregation.

Either way the rows come back aligned to the registry's valid cells, in the same
order as the clustering feature matrix, so supervision labels and features refer
to the same cell by position *and* by ``cell_id``.
"""

from __future__ import annotations

import glob
import os

import numpy as np
import pandas as pd

__all__ = ["DEFAULT_TRAITS", "DOMINANT_TYPE_COLUMN", "build_trait_features_df"]

#: Traits used by default — the continuous plant properties that drive the
#: vegetation model's carbon and water fluxes.
DEFAULT_TRAITS: tuple[str, ...] = ("sla", "wooddens", "height", "vegc")

#: The categorical trait, one-hot encoded rather than averaged.
DOMINANT_TYPE_COLUMN = "dominant_type"

# Column names that are cell identity, never a trait.
_METADATA_COLUMNS = frozenset({"cell_id", "latitude", "longitude", "lat", "lon"})

# Aliases people actually use for the same trait, so a directory of files named
# after the model's own conventions works without renaming anything.
_TRAIT_ALIASES: dict[str, tuple[str, ...]] = {
    "biomass": ("biomass", "vegc", "vegc_from_ind"),
    "vegc": ("vegc", "vegc_from_ind", "biomass"),
    "wooddens": ("wooddens", "wood_density", "wooddensity"),
    "sla": ("sla", "specific_leaf_area"),
    "height": ("height", "canopy_height"),
    "age": ("age", "stand_age"),
}


def _candidate_names(trait: str) -> tuple[str, ...]:
    """Return the file/variable-name spellings to accept for ``trait``."""
    return _TRAIT_ALIASES.get(trait.lower(), (trait.lower(),))


def _find_trait_file(directory: str, trait: str) -> str:
    """Return the NetCDF in ``directory`` holding ``trait``.

    Raises
    ------
    FileNotFoundError
        If no file matches, listing what the directory does contain — a missing
        trait must fail loudly rather than silently drop a supervision dimension.
    ValueError
        If several files match ambiguously.
    """
    available = sorted(os.path.basename(p) for p in glob.glob(os.path.join(directory, "*.nc")))
    matches: list[str] = []
    for name in _candidate_names(trait):
        matches += [f for f in available if f.lower().startswith(name)]
    if not matches:
        for name in _candidate_names(trait):
            matches += [f for f in available if name in f.lower()]

    unique = sorted(set(matches))
    if not unique:
        raise FileNotFoundError(
            f"No NetCDF for trait {trait!r} in {directory!r}. Looked for files starting with "
            f"{list(_candidate_names(trait))}; the directory contains {available or 'no .nc files'}."
        )
    if len(unique) > 1:
        raise ValueError(
            f"Ambiguous NetCDFs for trait {trait!r} in {directory!r}: {unique}. "
            "Keep one file per trait, or pass a prepared per-cell CSV instead."
        )
    return os.path.join(directory, unique[0])


def _reduce_over_time(da, *, categorical: bool):
    """Collapse a trait DataArray's non-spatial dims to one value per cell."""
    import xarray as xr

    time_dims = [
        d for d in da.dims if d.lower() not in ("lat", "latitude", "y", "lon", "longitude", "x")
    ]
    if not time_dims:
        return da

    if not categorical:
        return da.mean(dim=time_dims, skipna=True)

    # Categorical: the prevailing class, not an average of class codes.
    def _mode(values, axis=-1):
        values = np.moveaxis(values, axis, -1)
        out = np.full(values.shape[:-1], np.nan, dtype=float)
        flat_in = values.reshape(-1, values.shape[-1])
        flat_out = out.reshape(-1)
        for i, row in enumerate(flat_in):
            row = row[np.isfinite(row)]
            if row.size:
                codes, counts = np.unique(row, return_counts=True)
                flat_out[i] = codes[int(np.argmax(counts))]
        return out

    return xr.apply_ufunc(
        _mode,
        da,
        input_core_dims=[time_dims],
        kwargs={"axis": -1},
        vectorize=False,
        dask="forbidden",
    )


def _from_netcdf_dir(
    directory: str,
    registry,
    traits: list[str],
    include_dominant_type: bool,
) -> pd.DataFrame:
    """Build the per-cell trait table from a directory of gridded trait NetCDFs."""
    import xarray as xr

    cell_ids = registry.get_valid_original_ids()
    lats, lons = registry.get_valid_lat_lon()
    out = pd.DataFrame({"cell_id": cell_ids, "latitude": lats, "longitude": lons})

    wanted = [(t, False) for t in traits]
    if include_dominant_type:
        wanted.append((DOMINANT_TYPE_COLUMN, True))

    for trait, categorical in wanted:
        try:
            path = _find_trait_file(directory, trait)
        except FileNotFoundError:
            if categorical:
                # dominant_type is a bonus dimension, not a requirement: a model run
                # without it still yields perfectly good continuous supervision.
                continue
            raise

        with xr.open_dataset(path) as ds:
            var = _pick_variable(ds, trait, path)
            da = registry.align_to_original(ds[[var]].rename({var: trait}))[trait]
            reduced = _reduce_over_time(da, categorical=categorical)
            flat = registry.flatten_on_valid(reduced)

        flat = np.asarray(flat).reshape(len(out), -1)
        if flat.shape[1] != 1:
            raise ValueError(
                f"Trait {trait!r} from {os.path.basename(path)} reduced to {flat.shape[1]} values "
                "per cell; expected exactly 1. Check the file's non-spatial dimensions."
            )
        out[trait] = flat[:, 0]

    return out


def _pick_variable(ds, trait: str, path: str) -> str:
    """Return the data variable in ``ds`` holding ``trait``."""
    names = list(ds.data_vars)
    if not names:
        raise ValueError(f"{path!r} contains no data variables.")
    if len(names) == 1:
        return names[0]
    for candidate in _candidate_names(trait):
        for name in names:
            if name.lower() == candidate:
                return name
    for candidate in _candidate_names(trait):
        for name in names:
            if candidate in name.lower():
                return name
    raise ValueError(
        f"Cannot tell which variable in {path!r} holds trait {trait!r}: found {names}. "
        "Rename the variable to the trait name, or pass a prepared per-cell CSV."
    )


def _from_csv(
    csv_path: str,
    registry,
    traits: list[str],
    include_dominant_type: bool,
) -> pd.DataFrame:
    """Load and validate a prepared per-cell trait table, aligning it to the registry."""
    table = pd.read_csv(csv_path)

    if "cell_id" not in table.columns:
        raise ValueError(
            f"Trait CSV {csv_path!r} needs a 'cell_id' column so rows can be matched to grid "
            f"cells by identity; found columns {list(table.columns)}."
        )

    resolved: dict[str, str] = {}
    for trait in traits:
        match = next(
            (c for c in table.columns if c.lower() in _candidate_names(trait)),
            None,
        )
        if match is None:
            raise ValueError(
                f"Trait CSV {csv_path!r} has no column for trait {trait!r} (tried "
                f"{list(_candidate_names(trait))}); found columns {list(table.columns)}."
            )
        resolved[trait] = match

    keep = ["cell_id", *resolved.values()]
    if include_dominant_type and DOMINANT_TYPE_COLUMN in table.columns:
        keep.append(DOMINANT_TYPE_COLUMN)
    for coord in ("latitude", "longitude"):
        if coord in table.columns:
            keep.append(coord)

    table = table[keep].rename(columns={v: k for k, v in resolved.items()})

    # Reindex onto the registry's valid cells so the rows line up with the feature
    # matrix. An inner join would silently shrink the grid; a left join surfaces
    # exactly which cells the trait file is missing.
    cell_ids = registry.get_valid_original_ids()
    lats, lons = registry.get_valid_lat_lon()
    frame = pd.DataFrame({"cell_id": cell_ids, "latitude": lats, "longitude": lons})
    merged = frame.merge(
        table.drop(columns=[c for c in ("latitude", "longitude") if c in table.columns]),
        on="cell_id",
        how="left",
    )

    trait_cols = [c for c in merged.columns if c not in _METADATA_COLUMNS]
    missing = int(merged[trait_cols].isna().all(axis=1).sum())
    if missing:
        raise ValueError(
            f"Trait CSV {csv_path!r} covers only {len(merged) - missing} of {len(merged)} valid "
            f"grid cells ({missing} cells have no trait values at all). Supervision must cover "
            "every cell being clustered — check that the CSV's cell_id values come from the same "
            "grid as the climate input."
        )
    return merged

def build_trait_features_df(
    supervision_data: str,
    registry,
    *,
    traits: list[str] | tuple[str, ...] = DEFAULT_TRAITS,
    include_dominant_type: bool = True,
) -> pd.DataFrame:
    """Build a per-cell trait table from model output, in whichever form it arrives.

    Parameters
    ----------
    supervision_data : str
        Either a directory of per-trait NetCDF files, or a prepared per-cell CSV.
        The form is detected from the path (a ``.csv`` file vs a directory).
    registry : DataRegistry
        Registry for the grid being clustered. Defines which cells are valid and in
        what order, so the returned rows align with the clustering feature matrix.
    traits : sequence of str, optional
        Continuous traits to read. Defaults to :data:`DEFAULT_TRAITS`. Common
        aliases are accepted (``biomass`` matches a ``vegc`` file, and so on).
    include_dominant_type : bool, optional
        Also read the categorical ``dominant_type`` when present (default
        ``True``). It is optional in the input: a run without it still produces
        usable continuous supervision.

    Returns
    -------
    pandas.DataFrame
        Columns ``cell_id, latitude, longitude``, one column per trait, and
        ``dominant_type`` when available. One row per valid grid cell, in registry
        order.

    Raises
    ------
    FileNotFoundError
        If ``supervision_data`` does not exist, or a requested trait has no file.
    ValueError
        If a trait column/variable cannot be identified, or the input does not
        cover every valid cell.

    Examples
    --------
    >>> traits = build_trait_features_df(  # doctest: +SKIP
    ...     "/path/to/model_output/trait_grids",
    ...     registry,
    ...     traits=["sla", "wooddens", "height", "biomass"],
    ... )
    """
    traits = [str(t) for t in traits]
    if not traits:
        raise ValueError("At least one trait is required for trait supervision.")

    if not os.path.exists(supervision_data):
        raise FileNotFoundError(
            f"supervision_data path does not exist: {supervision_data!r}. Pass a directory of "
            "per-trait NetCDF files or a prepared per-cell CSV."
        )

    if os.path.isdir(supervision_data):
        return _from_netcdf_dir(supervision_data, registry, traits, include_dominant_type)

    if supervision_data.lower().endswith(".csv"):
        return _from_csv(supervision_data, registry, traits, include_dominant_type)

    raise ValueError(
        f"Cannot interpret supervision_data {supervision_data!r}: expected a directory of trait "
        "NetCDFs or a .csv per-cell table."
    )
