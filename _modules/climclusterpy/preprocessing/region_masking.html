Source code for climclusterpy.preprocessing.region_masking

"""
Region-based masking for climate and soil datasets.

This module provides functionality to mask out specific geographical regions
(defined by Köppen-Geiger climate zones or Ecoregion IDs) from the DataRegistry
and all associated datasets.  After masking, cells belonging to the specified
regions have their valid-mask entry set to ``False`` and the corresponding
positions in every dataset variable are replaced with NaN (or the appropriate
sentinel for integer variables).

Typical usage
-------------
>>> from climclusterpy.data.loader import load_climate_and_soil_data, DataPaths
>>> from climclusterpy.supervision.koppen_geiger import perform_koppen_geiger_assignment, KG_LABELS
>>> from climclusterpy.preprocessing.region_masking import mask_regions_by_koppen, mask_regions_by_ecoregion

>>> data, registry = load_climate_and_soil_data(paths)
>>> data, registry = mask_regions_by_koppen(
...     data, registry,
...     mask_kg_labels=["BWh", "BWk"],  # mask hot + cold desert
... )
"""

from __future__ import annotations

import datetime
import logging
import warnings

import numpy as np
import pandas as pd
import xarray as xr

from climclusterpy.data.loader import INT_SENTINEL
from climclusterpy.data.registry import DataRegistry, _find_coord_names

logger = logging.getLogger(__name__)

def _apply_mask_to_datasets(
    data: dict[str, xr.Dataset],
    registry: DataRegistry,
    cells_to_mask: np.ndarray,
) -> dict[str, xr.Dataset]:
    """Set masked cells to NaN/sentinel in all datasets and update registry valid mask.

    Args:
        data: Dict of datasets, each on the original grid.
        registry: DataRegistry whose valid mask will be modified in-place.
        cells_to_mask: 1D boolean array of length ``n_valid_cells`` (True = mask out).
            Alternatively a 1D integer array of *flat original_cell_id* values to remove.

    Returns
    -------
        Updated ``data`` dict with NaN/sentinel applied to masked cells.
    """
    valid_mask = registry.get_valid_mask()  # (nlat, nlon) bool

    # Build a 2D boolean array of cells to nullify on the full spatial grid
    nullify_2d = np.zeros_like(valid_mask, dtype=bool)

    if cells_to_mask.dtype == bool:
        # cells_to_mask indexes into *valid* cells (1D, size = n_valid)
        valid_flat_indices = np.flatnonzero(valid_mask)
        if len(cells_to_mask) != len(valid_flat_indices):
            raise ValueError(
                f"cells_to_mask bool length ({len(cells_to_mask)}) does not match "
                f"number of valid cells ({len(valid_flat_indices)})."
            )
        flat_to_nullify = valid_flat_indices[cells_to_mask]
    else:
        # cells_to_mask is an array of original_cell_id values (integers)
        orig = registry.get_original_grid()
        id_to_flat = {oid: fi for fi, oid in enumerate(orig.cell_ids.ravel())}
        flat_to_nullify = np.array(
            [id_to_flat[oid] for oid in cells_to_mask if oid in id_to_flat],
            dtype=np.intp,
        )

    # Update the 2D nullify mask using flat indices into the ravelled spatial grid
    nullify_flat = np.zeros(valid_mask.size, dtype=bool)
    nullify_flat[flat_to_nullify] = True
    nullify_2d = nullify_flat.reshape(valid_mask.shape)

    # Update registry valid mask in-place
    registry._valid_mask = valid_mask & (~nullify_2d)

    # Also update cell IDs: set masked cells to -1 in OriginalGrid
    orig = registry.get_original_grid()
    orig.cell_ids[nullify_2d] = -1
    # Re-assign consecutive IDs to the remaining valid cells
    new_valid_mask = registry._valid_mask
    new_ids = np.full_like(orig.cell_ids, -1)
    new_ids[new_valid_mask] = np.arange(int(new_valid_mask.sum()), dtype=np.int64)
    orig.cell_ids[:] = new_ids

    n_masked = int(nullify_2d.sum())
    n_remaining = int(new_valid_mask.sum())
    logger.info(f"[RegionMasking] Masked {n_masked} cells. {n_remaining} valid cells remain.")

    # Apply NaN / sentinel to every dataset variable at masked locations
    masked_data: dict[str, xr.Dataset] = {}
    for key, ds in data.items():
        masked_data[key] = _nullify_cells_in_dataset(ds, nullify_2d)

    return masked_data


def _nullify_cells_in_dataset(ds: xr.Dataset, nullify_2d: np.ndarray) -> xr.Dataset:
    """Set cells indicated by ``nullify_2d`` to NaN (or sentinel) in ``ds``."""
    lat_name, lon_name = _find_coord_names(ds)
    # Build a 2D DataArray mask aligned to ds
    mask_da = xr.DataArray(
        nullify_2d,
        dims=(lat_name, lon_name),
        coords={lat_name: ds[lat_name], lon_name: ds[lon_name]},
    )

    new_vars: dict[str, xr.DataArray] = {}
    for var_name, da in ds.data_vars.items():
        if lat_name not in da.dims or lon_name not in da.dims:
            new_vars[var_name] = da
            continue

        # Broadcast the 2D mask to match the full dimensions of da
        broadcast_mask, _ = xr.broadcast(mask_da, da)
        broadcast_mask = broadcast_mask.transpose(*da.dims)

        if np.issubdtype(da.dtype, np.integer) or var_name == "soil_code":
            # Use integer sentinel for soil_code / integer variables
            nulled = da.where(~broadcast_mask, other=INT_SENTINEL)
        else:
            nulled = da.where(~broadcast_mask, other=np.nan)

        new_vars[var_name] = nulled

    # Preserve non-data variables (e.g. original_cell_id) as-is
    for var_name in ds.data_vars:
        if var_name not in new_vars:
            new_vars[var_name] = ds[var_name]

    return ds.assign(new_vars)

def mask_regions_by_assignment(
    data: dict[str, xr.Dataset],
    registry: DataRegistry,
    assignment_df: pd.DataFrame,
    mask_cluster_labels: list[int],
    *,
    cell_id_column: str = "cell_id",
    cluster_label_column: str = "cluster_label",
) -> tuple[dict[str, xr.Dataset], DataRegistry]:
    """Mask out cells based on a pre-computed assignment table.

    Works with any region assignment table in the package's standard format, one row per
    valid cell: its ``cell_id`` (the registry's original cell id) and its integer
    ``cluster_label`` (a Köppen–Geiger code, an ecoregion id, ...).

    The function updates the registry's valid-mask and nullifies masked cells
    in *all* datasets.

    Parameters
    ----------
    data : dict of str to xarray.Dataset
        The datasets, aligned to the original grid.
    registry : DataRegistry
        The original grid.
    assignment_df : pandas.DataFrame
        At least the ``cell_id`` and ``cluster_label`` columns.
    mask_cluster_labels : sequence of int
        The labels to mask out.
    cell_id_column, cluster_label_column : str, optional
        The column names (default ``"cell_id"`` and ``"cluster_label"``).

    Returns
    -------
    tuple
        ``(updated_data, updated_registry)``.
    """
    if cell_id_column not in assignment_df.columns:
        raise KeyError(
            f"assignment_df missing required column '{cell_id_column}'. "
            f"Available: {list(assignment_df.columns)}"
        )
    if cluster_label_column not in assignment_df.columns:
        raise KeyError(
            f"assignment_df missing required column '{cluster_label_column}'. "
            f"Available: {list(assignment_df.columns)}"
        )

    if not mask_cluster_labels:
        warnings.warn("mask_cluster_labels is empty. Nothing masked.", UserWarning, stacklevel=2)
        return data, registry

    to_mask = assignment_df[cluster_label_column].isin(mask_cluster_labels)
    n_to_mask = int(to_mask.sum())
    logger.info(
        f"[RegionMasking] Masking {n_to_mask} of {len(assignment_df)} assigned cells "
        f"(labels: {sorted(set(mask_cluster_labels))})."
    )

    if n_to_mask == 0:
        warnings.warn(
            f"No cells found matching labels {mask_cluster_labels}. Nothing masked.",
            UserWarning,
            stacklevel=2,
        )
        return data, registry

    cell_ids_to_mask = assignment_df.loc[to_mask, cell_id_column].to_numpy()
    updated_data = _apply_mask_to_datasets(data, registry, cell_ids_to_mask)
    return updated_data, registry

def compute_koppen_classification_from_data(
    data: dict[str, xr.Dataset],
    registry: DataRegistry,
    reference_period: tuple[str, str] | None = None,
    use_last_n_years: int = 30,
    chunk_time: int | None = None,
    cold_threshold: float = 0.0,
) -> pd.DataFrame:
    """Compute Köppen-Geiger classification for all valid cells using temperature and
    precipitation data from ``data``.

    Computes monthly climatologies (mean over ``reference_period`` or full record)
    and delegates to :func:`climclusterpy.supervision.koppen_geiger.assign_cells_to_koppen_geiger`
    for the actual classification logic.

    Args:
        data: Dict containing ``"temperature"`` and ``"precipitation"`` datasets
            aligned to the original grid.
        registry: Initialized DataRegistry.
        reference_period: Optional (start_date, end_date) ISO strings to restrict the
            climatology period (e.g. ``("1981-01-01", "2010-12-31")``). If not set,
            defaults to the most recent ``use_last_n_years`` years based on the
            temperature dataset's time coordinate.
        use_last_n_years: Number of most recent years to use when ``reference_period``
            is not provided (default: 30). Set to 0 to use the full record.
        chunk_time: Optional chunk size (number of time steps) to apply before
            climatology computation. This can reduce memory usage for very large
            datasets. If None, leaves chunking as-is.
        cold_threshold: C/D boundary temperature threshold in °C (default 0.0).

    Returns
    -------
        DataFrame with columns: ``cell_id``, ``latitude``, ``longitude``,
        ``cluster_label`` (integer KG code 1–30).
    """
    from climclusterpy.features.climate import (
        monthly_climatology_from_daily,
        to_celsius,
    )
    from climclusterpy.supervision.koppen_geiger import assign_cells_to_koppen_geiger

    if "temperature" not in data:
        raise KeyError("'temperature' key required in data dict for KG classification.")
    if "precipitation" not in data:
        raise KeyError("'precipitation' key required in data dict for KG classification.")

    temp_ds = data["temperature"]
    prec_ds = data["precipitation"]

    if chunk_time is not None:
        # Keep spatial dims unchunked; only chunk along time
        temp_ds = temp_ds.chunk({"time": int(chunk_time)})
        prec_ds = prec_ds.chunk({"time": int(chunk_time)})

    # Restrict to reference period if requested; otherwise default to last N years
    if reference_period is not None:
        start, end = reference_period
        temp_ds = temp_ds.sel(time=slice(start, end))
        prec_ds = prec_ds.sel(time=slice(start, end))
        logger.info(
            f"[RegionMasking] Using reference period {start} – {end} for KG classification."
        )
    elif use_last_n_years and use_last_n_years > 0:
        # Works on both numpy dates and cftime calendars (e.g. noleap): both subtract a
        # timedelta and both reset the time of day with replace().
        time_index = temp_ds.indexes["time"]
        t1 = time_index[-1]
        try:
            start = (t1 - datetime.timedelta(days=int(use_last_n_years * 365))).replace(
                hour=0, minute=0, second=0, microsecond=0
            )
        except TypeError as e:
            raise ValueError(
                f"use_last_n_years={use_last_n_years} needs calendar dates on the temperature "
                f"time axis, but it holds {time_index.dtype} values. Decode the time axis "
                "(xarray.decode_cf), pass reference_period=(start, end), or set "
                "use_last_n_years=0 to use the full record."
            ) from e
        temp_ds = temp_ds.sel(time=slice(start, t1))
        prec_ds = prec_ds.sel(time=slice(start, t1))
        logger.info(
            f"[RegionMasking] Using most recent {use_last_n_years} years "
            f"({str(start)[:10]} – {str(t1)[:10]}) for KG classification."
        )
    else:
        logger.info("[RegionMasking] Using the full record for KG classification.")

    temp_var = next(v for v in temp_ds.data_vars if v != "original_cell_id")
    prec_var = next(v for v in prec_ds.data_vars if v != "original_cell_id")

    # Monthly climatologies via the shared preprocessing helper
    logger.info("[RegionMasking] Computing monthly climatologies for KG classification…")
    temp_monthly, prec_monthly = monthly_climatology_from_daily(
        to_celsius(temp_ds[temp_var]),
        prec_ds[prec_var],
    )
    temp_monthly = temp_monthly.compute()
    prec_monthly = prec_monthly.compute()

    # Ensure month is the first dimension
    temp_monthly = temp_monthly.transpose("month", *[d for d in temp_monthly.dims if d != "month"])
    prec_monthly = prec_monthly.transpose("month", *[d for d in prec_monthly.dims if d != "month"])

    # Get valid cell coordinates and IDs from registry
    valid_lats, valid_lons = registry.get_valid_lat_lon()
    valid_ids = registry.get_valid_original_ids()

    lat_dim = next(d for d in temp_monthly.dims if d != "month")
    lon_dim = [d for d in temp_monthly.dims if d != "month"][1]

    logger.info(f"[RegionMasking] Extracting monthly features for {len(valid_lats)} valid cells…")

    # Vectorised extraction of monthly values at valid cell coordinates
    temp_arr = temp_monthly.sel(
        {
            lat_dim: xr.DataArray(valid_lats, dims="cell"),
            lon_dim: xr.DataArray(valid_lons, dims="cell"),
        },
        method="nearest",
    ).values  # (12, n_cells)

    prec_arr = prec_monthly.sel(
        {
            lat_dim: xr.DataArray(valid_lats, dims="cell"),
            lon_dim: xr.DataArray(valid_lons, dims="cell"),
        },
        method="nearest",
    ).values  # (12, n_cells)

    # Build the features DataFrame expected by assign_cells_to_koppen_geiger
    feat_data: dict = {
        "cell_id": valid_ids,
        "latitude": valid_lats,
        "longitude": valid_lons,
    }
    for m_idx in range(12):
        feat_data[f"tas_month_{m_idx + 1:02d}"] = temp_arr[m_idx]
        feat_data[f"pr_month_{m_idx + 1:02d}"] = prec_arr[m_idx]

    features_df = pd.DataFrame(feat_data)

    # Delegate classification to the canonical implementation
    logger.info("[RegionMasking] Running KG classification…")
    return assign_cells_to_koppen_geiger(features_df, cold_threshold=cold_threshold)

def mask_regions_by_koppen(
    data: dict[str, xr.Dataset],
    registry: DataRegistry,
    mask_kg_labels: list[str],
    reference_period: tuple[str, str] | None = None,
    use_last_n_years: int = 30,
    chunk_time: int | None = None,
    cold_threshold: float = 0.0,
) -> tuple[dict[str, xr.Dataset], DataRegistry]:
    """Mask out cells belonging to specified Köppen-Geiger climate zones.

    Computes a KG climatology classification over the loaded datasets, identifies
    all valid cells that fall into ``mask_kg_labels``, removes them from the
    registry's valid mask, and sets their values to NaN (float) or the integer
    sentinel in all datasets.

    Args:
        data: Dict of datasets keyed by variable name, aligned to the original grid.
        registry: Initialized DataRegistry.
        mask_kg_labels: List of KG labels to mask out, e.g. ``["BWh", "BWk"]``.
        reference_period: Optional (start, end) ISO date strings to restrict the
            climatology used for classification.
        use_last_n_years: If ``reference_period`` is not provided, use the most
            recent N years (default 30). Set to 0 for full record.
        chunk_time: Optional dask chunk size along ``time`` used during KG
            climatology computation.
        cold_threshold: C/D boundary in °C (default 0.0).

    Returns
    -------
        Tuple of (updated_data, updated_registry).  The registry and data dicts
        are modified in-place; the same objects are also returned for convenience.
    """
    from climclusterpy.supervision.koppen_geiger import KG_LABELS, LABEL_TO_CODE

    # Validate labels
    valid_labels = set(KG_LABELS.values())
    unknown = [l for l in mask_kg_labels if l not in valid_labels]
    if unknown:
        raise ValueError(f"Unknown KG labels: {unknown}. Valid labels are: {sorted(valid_labels)}")

    logger.info(f"[RegionMasking] Masking KG zones: {mask_kg_labels}")

    kg_df = compute_koppen_classification_from_data(
        data,
        registry,
        reference_period=reference_period,
        use_last_n_years=use_last_n_years,
        chunk_time=chunk_time,
        cold_threshold=cold_threshold,
    )

    mask_codes = [LABEL_TO_CODE[l] for l in mask_kg_labels]
    return mask_regions_by_assignment(
        data,
        registry,
        kg_df,
        mask_cluster_labels=mask_codes,
    )

def mask_regions_by_ecoregion(
    data: dict[str, xr.Dataset],
    registry: DataRegistry,
    mask_ecoregion_ids: list[int],
    shapefile_path: str,
    ecoregion_id_column: str = "ECO_ID",
) -> tuple[dict[str, xr.Dataset], DataRegistry]:
    """Mask out cells belonging to specified ecoregion IDs.

    Performs a spatial join between the valid grid cells and ecoregion polygons,
    then removes cells whose ecoregion ID is in ``mask_ecoregion_ids``.

    Args:
        data: Dict of datasets aligned to the original grid.
        registry: Initialized DataRegistry.
        mask_ecoregion_ids: List of integer ecoregion IDs to mask out.
        shapefile_path: Path to the ecoregions shapefile (.shp).
        ecoregion_id_column: Column in the shapefile containing ecoregion IDs.

    Returns
    -------
        Tuple of (updated_data, updated_registry).
    """
    from climclusterpy.supervision.ecoregions import (
        assign_cells_to_ecoregions,
        load_ecoregions_shapefile,
    )

    logger.info(
        f"[RegionMasking] Masking ecoregion IDs: {mask_ecoregion_ids} "
        f"(column: {ecoregion_id_column})"
    )

    ecoregions_gdf = load_ecoregions_shapefile(shapefile_path)
    eco_df = assign_cells_to_ecoregions(registry, ecoregions_gdf, ecoregion_id_column)

    to_mask_bool = eco_df["cluster_label"].isin(mask_ecoregion_ids).values

    n_to_mask = int(to_mask_bool.sum())
    logger.info(
        f"[RegionMasking] {n_to_mask} of {len(eco_df)} valid cells will be masked "
        f"(ecoregion IDs: {mask_ecoregion_ids})."
    )

    if n_to_mask == 0:
        warnings.warn(
            f"No cells found matching ecoregion IDs {mask_ecoregion_ids}. Nothing masked.",
            UserWarning,
            stacklevel=2,
        )
        return data, registry

    updated_data = _apply_mask_to_datasets(data, registry, to_mask_bool)
    return updated_data, registry

def save_masked_datasets(
    data: dict[str, xr.Dataset],
    output_dir: str,
    compression_level: int = 4,
) -> dict[str, str]:
    """Save all masked datasets as NetCDF4 files.

    Args:
        data: Dict of datasets to save.
        output_dir: Directory in which to write NetCDF files.
        compression_level: zlib compression level 0–9 (default 4).

    Returns
    -------
        Dict mapping variable name to saved file path.
    """
    import os

    os.makedirs(output_dir, exist_ok=True)
    saved: dict[str, str] = {}

    for key, ds in data.items():
        out_path = os.path.join(output_dir, f"{key}.nc")
        logger.info(f"[RegionMasking] Saving {key} → {out_path}")

        # Build per-variable encoding
        encoding = {}
        for var in ds.data_vars:
            da = ds[var]
            enc: dict = {"zlib": True, "complevel": compression_level}
            if np.issubdtype(da.dtype, np.integer):
                enc["dtype"] = "int32"
                enc["_FillValue"] = INT_SENTINEL
            else:
                enc["dtype"] = "float32"
                enc["_FillValue"] = np.float32(np.nan)
            encoding[var] = enc

        # Compute dask arrays before writing to avoid scheduler issues
        ds_computed = ds.compute()
        ds_computed.to_netcdf(out_path, format="NETCDF4", encoding=encoding)
        saved[key] = out_path

    logger.info(f"[RegionMasking] All datasets saved to {output_dir}")
    return saved
