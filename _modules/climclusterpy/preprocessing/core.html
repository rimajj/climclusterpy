from __future__ import annotations

"""
Lightweight preprocessing helpers used by tests and initial CLI wiring.

Contracts (minimal):
- align_coords(ds: xr.Dataset, grid: GridSpecLike) -> xr.Dataset
  GridSpecLike: dict with keys {latitudes, longitudes} or an xr.Dataset with lat/lon coords.

- build_features(ds: xr.Dataset, spec: VariableSpecLike|list[VariableSpecLike]) -> xr.Dataset
  VariableSpecLike: dict with keys {name, temporal_agg? in {monthly, seasonal, yearly},
  normalize?: bool}. Returns a Dataset containing engineered feature variables with
  dims (latitude, longitude, <agg dims>). Uses simple mean aggregations.
"""
from collections.abc import Iterable, Mapping
from typing import Any

import numpy as np
import pandas as pd
import xarray as xr

from climclusterpy.data.coords import standardize_dataset_coordinates

from ..data.coords import LAT_NAMES, LON_NAMES


def _detect_coord_names(ds: xr.Dataset) -> tuple[str, str]:
    lat = next((n for n in LAT_NAMES if n in ds.coords), None)
    lon = next((n for n in LON_NAMES if n in ds.coords), None)
    if lat is None or lon is None:
        raise ValueError(f"Could not find lat/lon coords. Available: {list(ds.coords)}")
    return lat, lon


def _coerce_time_coord(da: xr.DataArray, time_dim: str = "time") -> xr.DataArray:
    """Return ``da`` with calendar dates on ``time_dim``.

    A time axis that is already dates (numpy or cftime) is kept. An integer index is read
    as consecutive months starting January 2000.
    """
    try:
        _ = da[time_dim].dt  # xarray raises AttributeError when the axis is not dates
        return da
    except (AttributeError, TypeError):
        # ints -> monthly datetimes
        periods = int(da.sizes.get(time_dim, da[time_dim].size))
        new_time = pd.date_range("2000-01-01", periods=periods, freq="MS")
        return da.assign_coords({time_dim: new_time})

def align_coords(ds: xr.Dataset, grid: Any) -> xr.Dataset:
    """Align dataset spatial coordinates to a provided grid.

    The dataset's coordinates are first standardized (renamed to ``latitude`` /
    ``longitude``, rounded, longitudes above 180 shifted into [-180, 180]; see
    :func:`~climclusterpy.data.coords.standardize_dataset_coordinates`). Every
    variable is then sampled at the target coordinates by **nearest-neighbour**
    selection (:meth:`xarray.Dataset.interp` with ``method="nearest"``): values are
    picked, never averaged or interpolated. A target point outside the dataset's
    coordinate range gets NaN.

    Parameters
    ----------
    ds : xarray.Dataset
        The data to align. Needs a latitude and a longitude coordinate under one of
        the accepted names (``latitude``, ``lat``, ``Latitude``, ``LAT`` and the
        longitude equivalents), in degrees.
    grid : dict or xarray.Dataset or xarray.DataArray
        The target grid, either

        * a mapping with keys ``"latitudes"`` and ``"longitudes"``, each a 1-D array
          of coordinate values in degrees, or
        * an xarray object whose latitude and longitude coordinates (under any
          accepted name) are taken as the target.

        The target values are used as given: they are not rounded or shifted, so
        give longitudes in [-180, 180] to match the standardized dataset.

    Returns
    -------
    xarray.Dataset
        ``ds`` on the target grid, with coordinates named ``latitude`` and
        ``longitude``.

    Raises
    ------
    ValueError
        If ``ds`` or an xarray ``grid`` has no recognizable latitude/longitude
        coordinate.
    TypeError
        If ``grid`` is neither a mapping nor an xarray object.
    """
    ds = standardize_dataset_coordinates(ds)
    lat_name, lon_name = _detect_coord_names(ds)

    if isinstance(grid, Mapping):
        if grid.get("latitudes") is None or grid.get("longitudes") is None:
            raise ValueError(
                f"grid dict must contain 'latitudes' and 'longitudes'; got keys {sorted(grid)}"
            )
        lats = np.asarray(grid["latitudes"])
        lons = np.asarray(grid["longitudes"])
    elif isinstance(grid, (xr.Dataset, xr.DataArray)):
        # detect names on grid as well
        if isinstance(grid, xr.DataArray):
            gds = grid.to_dataset(name="_tmp")
        else:
            gds = grid
        g_lat, g_lon = _detect_coord_names(gds)
        lats = gds[g_lat].values
        lons = gds[g_lon].values
    else:
        raise TypeError("grid must be a dict or xarray object with lat/lon coords")

    return ds.interp({lat_name: lats, lon_name: lons}, method="nearest")




def _aggregate_temporal(da: xr.DataArray, temporal_agg: str | None) -> xr.DataArray:
    if temporal_agg is None:
        return da
    da = _coerce_time_coord(da)
    if temporal_agg == "monthly":
        out = (
            da.assign_coords(month=da["time"].dt.month).groupby(["month"]).mean("time", skipna=True)
        )
        # order dims as (lat, lon, month)
        lat_name, lon_name = _detect_coord_names(out.to_dataset())
        return out.transpose(lat_name, lon_name, "month")
    if temporal_agg == "seasonal":
        season = (da["time"].dt.month - 1) // 3 + 1
        out = da.assign_coords(season=season).groupby(["season"]).mean("time", skipna=True)
        lat_name, lon_name = _detect_coord_names(out.to_dataset())
        return out.transpose(lat_name, lon_name, "season")
    if temporal_agg == "yearly":
        out = da.groupby(da["time"].dt.year).mean("time", skipna=True)
        lat_name, lon_name = _detect_coord_names(out.to_dataset())
        return out.transpose(lat_name, lon_name, "year")
    raise ValueError(f"Unknown temporal_agg: {temporal_agg}")

def build_features(ds: xr.Dataset, spec: Any) -> xr.Dataset:
    """Construct simple feature tensors from variables defined in ``spec``.

    ``spec`` can be a dict ``{name, temporal_agg?, normalize?}`` or a list of
    those. For each entry, produce a variable named ``name`` if no aggregation
    is given, otherwise ``f"{name}_{temporal_agg}"``. If ``normalize=True``,
    standardize each grid cell along its non-spatial axis (see ``spec`` below).

    Parameters
    ----------
    ds : xarray.Dataset
        Input data with a latitude and a longitude coordinate (any accepted name;
        they are standardized to ``latitude`` / ``longitude`` first) and, for any
        entry with a ``temporal_agg``, a ``time`` dimension. A ``time`` coordinate
        that is not datetime-like (e.g. plain integers) is replaced by monthly
        dates starting 2000-01-01, i.e. the steps are assumed to be consecutive
        months.
    spec : dict or list of dict
        One feature description, or a list of them. Each is a mapping with keys

        * ``"name"`` (str, required) — the data variable in ``ds`` to use;
        * ``"temporal_agg"`` (str, optional) — ``"monthly"`` (mean per calendar
          month, new dimension ``month`` 1–12), ``"seasonal"`` (mean per
          three-month block Jan–Mar, Apr–Jun, Jul–Sep, Oct–Dec, new dimension
          ``season`` 1–4) or ``"yearly"`` (mean per calendar year, new dimension
          ``year``). Means skip NaN. Omit it (or pass ``None``) to copy the
          variable unchanged;
        * ``"normalize"`` (bool, optional, default ``False``) — subtract the mean
          and divide by the (population) standard deviation of each grid cell along
          its one dimension that is neither latitude nor longitude: the new
          ``month``/``season``/``year`` axis after an aggregation, otherwise e.g.
          ``time``. A zero standard deviation is replaced by 1. A variable with only
          the two spatial dimensions is left unchanged. (Before 0.3.0 the **last**
          axis was used, which for a ``(time, latitude, longitude)`` variable
          without an aggregation is longitude.)

    Returns
    -------
    xarray.Dataset
        One variable per entry, aggregated variables with dimensions
        ``(latitude, longitude, <month|season|year>)``. Variables without an
        aggregation keep their original dimension order.

    Raises
    ------
    ValueError
        If an entry has no ``name`` or names a variable not in ``ds``, if
        ``temporal_agg`` is not one of the three accepted values, if ``ds`` has
        no recognizable latitude/longitude coordinate, or if ``normalize`` is asked
        for a variable with more than one non-spatial dimension.
    """
    if isinstance(spec, Mapping):
        items: Iterable[Mapping[str, Any]] = [spec]  # type: ignore
    else:
        items = spec  # type: ignore

    ds = standardize_dataset_coordinates(ds)
    out_vars: dict[str, tuple[tuple[str, ...], np.ndarray]] = {}
    coords: dict[str, Any] = {}
    lat_name, lon_name = _detect_coord_names(ds)

    for entry in items:
        name = entry.get("name")
        if not name or name not in ds:
            raise ValueError(f"Variable '{name}' not found in dataset")
        temporal_agg = entry.get("temporal_agg")
        normalize = bool(entry.get("normalize", False))

        da = ds[name]
        da = _aggregate_temporal(da, temporal_agg)
        # ensure dims order (lat, lon, [agg])
        dims = da.dims
        data = da.values

        # optional per-cell standardization along the one non-spatial axis, found by NAME:
        # by position it was the last axis, which is longitude for (time, lat, lon) data
        if normalize:
            other = [i for i, d in enumerate(dims) if d not in (lat_name, lon_name)]
            if len(other) > 1:
                raise ValueError(
                    f"normalize=True needs one non-spatial dimension to standardize along; "
                    f"'{name}' has {[dims[i] for i in other]}. Aggregate it first "
                    "(temporal_agg) or select one."
                )
            if other:
                axis = other[0]
                mu = np.nanmean(data, axis=axis, keepdims=True)
                sigma = np.nanstd(data, axis=axis, ddof=0, keepdims=True)
                sigma = np.where(sigma == 0, 1.0, sigma)
                data = (data - mu) / sigma

        var_out_name = name if temporal_agg is None else f"{name}_{temporal_agg}"
        out_vars[var_out_name] = (dims, data)
        # collect coords (shared)
        for c in da.coords:
            coords.setdefault(c, da.coords[c])

    return xr.Dataset(out_vars, coords=coords)
