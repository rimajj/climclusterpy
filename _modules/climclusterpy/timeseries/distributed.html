"""
Distributed Cell Selection for Clusters — Memory-Efficient Streaming Version.

This module implements the 'distributed' timeseries type where NPATCH different cells
are selected per cluster (instead of duplicating one representative cell N_REPEAT times).

Key design principles for memory efficiency:
- Only unique grid cells are extracted (deduplication), then mapped back to all slots
- Variables are processed ONE at a time (never all 5 in memory simultaneously)
- Data is streamed from dask chunks, never materializing the full 3D grid
- Parallel time-chunk extraction uses threads for fast I/O-bound reads

Memory footprint for n=18026, npatch=50:
  Old approach:  ~958 GB peak (full grid + extracted arrays for all vars)
  New approach:  ~12 GB peak (one var's unique cells + output buffer)
"""

import logging
import os
import time as _time

import numpy as np
import pandas as pd
import xarray as xr
from joblib import Parallel, delayed

from .. import _settings

logger = logging.getLogger(__name__)


def _select_cells_for_cluster(
    cluster_id: int,
    cluster_cells: pd.DataFrame,
    npatch: int,
    rng: np.random.Generator,
) -> pd.DataFrame:
    """
    Select NPATCH cells from a cluster, with repetition if needed.

    Logic:
    - If cluster has >= NPATCH cells: randomly select NPATCH unique cells
    - If cluster has < NPATCH cells:
        * Repeat each cell floor(NPATCH / n_cells) times
        * Randomly select (NPATCH % n_cells) additional cells for remainder

    Args:
        cluster_id: Cluster label
        cluster_cells: DataFrame with cells in this cluster (columns: cell_id, latitude, longitude)
        npatch: Number of cells to select
        rng: NumPy random generator for reproducible selection

    Returns
    -------
        DataFrame with NPATCH rows, each row is a selected cell with columns:
        cluster_label, cell_id, latitude, longitude, cell_index_in_cluster
    """
    n_cells = len(cluster_cells)

    if n_cells == 0:
        raise ValueError(f"Cluster {cluster_id} has no cells")

    selected_rows = []

    if n_cells >= npatch:
        # Enough cells: randomly sample NPATCH unique cells
        sampled_indices = rng.choice(n_cells, size=npatch, replace=False)
        for cell_idx, idx in enumerate(sampled_indices):
            row = cluster_cells.iloc[idx].copy()
            selected_rows.append(
                {
                    "cluster_label": cluster_id,
                    "cell_id": int(row["cell_id"]),
                    "latitude": float(row["latitude"]),
                    "longitude": float(row["longitude"]),
                    "cell_index_in_cluster": cell_idx,
                    "is_repeated": False,
                }
            )
    else:
        # Not enough cells: repeat strategically
        n_full_repeats = npatch // n_cells  # Each cell appears this many times
        n_remainder = npatch % n_cells  # Additional cells needed

        cell_idx = 0

        # Add each cell n_full_repeats times
        for _ in range(n_full_repeats):
            for i in range(n_cells):
                row = cluster_cells.iloc[i].copy()
                selected_rows.append(
                    {
                        "cluster_label": cluster_id,
                        "cell_id": int(row["cell_id"]),
                        "latitude": float(row["latitude"]),
                        "longitude": float(row["longitude"]),
                        "cell_index_in_cluster": cell_idx,
                        "is_repeated": True,
                    }
                )
                cell_idx += 1

        # Add remainder cells (randomly selected)
        if n_remainder > 0:
            remainder_indices = rng.choice(n_cells, size=n_remainder, replace=False)
            for idx in remainder_indices:
                row = cluster_cells.iloc[idx].copy()
                selected_rows.append(
                    {
                        "cluster_label": cluster_id,
                        "cell_id": int(row["cell_id"]),
                        "latitude": float(row["latitude"]),
                        "longitude": float(row["longitude"]),
                        "cell_index_in_cluster": cell_idx,
                        "is_repeated": True,
                    }
                )
                cell_idx += 1

    return pd.DataFrame(selected_rows)


def draw_distributed_cells(
    clusters, df_labels: pd.DataFrame, npatch: int, random_seed=None
) -> pd.DataFrame:
    """``npatch`` members per cluster, drawn with the repeat rule of `_select_cells_for_cluster`.

    Draws serially in the order ``clusters`` is given, from one generator seeded with
    ``random_seed``, so the same seed and the same labels always pick the same cells. This is the
    one copy of the draw: the extractor below and the public ``draw_cells`` both call it.
    """
    base_rng = np.random.default_rng(random_seed)
    all_selected_cells: list[pd.DataFrame] = []
    for cluster_id in clusters:
        cluster_cells = df_labels[df_labels["cluster_label"] == cluster_id]
        if len(cluster_cells) == 0:
            continue
        all_selected_cells.append(
            _select_cells_for_cluster(cluster_id, cluster_cells, npatch, base_rng)
        )
    return pd.concat(all_selected_cells, ignore_index=True)


def _find_data_var(ds_var: xr.Dataset, var_name: str) -> str:
    """Find the primary data variable name inside a single-var dataset."""
    data_var_names = list(ds_var.data_vars)
    if len(data_var_names) == 1:
        return data_var_names[0]
    if var_name in data_var_names:
        return var_name
    matches = [dv for dv in data_var_names if var_name in dv or dv in var_name]
    return matches[0] if matches else data_var_names[0]


def _values_on_grid(ds: xr.Dataset | None, field: str, n_lat: int, n_lon: int):
    """``ds``'s ``field`` as a ``(lat, lon)`` array when it sits on the climate grid, else None."""
    if not isinstance(ds, xr.Dataset) or not ds.data_vars:
        return None
    names = [v for v in ds.data_vars if v == field] or [v for v in ds.data_vars if "soil" in v]
    if field == "soil_depth":
        names = [v for v in names if "code" not in v] or names
    if not names:
        return None
    da = ds[names[0]].squeeze(drop=True)
    from climclusterpy.data.coords import LAT_NAMES, LON_NAMES

    lat = next((n for n in LAT_NAMES if n in da.dims), None)
    lon = next((n for n in LON_NAMES if n in da.dims), None)
    if lat is None or lon is None or da.ndim != 2:
        return None
    arr = np.asarray(da.transpose(lat, lon).values)
    return arr if arr.shape == (n_lat, n_lon) else None


def _read_direct(climate: dict, lat_idx: np.ndarray, lon_idx: np.ndarray, n_jobs: int):
    """The unique cells' series of every variable that can be read straight from its file.

    Only lazily opened (dask-backed) variables are tried: one already in memory is a single fancy
    index away. ``CLIMCLUSTERPY_DIRECT_READ=0`` turns the direct read off; both paths give
    identical results, so this only matters for speed or for checking one against the other.
    """
    if not _settings.direct_read():
        return {}, {}
    from ..io.cellread import direct_source, read_cell_series

    sources = {}
    for var_name, ds_var in climate.items():
        if "time" not in ds_var.dims:
            continue
        da = ds_var[_find_data_var(ds_var, var_name)]
        if not hasattr(da.data, "__dask_graph__"):
            continue
        source = direct_source(da, da["latitude"].values[lat_idx], da["longitude"].values[lon_idx])
        if source is not None:
            sources[var_name] = source
        else:
            logger.info(
                f"[Distributed] {var_name}: not readable straight from its file; using the "
                "general (slower) path."
            )
    return read_cell_series(sources, n_jobs=n_jobs)

def extract_distributed_cells(
    clusters: list[int],
    df_labels: pd.DataFrame,
    preprocessed_climate_data: dict[str, xr.Dataset],
    df_soil_code: pd.DataFrame,
    df_soil_depth: pd.DataFrame | None,
    soil_code_data: xr.Dataset,
    soildepth_data: xr.Dataset | None,
    npatch: int,
    n_jobs: int = -1,
    seed: int | None = None,
    preload: bool = False,
) -> xr.Dataset:
    """The climate and soil of ``npatch`` drawn member cells of every cluster.

    The extraction behind ``representative="distributed"``. It keeps memory low in four
    steps:

    1. draw every cluster's cells up front (metadata only, cheap);
    2. read each grid cell once, however many slots drew it -- many of the
       ``n_clusters x npatch`` slots share a cell;
    3. read each climate variable straight from its NetCDF file where that is exact (each
       stored block holding a drawn cell is decompressed once, and the blocks of all
       variables are spread over worker processes; on the global daily forcing about 8x
       faster per variable, with identical values), otherwise stream it in time chunks,
       never holding the full grid in memory;
    4. join the soil fields with a pandas merge.

    As a guide: 18 026 clusters x 50 cells, 5 variables and 43 435 time steps need about
    6.3 GB per variable for the unique cells and peak at about 12 GB.

    Parameters
    ----------
    clusters : list of int
        The cluster ids.
    df_labels : pandas.DataFrame
        ``cell_id``, ``cluster_label``, ``latitude`` and ``longitude`` of every cell.
    preprocessed_climate_data : dict of str to xarray.Dataset
        The climate variables (may be dask-backed).
    df_soil_code, df_soil_depth : pandas.DataFrame
        The soil fields as tables (``df_soil_depth`` may be ``None``).
    soil_code_data, soildepth_data : xarray.Dataset
        The same soil fields as grids (``soildepth_data`` may be ``None``).
    npatch : int
        How many cells to draw per cluster.
    n_jobs : int, optional
        Parallel workers: processes for the direct file read, threads otherwise.
    seed : int, optional
        Seeds the draw; the same seed and labels give the same cells.
    preload : bool, optional
        Load a variable that cannot be read directly from its file into memory whole
        before extracting from it, instead of streaming it (default ``False``).

    Returns
    -------
    xarray.Dataset
        Dimensions ``cell`` (``n_clusters * npatch``) and ``time``. When the expanded
        array would be too large, the climate variables use ``unique_cell`` instead and
        a ``_slot_to_unique`` variable maps each slot to its cell.
    """
    t_start = _time.time()
    n_total_expected = len(clusters) * npatch
    logger.info(
        f"[Distributed] Extracting {npatch} cells per cluster for {len(clusters)} clusters "
        f"({n_total_expected:,} total cell-slots)..."
    )

    # ── STEP 1: Select all cells for all clusters (metadata only) ──────────
    t1 = _time.time()
    metadata_df = draw_distributed_cells(clusters, df_labels, npatch, seed)
    n_total_cells = len(metadata_df)
    t2 = _time.time()
    logger.info(f"[Distributed] Selected {n_total_cells:,} cells total ({t2 - t1:.2f}s)")

    # ── STEP 2: Compute grid indices & deduplicate ─────────────────────
    # With no climate to read (only the grid and the soil are wanted) the soil map is the grid.
    first_var_key = next(iter(preprocessed_climate_data), None)
    grid_source = (
        preprocessed_climate_data[first_var_key] if first_var_key is not None else soil_code_data
    )
    grid_lats = grid_source["latitude"].values
    grid_lons = grid_source["longitude"].values

    cell_lats = metadata_df["latitude"].values
    cell_lons = metadata_df["longitude"].values

    # Chunked broadcast lookup to cap memory at ~2 GB
    _LOOKUP_CHUNK = max(1, int(2e9 / (max(grid_lats.size, grid_lons.size) * 8)))
    lat_indices = np.empty(n_total_cells, dtype=np.int32)
    lon_indices = np.empty(n_total_cells, dtype=np.int32)
    for _s in range(0, n_total_cells, _LOOKUP_CHUNK):
        _e = min(_s + _LOOKUP_CHUNK, n_total_cells)
        lat_indices[_s:_e] = np.argmin(
            np.abs(grid_lats[None, :] - cell_lats[_s:_e, None]),
            axis=1,
        )
        lon_indices[_s:_e] = np.argmin(
            np.abs(grid_lons[None, :] - cell_lons[_s:_e, None]),
            axis=1,
        )

    # Deduplicate: find unique (lat_idx, lon_idx) pairs
    max_lon = grid_lons.size
    packed = lat_indices.astype(np.int64) * max_lon + lon_indices.astype(np.int64)
    unique_packed, slot_to_unique = np.unique(packed, return_inverse=True)
    n_unique = len(unique_packed)
    unique_lat_idx = (unique_packed // max_lon).astype(np.int32)
    unique_lon_idx = (unique_packed % max_lon).astype(np.int32)

    dedup_ratio = n_total_cells / max(n_unique, 1)
    t3 = _time.time()
    logger.info(
        f"[Distributed] Grid indices computed. "
        f"Unique cells: {n_unique:,} / {n_total_cells:,} slots "
        f"(dedup {dedup_ratio:.1f}×) [{t3 - t2:.2f}s]"
    )

    # ── STEP 3: Extract climate data — ONE variable at a time ────────────
    if first_var_key is not None and "time" in preprocessed_climate_data[first_var_key].coords:
        time_coord = preprocessed_climate_data[first_var_key]["time"].values
    else:
        time_coord = np.array([0])
    n_time = len(time_coord)

    # Normalize longitudes for CLM compatibility
    longitude_values = metadata_df["longitude"].values.copy()
    longitude_values = np.where(longitude_values > 180, longitude_values - 360, longitude_values)

    data_vars: dict[str, tuple] = {
        "cluster_label": (["cell"], metadata_df["cluster_label"].values.astype(np.int32)),
        "latitude": (["cell"], metadata_df["latitude"].values.astype(np.float32)),
        "longitude": (["cell"], longitude_values.astype(np.float32)),
        "cell_id": (["cell"], metadata_df["cell_id"].values.astype(np.int32)),
    }

    # Time-chunk size for streaming extraction.
    # chunk_t × n_unique × 4B  e.g. 2000 × 36051 × 4 ≈ 275 MB (very manageable)
    TIME_CHUNK = _settings.distributed_time_chunk()

    effective_n_jobs = n_jobs if n_jobs > 0 else min(os.cpu_count() or 4, 64)

    # If expanded array for one variable exceeds threshold, use dedup mode.
    EXPAND_THRESHOLD_GB = _settings.distributed_expand_threshold_gb()

    is_dedup = False
    estimated_expanded_gb = 0.0  # will be set per-variable

    direct, direct_reads = _read_direct(
        preprocessed_climate_data, unique_lat_idx, unique_lon_idx, n_jobs
    )
    if direct:
        logger.info(
            f"[Distributed] Read {', '.join(direct)} straight from the input files "
            f"({sum(direct_reads.values()):,} stored blocks, {_time.time() - t3:.2f}s)"
        )

    for var_name, ds_var in preprocessed_climate_data.items():
        t_var = _time.time()
        data_var = _find_data_var(ds_var, var_name)
        da = ds_var[data_var]

        if "time" in ds_var.dims and var_name in direct:
            estimated_expanded_gb = n_time * n_total_cells * 4 / (1024**3)
            unique_data = direct.pop(var_name)
        elif "time" in ds_var.dims:
            estimated_expanded_gb = n_time * n_total_cells * 4 / (1024**3)
            if preload and hasattr(da.data, "__dask_graph__"):
                da = da.load()

            # Allocate output for unique cells only: (n_time, n_unique)
            unique_data = np.empty((n_time, n_unique), dtype=np.float32)

            # Check if data is already loaded (numpy-backed) vs dask-backed. The old test,
            # `hasattr(da.data, "dask_graph")`, is False for EVERY dask array (the attribute is
            # `__dask_graph__`), so a lazily opened global variable took the in-memory branch
            # and was loaded whole -- ~35 GB per variable, OOM at 64 GB on the global grid.
            is_loaded = not hasattr(da.data, "__dask_graph__")

            if is_loaded:
                # Data already in memory — single fast fancy-index
                full_arr = da.values
                unique_data[:] = full_arr[:, unique_lat_idx, unique_lon_idx]
            else:
                # Dask-backed — stream in time chunks, optionally parallel
                chunks_spec = list(range(0, n_time, TIME_CHUNK))

                def _extract_chunk(t0, _da=da, _ulat=unique_lat_idx, _ulon=unique_lon_idx):
                    t1 = min(t0 + TIME_CHUNK, n_time)
                    chunk = _da.isel(time=slice(t0, t1)).values
                    return t0, chunk[:, _ulat, _ulon].astype(np.float32)

                if len(chunks_spec) > 1 and effective_n_jobs > 1:
                    results = Parallel(
                        n_jobs=min(effective_n_jobs, len(chunks_spec)),
                        prefer="threads",
                        verbose=0,
                    )(delayed(_extract_chunk)(t0) for t0 in chunks_spec)
                    for t0, chunk_data in results:
                        t1 = min(t0 + TIME_CHUNK, n_time)
                        unique_data[t0:t1] = chunk_data
                    del results
                else:
                    for t0 in chunks_spec:
                        _, chunk_data = _extract_chunk(t0)
                        t1 = min(t0 + TIME_CHUNK, n_time)
                        unique_data[t0:t1] = chunk_data
                        del chunk_data

        if "time" in ds_var.dims:
            # Decide whether to expand or keep dedup
            if estimated_expanded_gb > EXPAND_THRESHOLD_GB:
                data_vars[var_name] = (["time", "unique_cell"], unique_data)
                is_dedup = True
                mode_str = f"unique-only ({n_unique:,} cells, {n_time * n_unique * 4 / 1e9:.1f} GB)"
            else:
                expanded = unique_data[:, slot_to_unique]
                data_vars[var_name] = (["time", "cell"], expanded)
                del unique_data
                mode_str = f"expanded ({n_total_cells:,} cells, {estimated_expanded_gb:.1f} GB)"

        else:
            # Static variable — always small
            full_data = da.values
            unique_static = full_data[unique_lat_idx, unique_lon_idx]
            expanded_static = unique_static[slot_to_unique]
            data_vars[var_name] = (["cell"], expanded_static.astype(np.float32))
            del full_data, unique_static
            mode_str = "static"

        elapsed = _time.time() - t_var
        logger.info(f"[Distributed]   - {var_name}: {elapsed:.2f}s [{mode_str}]")

    t4 = _time.time()
    logger.info(f"[Distributed] Extracted climate data ({t4 - t3:.2f}s)")

    # ── STEP 4: Soil data ─────────────────────────────────────────
    # Each slot takes the soil OF ITS OWN CELL. Read straight off the soil grid at the
    # chosen positions when it shares the climate grid (the loader aligns them); the
    # cell-id merge below is the fallback for tables that carry ids.
    for field, ds_soil in (("soil_code", soil_code_data), ("soil_depth", soildepth_data)):
        if field in data_vars or ds_soil is None:
            continue
        on_grid = _values_on_grid(ds_soil, field, grid_lats.size, grid_lons.size)
        if on_grid is not None:
            data_vars[field] = (["cell"], on_grid[unique_lat_idx, unique_lon_idx][slot_to_unique])

    cell_id_col = "cell_id" if "cell_id" in df_soil_code.columns else "original_cell_id"
    if "soil_code" not in data_vars and cell_id_col in df_soil_code.columns:
        soil_col = "soil_code" if "soil_code" in df_soil_code.columns else "soiltexture"
        if soil_col in df_soil_code.columns:
            soil_merge = metadata_df[["cell_id"]].merge(
                df_soil_code[[cell_id_col, soil_col]].rename(columns={cell_id_col: "cell_id"}),
                on="cell_id",
                how="left",
            )
            data_vars["soil_code"] = (["cell"], soil_merge[soil_col].values)

    if df_soil_depth is not None and "soil_depth" not in data_vars:
        depth_cell_id_col = "cell_id" if "cell_id" in df_soil_depth.columns else "original_cell_id"
        if depth_cell_id_col in df_soil_depth.columns:
            depth_col = [
                c for c in df_soil_depth.columns if "soil" in c.lower() and "code" not in c.lower()
            ]
            if depth_col:
                depth_merge = metadata_df[["cell_id"]].merge(
                    df_soil_depth[[depth_cell_id_col, depth_col[0]]].rename(
                        columns={depth_cell_id_col: "cell_id"}
                    ),
                    on="cell_id",
                    how="left",
                )
                data_vars["soil_depth"] = (["cell"], depth_merge[depth_col[0]].values)

    # ── Build output Dataset ───────────────────────────────────────
    coords: dict[str, np.ndarray] = {
        "cell": np.arange(n_total_cells),
        "time": time_coord,
    }
    if is_dedup:
        coords["unique_cell"] = np.arange(n_unique)

    ds = xr.Dataset(data_vars, coords=coords)

    # Attach dedup mapping so the pipeline can expand during on-disk NC writing
    # Use int (1/0) instead of bool — netCDF4 rejects bool (b1) attributes.
    if is_dedup:
        ds.attrs["_distributed_dedup"] = 1
        ds.attrs["_n_unique_cells"] = int(n_unique)
        ds.attrs["_n_total_cells"] = int(n_total_cells)
        ds["_slot_to_unique"] = ("cell", slot_to_unique.astype(np.int32))
        logger.info(
            f"[Distributed] \u26a1 Dedup mode: {n_unique:,} unique cells for "
            f"{n_total_cells:,} slots"
        )
    else:
        ds.attrs["_distributed_dedup"] = 0

    t_end = _time.time()
    logger.info(f"[Distributed] Total extraction: {t_end - t_start:.2f}s")

    return ds
