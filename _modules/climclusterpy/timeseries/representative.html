Source code for climclusterpy.timeseries.representative

"""Cluster *representative* timeseries: pick one real cell to stand for each cluster.

The counterpart to :mod:`climclusterpy.timeseries.mean`. A cluster mean is a series
no cell actually experiences; a representative is a real member, which keeps the
input physically consistent for a model with non-linear responses. How that member is
chosen is the ``selection_method``:

* ``"precomputed_features"`` (default) — nearest to the cluster centroid in feature
  space. Fast and needs no extra data beyond the feature table.
* ``"precomputed_cells"`` — a choice made upstream, which is how LMNN **medoids**
  (:func:`climclusterpy.clustering.medoid.select_medoids`) are honoured.
* ``"timeseries_distance"`` — nearest to the cluster mean in full timeseries space.
  The most faithful and the slowest; needs no feature table.
* ``"random"`` — one member drawn at random (:func:`draw_random_cells`); no
  representative is computed. Reproducible for a fixed ``seed`` at any ``n_jobs``,
  and the chosen cell's **own** soil is used rather than the cluster's most common one.

Clusters are processed in parallel with joblib over **threads**, so the caller is
expected to pass in arrays already loaded into memory (as :func:`climclusterpy.create_clusters`
does with ``preload_climate=True``).
"""

import logging
from typing import Literal

import numpy as np
import pandas as pd
import xarray as xr
from joblib import Parallel, delayed

from climclusterpy._log import joblib_verbosity
from climclusterpy.data.coords import COORD_PRECISION

logger = logging.getLogger(__name__)

SelectionMethod = Literal[
    "precomputed_features", "timeseries_distance", "random", "precomputed_cells"
]

def get_most_common_soil(values):
    """The most common soil code among a cluster's cells.

    Parameters
    ----------
    values : pandas.Series, xarray.DataArray or array-like
        The soil codes of the cluster's cells. Missing values (NaN) are ignored.

    Returns
    -------
    scalar
        The most frequent code, keeping the input's type; the smallest code on a tie;
        NaN when there is no code at all.
    """
    from scipy.stats import mode as scipy_mode

    arr = np.asarray(getattr(values, "values", values)).ravel()
    arr = arr[~pd.isna(arr)]
    if arr.size == 0:
        return np.nan
    mode_val = np.atleast_1d(scipy_mode(arr).mode)
    return mode_val[0] if mode_val.size > 0 else np.nan

def select_representative_by_features(
    cluster_cells: pd.DataFrame,
    features_df: pd.DataFrame,
    grid_lats: np.ndarray,
    grid_lons: np.ndarray,
) -> tuple[int, int, dict]:
    """The cluster member nearest the cluster's mean in feature space (the centroid choice).

    Parameters
    ----------
    cluster_cells : pandas.DataFrame
        The cluster's cells, with ``latitude`` and ``longitude`` columns.
    features_df : pandas.DataFrame
        One row per cell: ``latitude``, ``longitude`` and the feature columns. Every
        numeric column that is not cell metadata (``cell_id``, ``latitude``, ``longitude``,
        ``lat``, ``lon``, ``ecoregion_id``, ``cluster_label``) is a feature; non-numeric
        columns are ignored. Joined to ``cluster_cells`` on the coordinates, which must
        match exactly.
    grid_lats, grid_lons : numpy.ndarray
        The grid's coordinate axes.

    Returns
    -------
    tuple
        ``(lat_index, lon_index, {"latitude": ..., "longitude": ...})`` of the chosen cell.

    Raises
    ------
    ValueError
        If no cluster cell has features, or ``features_df`` has no feature columns.
    """
    from climclusterpy.features.assemble import METADATA_COLUMNS

    # Pick the feature columns from the feature table alone, and take nothing but the
    # coordinates from the cluster's cells. Merging the two whole tables renamed a column
    # both carry (``cell_id``) to ``cell_id_x``/``cell_id_y``, which then passed as
    # features, so the "centroid" cell was chosen largely by its cell number.
    feature_cols = [
        col
        for col in features_df.columns
        if col not in METADATA_COLUMNS and pd.api.types.is_numeric_dtype(features_df[col])
    ]
    if len(feature_cols) == 0:
        raise ValueError(
            "No feature columns found in features_df: every column is cell metadata "
            f"({sorted(METADATA_COLUMNS)}) or not numeric."
        )

    merged = pd.merge(
        cluster_cells[["latitude", "longitude"]],
        features_df[["latitude", "longitude", *feature_cols]],
        on=["latitude", "longitude"],
        how="inner",
    )

    if len(merged) == 0:
        raise ValueError("No features found for cluster cells - coordinate mismatch?")

    # Compute cluster centroid in feature space
    cluster_centroid = merged[feature_cols].mean().values

    # Compute Euclidean distance to centroid for each cell
    X = merged[feature_cols].values
    from scipy.spatial.distance import cdist

    distances = cdist(X, cluster_centroid.reshape(1, -1), metric="euclidean").flatten()

    # Select cell closest to centroid
    rep_idx = int(np.argmin(distances))
    rep_cell = merged.iloc[rep_idx]

    # Find grid indices
    rep_lat_idx = int(np.where(grid_lats == rep_cell["latitude"])[0][0])
    rep_lon_idx = int(np.where(grid_lons == rep_cell["longitude"])[0][0])

    return (
        rep_lat_idx,
        rep_lon_idx,
        {"latitude": rep_cell["latitude"], "longitude": rep_cell["longitude"]},
    )




def _select_representative_by_precomputed_cells(
    cluster_id: int,
    representative_cells: pd.DataFrame,
    grid_lats: np.ndarray,
    grid_lons: np.ndarray,
) -> tuple[int, int, dict]:
    """Use a representative cell chosen upstream (e.g. an LMNN medoid).

    Parameters
    ----------
    cluster_id : int
        Cluster whose representative is wanted.
    representative_cells : pandas.DataFrame
        One row per cluster with columns ``cluster_label``, ``latitude``,
        ``longitude`` — typically built from
        :func:`climclusterpy.clustering.medoid.select_medoids`.
    grid_lats, grid_lons : numpy.ndarray
        Grid coordinate arrays.

    Returns
    -------
    tuple
        ``(rep_lat_idx, rep_lon_idx, rep_info_dict)``.
    """
    row = representative_cells[representative_cells["cluster_label"] == cluster_id]
    if row.empty:
        raise ValueError(
            f"No precomputed representative cell for cluster {cluster_id}. "
            "representative_cells must cover every cluster label."
        )
    rep = row.iloc[0]
    lat_hits = np.where(grid_lats == rep["latitude"])[0]
    lon_hits = np.where(grid_lons == rep["longitude"])[0]
    if lat_hits.size == 0 or lon_hits.size == 0:
        raise ValueError(
            f"Representative cell for cluster {cluster_id} at "
            f"({rep['latitude']}, {rep['longitude']}) is not on the climate grid. "
            "Representative coordinates must come from the same grid as the input data."
        )
    return (
        int(lat_hits[0]),
        int(lon_hits[0]),
        {"latitude": rep["latitude"], "longitude": rep["longitude"]},
    )


def _select_representative_by_timeseries_distance(
    cluster_cells: pd.DataFrame,
    preprocessed_climate_data: dict[str, xr.Dataset],
    grid_lats: np.ndarray,
    grid_lons: np.ndarray,
    cell_indices: list,
    distance_metric: str = "euclidean",
) -> tuple[int, int, dict]:
    """
    Most accurate method: Select representative based on full timeseries distance.

    Uses scipy.spatial.distance.cdist for efficient pairwise distance computation.
    Concatenates all climate variable timeseries and computes distance in full timeseries space.

    Args:
            distance_metric: scipy distance metric ('euclidean', 'correlation', 'cityblock', etc.)

    Returns
    -------
            (rep_lat_idx, rep_lon_idx, rep_info_dict)
    """
    from scipy.spatial.distance import cdist

    # Build timeseries matrix: (n_cells, n_timesteps_all_variables)
    timeseries_list = []
    for lat_idx, lon_idx in cell_indices:
        cell_ts_parts = []
        for var_name, ds_var in preprocessed_climate_data.items():
            data_var_names = list(ds_var.data_vars)
            if len(data_var_names) == 1:
                data_var = data_var_names[0]
            elif var_name in data_var_names:
                data_var = var_name
            else:
                matches = [dv for dv in data_var_names if var_name in dv or dv in var_name]
                data_var = matches[0] if matches else data_var_names[0]

            ts = ds_var.isel(latitude=lat_idx, longitude=lon_idx)[data_var]
            ts_np = np.asarray(ts.values).flatten()
            cell_ts_parts.append(ts_np)

        # Concatenate all variables for this cell
        cell_ts_full = np.concatenate(cell_ts_parts)
        timeseries_list.append(cell_ts_full)

    # Stack into matrix: (n_cells, n_timesteps_total)
    ts_matrix = np.vstack(timeseries_list).astype(np.float32)

    # Compute cluster centroid (mean timeseries)
    cluster_centroid = np.nanmean(ts_matrix, axis=0, keepdims=True)

    # Compute distances to centroid using scipy (highly optimized C code)
    distances = cdist(ts_matrix, cluster_centroid, metric=distance_metric).flatten()

    # Select cell with minimum distance
    rep_cell_idx = int(np.argmin(distances))
    rep_lat_idx, rep_lon_idx = cell_indices[rep_cell_idx]

    return (
        rep_lat_idx,
        rep_lon_idx,
        {"latitude": grid_lats[rep_lat_idx], "longitude": grid_lons[rep_lon_idx]},
    )


def _select_representative_by_random(
    cluster_cells: pd.DataFrame,
    grid_lats: np.ndarray,
    grid_lons: np.ndarray,
    rng: np.random.Generator,
) -> tuple[int, int, dict]:
    """
    Simplest method: Randomly select one cell from the cluster.

    No feature computation or timeseries distance calculation needed.
    Just pick a random cell and use its timeseries.

    Args:
            cluster_cells: DataFrame with cluster cells (must have latitude, longitude)
            grid_lats, grid_lons: Grid coordinate arrays
            rng: NumPy random generator for reproducible selection

    Returns
    -------
            (rep_lat_idx, rep_lon_idx, rep_info_dict)
    """
    # Randomly select one cell from cluster
    rep_cell = cluster_cells.sample(n=1, random_state=rng.bit_generator).iloc[0]

    # Find grid indices
    rep_lat_idx = int(np.where(grid_lats == rep_cell["latitude"])[0][0])
    rep_lon_idx = int(np.where(grid_lons == rep_cell["longitude"])[0][0])

    return (
        rep_lat_idx,
        rep_lon_idx,
        {"latitude": rep_cell["latitude"], "longitude": rep_cell["longitude"]},
    )


def _soil_depth_variable(soildepth_data: xr.Dataset) -> str | None:
    """The soil-depth variable of ``soildepth_data``; None when the dataset holds no variable.

    Taken by name: ``soil_depth`` itself, else the first variable whose name contains
    ``"soil"`` (e.g. LPJmL's ``soildepth``), else the dataset's only variable whatever its
    name (e.g. ``depth``).

    Raises
    ------
    ValueError
        When several variables are present and none of them is named like a soil depth, so
        there is no telling which one to read.
    """
    names = list(soildepth_data.data_vars)
    if not names:
        return None
    if "soil_depth" in names:
        return "soil_depth"
    soil_like = [v for v in names if "soil" in v]
    if soil_like:
        return soil_like[0]
    if len(names) == 1:
        return names[0]
    raise ValueError(
        f"Cannot tell which variable of the soil-depth dataset is the soil depth: it holds "
        f"{names}. Name it 'soil_depth', or pass a dataset holding only that variable."
    )


def process_single_cluster(
    cluster_id,
    cluster_cells,
    preprocessed_climate_data,
    df_soil_code=None,
    df_soil_depth=None,
    soil_code_data=None,
    soildepth_data=None,
    selection_method: SelectionMethod = "precomputed_features",
    features_df: pd.DataFrame | None = None,
    distance_metric: str = "euclidean",
    rng: np.random.Generator | None = None,
    representative_cells: pd.DataFrame | None = None,
):
    """
    Process a single cluster to extract representative cell and timeseries.

    Args:
            selection_method: Method to use for selecting representative cell:
                    - 'precomputed_features': Use pre-computed LMNN features (DEFAULT, requires features_df)
                    - 'timeseries_distance': Full timeseries distance (most accurate, slower)
                    - 'random': Randomly select one cell (fastest, no features needed)
                    - 'precomputed_cells': Use cells chosen upstream (requires representative_cells),
                      e.g. LMNN medoids from climclusterpy.clustering.medoid.select_medoids
            features_df: Pre-computed features DataFrame (required for 'precomputed_features' method)
            distance_metric: Distance metric for 'timeseries_distance' method
            rng: NumPy random generator for 'random' method (if None, will create one)
            representative_cells: cluster_label/latitude/longitude table (required for
                    'precomputed_cells' method)
    """
    rep_info = {"cluster_label": cluster_id}

    # Use median of absolute latitude values to avoid issues with clusters spanning the equator
    abs_lats = np.abs(cluster_cells["latitude"].values)
    center_lat = float(np.median(abs_lats))
    # Keep the sign of the latitude with the median absolute value
    # Find the actual latitude value closest to the median absolute value
    closest_idx = np.argmin(np.abs(np.abs(cluster_cells["latitude"].values) - center_lat))
    center_lat = float(cluster_cells["latitude"].iloc[closest_idx])
    center_lon = cluster_cells["longitude"].mean()

    # All datasets now have 'latitude' and 'longitude' as coordinate names and are on the same grid
    grid_lats = preprocessed_climate_data[next(iter(preprocessed_climate_data.keys()))][
        "latitude"
    ].values
    grid_lons = preprocessed_climate_data[next(iter(preprocessed_climate_data.keys()))][
        "longitude"
    ].values

    # Select representative cell based on chosen method
    if selection_method == "precomputed_features":
        if features_df is None:
            raise ValueError(
                "features_df must be provided for 'precomputed_features' selection method"
            )
        rep_lat_idx, rep_lon_idx, coord_info = select_representative_by_features(
            cluster_cells, features_df, grid_lats, grid_lons
        )
    elif selection_method == "timeseries_distance":
        # Build cell indices only for timeseries_distance method (expensive operation)
        cell_indices = [
            (
                np.where(grid_lats == row["latitude"])[0][0],
                np.where(grid_lons == row["longitude"])[0][0],
            )
            for _, row in cluster_cells.iterrows()
        ]
        rep_lat_idx, rep_lon_idx, coord_info = _select_representative_by_timeseries_distance(
            cluster_cells,
            preprocessed_climate_data,
            grid_lats,
            grid_lons,
            cell_indices,
            distance_metric,
        )
    elif selection_method == "precomputed_cells":
        if representative_cells is None:
            raise ValueError(
                "representative_cells must be provided for 'precomputed_cells' selection method"
            )
        rep_lat_idx, rep_lon_idx, coord_info = _select_representative_by_precomputed_cells(
            cluster_id, representative_cells, grid_lats, grid_lons
        )
    elif selection_method == "random":
        if representative_cells is not None:
            # Drawn up front by extract_representative_cells (reproducible).
            rep_lat_idx, rep_lon_idx, coord_info = _select_representative_by_precomputed_cells(
                cluster_id, representative_cells, grid_lats, grid_lons
            )
        else:
            if rng is None:
                rng = np.random.default_rng()
            rep_lat_idx, rep_lon_idx, coord_info = _select_representative_by_random(
                cluster_cells, grid_lats, grid_lons, rng
            )
    else:
        raise ValueError(
            f"Unknown selection_method: {selection_method}. Use 'precomputed_features', "
            "'precomputed_cells', 'timeseries_distance', or 'random'"
        )

    rep_info.update(coord_info)

    # Extract timeseries for representative cell
    rep_timeseries = {}
    for var_name, ds_var in preprocessed_climate_data.items():
        data_var_names = list(ds_var.data_vars)
        if len(data_var_names) == 1:
            data_var = data_var_names[0]
        elif var_name in data_var_names:
            data_var = var_name
        else:
            matches = [dv for dv in data_var_names if var_name in dv or dv in var_name]
            data_var = matches[0] if matches else data_var_names[0]

        # Use .load() on single cell selection to avoid repeated lazy reads
        ts = ds_var.isel(latitude=rep_lat_idx, longitude=rep_lon_idx)[data_var].load()
        rep_timeseries[var_name] = ts.values

    # Determine soil_code for the cluster using robust ID-based mapping when available
    # For random selection, just use the selected cell's soil code (no need to compute mode)
    if soil_code_data is None:
        # No soil-code-like static field was supplied (a non-LPJmL input): nothing to report.
        rep_info["soil_code"] = np.nan
    elif selection_method == "random":
        # Direct lookup: use the selected representative cell's soil code
        soil_var = (
            "soil_code"
            if "soil_code" in soil_code_data.data_vars
            else ("soiltexture" if "soiltexture" in soil_code_data.data_vars else None)
        )
        if soil_var is not None and soil_var in soil_code_data.data_vars:
            rep_info["soil_code"] = (
                soil_code_data[soil_var]
                .isel(latitude=rep_lat_idx, longitude=rep_lon_idx)
                .values.item()
            )
        else:
            # try to find any soil-like var
            soil_vars = [v for v in soil_code_data.data_vars if "soil" in v]
            if soil_vars:
                soil_var = soil_vars[0]
                rep_info["soil_code"] = (
                    soil_code_data[soil_var]
                    .isel(latitude=rep_lat_idx, longitude=rep_lon_idx)
                    .values.item()
                )
            else:
                rep_info["soil_code"] = np.nan
    else:
        # For mean/representative methods, compute mode over cluster cells
        soil_mode = np.nan
        if "cell_id" in cluster_cells.columns:
            # Prefer mapping by original_cell_id to avoid coordinate precision issues
            id_col = (
                "original_cell_id"
                if "original_cell_id" in df_soil_code.columns
                else ("cell_id" if "cell_id" in df_soil_code.columns else None)
            )
            if id_col is not None:
                soil_col = (
                    "soil_code"
                    if "soil_code" in df_soil_code.columns
                    else ("soiltexture" if "soiltexture" in df_soil_code.columns else None)
                )
                if soil_col is not None:
                    id_to_soil = dict(zip(df_soil_code[id_col].astype(int), df_soil_code[soil_col]))
                    soil_values = cluster_cells["cell_id"].map(id_to_soil)
                    soil_mode = get_most_common_soil(soil_values)
            else:
                # Fallback: merge on rounded coordinates
                cc = cluster_cells.copy()
                dfsc = df_soil_code.copy()
                cc["latitude"] = cc["latitude"].round(COORD_PRECISION)
                cc["longitude"] = cc["longitude"].round(COORD_PRECISION)
                dfsc["latitude"] = dfsc["latitude"].round(COORD_PRECISION)
                dfsc["longitude"] = dfsc["longitude"].round(COORD_PRECISION)
                merged = pd.merge(cc, dfsc, on=["latitude", "longitude"], how="left")
                soil_col = (
                    "soil_code"
                    if "soil_code" in merged.columns
                    else ("soiltexture" if "soiltexture" in merged.columns else None)
                )
                soil_mode = (
                    get_most_common_soil(merged[soil_col])
                    if not merged.empty and soil_col in merged.columns
                    else np.nan
                )
        else:
            # No cell IDs? Fallback to coordinate merge
            cc = cluster_cells.copy()
            dfsc = df_soil_code.copy()
            cc["latitude"] = cc["latitude"].round(COORD_PRECISION)
            cc["longitude"] = cc["longitude"].round(COORD_PRECISION)
            dfsc["latitude"] = dfsc["latitude"].round(COORD_PRECISION)
            dfsc["longitude"] = dfsc["longitude"].round(COORD_PRECISION)
            merged = pd.merge(cc, dfsc, on=["latitude", "longitude"], how="left")
            soil_col = (
                "soil_code"
                if "soil_code" in merged.columns
                else ("soiltexture" if "soiltexture" in merged.columns else None)
            )
            soil_mode = (
                get_most_common_soil(merged[soil_col])
                if not merged.empty and soil_col in merged.columns
                else np.nan
            )
        # Fallback: if soil_mode is NaN, sample soil_code at the representative cell directly
        if (isinstance(soil_mode, float) and np.isnan(soil_mode)) or (pd.isna(soil_mode)):
            soil_var = (
                "soil_code"
                if "soil_code" in soil_code_data.data_vars
                else ("soiltexture" if "soiltexture" in soil_code_data.data_vars else None)
            )
            if soil_var is not None and soil_var in soil_code_data.data_vars:
                rep_info["soil_code"] = (
                    soil_code_data[soil_var]
                    .isel(latitude=rep_lat_idx, longitude=rep_lon_idx)
                    .values.item()
                )
            else:
                # try to find any soil-like var
                soil_vars = [v for v in soil_code_data.data_vars if "soil" in v]
                if soil_vars:
                    soil_var = soil_vars[0]
                    rep_info["soil_code"] = (
                        soil_code_data[soil_var]
                        .isel(latitude=rep_lat_idx, longitude=rep_lon_idx)
                        .values.item()
                    )
                else:
                    rep_info["soil_code"] = np.nan
        else:
            rep_info["soil_code"] = soil_mode

    # Soil depth (direct index selection, no nearest-neighbor)
    if soildepth_data is not None:
        depth_var = _soil_depth_variable(soildepth_data)
        if depth_var is not None:
            # Detect coordinate names for soil depth dataset (might be 'Latitude', 'Longitude')
            from climclusterpy.data.coords import LAT_NAMES, LON_NAMES

            depth_lat_name = next((n for n in LAT_NAMES if n in soildepth_data.coords), "latitude")
            depth_lon_name = next((n for n in LON_NAMES if n in soildepth_data.coords), "longitude")
            rep_lat_idx = np.where(grid_lats == rep_info["latitude"])[0][0]
            rep_lon_idx = np.where(grid_lons == rep_info["longitude"])[0][0]
            rep_depth = (
                soildepth_data[depth_var]
                .isel({depth_lat_name: rep_lat_idx, depth_lon_name: rep_lon_idx})
                .values.item()
            )
            rep_info["soil_depth"] = rep_depth
        else:
            rep_info["soil_depth"] = np.nan
    else:
        rep_info["soil_depth"] = np.nan
    return rep_info, rep_timeseries, center_lat, center_lon


def draw_random_cells(clusters, df_labels: pd.DataFrame, random_seed=None) -> pd.DataFrame:
    """One member per cluster, drawn uniformly at random — reproducibly for a given seed.

    Draws serially in the order ``clusters`` is given, from one generator seeded with
    ``random_seed``, so the same seed and the same labels always pick the same cells.

    Returns
    -------
    pandas.DataFrame
        One row per cluster: ``cluster_label``, ``latitude``, ``longitude`` (and ``cell_id``
        when ``df_labels`` carries it) — the ``representative_cells`` table the
        ``"precomputed_cells"`` path reads.
    """
    rng = np.random.default_rng(random_seed)
    rows = []
    for cluster_id in clusters:
        members = df_labels[df_labels["cluster_label"] == cluster_id]
        if members.empty:
            raise ValueError(f"Cluster {cluster_id} has no cells to draw from.")
        pick = members.iloc[int(rng.integers(len(members)))]
        row = {
            "cluster_label": cluster_id,
            "latitude": pick["latitude"],
            "longitude": pick["longitude"],
        }
        if "cell_id" in members.columns:
            row["cell_id"] = int(pick["cell_id"])
        rows.append(row)
    return pd.DataFrame(rows)

def extract_representative_cells(
    clusters,
    df_labels,
    preprocessed_climate_data,
    df_soil_code=None,
    df_soil_depth=None,
    soil_code_data=None,
    soildepth_data=None,
    n_jobs=4,
    n_samples=None,
    min_samples=1,
    seed=None,
    selection_method: SelectionMethod = "precomputed_features",
    features_df: pd.DataFrame | None = None,
    distance_metric: str = "euclidean",
    representative_cells: pd.DataFrame | None = None,
):
    """One real cell's climate and soil for each cluster, clusters processed in parallel.

    Parameters
    ----------
    clusters : list of int
        The cluster ids.
    df_labels : pandas.DataFrame
        ``cell_id``, ``cluster_label``, ``latitude`` and ``longitude`` of every cell.
    preprocessed_climate_data : dict of str to xarray.Dataset
        The climate variables, already loaded into memory.
    df_soil_code, df_soil_depth : pandas.DataFrame or None
        The soil fields as tables. ``None`` for an input without them (a model other than
        LPJmL): the output's soil columns are then all NaN.
    soil_code_data, soildepth_data : xarray.Dataset or None
        The same soil fields as grids, or ``None`` as above.
    n_jobs : int, optional
        Parallel threads.
    n_samples, min_samples : optional
        Choose among a random subset of each cluster's cells instead of all of them, as in
        :func:`sample_cluster_indices`.
    seed : int, optional
        Seeds that subset and the ``"random"`` draw.
    selection_method : str, optional
        How the cell is chosen:

        * ``"precomputed_features"`` (default) -- nearest the cluster centroid in feature
          space; needs ``features_df``;
        * ``"timeseries_distance"`` -- nearest the cluster mean in full timeseries space;
          slower, needs no features;
        * ``"random"`` -- one member drawn at random;
        * ``"precomputed_cells"`` -- the cells given in ``representative_cells`` (how
          medoids of a learned metric are honoured).
    features_df : pandas.DataFrame, optional
        The per-cell features, for ``"precomputed_features"``.
    distance_metric : str, optional
        The distance of ``"timeseries_distance"`` (``"euclidean"``, ``"correlation"``, ...).
    representative_cells : pandas.DataFrame, optional
        ``cluster_label``, ``latitude`` and ``longitude``, one row per cluster, for
        ``"precomputed_cells"``.

    Returns
    -------
    xarray.Dataset
        The chosen cells and their timeseries, one entry per cluster.
    """
    # Validate selection method
    if selection_method == "precomputed_features" and features_df is None:
        raise ValueError(
            "features_df must be provided when selection_method='precomputed_features'.\n"
            "Either:\n"
            "  1. Pass features_df (pre-computed features DataFrame), or\n"
            "  2. Use selection_method='timeseries_distance' (slower, no features needed), or\n"
            "  3. Use selection_method='random' (fastest, no features needed)"
        )
    if selection_method == "precomputed_cells" and representative_cells is None:
        raise ValueError(
            "representative_cells must be provided when selection_method='precomputed_cells'. "
            "Build it from climclusterpy.clustering.medoid.select_medoids, or choose another "
            "selection_method."
        )

    # Random picks are drawn HERE, serially and in cluster order, before the threaded
    # fan-out. Handing one generator to every thread made the draw depend on which thread
    # reached it first, so a fixed seed did not reproduce the same cells at n_jobs > 1.
    rng = None
    if selection_method == "random" and representative_cells is None:
        representative_cells = draw_random_cells(clusters, df_labels, seed)

    # Apply sampling to cluster cells before processing
    # For random selection, skip sampling entirely since we only pick 1 cell per cluster anyway
    if n_samples is not None and selection_method != "random":
        import logging

        from climclusterpy.timeseries.sampling import sample_cluster_indices

        logger = logging.getLogger(__name__)
        logger.info(
            f"Sampling cells for representative extraction: n_samples={n_samples}, min_samples={min_samples}"
        )

        # Build cluster_to_indices mapping (using DataFrame row indices)
        cluster_to_row_indices = {}
        for cluster_id in clusters:
            mask = df_labels["cluster_label"] == cluster_id
            cluster_to_row_indices[cluster_id] = df_labels[mask].index.tolist()

        # Sample indices
        sampled_indices = sample_cluster_indices(
            cluster_to_row_indices,
            n_samples=n_samples,
            min_samples=min_samples,
            seed=seed,
        )

        # Create a sampled df_labels by selecting only sampled rows
        all_sampled_rows = []
        for cluster_id in clusters:
            all_sampled_rows.extend(sampled_indices[cluster_id])
        df_labels = df_labels.loc[all_sampled_rows].copy()

    results = Parallel(n_jobs=n_jobs, prefer="threads", verbose=joblib_verbosity())(
        delayed(process_single_cluster)(
            cluster_id,
            df_labels[df_labels["cluster_label"] == cluster_id],
            preprocessed_climate_data,
            df_soil_code,
            df_soil_depth,
            soil_code_data,
            soildepth_data,
            selection_method,
            features_df,
            distance_metric,
            rng,
            representative_cells,
        )
        for cluster_id in clusters
    )
    # Collect results
    rep_properties = {
        k: [] for k in ["cluster_label", "latitude", "longitude", "soil_code", "soil_depth"]
    }
    rep_timeseries = {var: [] for var in preprocessed_climate_data.keys()}
    center_lat_list = []
    center_lon_list = []
    for rep_info, rep_ts, center_lat, center_lon in results:
        for k in rep_properties:
            rep_properties[k].append(rep_info[k])
        for var in rep_timeseries:
            rep_timeseries[var].append(rep_ts[var])
        center_lat_list.append(center_lat)
        center_lon_list.append(center_lon)
    # Build xarray Dataset
    time_dim = next(iter(preprocessed_climate_data.values()))["time"].values
    ds_out = xr.Dataset()
    ds_out["cluster_label"] = (("cluster",), rep_properties["cluster_label"])
    ds_out["latitude"] = (("cluster",), rep_properties["latitude"])
    ds_out["longitude"] = (("cluster",), rep_properties["longitude"])
    ds_out["soil_code"] = (("cluster",), rep_properties["soil_code"])
    ds_out["soil_depth"] = (("cluster",), rep_properties["soil_depth"])
    ds_out["center_lat"] = (("cluster",), center_lat_list)
    ds_out["center_lon"] = (("cluster",), center_lon_list)
    ds_out["center_lat"].attrs = {
        "units": "degrees_north",
        "description": "Latitude of cluster center",
        "_FillValue": np.nan,
    }
    ds_out["center_lon"].attrs = {
        "units": "degrees_east",
        "description": "Longitude of cluster center",
        "_FillValue": np.nan,
    }
    for var_name, arrs in rep_timeseries.items():
        ds_out[var_name] = (("cluster", "time"), np.stack(arrs))
    ds_out["time"] = (("time",), time_dim)
    return ds_out




def _place_labels_by_cell_id(df_labels: pd.DataFrame, registry) -> pd.DataFrame:
    """``df_labels`` with ``latitude``/``longitude`` taken from the registry by ``cell_id``.

    Raises
    ------
    ValueError
        When a ``cell_id`` is not a valid cell of the registry's original grid.
    """
    # Cells outside the valid mask carry id -1 in the registry; no label can refer to them.
    orig = registry.get_original_grid()
    lon2d, lat2d = np.meshgrid(orig.longitudes, orig.latitudes)
    flat_ids = orig.cell_ids.ravel()
    valid = flat_ids >= 0
    flat_ids = flat_ids[valid]
    id_to_lat = dict(zip(flat_ids.tolist(), lat2d.ravel()[valid].tolist()))
    id_to_lon = dict(zip(flat_ids.tolist(), lon2d.ravel()[valid].tolist()))
    # A label for a cell the grid does not have would get no position (or, for -1, the
    # position of a cell outside the land mask): refuse it rather than carry it along.
    unknown = ~df_labels["cell_id"].isin(flat_ids)
    if unknown.any():
        bad = sorted(pd.unique(df_labels.loc[unknown, "cell_id"]).tolist())
        raise ValueError(
            f"The label table names {int(unknown.sum())} cell(s) that are not valid cells of "
            f"the original grid (cell_id {bad[:10]}{' ...' if len(bad) > 10 else ''}; the "
            f"grid has {len(flat_ids)} valid cells). The labels must come from the same grid "
            "as the input data."
        )
    df_labels = df_labels.copy()
    df_labels["latitude"] = df_labels["cell_id"].map(id_to_lat)
    df_labels["longitude"] = df_labels["cell_id"].map(id_to_lon)
    return df_labels

def load_and_clean_cluster_labels(labels_file, registry: object):
    """Load label CSV and map `cell_id` to latitude/longitude using registry's original grid."""
    logger.info(f"Loading cluster labels from: {labels_file}")
    df_labels = pd.read_csv(labels_file)
    required_cols = ["latitude", "longitude", "cluster_label"]
    missing_cols = [col for col in required_cols if col not in df_labels.columns]
    if missing_cols:
        raise ValueError(f"Missing required columns in label file: {missing_cols}")
    initial_count = len(df_labels)
    df_labels["cluster_label"] = pd.to_numeric(df_labels["cluster_label"], errors="coerce")
    df_labels = df_labels.dropna(subset=["cluster_label"])
    df_labels["cluster_label"] = df_labels["cluster_label"].astype(int)
    df_labels = df_labels[df_labels["cluster_label"] >= 0]
    # concise status
    logger.info(
        f"Labels cleaned: {len(df_labels)} valid (removed {initial_count - len(df_labels)})"
    )
    # Map cell_id to lat/lon using registry
    if "cell_id" not in df_labels.columns:
        raise ValueError("Expected label CSV to include 'cell_id' column")
    df_labels = _place_labels_by_cell_id(df_labels, registry)
    n_duplicates = df_labels.duplicated(subset=["latitude", "longitude"]).sum()
    if n_duplicates > 0:
        logger.warning(
            f"WARNING: Found {n_duplicates} duplicate lat/lon combinations. Keeping first occurrence."
        )
        df_labels = df_labels.drop_duplicates(subset=["latitude", "longitude"], keep="first")
    return df_labels
