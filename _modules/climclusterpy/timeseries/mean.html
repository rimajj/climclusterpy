"""Cluster *mean* timeseries: average every member cell of a cluster.

One of the two ways to give a cluster a timeseries. Averaging over all members is
the statistically natural choice and smooths within-cluster noise, but the result is
a synthetic series no real cell exhibits — which matters for a vegetation model whose
responses are non-linear. The alternative, picking one real member, is in
:mod:`climclusterpy.timeseries.representative`.

Built on xarray throughout (no pandas round-trips) so the reduction stays lazy and
chunk-parallel over large grids.
"""

import logging
import time

import numpy as np
import xarray as xr
from scipy.stats import mode as scipy_mode

# Import shared coordinate utilities
from climclusterpy.data.coords import (
    COORD_PRECISION,
    LAT_NAMES,
    LON_NAMES,
)
from climclusterpy.data.registry import DataRegistry

logger = logging.getLogger(__name__)

def find_center_cell(ds_cluster):
    """The member of one cluster nearest its median position.

    The median is taken of the *absolute* latitude, so a cluster spanning the equator
    (members at 30° N and 30° S) is centred at 30°, not at 0° where it has no member.

    Parameters
    ----------
    ds_cluster : xarray.Dataset
        One cluster's cells along a ``point`` dimension, with ``latitude`` and
        ``longitude`` (degrees), as :func:`build_cluster_mean_timeseries` groups them.

    Returns
    -------
    dict
        ``{"center_lat": ..., "center_lon": ...}`` of that member, rounded to the package's
        coordinate precision; both NaN for an empty cluster.
    """
    if ds_cluster.sizes.get("point", 0) == 0:
        return {"center_lat": np.nan, "center_lon": np.nan}

    abs_lats = np.abs(ds_cluster.latitude.values)
    lat_med_abs = np.median(abs_lats)
    lon_med = ds_cluster.longitude.median()
    distances = np.sqrt(
        (np.abs(ds_cluster.latitude) - lat_med_abs) ** 2 + (ds_cluster.longitude - lon_med) ** 2
    )
    min_idx = distances.argmin()
    center_lat = float(ds_cluster.latitude.isel(point=min_idx))
    center_lon = float(ds_cluster.longitude.isel(point=min_idx))
    return {
        "center_lat": np.round(center_lat, COORD_PRECISION),
        "center_lon": np.round(center_lon, COORD_PRECISION),
    }

def compute_mean_timeseries(
    ds: xr.Dataset,
    var: str,
    labels: xr.Dataset,
    output_var_name: str | None = None,
    n_samples: int | float | None = None,
    min_samples: int = 1,
    seed: int | None = None,
) -> xr.Dataset:
    """Compute mean timeseries per cluster given a small dataset and label points.

    Each labelled point is matched to a grid cell of ``ds``: **by id** when the labels carry
    ``cell_id`` and ``ds`` carries the grid's ``original_cell_id`` map (which
    :meth:`~climclusterpy.data.registry.DataRegistry.attach_original_ids` adds), otherwise
    to the cell whose latitude and whose longitude are nearest to it (each axis separately,
    so a point off the grid is snapped to the closest cell rather than rejected). The
    cluster's series is the NaN-skipping mean over its matched cells at every time step.

    Parameters
    ----------
    ds : xarray.Dataset
        Gridded data with coordinates named exactly ``latitude`` and ``longitude``
        (standardize them first if needed), and optionally an ``original_cell_id``
        variable on ``(latitude, longitude)`` giving each cell's id (``-1`` off the grid).
        The variable ``var`` must have dimensions
        ``(time, latitude, longitude)``, ``(latitude, longitude, time)`` or
        ``(latitude, longitude)``; a 2-D variable is treated as a single time step.
        Any other dimension order is read as ``(time, latitude, longitude)``
        without a check.
    var : str
        Name of the data variable in ``ds`` to average.
    labels : xarray.Dataset
        One entry per cell along a ``point`` dimension, with variables ``latitude``,
        ``longitude`` (degrees) and ``cluster_label`` (integer). Points with a label
        of 0 or below are ignored. An optional ``cell_id`` variable is used to match
        points when ``ds`` has ``original_cell_id``; before 0.3.0 it was ignored and
        matching was always by coordinates.
    output_var_name : str, optional
        Name of the output variable. Default ``None``: ``f"{var}_mean"``.
    n_samples : int or float, optional
        Average over a random subset of each cluster's cells instead of all of them:
        an int is a fixed number of cells per cluster (must be positive), a float in
        (0, 1) is a fraction of each cluster's cells, rounded up and at least 1.
        ``1.0`` is rejected; use ``None`` for all cells. Default ``None``. Cells are
        drawn without replacement; a cluster with fewer cells than requested uses all
        of them.
    min_samples : int, optional
        Lower bound on the number of cells drawn per cluster when ``n_samples`` is
        given (default 1, must be at least 1). A cluster with fewer cells uses all of
        them and a warning is logged. Has no effect when ``n_samples`` is ``None``.
    seed : int, optional
        Seed for the cell draw when ``n_samples`` is given. Default ``None``: a
        different draw on every call.

    Returns
    -------
    xarray.Dataset
        One variable (``output_var_name``, or ``f"{var}_mean"``) with dimensions
        ``("time", "cluster")``. The ``cluster`` coordinate lists the positive labels
        in ascending order; the ``time`` coordinate is copied from ``ds`` (or is
        ``0, 1, …`` when ``ds`` has none).

    Raises
    ------
    ValueError
        If ``var`` is not in ``ds``, ``n_samples``/``min_samples`` is invalid, a
        ``cell_id`` is not on the grid of ``ds``, or no point has a positive label.
    """
    if var not in ds:
        raise ValueError(f"Variable '{var}' not found in dataset")
    lat_name = "latitude"
    lon_name = "longitude"

    lat_vals = ds[lat_name].values
    lon_vals = ds[lon_name].values
    time_vals = ds.get(
        "time", xr.DataArray(np.arange(ds.sizes.get("time", 1)), dims=["time"])
    ).values

    # Prepare index list per cluster
    clabs = labels["cluster_label"].values
    # Match by id when both sides carry one; coordinates only when they cannot.
    id_to_index: dict[int, tuple[int, int]] | None = None
    if "cell_id" in labels and "original_cell_id" in ds:
        id_grid = ds["original_cell_id"].transpose(lat_name, lon_name).values
        rows, cols = np.nonzero(id_grid >= 0)
        id_to_index = {int(i): (int(r), int(c)) for i, r, c in zip(id_grid[rows, cols], rows, cols)}
        ids = np.asarray(labels["cell_id"].values).astype(int)
        unknown = sorted({int(i) for i, c in zip(ids, clabs) if int(c) > 0} - set(id_to_index))
        if unknown:
            raise ValueError(
                f"{len(unknown)} labelled cell_id(s) are not on the grid of ds (first: "
                f"{unknown[:5]}). The labels and ds must describe the same grid."
            )
    lats = labels["latitude"].values
    lons = labels["longitude"].values
    clusters = sorted({int(c) for c in clabs if int(c) > 0})
    if not clusters:
        raise ValueError(
            "No labelled cell has a positive cluster label (labels of 0 or below are "
            "ignored), so there is no cluster to average."
        )

    cluster_to_indices: dict[int, list[tuple[int, int]]] = {c: [] for c in clusters}
    for i in range(len(clabs)):
        c = int(clabs[i])
        if c <= 0:
            continue
        if id_to_index is not None:
            cluster_to_indices[c].append(id_to_index[int(ids[i])])
            continue
        lat_idx = int(np.argmin(np.abs(lat_vals - lats[i])))
        lon_idx = int(np.argmin(np.abs(lon_vals - lons[i])))
        cluster_to_indices[c].append((lat_idx, lon_idx))

    # Apply sampling if requested
    if n_samples is not None:
        from climclusterpy.timeseries.sampling import sample_cluster_indices

        cluster_to_indices = sample_cluster_indices(
            cluster_to_indices,
            n_samples=n_samples,
            min_samples=min_samples,
            seed=seed,
        )

    data = ds[var].values  # shape (time, lat, lon) or (lat, lon, time)
    # normalize to (time, lat, lon)
    if ds[var].dims[:3] == (lat_name, lon_name, "time"):
        data = np.transpose(data, (2, 0, 1))
    elif ds[var].dims[:2] == (lat_name, lon_name) and len(ds[var].dims) == 2:
        data = data[None, ...]

    means = []
    for c in clusters:
        idxs = cluster_to_indices.get(c, [])
        if not idxs:
            means.append(np.full((len(time_vals),), np.nan))
            continue
        lat_idx = [i for i, _ in idxs]
        lon_idx = [j for _, j in idxs]
        vals = data[:, lat_idx, lon_idx]  # (time, points)
        means.append(np.nanmean(vals, axis=1))

    arr = np.stack(means, axis=1) if means else np.empty((len(time_vals), 0))
    name = output_var_name or f"{var}_mean"
    return xr.Dataset(
        {name: (("time", "cluster"), arr)}, coords={"time": time_vals, "cluster": clusters}
    )

def build_cluster_mean_timeseries(
    climate_data: dict[str, xr.Dataset],
    registry: DataRegistry,
    labels_csv: str,
    n_samples: int | float | None = None,
    min_samples: int = 1,
    seed: int | None = None,
) -> xr.Dataset:
    """The mean timeseries of every climate input, per cluster.

    :func:`compute_mean_timeseries` for each climate dataset of ``climate_data`` at once,
    plus each cluster's centre coordinates and, when ``climate_data`` holds them, its most
    common ``soil_code`` and mean ``soil_depth``. The result can be handed to
    :func:`climclusterpy.build_reduced_grid`.

    Parameters
    ----------
    climate_data : dict of str to xarray.Dataset
        The loaded inputs, keyed by field name (``"temperature"``, ``"precipitation"``, ...),
        as :func:`climclusterpy.data.load_climate_and_soil_data` returns them. Keys other
        than the standard climate fields and the two soil fields are averaged under their
        own name.
    registry : DataRegistry
        The original grid; places each labelled ``cell_id``.
    labels_csv : str
        The per-cell label table (``cell_id``, ``latitude``, ``longitude``,
        ``cluster_label``); labels of 0 or below are ignored. A table with no positive
        label, or with a ``cell_id`` that is not a valid cell of ``registry``, is refused
        with a ``ValueError``.
    n_samples, min_samples, seed
        As :func:`compute_mean_timeseries`: average over a random subset of each
        cluster's cells instead of all of them.

    Returns
    -------
    xarray.Dataset
        One variable per climate field over ``("time", "cluster")``, plus ``cluster_label``,
        ``center_lat``/``center_lon`` (also as ``latitude``/``longitude``) and the soil
        fields over ``("cluster",)``.
    """
    ds_labels = load_cluster_labels_as_xarray(labels_csv, registry)

    # Map of climate dict keys to (source var name, standardized output name)
    var_map = {
        "precipitation": ("tpr", "precipitation"),
        "temperature": ("TMean", "temperature"),
        "short_wave_radiation": ("SWR", "short_wave_radiation"),
        "long_wave_radiation": ("LWR", "long_wave_radiation"),
        "humid": ("humid", "humid"),
    }

    out_vars = {}
    for key, (src_var_expected, out_name) in var_map.items():
        if key not in climate_data:
            continue
        ds = climate_data[key]
        var_candidates = list(ds.data_vars)
        # Prefer canonical out_name in dataset, then expected src var from original naming
        src_var = None
        if out_name in var_candidates:
            src_var = out_name
        elif src_var_expected in var_candidates:
            src_var = src_var_expected
        else:
            # Filter to time-varying variables only
            timed = [v for v in var_candidates if "time" in ds[v].dims]
            if len(timed) == 1:
                src_var = timed[0]
            else:
                # Try best textual match among time-varying vars
                prefer = [v for v in timed if (out_name in v) or (v in out_name)]
                src_var = (
                    prefer[0]
                    if prefer
                    else (timed[0] if timed else (var_candidates[0] if var_candidates else None))
                )
        if src_var is None:
            continue
        res = compute_mean_timeseries(
            ds,
            src_var,
            ds_labels,
            output_var_name=out_name,
            n_samples=n_samples,
            min_samples=min_samples,
            seed=seed,
        )
        out_vars[out_name] = res[out_name]

    # Anything in climate_data that wasn't one of the known keys above -- a custom climate
    # variable from a non-LPJmL input -- gets the same per-cluster mean treatment, named
    # after its own dict key instead of being silently dropped.
    for key, ds in climate_data.items():
        if key in var_map or key in out_vars or key in ("soil_code", "soil_depth"):
            continue
        var_candidates = list(ds.data_vars)
        if key in var_candidates:
            src_var = key
        else:
            timed = [v for v in var_candidates if "time" in ds[v].dims]
            src_var = timed[0] if timed else (var_candidates[0] if var_candidates else None)
        if src_var is None:
            continue
        res = compute_mean_timeseries(
            ds,
            src_var,
            ds_labels,
            output_var_name=key,
            n_samples=n_samples,
            min_samples=min_samples,
            seed=seed,
        )
        out_vars[key] = res[key]

    # Merge into single dataset
    ds_out = xr.Dataset(out_vars)

    # Ensure we have a 'cluster' coordinate present; if not, derive from labels
    if not ds_out.data_vars:
        # Construct a structural-only dataset from labels to satisfy downstream expectations
        clusters_from_labels = sorted(
            {int(c) for c in ds_labels["cluster_label"].values if int(c) > 0}
        )
        if clusters_from_labels:
            ds_out = ds_out.assign_coords(cluster=clusters_from_labels)
    else:
        # Pick the first variable to copy coords from
        first_var = next(iter(ds_out.data_vars))
        if "cluster" not in ds_out.coords and "cluster" in ds_out[first_var].coords:
            ds_out = ds_out.assign_coords(cluster=ds_out[first_var].coords["cluster"])

    # Normalize cluster labels and center coordinates for structural parity
    if ("cluster" in ds_out.dims) or ("cluster" in ds_out.coords):
        if "cluster" not in ds_out.coords:
            ds_out = ds_out.assign_coords(cluster=np.arange(ds_out.sizes.get("cluster", 0)))
        clusters = [int(c) for c in np.asarray(ds_out["cluster"].values).tolist()]
        if "cluster_label" not in ds_out:
            ds_out["cluster_label"] = (("cluster",), clusters)

        # Cluster centres from the labels; a cluster without one gets NaN coordinates
        centers_ds = calculate_cluster_centers_xarray(ds_labels)
        center_lat_map = {
            int(lbl): float(lat)
            for lbl, lat in zip(
                centers_ds["cluster_label"].values.tolist(),
                centers_ds["center_lat"].values.tolist(),
            )
        }
        center_lon_map = {
            int(lbl): float(lon)
            for lbl, lon in zip(
                centers_ds["cluster_label"].values.tolist(),
                centers_ds["center_lon"].values.tolist(),
            )
        }
        center_lats = [center_lat_map.get(int(c), np.nan) for c in clusters]
        center_lons = [center_lon_map.get(int(c), np.nan) for c in clusters]

        ds_out["center_lat"] = (("cluster",), center_lats)
        ds_out["center_lon"] = (("cluster",), center_lons)
        ds_out["center_lat"].attrs = {
            "units": "degrees_north",
            "description": "Latitude of cluster center",
            "_FillValue": np.nan,
        }
        ds_out["center_lon"].attrs = {
            "units": "degrees_east",
            "description": "Longitude of cluster center",
            "_FillValue": np.nan,
        }
        # Also expose latitude/longitude per cluster
        ds_out["latitude"] = (("cluster",), center_lats)
        ds_out["longitude"] = (("cluster",), center_lons)

        # Also provide soil fields for mean mode: soil_code as mode across cluster, soil_depth as mean across cluster
        soil_code_ds = climate_data.get("soil_code")
        if soil_code_ds is not None and isinstance(soil_code_ds, xr.Dataset):
            # Determine variable name for soil code
            soil_code_var = None
            for v in soil_code_ds.data_vars:
                if (
                    ("soil" in v.lower())
                    or ("texture" in v.lower())
                    or (v.lower() in ("soil_code",))
                ):
                    soil_code_var = v
                    break
            if soil_code_var is None and len(soil_code_ds.data_vars) == 1:
                soil_code_var = next(iter(soil_code_ds.data_vars))
            # Build mapping cluster -> indices
            sc_lat_name = next((n for n in LAT_NAMES if n in soil_code_ds.coords), None)
            sc_lon_name = next((n for n in LON_NAMES if n in soil_code_ds.coords), None)
            if sc_lat_name is None or sc_lon_name is None:
                raise ValueError(
                    f"Soil code dataset missing lat/lon coords. Available: {list(soil_code_ds.coords)}"
                )
            lat_vals = soil_code_ds[sc_lat_name].values
            lon_vals = soil_code_ds[sc_lon_name].values
            clabs = ds_labels["cluster_label"].values
            lats = ds_labels["latitude"].values
            lons = ds_labels["longitude"].values
            cluster_to_idx: dict[int, list[tuple[int, int]]] = {c: [] for c in clusters}
            for i in range(len(clabs)):
                c = int(clabs[i])
                if c not in cluster_to_idx:
                    continue
                # Strict equality on standardized grids
                lat_matches = np.where(lat_vals == lats[i])[0]
                lon_matches = np.where(lon_vals == lons[i])[0]
                if lat_matches.size == 0 or lon_matches.size == 0:
                    raise ValueError(
                        "Label coordinate not found on soil_code grid; grids must be standardized"
                    )
                cluster_to_idx[c].append((int(lat_matches[0]), int(lon_matches[0])))
            soil_codes = []
            for c in clusters:
                idxs = cluster_to_idx.get(c, [])
                if not idxs or soil_code_var is None:
                    soil_codes.append(np.nan)
                    continue
                vals = [
                    soil_code_ds[soil_code_var].isel({sc_lat_name: i, sc_lon_name: j}).values.item()
                    for i, j in idxs
                ]
                vals_arr = np.array(vals)
                vals_arr = vals_arr[~np.isnan(vals_arr)] if vals_arr.size else vals_arr
                if vals_arr.size == 0:
                    soil_codes.append(np.nan)
                else:
                    mode_res = scipy_mode(vals_arr, keepdims=False)
                    mode_val = (
                        mode_res.mode
                        if hasattr(mode_res, "mode")
                        else (mode_res[0] if len(mode_res) else np.nan)
                    )
                    soil_codes.append(
                        float(mode_val)
                        if not (isinstance(mode_val, float) and np.isnan(mode_val))
                        else np.nan
                    )
            ds_out["soil_code"] = (("cluster",), soil_codes)

        soil_depth_ds = climate_data.get("soil_depth")
        if soil_depth_ds is not None and isinstance(soil_depth_ds, xr.Dataset):
            # Determine variable name for soil depth
            depth_var = None
            for v in soil_depth_ds.data_vars:
                if ("soil" in v.lower()) or ("depth" in v.lower()):
                    depth_var = v
                    break
            if depth_var is None and len(soil_depth_ds.data_vars) == 1:
                depth_var = next(iter(soil_depth_ds.data_vars))
            dd_lat_name = next((n for n in LAT_NAMES if n in soil_depth_ds.coords), None)
            dd_lon_name = next((n for n in LON_NAMES if n in soil_depth_ds.coords), None)
            if dd_lat_name is None or dd_lon_name is None:
                raise ValueError(
                    f"Soil depth dataset missing lat/lon coords. Available: {list(soil_depth_ds.coords)}"
                )
            lat_vals = soil_depth_ds[dd_lat_name].values
            lon_vals = soil_depth_ds[dd_lon_name].values
            clabs = ds_labels["cluster_label"].values
            lats = ds_labels["latitude"].values
            lons = ds_labels["longitude"].values
            cluster_to_idx: dict[int, list[tuple[int, int]]] = {c: [] for c in clusters}
            for i in range(len(clabs)):
                c = int(clabs[i])
                if c not in cluster_to_idx:
                    continue
                lat_matches = np.where(lat_vals == lats[i])[0]
                lon_matches = np.where(lon_vals == lons[i])[0]
                if lat_matches.size == 0 or lon_matches.size == 0:
                    raise ValueError(
                        "Label coordinate not found on soil_depth grid; grids must be standardized"
                    )
                cluster_to_idx[c].append((int(lat_matches[0]), int(lon_matches[0])))
            soil_depths = []
            for c in clusters:
                idxs = cluster_to_idx.get(c, [])
                if not idxs or depth_var is None:
                    soil_depths.append(np.nan)
                    continue
                vals = [
                    soil_depth_ds[depth_var].isel({dd_lat_name: i, dd_lon_name: j}).values.item()
                    for i, j in idxs
                ]
                vals_arr = np.array(vals)
                vals_arr = vals_arr[~np.isnan(vals_arr)] if vals_arr.size else vals_arr
                soil_depths.append(float(np.nanmean(vals_arr)) if vals_arr.size else np.nan)
            ds_out["soil_depth"] = (("cluster",), soil_depths)

    return ds_out




def load_cluster_labels_as_xarray(labels_file, registry: object):
    """Load cluster labels and convert to xarray Dataset with proper coordinate handling."""
    logger.info(f"Loading cluster labels from: {labels_file}")

    # Read the CSV file into a pandas DataFrame first (minimal pandas usage)
    import pandas as pd

    df_labels = pd.read_csv(labels_file)

    # Check required columns
    required_cols = ["latitude", "longitude", "cluster_label"]
    missing_cols = [col for col in required_cols if col not in df_labels.columns]
    if missing_cols:
        raise ValueError(f"Missing required columns in label file: {missing_cols}")

    # Convert cluster_label to numeric and filter out non-numeric values
    initial_count = len(df_labels)
    df_labels["cluster_label"] = pd.to_numeric(df_labels["cluster_label"], errors="coerce")
    df_labels = df_labels.dropna(subset=["cluster_label"])
    df_labels["cluster_label"] = df_labels["cluster_label"].astype(int)

    # Remove any cluster_label = 0 which might be from header issues
    df_labels = df_labels[df_labels["cluster_label"] > 0]

    logger.info(
        f"After cleaning: {len(df_labels)} valid cluster entries (removed {initial_count - len(df_labels)} invalid entries)"
    )

    if df_labels.empty:
        raise ValueError(
            f"The label table {labels_file} has no cell with a positive cluster label (labels "
            "of 0 or below are ignored), so there is no cluster to average."
        )

    # Map cell_id to lat/lon via registry
    if "cell_id" not in df_labels.columns:
        raise ValueError("Expected label CSV to include 'cell_id' column")
    from climclusterpy.timeseries.representative import _place_labels_by_cell_id

    df_labels = _place_labels_by_cell_id(df_labels, registry)

    # Remove duplicates
    n_duplicates = df_labels.duplicated(subset=["latitude", "longitude"]).sum()
    if n_duplicates > 0:
        logger.warning(
            f"WARNING: Found {n_duplicates} duplicate lat/lon combinations. Keeping first occurrence."
        )
        df_labels = df_labels.drop_duplicates(subset=["latitude", "longitude"], keep="first")

    # Convert to xarray Dataset
    # Create a point dimension for spatial indexing
    df_labels = df_labels.reset_index(drop=True)
    df_labels["point"] = df_labels.index

    data_vars = {
        "cluster_label": (("point",), df_labels["cluster_label"].values),
        "latitude": (("point",), df_labels["latitude"].values),
        "longitude": (("point",), df_labels["longitude"].values),
        "cell_id": (("point",), df_labels["cell_id"].values),
    }
    ds_labels = xr.Dataset(data_vars)

    # Set coordinates
    ds_labels = ds_labels.set_coords(["latitude", "longitude"])

    # Add metadata
    ds_labels.attrs["description"] = "Cluster labels for climate data points"
    ds_labels.attrs["source_file"] = labels_file
    ds_labels["cluster_label"].attrs["description"] = "Cluster assignment for each spatial point"
    ds_labels["latitude"].attrs["units"] = "degrees_north"
    ds_labels["longitude"].attrs["units"] = "degrees_east"

    unique_clusters = np.unique(ds_labels["cluster_label"].values)
    logger.info(
        f"Found {len(unique_clusters)} unique clusters ranging from {min(unique_clusters)} to {max(unique_clusters)}"
    )

    return ds_labels
