Source code for climclusterpy.supervision.ecoregions

"""Supervision from ecoregion polygons (Ecoregions2017 or your own shapefile).

Assigns each grid cell the ecoregion whose polygon it falls in, producing the same
``cell_id, latitude, longitude, cluster_label`` table as the other supervision
sources. Use this when you have an authoritative external delineation of ecological
regions and want the learned metric to reproduce *those* boundaries.

Ecoregions with only a few cells on the grid make poor training classes: metric learning
draws several same-class neighbours for every cell, which a region of three cells cannot
supply. :func:`merge_small_ecoregions` therefore absorbs each region below a minimum size into
the nearest larger region, preferring one of the same biome and realm within a distance cap.

A label CSV is reused when it was written for the same grid, shapefile and id column,
because the point-in-polygon join over a global grid is slow and its result never changes
for the same inputs. Its inputs are recorded beside it, so other inputs are never served it.

``geopandas`` (the ``[ecoregions]`` extra) is imported **inside** the functions that
need it, so this module can be imported — and documented by autodoc — without the
geo stack installed.
"""

from __future__ import annotations

import glob
import logging
import os
from typing import TYPE_CHECKING

import numpy as np
import pandas as pd

from climclusterpy._cache import file_identity, fingerprint, read_fingerprint, write_fingerprint
from climclusterpy.data.registry import DataRegistry

logger = logging.getLogger(__name__)

if TYPE_CHECKING:  # annotations only; the runtime import is inside each function
    import geopandas as gpd

__all__ = [
    "DEFAULT_MERGE_MAX_KM",
    "assign_cells_to_ecoregions",
    "compute_ecoregion_centers",
    "load_ecoregions_shapefile",
    "merge_small_ecoregions",
    "perform_ecoregion_assignment",
]

#: Distance cap, in km, within which :func:`merge_small_ecoregions` looks for a merge target
#: before it falls back to any distance. Neighbouring cells share a climate; a same-biome region
#: on another continent does not, so a nearby region of another biome is the better target.
DEFAULT_MERGE_MAX_KM = 1500.0

_EARTH_RADIUS_KM = 6371.0

#: Where a small ecoregion may go, best first: ``(name, same_biome, same_realm, within_cap)``.
#: Distance is a gate, not a tie-break: biome and realm only choose among the regions within
#: the cap, and the uncapped rungs are reached only when nothing large is within it (a remote
#: island).
_MERGE_RUNGS = (
    ("near_same_biome_same_realm", True, True, True),
    ("near_same_realm_any_biome", False, True, True),
    ("near_any", False, False, True),
    ("far_same_biome", True, False, False),
    ("far_any", False, False, False),
)

#: The attribute columns of the Ecoregions2017 shapefile that carry its own hierarchy.
_BIOME_COLUMN = "BIOME_NUM"
_REALM_COLUMN = "REALM"


def _import_geopandas():
    """Import geopandas, or explain which extra provides it."""
    try:
        import geopandas as gpd
    except ImportError as exc:  # pragma: no cover - exercised by the bare-core gate
        raise ImportError(
            "Ecoregion supervision needs the [ecoregions] extra (geopandas). Install with: "
            "pip install 'climclusterpy[ecoregions]' — or, if the pip build fails on GEOS/PROJ, "
            "conda install -c conda-forge geopandas"
        ) from exc
    return gpd

def load_ecoregions_shapefile(shapefile_path: str) -> gpd.GeoDataFrame:
    """Load ecoregions shapefile.

    Args:
        shapefile_path: Path to the .shp file

    Returns
    -------
        GeoDataFrame with ecoregion polygons and attributes

    Raises
    ------
        FileNotFoundError: If shapefile doesn't exist
        ValueError: If shapefile is empty or invalid
    """
    gpd = _import_geopandas()

    if not os.path.exists(shapefile_path):
        raise FileNotFoundError(f"Ecoregions shapefile not found: {shapefile_path}")

    logger.info(f"[Ecoregions] Loading shapefile: {shapefile_path}")
    gdf = gpd.read_file(shapefile_path)

    if gdf.empty:
        raise ValueError(f"Ecoregions shapefile is empty: {shapefile_path}")

    logger.info(f"[Ecoregions] Loaded {len(gdf)} ecoregions")
    logger.info(f"[Ecoregions] Columns: {list(gdf.columns)}")
    logger.info(f"[Ecoregions] CRS: {gdf.crs}")

    return gdf




def _one_label_per_cell(df: pd.DataFrame) -> pd.DataFrame:
    """Keep one ecoregion per cell: the lowest id when a cell matched several.

    A cell centre on the border of two polygons intersects both, and a cell in no polygon can
    be equally near two of them. The lowest id is arbitrary but deterministic and independent
    of the order of the shapefile's rows.
    """
    n_matches = df.groupby("cell_id")["cluster_label"].transform("size")
    tied = df[n_matches > 1]
    if not tied.empty:
        examples = [
            f"({grp['latitude'].iloc[0]:.2f}, {grp['longitude'].iloc[0]:.2f}): "
            f"{sorted(int(v) for v in grp['cluster_label'])}"
            for _, grp in list(tied.groupby("cell_id"))[:5]
        ]
        logger.warning(
            f"[Ecoregions] {tied['cell_id'].nunique()} cell(s) matched more than one ecoregion "
            f"(on a shared border, or equally near two when no polygon covers them); each "
            f"keeps the lowest id. First: {'; '.join(examples)}"
        )
    return (
        df.sort_values(["cell_id", "cluster_label"], kind="stable")
        .drop_duplicates("cell_id", keep="first")
        .reset_index(drop=True)
    )

def assign_cells_to_ecoregions(
    registry: DataRegistry,
    ecoregions_gdf: gpd.GeoDataFrame,
    ecoregion_id_column: str = "ECO_ID",
) -> pd.DataFrame:
    """Assign each valid grid cell to an ecoregion based on spatial intersection.

    Uses vectorized spatial join (geopandas.sjoin) for fast assignment.

    Args:
        registry: DataRegistry containing grid information and valid cell mask
        ecoregions_gdf: GeoDataFrame with ecoregion polygons
        ecoregion_id_column: Column name in the shapefile containing ecoregion IDs

    Returns
    -------
        DataFrame with columns: cell_id, latitude, longitude, cluster_label
        where cluster_label is the ecoregion ID. Exactly one row per valid cell: a cell that
        matches several ecoregions (on a shared border, or equally near two) keeps the lowest
        id, with a warning.

    Raises
    ------
        ValueError: If ecoregion_id_column not found in shapefile
    """
    gpd = _import_geopandas()

    if ecoregion_id_column not in ecoregions_gdf.columns:
        raise ValueError(
            f"Column '{ecoregion_id_column}' not found in ecoregions shapefile. "
            f"Available columns: {list(ecoregions_gdf.columns)}"
        )

    logger.info(
        f"[Ecoregions] Assigning cells to ecoregions using column '{ecoregion_id_column}'..."
    )

    # Get valid lat/lon coordinates from registry
    valid_lats, valid_lons = registry.get_valid_lat_lon()
    n_cells = len(valid_lats)

    logger.info(
        f"[Ecoregions] Processing {n_cells} valid grid cells using vectorized spatial join..."
    )

    # Ensure ecoregions are in a geographic CRS (lat/lon)
    if ecoregions_gdf.crs is None:
        logger.warning("[Ecoregions] Warning: No CRS defined. Assuming EPSG:4326 (WGS84)")
        ecoregions_gdf = ecoregions_gdf.set_crs("EPSG:4326")
    elif not ecoregions_gdf.crs.is_geographic:
        logger.info(f"[Ecoregions] Converting from {ecoregions_gdf.crs} to EPSG:4326...")
        ecoregions_gdf = ecoregions_gdf.to_crs("EPSG:4326")

    # Normalize grid longitudes to [-180, 180) to match standard ecoregion shapefiles
    logger.info("[Ecoregions] Normalizing grid coordinates to [-180, 180)...")
    normalized_lons = np.array(valid_lons, dtype=float)
    # Bring longitudes into [-180, 180)
    normalized_lons = ((normalized_lons + 180) % 360) - 180

    # Check and report coordinate ranges for debugging
    logger.info(f"[Ecoregions] Grid lat range: [{valid_lats.min():.2f}, {valid_lats.max():.2f}]")
    logger.info(
        f"[Ecoregions] Grid lon range (normalized): [{normalized_lons.min():.2f}, {normalized_lons.max():.2f}]"
    )

    # Get ecoregion bounds for comparison
    eco_bounds = ecoregions_gdf.total_bounds  # [minx, miny, maxx, maxy]
    logger.info(
        f"[Ecoregions] Ecoregion bounds: lon [{eco_bounds[0]:.2f}, {eco_bounds[2]:.2f}], lat [{eco_bounds[1]:.2f}, {eco_bounds[3]:.2f}]"
    )

    # Create GeoDataFrame of all grid points
    logger.info("[Ecoregions] Creating points GeoDataFrame...")
    points_df = pd.DataFrame(
        {
            "cell_id": list(range(n_cells)),
            "latitude": valid_lats,
            "longitude": normalized_lons,
        }
    )

    # Create Point geometries from coordinates
    points_gdf = gpd.GeoDataFrame(
        points_df,
        geometry=gpd.points_from_xy(points_df.longitude, points_df.latitude),
        crs="EPSG:4326",
    )

    # Perform vectorized spatial join (point-in-polygon)
    # Use 'intersects' to include points on polygon boundaries (more inclusive than 'within')
    logger.info(
        "[Ecoregions] Performing vectorized spatial join with 'intersects' predicate (includes boundary points)..."
    )
    matched = gpd.sjoin(
        points_gdf,
        ecoregions_gdf[[ecoregion_id_column, "geometry"]],
        how="left",
        predicate="intersects",
    )

    # Filter to only successfully matched points and rename ecoregion column
    df_assigned = matched.dropna(subset=[ecoregion_id_column]).copy()
    df_assigned = df_assigned.rename(columns={ecoregion_id_column: "cluster_label"})

    # Keep only the needed columns
    df = df_assigned[["cell_id", "latitude", "longitude", "cluster_label"]].copy()

    # Ensure cluster_label is int
    df["cluster_label"] = df["cluster_label"].astype(int)
    df = _one_label_per_cell(df)

    n_assigned = len(df)
    n_unassigned = n_cells - n_assigned
    logger.info(f"[Ecoregions] Assigned {n_assigned} cells to ecoregions")
    logger.info(f"[Ecoregions] Unassigned cells after intersects join: {n_unassigned}")

    if n_unassigned > 0:
        logger.info(
            "[Ecoregions] Running nearest-neighbor fallback for unassigned cells "
            "(ensures every valid cell receives an ecoregion label)..."
        )
        assigned_ids = set(df["cell_id"].tolist())
        unassigned_points = points_gdf[~points_gdf["cell_id"].isin(assigned_ids)].copy()

        nearest = gpd.sjoin_nearest(
            unassigned_points,
            ecoregions_gdf[[ecoregion_id_column, "geometry"]],
            how="left",
            distance_col="distance_deg",
        )

        if nearest[ecoregion_id_column].isna().any():
            raise RuntimeError(
                "Nearest-neighbor assignment failed for some cells; "
                "please verify the ecoregion shapefile covers the study area."
            )

        nearest = nearest.rename(columns={ecoregion_id_column: "cluster_label"})
        nearest = nearest[["cell_id", "latitude", "longitude", "cluster_label"]]
        nearest["cluster_label"] = nearest["cluster_label"].astype(int)
        nearest = _one_label_per_cell(nearest)

        df = pd.concat([df, nearest], ignore_index=True)
        # Restore original ordering by cell_id to remain compatible with downstream code
        df = df.sort_values("cell_id").reset_index(drop=True)

        logger.info(
            f"[Ecoregions] Fallback assigned {len(nearest)} additional cells "
            "(all valid cells now labeled)."
        )

    return df

def compute_ecoregion_centers(df_labels: pd.DataFrame) -> pd.DataFrame:
    """Compute center coordinates for each ecoregion.

    Args:
        df_labels: DataFrame with columns: cell_id, latitude, longitude, cluster_label

    Returns
    -------
        DataFrame with columns: cluster_label, center_lat, center_lon
    """
    logger.info("[Ecoregions] Computing ecoregion centers...")

    centers = []
    for ecoregion_id in sorted(df_labels["cluster_label"].unique()):
        cells_in_ecoregion = df_labels[df_labels["cluster_label"] == ecoregion_id]

        # Use median of absolute latitude to handle equator-spanning regions
        abs_lats = np.abs(cells_in_ecoregion["latitude"].values)
        center_lat_abs = float(np.median(abs_lats))

        # Find the actual latitude closest to the median absolute value
        closest_idx = np.argmin(
            np.abs(np.abs(cells_in_ecoregion["latitude"].values) - center_lat_abs)
        )
        center_lat = float(cells_in_ecoregion["latitude"].iloc[closest_idx])

        # Use mean longitude
        center_lon = float(cells_in_ecoregion["longitude"].mean())

        centers.append(
            {
                "cluster_label": ecoregion_id,
                "center_lat": center_lat,
                "center_lon": center_lon,
            }
        )

    df_centers = pd.DataFrame(centers)
    logger.info(f"[Ecoregions] Computed centers for {len(df_centers)} ecoregions")

    return df_centers




def _region_attribute(
    ecoregions_gdf: gpd.GeoDataFrame, id_column: str, column: str
) -> dict[int, object] | None:
    """``{region id: value}`` of one attribute column, or ``None`` when the file lacks it."""
    if column not in ecoregions_gdf.columns:
        return None
    table = ecoregions_gdf[[id_column, column]].drop_duplicates()
    conflicting = table[table.duplicated(id_column, keep=False)]
    if not conflicting.empty:
        raise ValueError(
            f"The ecoregion shapefile gives region(s) "
            f"{sorted(int(v) for v in conflicting[id_column].unique())[:10]} more than one "
            f"{column!r} value; one region must have one {column}."
        )
    return {int(k): v for k, v in zip(table[id_column], table[column])}


def _unit_vectors(lat: np.ndarray, lon: np.ndarray) -> np.ndarray:
    """Cells as unit vectors, so great-circle nearness becomes a dot product."""
    la, lo = np.deg2rad(lat), np.deg2rad(lon)
    return np.stack([np.cos(la) * np.cos(lo), np.cos(la) * np.sin(lo), np.sin(la)], axis=1)

def merge_small_ecoregions(
    labels: pd.DataFrame,
    min_cells: int = 10,
    *,
    biome: dict | pd.Series | None = None,
    realm: dict | pd.Series | None = None,
    max_km: float = DEFAULT_MERGE_MAX_KM,
) -> tuple[pd.DataFrame, pd.DataFrame]:
    """Absorb every ecoregion with fewer than ``min_cells`` cells into a larger neighbour.

    Metric learning draws several same-class neighbours for every cell, which a region with
    only a few cells on the grid cannot supply. Each such region is merged, in one pass, into
    the *nearest* region that has at least ``min_cells`` cells, where nearest is the smallest
    great-circle distance between any cell of the one and any cell of the other (so a region
    that touches the target wins over one whose middle happens to be closer). Among the
    regions within ``max_km``, one of the same biome and realm is preferred, then one of the
    same realm, then any; only when no large region lies within ``max_km`` (a remote island)
    is the nearest one of the same biome, then any, taken at any distance. Only regions that
    are already large enough are targets, so no chains form and the order the small regions
    are processed in does not matter.

    Parameters
    ----------
    labels : pandas.DataFrame
        One row per grid cell with ``latitude``, ``longitude`` (degrees) and
        ``cluster_label`` (the ecoregion id), e.g. from :func:`assign_cells_to_ecoregions`.
        Other columns are carried through unchanged.
    min_cells : int, optional
        Smallest number of cells a region keeps its own class with (default 10). ``0`` or
        ``1`` merges nothing.
    biome, realm : dict or pandas.Series, optional
        Each region's biome and realm (``{region id: value}``), e.g. the ``BIOME_NUM`` and
        ``REALM`` attributes of Ecoregions2017. Without one, the preference for it is left out
        and distance alone decides among the remaining candidates.
    max_km : float, optional
        Distance cap of the preferred targets, in km (default 1,500).

    Returns
    -------
    merged : pandas.DataFrame
        ``labels`` with ``cluster_label`` replaced by the merged class (the id of the region
        it was absorbed into, or its own).
    merges : pandas.DataFrame
        One row per absorbed region: ``source_id``, ``source_cells``, ``target_id``,
        ``target_cells``, ``rule`` (which preference chose the target), ``distance_km`` and,
        when given, the biome and realm of both.

    Raises
    ------
    ValueError
        If ``min_cells`` is negative, a required column is missing, a region lacks a biome or
        realm entry, or no region has ``min_cells`` cells.
    """
    if int(min_cells) != min_cells or min_cells < 0:
        raise ValueError(f"min_cells must be a whole number >= 0, got {min_cells!r}.")
    missing = {"latitude", "longitude", "cluster_label"} - set(labels.columns)
    if missing:
        raise ValueError(f"labels lacks column(s) {sorted(missing)}.")

    eco = labels["cluster_label"].to_numpy().astype(np.int64)
    ids, counts = np.unique(eco, return_counts=True)
    size = dict(zip(ids.tolist(), counts.tolist()))
    small = [i for i in ids.tolist() if size[i] < min_cells]
    merge_columns = [
        "source_id", "source_cells", "target_id", "target_cells", "rule", "distance_km",
    ]  # fmt: skip
    if not small:
        return labels.copy(), pd.DataFrame(columns=merge_columns)
    if len(small) == len(ids):
        raise ValueError(
            f"No ecoregion has {min_cells} or more cells on this grid (the largest has "
            f"{int(counts.max())}), so there is nothing to merge into. Lower min_cells."
        )

    def _codes(attribute, name):
        if attribute is None:
            return None
        attribute = dict(attribute)
        unknown = [i for i in ids.tolist() if i not in attribute]
        if unknown:
            raise ValueError(f"No {name} given for ecoregion(s) {unknown[:10]}.")
        codes, _ = pd.factorize(pd.Series([attribute[i] for i in ids.tolist()], dtype=object))
        return attribute, codes

    biome_info, realm_info = _codes(biome, "biome"), _codes(realm, "realm")

    # Members of each region from one stable sort; `ids` is sorted, so the group starts are
    # where each id first appears.
    order = np.argsort(eco, kind="stable")
    starts = np.searchsorted(eco[order], ids)
    ends = np.append(starts[1:], len(order))
    members = {rid: order[a:b] for rid, a, b in zip(ids.tolist(), starts, ends)}
    is_large = counts >= min_cells

    xyz = _unit_vectors(
        labels["latitude"].to_numpy(dtype=float), labels["longitude"].to_numpy(dtype=float)
    )
    # The great-circle distance falls as the dot product rises, so the cap is a minimum dot.
    min_dot = float(np.cos(min(max_km / _EARTH_RADIUS_KM, np.pi)))

    merged = eco.copy()
    records = []
    for rid in small:
        pos = int(np.searchsorted(ids, rid))
        # Nearest distance from this region to every region in one pass: the best dot product
        # between any of its cells and each grid cell, then the maximum per region.
        best_per_cell = (xyz[members[rid]] @ xyz.T).max(axis=0)
        region_dot = np.maximum.reduceat(best_per_cell[order], starts)

        for rule, same_biome, same_realm, within_cap in _MERGE_RUNGS:
            ok = is_large.copy()
            if same_biome:
                if biome_info is None:
                    continue
                ok &= biome_info[1] == biome_info[1][pos]
            if same_realm:
                if realm_info is None:
                    continue
                ok &= realm_info[1] == realm_info[1][pos]
            if within_cap:
                ok &= region_dot >= min_dot
            if ok.any():
                chosen = (rule, int(np.argmax(np.where(ok, region_dot, -np.inf))))
                break
        # The last rule has no condition and at least one region is large, so one always fits.
        rule, best = chosen
        target = int(ids[best])
        merged[members[rid]] = target
        record = {
            "source_id": int(rid),
            "source_cells": int(size[rid]),
            "target_id": target,
            "target_cells": int(size[target]),
            "rule": rule,
            "distance_km": round(
                _EARTH_RADIUS_KM * float(np.arccos(np.clip(region_dot[best], -1.0, 1.0))), 3
            ),
        }
        for name, info in (("biome", biome_info), ("realm", realm_info)):
            if info is not None:
                record[f"source_{name}"] = info[0][int(rid)]
                record[f"target_{name}"] = info[0][target]
        records.append(record)

    merges = pd.DataFrame(records)
    logger.info(
        f"[Ecoregions] Merged {len(merges)} of {len(ids)} ecoregions with fewer than "
        f"{min_cells} cells into a larger neighbour: {len(np.unique(merged))} classes; "
        f"rules used {merges['rule'].value_counts().to_dict()}; largest distance "
        f"{merges['distance_km'].max():.0f} km."
    )
    out = labels.copy()
    out["cluster_label"] = merged
    return out, merges

def perform_ecoregion_assignment(
    registry: DataRegistry,
    ecoregions_shapefile: str,
    output_dir: str,
    aggregation: str = "monthly",
    ecoregion_id_column: str = "ECO_ID",
    force_recompute: bool = False,
    min_cells: int = 0,
    max_merge_km: float = DEFAULT_MERGE_MAX_KM,
) -> dict[str, any]:
    """Perform ecoregion-based cell assignment (replaces clustering).

    Returns the same structure as perform_hierarchical_clustering, so the ecoregions can
    stand in for a clustering.

    Reuses an existing labels CSV in ``output_dir``, skipping the expensive spatial
    assignment, only when it was written for the same grid, the same shapefile (path, size,
    modification time) and the same ``ecoregion_id_column`` -- recorded in a ``.inputs`` file
    beside it. A labels CSV without that record, or with another one, is never reused.

    Args:
        registry: DataRegistry containing grid information
        ecoregions_shapefile: Path to ecoregions .shp file
        output_dir: Directory to save labels CSV
        aggregation: Aggregation type (used for filename compatibility)
        ecoregion_id_column: Column name for ecoregion IDs in shapefile
        force_recompute: Assign afresh even when a matching labels CSV exists.
        min_cells: Merge every ecoregion with fewer cells than this into a larger neighbour
            (see :func:`merge_small_ecoregions`); the shapefile's ``BIOME_NUM`` and ``REALM``
            columns, when present, steer the merge. Default 0: no merge, one class per
            ecoregion. The merges are listed in ``ecoregion_merges_min<min_cells>.csv``
            beside the labels.
        max_merge_km: Distance cap of the preferred merge targets, in km.

    Returns
    -------
        dict with keys: 'n_clusters' (number of classes, after any merge), 'method',
        'labels_csv', 'ecoregion_id_column', 'min_cells', and 'merges_csv' (``None`` without
        a merge)
    """
    if int(min_cells) != min_cells or min_cells < 0:
        raise ValueError(f"min_cells must be a whole number >= 0, got {min_cells!r}.")
    os.makedirs(output_dir, exist_ok=True)

    grid = registry.get_original_grid()
    key = fingerprint(
        np.asarray(grid.latitudes, dtype=float),
        np.asarray(grid.longitudes, dtype=float),
        np.asarray(grid.cell_ids),
        file_identity(ecoregions_shapefile) if os.path.exists(ecoregions_shapefile) else None,
        ecoregion_id_column,
        # Only a merge changes the labels, so an unmerged run keeps the key it always had.
        *((int(min_cells), float(max_merge_km)) if min_cells > 1 else ()),
    )
    merges_csv = (
        os.path.join(output_dir, f"ecoregion_merges_min{int(min_cells)}.csv")
        if min_cells > 1
        else None
    )

    # Reuse an earlier assignment only when it was made from these same inputs.
    # Pattern: ecoregion_labels_{aggregation}_{n}.csv, with the fingerprint beside it.
    existing_labels = [
        path
        for path in sorted(
            glob.glob(os.path.join(output_dir, f"ecoregion_labels_{aggregation}_*.csv"))
        )
        if read_fingerprint(path) == key
    ]

    if existing_labels and not force_recompute:
        labels_csv = existing_labels[0]
        logger.info(f"[Ecoregions] Found existing labels for these inputs: {labels_csv}")
        logger.info("[Ecoregions] Skipping spatial assignment (reusing existing results)")

        # Load to get ecoregion count and prepare final output
        df_output = pd.read_csv(labels_csv)
        n_ecoregions = int(df_output["cluster_label"].nunique())

        logger.info(
            f"[Ecoregions] Loaded {len(df_output)} assignments for {n_ecoregions} ecoregions"
        )

        return {
            "n_clusters": n_ecoregions,
            "method": "ecoregions",
            "labels_csv": labels_csv,
            "ecoregion_id_column": ecoregion_id_column,
            "min_cells": int(min_cells),
            "merges_csv": merges_csv,
        }

    # No existing labels — run spatial assignment
    logger.info("[Ecoregions] No labels for these inputs yet — running spatial assignment")

    # Load ecoregions shapefile
    ecoregions_gdf = load_ecoregions_shapefile(ecoregions_shapefile)

    # Assign cells to ecoregions (vectorized)
    df_labels = assign_cells_to_ecoregions(registry, ecoregions_gdf, ecoregion_id_column)
    n_unmerged = int(df_labels["cluster_label"].nunique())
    if min_cells > 1:
        df_labels, merges = merge_small_ecoregions(
            df_labels,
            min_cells,
            biome=_region_attribute(ecoregions_gdf, ecoregion_id_column, _BIOME_COLUMN),
            realm=_region_attribute(ecoregions_gdf, ecoregion_id_column, _REALM_COLUMN),
            max_km=max_merge_km,
        )
        merges.to_csv(merges_csv, index=False)

    # Compute ecoregion centers
    df_centers = compute_ecoregion_centers(df_labels)

    # Merge labels with centers
    df_output = df_labels.merge(df_centers, on="cluster_label", how="left")

    # Determine number of ecoregions
    n_ecoregions = int(df_labels["cluster_label"].nunique())

    logger.info(
        f"[Ecoregions] {n_unmerged} ecoregions on the grid"
        + (f", {n_ecoregions} classes after merging" if min_cells > 1 else "")
    )

    # Save labels CSV (format compatible with clustering)
    stem = f"ecoregion_labels_{aggregation}" + (f"_min{int(min_cells)}" if min_cells > 1 else "")
    labels_csv = os.path.join(output_dir, f"{stem}_{n_ecoregions}.csv")
    df_output.to_csv(labels_csv, index=False)
    write_fingerprint(labels_csv, key)
    logger.info(f"[Ecoregions] Saved labels to: {labels_csv}")

    return {
        "n_clusters": n_ecoregions,  # Number of classes acts as cluster count
        "method": "ecoregions",
        "labels_csv": labels_csv,
        "ecoregion_id_column": ecoregion_id_column,
        "min_cells": int(min_cells),
        "merges_csv": merges_csv,
    }
