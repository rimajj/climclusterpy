"""Supervision from Köppen–Geiger climate zones.

Assigns each grid cell one of the 30 Köppen–Geiger subclasses (Peel et al.,
2007) from its monthly temperature and precipitation climatology, and writes the
standard ``cell_id, latitude, longitude, cluster_label`` supervision table. This
is the zero-extra-data option: the labels come from the same climate input that
is being clustered, so no shapefile or model output is required.

Input is the monthly climatology already computed by
:mod:`climclusterpy.features`. To classify straight from
daily files, reduce them first with
:func:`climclusterpy.features.climate.monthly_climatology_from_daily`.
"""

from __future__ import annotations

import logging
import os

import numpy as np
import pandas as pd

logger = logging.getLogger(__name__)

#: Integer code -> Köppen–Geiger subclass label (Peel et al. 2007, 30 classes).
KG_LABELS = {
    1: "Af",
    2: "Am",
    3: "Aw",
    4: "BWh",
    5: "BWk",
    6: "BSh",
    7: "BSk",
    8: "Csa",
    9: "Csb",
    10: "Csc",
    11: "Cfa",
    12: "Cfb",
    13: "Cfc",
    14: "Cwa",
    15: "Cwb",
    16: "Cwc",
    17: "Dsa",
    18: "Dsb",
    19: "Dsc",
    20: "Dsd",
    21: "Dfa",
    22: "Dfb",
    23: "Dfc",
    24: "Dfd",
    25: "Dwa",
    26: "Dwb",
    27: "Dwc",
    28: "Dwd",
    29: "ET",
    30: "EF",
}

# Reverse mapping
LABEL_TO_CODE = {v: k for k, v in KG_LABELS.items()}

def classify_koppen_from_features(
    features_df: pd.DataFrame,
    cold_threshold: float = 0.0,
) -> np.ndarray:
    """
    Classify cells into Köppen-Geiger zones using pre-computed monthly features.

    This function expects the features DataFrame to contain:
    - tas_month_01 through tas_month_12: Monthly mean temperature (°C)
    - pr_month_01 through pr_month_12: Monthly mean precipitation (mm/month)
    - latitude: Cell latitude (for hemisphere determination)

    Args:
        features_df: DataFrame with monthly climatology features from summarize_climate_features
        cold_threshold: Temperature threshold (°C) separating C and D climates.
            Common values: 0.0 (standard) or -3.0 (variant).

    Returns
    -------
        1D array of KG class codes (integers 1-30) for each cell.
        0 indicates unclassified/invalid cells.

    Raises
    ------
        ValueError: If required monthly features are missing from DataFrame.
    """
    n_cells = len(features_df)

    # Verify required columns exist
    temp_cols = [f"tas_month_{m:02d}" for m in range(1, 13)]
    prec_cols = [f"pr_month_{m:02d}" for m in range(1, 13)]

    missing_temp = [c for c in temp_cols if c not in features_df.columns]
    missing_prec = [c for c in prec_cols if c not in features_df.columns]

    if missing_temp:
        raise ValueError(
            f"Missing temperature monthly features: {missing_temp}. "
            f"Ensure include_monthly_climatology=True in summarize_climate_features()."
        )
    if missing_prec:
        raise ValueError(
            f"Missing precipitation monthly features: {missing_prec}. "
            f"Ensure include_monthly_climatology=True in summarize_climate_features()."
        )

    if "latitude" not in features_df.columns:
        raise ValueError("Missing 'latitude' column needed for hemisphere determination.")

    # Extract monthly data as arrays: shape (n_cells, 12)
    Tm = np.column_stack([features_df[c].values for c in temp_cols])  # (n_cells, 12)
    Pm = np.column_stack([features_df[c].values for c in prec_cols])  # (n_cells, 12)
    lat = features_df["latitude"].values  # (n_cells,)

    # Initialize output
    kg = np.zeros(n_cells, dtype=np.int16)

    # Basic temperature statistics
    T_ann = np.nanmean(Tm, axis=1)  # Annual mean temperature
    T_warm = np.nanmax(Tm, axis=1)  # Warmest month temperature
    T_cold = np.nanmin(Tm, axis=1)  # Coldest month temperature

    # Basic precipitation statistics
    P_ann = np.nansum(Pm, axis=1)  # Annual total precipitation (mm/year)
    P_min = np.nanmin(Pm, axis=1)  # Driest month
    np.nanmax(Pm, axis=1)  # Wettest month

    # Valid cells mask (no NaN in critical variables)
    mask_valid = ~(np.isnan(T_ann) | np.isnan(P_ann))

    # --- A: Tropical climates (T_cold >= 18°C) ---
    is_A = T_cold >= 18.0
    # Af: No dry season (P_min >= 60 mm)
    Af = is_A & (P_min >= 60.0)
    # Am: Monsoon (P_min < 60 but P_min >= 100 - P_ann/25)
    Am = is_A & (~Af) & (P_min >= (100.0 - P_ann / 25.0))
    # Aw: Savanna (everything else in A)
    Aw = is_A & (~Af) & (~Am)

    # --- B: Arid climates ---
    # Aridity threshold depends on precipitation seasonality
    # Determine warm-half months based on hemisphere
    # Northern hemisphere: Apr-Sep (months 4-9), Southern: Oct-Mar (10-12, 1-3)
    wh_mask = np.zeros((n_cells, 12), dtype=bool)
    for i in range(n_cells):
        if lat[i] >= 0:
            wh_mask[i, 3:9] = True  # months 4-9 (0-indexed: 3-8)
        else:
            wh_mask[i, [9, 10, 11, 0, 1, 2]] = True  # months 10-12, 1-3

    # Fraction of precipitation in warm half
    P_warm_half = np.nansum(Pm * wh_mask, axis=1)
    frac_warm = np.where(P_ann > 0, P_warm_half / P_ann, 0.5)

    # Aridity threshold
    adjust = np.zeros(n_cells)
    adjust[frac_warm >= 0.7] = 280.0
    adjust[frac_warm < 0.3] = 0.0
    adjust[(frac_warm >= 0.3) & (frac_warm < 0.7)] = 140.0
    P_th = 20.0 * T_ann + adjust
    P_th = np.maximum(P_th, 0.0)

    is_B = P_ann < P_th
    is_BW = is_B & (P_ann < 0.5 * P_th)  # Desert
    is_BS = is_B & (~is_BW)  # Steppe

    # --- E: Polar climates (T_warm < 10°C) ---
    is_E = T_warm < 10.0
    is_ET = is_E & (T_warm >= 0.0)  # Tundra
    is_EF = is_E & (T_warm < 0.0)  # Ice cap

    # --- C and D: Temperate and Continental ---
    remaining = ~(is_A | is_B | is_E)
    is_C = remaining & (T_cold > cold_threshold)
    is_D = remaining & (~is_C)

    # Precipitation seasonality for C and D
    p_summer_only = np.where(wh_mask, Pm, np.nan)
    p_winter_only = np.where(~wh_mask, Pm, np.nan)
    pmin_summer = np.nanmin(p_summer_only, axis=1)
    pmin_winter = np.nanmin(p_winter_only, axis=1)
    pmax_summer = np.nanmax(p_summer_only, axis=1)
    pmax_winter = np.nanmax(p_winter_only, axis=1)

    # C precipitation subtypes
    c_is_s = (pmin_summer < 40.0) & (pmin_summer < (P_ann / 3.0))  # Summer dry
    c_is_w = (pmin_winter < 10.0) & (pmin_winter < (P_ann / 10.0))  # Winter dry
    c_is_f = ~(c_is_s | c_is_w)  # No dry season

    # D precipitation subtypes (stricter criteria)
    d_is_w = (pmax_summer >= 10.0 * pmin_winter) & (pmin_winter < 30.0)
    d_is_s = (pmax_winter >= 3.0 * pmin_summer) & (pmin_summer < 30.0)
    d_is_f = ~(d_is_s | d_is_w)

    # Temperature subtypes for C and D
    months_above_10 = np.sum(Tm > 10.0, axis=1)
    temp_a = T_warm >= 22.0  # Hot summer
    temp_b = (~temp_a) & (months_above_10 >= 4)  # Warm summer
    temp_c = (~temp_a) & (months_above_10 >= 1) & (months_above_10 < 4)  # Cold summer
    temp_d = T_cold < -38.0  # Extreme cold (for D only)

    # --- Assign class codes ---
    # A classes
    kg[Af & mask_valid] = LABEL_TO_CODE["Af"]
    kg[Am & mask_valid] = LABEL_TO_CODE["Am"]
    kg[Aw & mask_valid] = LABEL_TO_CODE["Aw"]

    # B classes
    kg[(is_BW & (T_ann >= 18.0)) & mask_valid] = LABEL_TO_CODE["BWh"]
    kg[(is_BW & (T_ann < 18.0)) & mask_valid] = LABEL_TO_CODE["BWk"]
    kg[(is_BS & (T_ann >= 18.0)) & mask_valid] = LABEL_TO_CODE["BSh"]
    kg[(is_BS & (T_ann < 18.0)) & mask_valid] = LABEL_TO_CODE["BSk"]

    # E classes
    kg[is_ET & mask_valid] = LABEL_TO_CODE["ET"]
    kg[is_EF & mask_valid] = LABEL_TO_CODE["EF"]

    # C classes (9 combinations: s/f/w × a/b/c)
    c_mask = is_C & mask_valid
    kg[c_mask & c_is_s & temp_a] = LABEL_TO_CODE.get("Csa", 0)
    kg[c_mask & c_is_s & temp_b] = LABEL_TO_CODE.get("Csb", 0)
    kg[c_mask & c_is_s & temp_c] = LABEL_TO_CODE.get("Csc", 0)
    kg[c_mask & c_is_f & temp_a] = LABEL_TO_CODE.get("Cfa", 0)
    kg[c_mask & c_is_f & temp_b] = LABEL_TO_CODE.get("Cfb", 0)
    kg[c_mask & c_is_f & temp_c] = LABEL_TO_CODE.get("Cfc", 0)
    kg[c_mask & c_is_w & temp_a] = LABEL_TO_CODE.get("Cwa", 0)
    kg[c_mask & c_is_w & temp_b] = LABEL_TO_CODE.get("Cwb", 0)
    kg[c_mask & c_is_w & temp_c] = LABEL_TO_CODE.get("Cwc", 0)

    # D classes (12 combinations: s/f/w × a/b/c/d)
    d_mask = is_D & mask_valid
    kg[d_mask & d_is_f & temp_a] = LABEL_TO_CODE.get("Dfa", 0)
    kg[d_mask & d_is_f & temp_b] = LABEL_TO_CODE.get("Dfb", 0)
    kg[d_mask & d_is_f & temp_c] = LABEL_TO_CODE.get("Dfc", 0)
    kg[d_mask & d_is_f & temp_d] = LABEL_TO_CODE.get("Dfd", 0)
    kg[d_mask & d_is_w & temp_a] = LABEL_TO_CODE.get("Dwa", 0)
    kg[d_mask & d_is_w & temp_b] = LABEL_TO_CODE.get("Dwb", 0)
    kg[d_mask & d_is_w & temp_c] = LABEL_TO_CODE.get("Dwc", 0)
    kg[d_mask & d_is_w & temp_d] = LABEL_TO_CODE.get("Dwd", 0)
    kg[d_mask & d_is_s & temp_a] = LABEL_TO_CODE.get("Dsa", 0)
    kg[d_mask & d_is_s & temp_b] = LABEL_TO_CODE.get("Dsb", 0)
    kg[d_mask & d_is_s & temp_c] = LABEL_TO_CODE.get("Dsc", 0)
    kg[d_mask & d_is_s & temp_d] = LABEL_TO_CODE.get("Dsd", 0)

    return kg

def assign_cells_to_koppen_geiger(
    features_df: pd.DataFrame,
    cold_threshold: float = 0.0,
) -> pd.DataFrame:
    """
    Assign each cell to a Köppen-Geiger climate zone.

    Uses the monthly climatology features from the features DataFrame to
    classify cells. The output format matches the ecoregion assignment format.

    Args:
        features_df: DataFrame with columns:
            - cell_id, latitude, longitude
            - tas_month_01..tas_month_12 (monthly mean temperature °C)
            - pr_month_01..pr_month_12 (monthly precipitation mm/month)
        cold_threshold: Temperature threshold for C/D boundary (default 0.0°C)

    Returns
    -------
        DataFrame with columns: cell_id, latitude, longitude, cluster_label
        where cluster_label is the KG class code (1-30).
        Cells with code 0 (unclassified) are excluded.
    """
    logger.info(f"[KöppenGeiger] Classifying {len(features_df)} cells into KG climate zones...")
    logger.info(f"[KöppenGeiger] Using cold-month threshold: {cold_threshold}°C")

    # Classify cells
    kg_codes = classify_koppen_from_features(features_df, cold_threshold)

    # Build output DataFrame
    result = pd.DataFrame(
        {
            "cell_id": features_df["cell_id"].values,
            "latitude": features_df["latitude"].values,
            "longitude": features_df["longitude"].values,
            "cluster_label": kg_codes,
        }
    )

    # Filter out unclassified cells (code 0)
    n_unclassified = (kg_codes == 0).sum()
    if n_unclassified > 0:
        logger.warning(
            f"[KöppenGeiger] Warning: {n_unclassified} cells could not be classified (NaN values)"
        )

    result = result[result["cluster_label"] > 0].reset_index(drop=True)

    # Report statistics
    n_zones = result["cluster_label"].nunique()
    zone_counts = result["cluster_label"].value_counts().sort_index()

    logger.info(f"[KöppenGeiger] Assigned {len(result)} cells to {n_zones} KG zones")
    logger.info("[KöppenGeiger] Zone distribution:")
    for code, count in zone_counts.items():
        label = KG_LABELS.get(code, "?")
        logger.info(f"  {label} ({code}): {count} cells")

    return result

def perform_koppen_geiger_assignment(
    features_df: pd.DataFrame,
    output_dir: str,
    aggregation: str = "koppen_geiger",
    cold_threshold: float = 0.0,
) -> dict:
    """
    Perform Köppen-Geiger assignment and save results.

    This function mirrors the interface of perform_ecoregion_assignment()
    to allow seamless substitution in the LMNN pipeline.

    Args:
        features_df: DataFrame with cell features including monthly climatologies
        output_dir: Directory to save output files
        aggregation: Output subfolder name (default "koppen_geiger")
        cold_threshold: Temperature threshold for C/D boundary

    Returns
    -------
        Dict with keys:
        - labels_csv: Path to saved labels CSV file
        - n_clusters: Number of KG zones found
        - zone_counts: Dict mapping KG codes to cell counts
    """
    os.makedirs(output_dir, exist_ok=True)

    # Assign cells
    labels_df = assign_cells_to_koppen_geiger(features_df, cold_threshold)

    # Save labels CSV
    labels_file = os.path.join(output_dir, f"koppen_geiger_labels_{aggregation}.csv")
    labels_df.to_csv(labels_file, index=False)
    logger.info(f"[KöppenGeiger] Saved labels to: {labels_file}")

    # Compute statistics
    n_zones = labels_df["cluster_label"].nunique()
    zone_counts = labels_df["cluster_label"].value_counts().to_dict()

    return {
        "labels_csv": labels_file,
        "n_clusters": n_zones,
        "zone_counts": zone_counts,
        "zone_labels": KG_LABELS,
    }
