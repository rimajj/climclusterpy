Source code for climclusterpy.supervision.traits

"""Supervision from vegetation-model output traits.

Groups grid cells by what the vegetation model actually produces there — a set
of continuous traits (SLA, wood density, height, biomass, …) plus a categorical
dominant plant type — and uses those groups as LMNN class labels. Supervising on
the model's own output is the strongest signal available: it teaches the metric
to keep together exactly the cells whose simulated vegetation is alike, which is
what the reduced grid has to reproduce.

Latitude/longitude are deliberately *not* features: grouping is purely
attribute-based, so spatially distant cells with the same vegetation share a
label.

Feature construction per valid cell:

1. z-score each continuous trait independently, so no trait dominates by unit;
2. one-hot encode the categorical dominant type, left unscaled so each type is a
   unit indicator comparable in magnitude to a z-scored continuous feature;
3. column-stack ``[continuous_z..., type_onehot...]``.

Default linkage is average (UPGMA), which is robust to the elongated trait
clusters this data produces.
"""

from __future__ import annotations

import logging
import os
from collections.abc import Sequence

import numpy as np
import pandas as pd
from scipy.cluster.hierarchy import fcluster, linkage
from sklearn.preprocessing import StandardScaler

from climclusterpy._cache import fingerprint
from climclusterpy.features.soil import encode_soil_code

logger = logging.getLogger(__name__)

def build_trait_feature_matrix(
    features_df: pd.DataFrame,
    continuous_cols: Sequence[str],
    categorical_col: str | None = None,
) -> np.ndarray:
    """Build the multivariate feature matrix for clustering.

    Steps:
        1. Z-score standardize each continuous column independently.
        2. One-hot encode the categorical column (if given), unscaled.
        3. Column-stack: [continuous_z..., type_onehot...].

    Args:
        features_df: DataFrame with one row per valid cell, containing at least
            the columns in ``continuous_cols`` (and ``categorical_col`` if set).
        continuous_cols: Names of continuous feature columns to z-score.
        categorical_col: Optional name of the categorical (integer) column to
            one-hot encode.

    Returns
    -------
        Feature matrix of shape (n_cells, n_continuous + n_types).
    """
    blocks: list[np.ndarray] = []
    names: list[str] = []

    # 1. Continuous features (z-scored independently). NaN -> column mean so
    #    cells missing a trait (e.g. no woody biomass) are treated as neutral.
    for col in continuous_cols:
        vals = features_df[col].to_numpy(dtype=np.float64).reshape(-1, 1)
        vals = np.where(np.isnan(vals), np.nanmean(vals), vals)
        z = StandardScaler().fit_transform(vals).astype(np.float32)
        blocks.append(z)
        names.append(col)

    # 2. Categorical one-hot (unscaled unit indicators)
    if categorical_col is not None:
        codes = features_df[categorical_col].to_numpy(dtype=np.float64)
        ohe_df, _ = encode_soil_code(codes)  # columns named soil_code_<value>
        ohe = ohe_df.to_numpy(dtype=np.float32)
        blocks.append(ohe)
        names.extend([f"{categorical_col}_{c}" for c in ohe_df.columns])

    X = np.column_stack(blocks).astype(np.float32)
    logger.info(
        f"[TraitFeatures] Feature matrix shape: {X.shape} "
        f"({len(continuous_cols)} continuous, "
        f"{X.shape[1] - len(continuous_cols)} one-hot type)"
    )
    logger.debug(f"[TraitFeatures] Feature columns: {names}")
    return X

def build_trait_feature_table(
    features_df: pd.DataFrame,
    cell_ids: np.ndarray,
    latitudes: np.ndarray,
    longitudes: np.ndarray,
    continuous_cols: Sequence[str],
    categorical_col: str | None = None,
) -> pd.DataFrame:
    """Build a features DataFrame for representative-cell selection.

    Used with the existing ``precomputed_features`` selection method, which
    merges on (latitude, longitude) and picks the cell whose feature vector is
    closest (raw Euclidean) to the cluster centroid. Because that selector does
    NOT standardize, the continuous columns are z-scored here so that the
    representative is chosen in the same balanced, scale-free feature space used
    for clustering. Latitude/longitude are included only as plain metadata
    columns (for the merge), not as clustering features. The one-hot type
    columns are unit indicators, already comparable in magnitude to the
    z-scored continuous features.

    Returns
    -------
        DataFrame with columns: cell_id, latitude, longitude,
        <continuous_cols...> (z-scored), and <categorical>_<value>... one-hot
        columns.
    """
    out = pd.DataFrame(
        {
            "cell_id": cell_ids,
            "latitude": latitudes,
            "longitude": longitudes,
        }
    )
    for col in continuous_cols:
        vals = features_df[col].to_numpy(dtype=np.float64).reshape(-1, 1)
        vals = np.where(np.isnan(vals), np.nanmean(vals), vals)
        out[col] = StandardScaler().fit_transform(vals).astype(np.float32).ravel()
    if categorical_col is not None:
        codes = features_df[categorical_col].to_numpy(dtype=np.float64)
        ohe_df, _ = encode_soil_code(codes)
        # encode_soil_code names columns soil_code_<value>; rename to the real
        # categorical variable for clarity in the persisted features CSV.
        ohe_df = ohe_df.rename(columns=lambda c: c.replace("soil_code", categorical_col))
        ohe_df.index = out.index
        out = pd.concat([out, ohe_df], axis=1)
    return out

def perform_trait_assignment(
    features_df: pd.DataFrame,
    cell_ids: np.ndarray,
    latitudes: np.ndarray,
    longitudes: np.ndarray,
    n_clusters: int,
    output_dir: str,
    continuous_cols: Sequence[str],
    categorical_col: str | None = None,
    linkage_method: str = "complete",
    force_recompute: bool = False,
) -> str:
    """Perform hierarchical agglomerative clustering on output-derived features.

    Uses direct Euclidean linkage on the multivariate feature matrix. The
    feature matrix does not depend on n_clusters, so the dendrogram is identical for
    every n_clusters value: it is saved in ``output_dir`` and re-cut by later calls.
    It is reused only for the same feature matrix and linkage method -- its file name
    carries a fingerprint of both -- so a call with other traits, cells or values in
    the same folder builds its own. ``force_recompute=True`` rebuilds it regardless.

    Returns
    -------
        Absolute path to the cluster labels CSV with columns:
        cell_id, latitude, longitude, cluster_label.
    """
    os.makedirs(output_dir, exist_ok=True)

    X = build_trait_feature_matrix(
        features_df,
        continuous_cols=continuous_cols,
        categorical_col=categorical_col,
    )
    # The linkage is the expensive part; the matrix is cheap, so it is always built and its
    # fingerprint decides whether a saved dendrogram belongs to it. The fingerprint is in the
    # FILE NAME rather than beside it, so a dendrogram some other tool pinned under the plain
    # name (``dendrogram_output_<linkage>.npy``) is never reused for other inputs and never
    # overwritten.
    key = fingerprint(np.asarray(X, dtype=float), linkage_method)
    dendrogram_path = os.path.join(output_dir, f"dendrogram_output_{linkage_method}_{key}.npy")
    labels_csv_path = os.path.join(output_dir, f"cluster_labels_output_{n_clusters}.csv")

    n_cells = len(features_df)
    logger.info(
        f"[TraitSupervision] n_cells={n_cells}, n_clusters={n_clusters}, linkage={linkage_method}"
    )

    if not force_recompute and os.path.exists(dendrogram_path):
        logger.info(f"[TraitSupervision] Loading cached dendrogram: {dendrogram_path}")
        Z = np.load(dendrogram_path)
    else:
        logger.info(
            f"[TraitSupervision] Features built: shape={X.shape}. "
            f"Computing {linkage_method} linkage…"
        )
        Z = linkage(X, method=linkage_method, metric="euclidean")
        np.save(dendrogram_path, Z)
        logger.info(f"[TraitSupervision] Dendrogram saved: {dendrogram_path}")

    logger.info(f"[TraitSupervision] Cutting dendrogram -> {n_clusters} clusters…")
    labels = fcluster(Z, t=n_clusters, criterion="maxclust")

    df = pd.DataFrame(
        {
            "cell_id": cell_ids,
            "latitude": latitudes,
            "longitude": longitudes,
            "cluster_label": labels,
        }
    )
    df.to_csv(labels_csv_path, index=False)

    actual_n = df["cluster_label"].nunique()
    logger.info(
        f"[TraitSupervision] Saved labels -> {labels_csv_path} "
        f"(requested={n_clusters}, actual={actual_n})"
    )
    return labels_csv_path
