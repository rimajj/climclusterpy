Source code for climclusterpy.grid.distributed

"""Writing a *distributed* reduced grid: several real cells per cluster, each where it lies.

The counterpart to :mod:`climclusterpy.grid.build` and :mod:`climclusterpy.grid.native` for
the ``"distributed"`` cell choice. Those two write **one** cell per cluster on a packed grid;
this writes every chosen cell of every cluster at **its own position on the source grid**, so
no representative has to be computed and no cell is displaced:

* each chosen cell goes to its original ``(latitude, longitude)`` slot on the source lattice;
* a cell chosen more than once (a cluster smaller than ``cells_per_cluster``) is written again
  at the next free longitude of the **same latitude row**, and a full row gets extra columns
  east of the grid at the same spacing. Latitude is what reaches the vegetation (day length);
  longitude is inert when radiation comes from a forcing file, so a repeat keeps the one
  coordinate that matters and moves the one that does not;
* the axes are the source grid's own, so the step the CLM converter writes into its header —
  what the model reads back as its resolution, and what scales every area-weighted total — is
  the source cell size.

The workflow code this was moved out of handled a full row by re-spacing the whole longitude
axis over 360°, which silently changes the declared resolution — what scales every
area-weighted total — and every cell's longitude. Here the extension keeps the source step, and
a row that would need to run past 180° east is an error, because a wrong resolution is not a
warning.

The projection back is :func:`climclusterpy.grid.reconstruct.reconstruct_from_distributed`,
driven by the ``cluster_distributed_mapping_{K}.csv`` written here.
"""

from __future__ import annotations

import os

import numpy as np
import pandas as pd
import xarray as xr

__all__ = ["write_distributed_grid"]

#: Written into every distributed NetCDF so the model-run side can read what a cell stands for.
DISTRIBUTED_FLAG_ATTR = "climclusterpy_cell_selection"


def distributed_filename(
    variable: str,
    n_clusters: int,
    n_cells: int,
    *,
    aggregation_type: str = "clustering",
    cluster_method: str | None = None,
) -> str:
    """``{var}_{K}_clusters_{agg}_ncells_{n}_distributed[_{method}].nc`` — one naming rule."""
    suffix = f"_{cluster_method}" if cluster_method else ""
    return (
        f"{variable}_{n_clusters}_clusters_{aggregation_type}_ncells_{n_cells}"
        f"_distributed{suffix}.nc"
    )


def _place_slots(ds_ts: xr.Dataset, registry) -> tuple[list[tuple[int, int, int]], np.ndarray]:
    """``([(slot, lat_idx, lon_idx), ...], lon_axis)`` — one entry per chosen slot.

    ``lon_axis`` is the source longitude axis, extended eastwards **at the source spacing**
    when a latitude row holds more slots than the source grid has longitudes (a small regional
    grid with small, repeated clusters). The step stays the source cell size, so the resolution
    the model reads is unchanged; only the extra columns' (inert) longitudes are new.
    """
    orig = registry.get_original_grid()
    orig_lons = np.asarray(orig.longitudes, dtype=float)
    ids = np.asarray(orig.cell_ids)
    step = float(np.min(np.abs(np.diff(orig_lons)))) if orig_lons.size > 1 else 1.0

    rows, cols = np.nonzero(ids >= 0)
    cell_id_to_pos = {int(ids[i, j]): (int(i), int(j)) for i, j in zip(rows, cols)}

    n_cols = orig_lons.size
    slot_ids = np.asarray(ds_ts["cell_id"].values).astype(int)
    for slot, cid in enumerate(slot_ids):
        if cid not in cell_id_to_pos:
            raise ValueError(
                f"Chosen cell {cid} (slot {slot}) is not a valid cell of the original grid; the "
                "timeseries dataset and the registry describe different grids."
            )

    # Two passes. First every chosen cell takes its own position (its first slot does); only
    # then do the repeats look for free columns. In one pass a repeat placed early could take
    # the own position of a cell drawn later and push that cell off it -- measured on a
    # crowded synthetic row, where most drawn cells ended up away from their own coordinates.
    occupied: set[tuple[int, int]] = set()
    placed: dict[int, tuple[int, int]] = {}
    repeats: list[int] = []
    for slot, cid in enumerate(slot_ids):
        pos = cell_id_to_pos[int(cid)]
        if pos in occupied:
            repeats.append(slot)
            continue
        occupied.add(pos)
        placed[slot] = pos
    for slot in repeats:
        lat_idx = cell_id_to_pos[int(slot_ids[slot])][0]
        # Same latitude row, first free longitude -- extending the axis if needed.
        lon_idx = next((j for j in range(n_cols) if (lat_idx, j) not in occupied), n_cols)
        if lon_idx == n_cols:
            n_cols += 1
        occupied.add((lat_idx, lon_idx))
        placed[slot] = (lat_idx, lon_idx)
    positions = [(slot, *placed[slot]) for slot in range(len(slot_ids))]

    lon_axis = np.concatenate(
        [orig_lons, orig_lons[-1] + step * np.arange(1, n_cols - orig_lons.size + 1)]
    )
    if lon_axis[-1] >= 180.0 or n_cols * step > 360.0 + 1e-9:
        raise ValueError(
            f"A latitude row needs {n_cols} columns at the source spacing of {step:g} deg, which "
            "runs past 180 deg east — the limit the CLM tools accept. Re-spacing the axis to fit "
            "would change the resolution the model reads, so this is refused. Use fewer "
            "cells_per_cluster."
        )
    return positions, lon_axis

def write_distributed_grid(
    timeseries_dataset: xr.Dataset,
    registry,
    output_dir: str,
    *,
    n_clusters: int,
    artifacts_dir: str | None = None,
    aggregation_type: str = "clustering",
    cluster_method: str | None = None,
    extra_attrs: dict | None = None,
    overwrite: bool = True,
    time_chunk: int = 2000,
) -> dict[str, str]:
    """Write one NetCDF per forcing variable, every chosen cell at its own grid position.

    Parameters
    ----------
    timeseries_dataset : xarray.Dataset
        What :func:`climclusterpy.timeseries.extract_distributed_cells`
        returns: dimension ``cell`` (one entry per chosen slot) carrying ``cluster_label``,
        ``cell_id``, ``latitude``, ``longitude``, the forcing variables and the soil fields.
    registry : DataRegistry
        The original grid; its lattice becomes the written grid's axes.
    output_dir : str
        Where the NetCDFs go.
    n_clusters : int
        Number of clusters, for the file names.
    artifacts_dir : str, optional
        Where ``cluster_distributed_mapping_{K}.csv`` goes; defaults to ``output_dir``.
    aggregation_type, cluster_method : str, optional
        Carried into the file names (see :func:`distributed_filename`).
    extra_attrs : dict, optional
        Global attributes written into every file (the public API records the patch
        arithmetic here).
    overwrite : bool, optional
        When False, a variable whose NetCDF already exists and is non-empty is skipped.
    time_chunk : int, optional
        Time steps per write batch; bounds peak memory on long forcing series.

    Returns
    -------
    dict
        Variable name to written path, plus ``"__distributed_mapping__"``.

    Raises
    ------
    ValueError
        If a chosen cell is not on the registry's grid, or a latitude row is over-full.
    """
    import netCDF4 as nc4
    from cftime import date2num

    ds_ts = timeseries_dataset
    artifacts_dir = artifacts_dir if artifacts_dir is not None else output_dir
    os.makedirs(output_dir, exist_ok=True)
    os.makedirs(artifacts_dir, exist_ok=True)

    positions, orig_lons = _place_slots(ds_ts, registry)
    orig_lats = np.asarray(registry.get_original_grid().latitudes, dtype=float)
    slot_idx = np.array([p[0] for p in positions])
    lat_idx = np.array([p[1] for p in positions])
    lon_idx = np.array([p[2] for p in positions])

    mapping = pd.DataFrame(
        {
            "cell_index": slot_idx,
            "cluster_label": np.asarray(ds_ts["cluster_label"].values)[slot_idx].astype(int),
            "cell_id": np.asarray(ds_ts["cell_id"].values)[slot_idx].astype(int),
            "original_latitude": np.asarray(ds_ts["latitude"].values)[slot_idx].astype(float),
            "original_longitude": np.asarray(ds_ts["longitude"].values)[slot_idx].astype(float),
            "grid_latitude_index": lat_idx,
            "grid_longitude_index": lon_idx,
            "grid_latitude": orig_lats[lat_idx].astype(float),
            "grid_longitude": orig_lons[lon_idx].astype(float),
        }
    )
    mapping_path = os.path.join(artifacts_dir, f"cluster_distributed_mapping_{n_clusters}.csv")
    mapping.to_csv(mapping_path, index=False)

    exclude = {"cell_id", "latitude", "longitude", "cluster_label", "cell_index", "_slot_to_unique"}
    variables = [v for v in ds_ts.data_vars if v not in exclude]
    is_dedup = bool(ds_ts.attrs.get("_distributed_dedup", False))
    slot_to_unique = ds_ts["_slot_to_unique"].values if "_slot_to_unique" in ds_ts else None

    n_cells = int(ds_ts.sizes["cell"])
    files: dict[str, str] = {}
    for var in variables:
        da = ds_ts[var]
        has_time = "time" in da.dims
        if var == "soil_code":
            dtype, fill = np.int32, np.int32(-9999)
        else:
            dtype, fill = np.float32, np.float32(-1e32)

        path = os.path.join(
            output_dir,
            distributed_filename(
                var,
                n_clusters,
                n_cells,
                aggregation_type=aggregation_type,
                cluster_method=cluster_method,
            ),
        )
        if not overwrite and os.path.exists(path) and os.path.getsize(path) > 0:
            files[var] = path
            continue

        # netCDF4 directly, not xarray: the grid is mostly fill, and writing only the
        # populated cells in time chunks never allocates the full (time, lat, lon) array.
        ncf = nc4.Dataset(path, "w", format="NETCDF4_CLASSIC")
        try:
            ncf.createDimension("latitude", orig_lats.size)
            ncf.createDimension("longitude", orig_lons.size)
            ncf.setncattr(DISTRIBUTED_FLAG_ATTR, "distributed")
            for key, value in (extra_attrs or {}).items():
                ncf.setncattr(key, value)

            if has_time:
                time_vals = ds_ts["time"].values
                n_time = len(time_vals)
                ncf.createDimension("time", n_time)
                # Compressed, one time step per chunk: the grid is almost all fill (1 675 of
                # 201 600 cells at K = 67 x 25 on the half-degree globe), so an uncompressed
                # daily 1901-2019 variable would be ~35 GB of fill; and the CLM converter
                # reads a time step at a time, so a one-step chunk is never decompressed twice.
                out = ncf.createVariable(
                    var,
                    dtype,
                    ("time", "latitude", "longitude"),
                    fill_value=fill,
                    zlib=True,
                    complevel=4,
                    shuffle=True,
                    chunksizes=(1, orig_lats.size, orig_lons.size),
                )
                tvar = ncf.createVariable("time", "f8", ("time",))
                units = "days since 1901-01-01"
                cal = ds_ts["time"].encoding.get("calendar") or ds_ts["time"].attrs.get("calendar")
                if cal is None:
                    cal = getattr(time_vals[0], "calendar", "standard")
                tvar.units, tvar.calendar = units, cal
                if np.issubdtype(np.asarray(time_vals).dtype, np.number):
                    tvar[:] = np.asarray(time_vals, dtype="f8")
                else:
                    tvar[:] = date2num(list(time_vals), units=units, calendar=cal)
                var_is_dedup = is_dedup and "unique_cell" in da.dims
                for t0 in range(0, n_time, time_chunk):
                    t1 = min(t0 + time_chunk, n_time)
                    if var_is_dedup:
                        chunk = da.isel(time=slice(t0, t1)).values[:, slot_to_unique[slot_idx]]
                    else:
                        chunk = da.isel(
                            time=slice(t0, t1), cell=xr.DataArray(slot_idx, dims="_pts")
                        ).values
                    buf = np.full((t1 - t0, orig_lats.size, orig_lons.size), fill, dtype=dtype)
                    buf[:, lat_idx, lon_idx] = chunk
                    out[t0:t1, :, :] = buf
            else:
                out = ncf.createVariable(
                    var, dtype, ("latitude", "longitude"), fill_value=fill, zlib=True, complevel=4
                )
                grid = np.full((orig_lats.size, orig_lons.size), fill, dtype=dtype)
                values = np.asarray(da.values)[slot_idx]
                if var == "soil_code":
                    ok = np.isfinite(values.astype(float))
                    grid[lat_idx[ok], lon_idx[ok]] = values[ok].astype(np.int32)
                else:
                    grid[lat_idx, lon_idx] = values
                out[:, :] = grid
            # Coordinates AFTER the data variable: cdf2clm converts the FIRST variable in the
            # file unless told otherwise, and a 1-D latitude first makes it fail with "Invalid
            # number 1 of dimensions" (found on a real 48-cell grid, not by the unit tests).
            for name, values, units in (
                ("latitude", orig_lats, "degrees_north"),
                ("longitude", orig_lons, "degrees_east"),
            ):
                coord = ncf.createVariable(name, "f8", (name,))
                coord[:] = values
                coord.units = units
            ncf.sync()
        finally:
            ncf.close()
        files[var] = path

    files["__distributed_mapping__"] = mapping_path
    return files
