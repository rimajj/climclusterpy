"""Projecting output on the reduced grid back onto every cell of the original grid.

Two cases. With one simulated cell per cluster (:func:`reconstruct_output_netcdf`), the
two mapping CSVs written with the grid say where each cluster's cells sit on the reduced
grid and which original cells belong to it. With several cells per cluster
(:func:`reconstruct_from_distributed`), the distributed mapping names each simulated cell
and the cluster's value is their mean or median. Either way every original cell is found
by its ``cell_id`` through :class:`~climclusterpy.data.DataRegistry`, never by matching
coordinates.
"""

from __future__ import annotations

import logging
import os
import warnings

import numpy as np
import pandas as pd
import xarray as xr

from climclusterpy.data.registry import DataRegistry

logger = logging.getLogger(__name__)

#: Variable attributes carried from the reduced-grid input onto the reconstructed output.
#:
#: Reconstruction changes which cells a value sits in, never what the value MEANS, so the
#: description of the quantity has to survive it. Before this it did not: a reconstructed
#: vegetation-carbon field arrived with no units at all, and a consumer had to know out of band
#: that LPJmL writes gC/m2 while the tolerance it is scored against is quoted in kg/m2.
#:
#: It is an allow-list rather than a copy-everything-except, because of what must NOT come
#: across: the reconstructed array uses NaN for every cell outside the original grid's land
#: mask, so a `_FillValue` or `missing_value` of -1e32 inherited from an LPJmL input would
#: misdescribe it, and xarray would then encode those NaNs as -1e32 on write -- a different
#: file, with a different missing convention, from the one every existing consumer reads.
#: Under xarray's default decoding those two attributes already sit in ``encoding`` rather than
#: ``attrs``, so today they would not be copied anyway; the allow-list is what keeps that true
#: if how the input is opened ever changes.
_CARRIED_VAR_ATTRS = ("units", "long_name", "standard_name")


def _carried_attrs(source: xr.DataArray) -> dict:
    """The subset of ``source``'s attributes that survive reconstruction."""
    return {k: v for k, v in source.attrs.items() if k in _CARRIED_VAR_ATTRS}

def validate_reconstruction_grid(
    reconstructed_nc: str, registry: DataRegistry, var_name: str = "VegC"
) -> bool:
    """
    Validate that reconstructed NetCDF has identical grid structure to original soilcode grid.

    Tests:
    1. Grid dimensions match original grid
    2. Coordinates match original grid
    3. Valid/invalid (NaN) cells are in exactly the same locations as original grid

    Args:
        reconstructed_nc: Path to reconstructed NetCDF file
        registry: DataRegistry with original grid information
        var_name: Variable name to check in the reconstructed file

    Returns
    -------
        True if validation passes, False otherwise. The reason for a failure is logged as a
        warning.
    """
    logger.info(f"\n[Validation] Checking reconstructed file: {os.path.basename(reconstructed_nc)}")

    # Load reconstructed data
    ds_recon = xr.open_dataset(reconstructed_nc)

    # Get original grid
    orig_grid = registry.get_original_grid()
    orig_lats = orig_grid.latitudes
    orig_lons = orig_grid.longitudes
    orig_mask = registry.get_valid_mask()  # True where cells are valid

    # Check 1: Grid dimensions
    if var_name not in ds_recon:
        logger.warning(f"  ✗ FAIL: Variable '{var_name}' not found in reconstructed file")
        ds_recon.close()
        return False

    recon_lats = ds_recon["latitude"].values
    recon_lons = ds_recon["longitude"].values

    if recon_lats.shape != orig_lats.shape or recon_lons.shape != orig_lons.shape:
        logger.warning("  ✗ FAIL: Grid shape mismatch")
        logger.warning(f"    Original: lat={orig_lats.shape}, lon={orig_lons.shape}")
        logger.warning(f"    Reconstructed: lat={recon_lats.shape}, lon={recon_lons.shape}")
        ds_recon.close()
        return False

    # Check 2: Coordinates match
    if not np.allclose(recon_lats, orig_lats, rtol=1e-5):
        logger.warning("  ✗ FAIL: Latitude coordinates don't match")
        max_diff = np.max(np.abs(recon_lats - orig_lats))
        logger.warning(f"    Max difference: {max_diff}")
        ds_recon.close()
        return False

    if not np.allclose(recon_lons, orig_lons, rtol=1e-5):
        logger.warning("  ✗ FAIL: Longitude coordinates don't match")
        max_diff = np.max(np.abs(recon_lons - orig_lons))
        logger.warning(f"    Max difference: {max_diff}")
        ds_recon.close()
        return False

    # Check 3: Valid/invalid cells match original grid
    # Get first time step if temporal data
    if "time" in ds_recon[var_name].dims:
        recon_data = ds_recon[var_name].isel(time=0).values
    else:
        recon_data = ds_recon[var_name].values

    # Build reconstructed validity mask (non-NaN cells)
    # Account for fill values
    fill_value = ds_recon[var_name].encoding.get(
        "_FillValue", ds_recon[var_name].attrs.get("_FillValue", -1e32)
    )

    recon_valid = ~np.isnan(recon_data) & (recon_data != fill_value)

    # Compare masks
    if recon_valid.shape != orig_mask.shape:
        logger.warning("  ✗ FAIL: Mask shape mismatch")
        logger.warning(f"    Original: {orig_mask.shape}")
        logger.warning(f"    Reconstructed: {recon_valid.shape}")
        ds_recon.close()
        return False

    # Check if valid/invalid pattern matches exactly
    matches = recon_valid == orig_mask
    if not np.all(matches):
        mismatches = np.sum(~matches)

        # Show which type of mismatch
        extra_valid = np.sum(
            recon_valid & ~orig_mask
        )  # Cells that are valid in recon but invalid in orig
        missing_valid = np.sum(
            ~recon_valid & orig_mask
        )  # Cells that are invalid in recon but valid in orig

        # If cells are only missing (subset was clustered), this is acceptable - just warn
        if extra_valid == 0 and missing_valid > 0:
            logger.warning("  ⚠ WARNING: Reconstructed grid has fewer valid cells than original")
            logger.info(f"    Original valid cells: {np.sum(orig_mask)}")
            logger.info(f"    Reconstructed valid cells: {np.sum(recon_valid)}")
            logger.info(
                f"    Missing cells: {missing_valid} ({100 * missing_valid / orig_mask.size:.2f}%)"
            )
            logger.info(
                "    This is expected when clustering was done on a subset of cells (e.g., filtered by ecoregion)"
            )
            # Not a failure - clustering was done on a subset
        else:
            # Actual error - cells filled that shouldn't be, or unexpected pattern
            logger.warning("  ✗ FAIL: Valid/invalid cell pattern doesn't match original grid")
            logger.warning(f"    Total cells: {orig_mask.size}")
            logger.warning(
                f"    Mismatched cells: {mismatches} ({100 * mismatches / orig_mask.size:.2f}%)"
            )
            logger.warning(f"    Cells incorrectly filled (should be NaN): {extra_valid}")
            logger.warning(f"    Cells incorrectly NaN (should be filled): {missing_valid}")

            ds_recon.close()
            return False

    # All checks passed
    recon_valid_count = np.sum(recon_valid)
    logger.info("  ✓ PASS: Grid structure matches original")
    logger.info(f"    Grid shape: {orig_lats.shape[0]} lats × {orig_lons.shape[0]} lons")
    logger.info(f"    Valid cells: {recon_valid_count} / {orig_mask.size}")

    ds_recon.close()
    return True




def _read_position_table(latrow_map_csv: str) -> pd.DataFrame:
    if not os.path.exists(latrow_map_csv):
        raise FileNotFoundError(
            f"Position table (cluster_compact_map_<K>.csv) not found: {latrow_map_csv}"
        )
    ldf = pd.read_csv(latrow_map_csv)
    missing = {"cluster_label", "lat_index"} - set(ldf.columns)
    if missing:
        raise ValueError(
            f"{latrow_map_csv} is not a position table: it lacks the column(s) {sorted(missing)} "
            f"(found {list(ldf.columns)}). Pass the cluster_compact_map_<K>.csv written with the grid."
        )
    return ldf


def _require_mappings_agree(
    id_map: pd.DataFrame, ldf: pd.DataFrame, id_mapping_csv: str, latrow_map_csv: str
) -> None:
    """The ID mapping and the position table must describe the same set of clusters."""
    missing = {"original_cell_id", "cluster_index", "cluster_label"} - set(id_map.columns)
    if missing:
        raise ValueError(
            f"{id_mapping_csv} is not an ID mapping: it lacks the column(s) {sorted(missing)} "
            f"(found {list(id_map.columns)}). Pass the cluster_id_mapping_<K>.csv written with the grid."
        )
    in_ids = set(id_map["cluster_label"].astype(int))
    in_pos = set(ldf["cluster_label"].astype(int))
    if in_ids != in_pos:
        raise ValueError(
            "The two mapping files do not belong to the same reduced grid: "
            f"{os.path.basename(id_mapping_csv)} names {len(in_ids)} clusters, "
            f"{os.path.basename(latrow_map_csv)} names {len(in_pos)}, and only "
            f"{len(in_ids & in_pos)} are in both. Use the pair written by the same create_clusters "
            "run (they sit side by side in its output folder)."
        )


def _cluster_cells(
    ds: xr.Dataset, lat_name: str, lon_name: str, ldf: pd.DataFrame, where: str
) -> dict[int, tuple[int, int, int]]:
    """``{cluster_label: (lat_row, first_col, last_col)}`` on the model output's own axes.

    A position table that carries the grid coordinates (every table written since 0.3.0) is
    matched by coordinate, which also survives a model that trims empty margins or flips the
    latitude axis, and refuses output that is not on this grid at all. An older table has only
    array indices, which are checked against the output's shape.
    """
    n_lat, n_lon = ds.sizes[lat_name], ds.sizes[lon_name]
    has_range = {"cluster_index_start", "cluster_index_end"} <= set(ldf.columns)
    labels = ldf["cluster_label"].astype(int).to_numpy()
    if {"grid_latitude", "grid_longitude_start", "grid_longitude_end"} <= set(ldf.columns):
        what = f"the model output {where}"
        rows = _axis_positions(ds[lat_name].values, ldf["grid_latitude"].values, "latitude", what)
        j0 = _axis_positions(
            ds[lon_name].values, ldf["grid_longitude_start"].values, "longitude", what
        )
        j1 = _axis_positions(
            ds[lon_name].values, ldf["grid_longitude_end"].values, "longitude", what
        )
        first, last = np.minimum(j0, j1), np.maximum(j0, j1)
    else:
        rows = ldf["lat_index"].astype(int).to_numpy()
        if has_range:
            first = ldf["cluster_index_start"].astype(int).to_numpy()
            last = ldf["cluster_index_end"].astype(int).to_numpy()
        else:
            logger.warning(
                "[Reconstruct] Position table has no column ranges; averaging whole latitude rows"
            )
            first, last = np.zeros_like(rows), np.full_like(rows, n_lon - 1)
        if (
            rows.max(initial=-1) >= n_lat
            or last.max(initial=-1) >= n_lon
            or rows.min(initial=0) < 0
        ):
            raise ValueError(
                f"The model output {where} has {n_lat} x {n_lon} (latitude x longitude) cells, but "
                f"the position table places clusters up to row {rows.max()} and column "
                f"{last.max()}. The output was not produced on the reduced grid these mapping "
                "files describe."
            )
    return {int(lbl): (int(i), int(a), int(b)) for lbl, i, a, b in zip(labels, rows, first, last)}


def _has_value(da: xr.DataArray, lat_name: str, lon_name: str) -> np.ndarray:
    """2-D mask of cells holding a real value at any time/level (NaN and |x| >= 1e30 are missing)."""
    vals = np.asarray(da.transpose(..., lat_name, lon_name).values, dtype="float64")
    ok = np.isfinite(vals) & (np.abs(vals) < 1e30)
    fv = da.attrs.get("_FillValue", da.encoding.get("_FillValue"))
    if fv is not None:
        ok &= vals != fv
    return ok.reshape(-1, *vals.shape[-2:]).any(axis=0)


def _require_output_on_grid(
    ds: xr.Dataset,
    names: list[str],
    lat_name: str,
    lon_name: str,
    cells: dict[int, tuple[int, int, int]],
    where: str,
) -> None:
    """Refuse model output that holds values where this reduced grid has no cluster."""
    occupied = np.zeros((ds.sizes[lat_name], ds.sizes[lon_name]), dtype=bool)
    for i, a, b in cells.values():
        occupied[i, a : b + 1] = True
    for name in names:
        if name == "cluster_label":
            continue
        stray = _has_value(ds[name], lat_name, lon_name) & ~occupied
        if stray.any():
            raise ValueError(
                f"The model output {where} has values for '{name}' in {int(stray.sum())} grid "
                f"cell(s) where the reduced grid these mapping files describe has no cluster "
                f"({len(cells)} clusters, {int(occupied.sum())} occupied cells). It was produced on "
                "a different reduced grid -- check that the output and the mapping files come from "
                "the same create_clusters run (same number of clusters, same output folder). "
                "If your model writes a placeholder such as 0 in cells it did not simulate, set "
                "those cells to missing (NaN or the variable's _FillValue) first."
            )

def reconstruct_output_netcdf(
    clustered_nc: str,
    registry: DataRegistry,
    id_mapping_csv: str,
    latrow_map_csv: str,
    output_path: str | None = None,
) -> str:
    """Reconstruct a model output NetCDF (on the reduced grid) back to the original grid.

    Places values strictly through the ID mapping and the position table. Before copying
    anything it checks that the three inputs belong together: both mapping files describe the
    same clusters, every cluster's cells exist in the output, the output holds no values where
    this grid has no cluster, and every mapped cell exists on the original grid.

    - clustered_nc: model output on the reduced grid; dimensions ``lat``/``lon`` (as LPJmL
      writes them) or ``latitude``/``longitude`` (as this package writes them), plus any
      leading dimensions such as ``time``.
    - registry: the original grid (see :meth:`DataRegistry.load`).
    - id_mapping_csv: ``cluster_id_mapping_<K>.csv``.
    - latrow_map_csv: ``cluster_compact_map_<K>.csv``.
    - output_path: where to write; default next to the input with a ``_reconstructed`` suffix.
    """
    if not os.path.exists(clustered_nc):
        raise FileNotFoundError(f"Model output not found: {clustered_nc}")
    if id_mapping_csv is None or not os.path.exists(id_mapping_csv):
        raise FileNotFoundError(
            f"ID mapping (cluster_id_mapping_<K>.csv) not found: {id_mapping_csv}"
        )
    if latrow_map_csv is None:
        raise ValueError(
            "The position table (cluster_compact_map_<K>.csv) is required to reconstruct model "
            "output; it is written beside the ID mapping."
        )
    id_map = pd.read_csv(id_mapping_csv)
    ldf = _read_position_table(latrow_map_csv)
    _require_mappings_agree(id_map, ldf, id_mapping_csv, latrow_map_csv)

    where = f"'{os.path.basename(clustered_nc)}'"
    ds = xr.open_dataset(clustered_nc)
    try:
        lat_name, lon_name = _spatial_names(ds)
        k_file = ds.attrs.get("n_clusters")
        if k_file is not None and int(k_file) != len(ldf):
            raise ValueError(
                f"The model output {where} was written for {int(k_file)} clusters, but the mapping "
                f"files describe {len(ldf)}. Use the mapping files of the run that produced it."
            )
        names = [v for v in ds.data_vars if {lat_name, lon_name} <= set(ds[v].dims)]
        names = [v for v in names if v != "cluster_label_for_lat"]
        if not names:
            raise ValueError(
                f"The model output {where} has no variable on its {lat_name}/{lon_name} grid "
                f"(variables: {list(ds.data_vars)})."
            )
        cells = _cluster_cells(ds, lat_name, lon_name, ldf, where)
        _require_output_on_grid(ds, names, lat_name, lon_name, cells, where)

        orig = registry.get_original_grid()
        orig_valid_mask = registry.get_valid_mask()
        cell_ids = orig.cell_ids
        rows_i, cols_j = np.nonzero(cell_ids >= 0)
        id_to_pos = {int(cell_ids[i, j]): (int(i), int(j)) for i, j in zip(rows_i, cols_j)}
        unknown = int((~id_map["original_cell_id"].astype(int).isin(id_to_pos)).sum())
        if unknown:
            raise ValueError(
                f"{unknown} of {len(id_map)} cells in {os.path.basename(id_mapping_csv)} do not "
                "exist on the original grid given (the registry). The registry describes a "
                "different grid than the one that was clustered -- rebuild it from the same soil "
                "file, or load the one saved with the run (load_clusters / DataRegistry.load)."
            )

        results: dict[str, xr.DataArray] = {}
        logger.info(f"[Reconstruct] Reading clustered dataset: {clustered_nc}")
        for var in names:
            da = ds[var].transpose(..., lat_name, lon_name)
            lead_dims = da.dims[:-2]
            vals = np.asarray(da.values, dtype="float64")
            fv = da.attrs.get("_FillValue", -1e32)
            vals = np.where((vals == fv) | (np.abs(vals) >= 1e30), np.nan, vals)
            per_cluster: dict[int, np.ndarray] = {}
            with warnings.catch_warnings():
                warnings.simplefilter("ignore", RuntimeWarning)
                for lbl, (i, a, b) in cells.items():
                    per_cluster[lbl] = np.nanmean(vals[..., i, a : b + 1], axis=-1)
            empty = [lbl for lbl, v in per_cluster.items() if np.isnan(v).all()]
            if empty:
                warnings.warn(
                    f"'{var}': {len(empty)} of {len(per_cluster)} clusters have no value in the "
                    f"model output and stay NaN on the full grid (first: {sorted(empty)[:10]}).",
                    UserWarning,
                    stacklevel=3,
                )

            out = np.full((*vals.shape[:-2], orig.latitudes.size, orig.longitudes.size), np.nan)
            filled = 0
            for oid, lbl in zip(
                id_map["original_cell_id"].astype(int), id_map["cluster_label"].astype(int)
            ):
                i, j = id_to_pos[oid]
                if not orig_valid_mask[i, j]:
                    continue
                out[..., i, j] = per_cluster[lbl]
                filled += 1

            coords = {d: ds[d].values for d in lead_dims if d in ds.coords}
            coords.update({"latitude": orig.latitudes, "longitude": orig.longitudes})
            results[var] = xr.DataArray(
                out,
                dims=(*lead_dims, "latitude", "longitude"),
                coords=coords,
                name=var,
                attrs=_carried_attrs(da),
            )
            logger.info(f"[Reconstruct] Var '{var}': filled {filled} of {len(id_map)} cells")
    finally:
        ds.close()

    ds_out = xr.Dataset(results)
    if output_path is None:
        base, ext = os.path.splitext(clustered_nc)
        output_path = base + "_reconstructed.nc"
    ds_out.to_netcdf(output_path)
    return output_path

def _spatial_names(ds: xr.Dataset) -> tuple[str, str]:
    """``(lat_name, lon_name)`` — LPJmL writes ``lat``/``lon``, this package ``latitude``/..."""
    for lat, lon in (("latitude", "longitude"), ("lat", "lon")):
        if lat in ds.dims and lon in ds.dims:
            return lat, lon
    raise ValueError(
        f"Cannot find latitude/longitude dimensions in the model output (dims: {dict(ds.sizes)}). "
        "Expected 'lat'/'lon' (as LPJmL writes them) or 'latitude'/'longitude'."
    )


def _axis_positions(axis: np.ndarray, wanted: np.ndarray, what: str, where: str) -> np.ndarray:
    """Index of each wanted coordinate on ``axis``, refusing anything off the lattice."""
    axis = np.asarray(axis, dtype=float)
    wanted = np.asarray(wanted, dtype=float)
    if axis.size == 1:
        idx = np.zeros(wanted.size, dtype=int)
        step = 1.0
    else:
        order = np.argsort(axis)
        srt = axis[order]
        pos = np.clip(np.searchsorted(srt, wanted), 1, srt.size - 1)
        nearer_left = np.abs(wanted - srt[pos - 1]) <= np.abs(wanted - srt[pos])
        idx = order[np.where(nearer_left, pos - 1, pos)]
        step = float(np.min(np.diff(srt)))
    off = np.abs(axis[idx] - wanted)
    bad = off > 0.25 * step
    if bad.any():
        raise ValueError(
            f"{int(bad.sum())} cell(s) the mapping names have a {what} that is not on the "
            f"{what} axis of {where} (worst {float(off.max()):.6g} deg off; the axis runs "
            f"{float(axis.min()):.6g} to {float(axis.max()):.6g}). The model output was not "
            "produced on the reduced grid this mapping was written for."
        )
    return idx

def reconstruct_from_distributed(
    cluster_file_path: str,
    distributed_mapping_file: str,
    output_dir: str | None,
    registry: DataRegistry,
    labels_csv: str | None = None,
    *,
    variable: str | None = None,
    aggregate: str = "mean",
    output_path: str | None = None,
) -> str:
    """Project output from a distributed grid back onto every cell of the original grid.

    In the ``"distributed"`` cell choice each cluster was simulated on several of its own
    member cells, each with a fraction of the patch budget. This combines those cells per
    cluster — at every time step and for every non-spatial index — and writes the result to
    **every** original cell of the cluster.

    Parameters
    ----------
    cluster_file_path : str
        A NetCDF on the distributed grid: model output (``lat``/``lon``, as LPJmL writes it)
        or a forcing file this package wrote (``latitude``/``longitude``). The file name is
        not parsed; any name works. A model run may trim empty margins from the grid —
        chosen cells are located by coordinate, so that is harmless.
    distributed_mapping_file : str
        The ``cluster_distributed_mapping_{K}.csv`` written with the grid.
    output_dir : str or None
        Directory for the output when ``output_path`` is not given (default: beside the
        input).
    registry : DataRegistry
        The original grid.
    labels_csv : str
        The per-cell ``cell_id,cluster_label`` table for the whole original grid.
    variable : str, optional
        Reconstruct only this variable. Default: every variable with both spatial
        dimensions.
    aggregate : {"mean", "median"}, optional
        How a cluster's chosen cells are combined. ``"mean"`` (default) is what the
        vegetation model does across the patches inside one cell, so ``n`` cells × ``p``
        patches then corresponds to one cell × ``n·p`` patches. ``"median"`` reproduces
        the reconstruction the method was originally published with. Missing values (NaN, or a fill value
        of magnitude ≥ 1e30) are ignored.
    output_path : str, optional
        Exact output path; overrides ``output_dir``.

    Returns
    -------
    str
        Path to the reconstructed NetCDF. Cells outside the original land mask, and every
        cell of a cluster none of whose chosen cells has a value, are NaN.

    Raises
    ------
    FileNotFoundError
        If an input is missing.
    ValueError
        On an unknown ``aggregate``, a missing ``labels_csv``, or a model output whose grid
        does not hold the cells the mapping names.
    """
    if aggregate not in DISTRIBUTED_AGGREGATES:
        raise ValueError(f"aggregate must be one of {DISTRIBUTED_AGGREGATES}; got {aggregate!r}.")
    if not os.path.exists(cluster_file_path):
        raise FileNotFoundError(cluster_file_path)
    if not os.path.exists(distributed_mapping_file):
        raise FileNotFoundError(f"Distributed mapping not found: {distributed_mapping_file}")
    if labels_csv is None or not os.path.exists(labels_csv):
        raise ValueError(
            "labels_csv is required for distributed reconstruction: it says which original "
            "cells belong to each cluster."
        )

    mapping = pd.read_csv(distributed_mapping_file)
    labels = pd.read_csv(labels_csv)
    id_col = "cell_id" if "cell_id" in labels.columns else "original_cell_id"

    orig = registry.get_original_grid()
    orig_lats = np.asarray(orig.latitudes)
    orig_lons = np.asarray(orig.longitudes)
    ids = np.asarray(orig.cell_ids)
    rows, cols = np.nonzero(ids >= 0)
    pos = pd.DataFrame({"cell_id": ids[rows, cols].astype(int), "i": rows, "j": cols})

    # Every original cell of every cluster, with its grid position.
    members = labels[[id_col, "cluster_label"]].rename(columns={id_col: "cell_id"})
    members = members[members["cluster_label"] >= 0].merge(
        pos, on="cell_id", how="left", indicator=True
    )
    # A labelled cell the registry does not have, or a cluster the mapping does not cover,
    # would leave cells NaN that the model did simulate: refuse both, as the one-cell route does.
    off_grid = members["_merge"] != "both"
    if off_grid.any():
        bad = members.loc[off_grid, "cell_id"].tolist()
        raise ValueError(
            f"{int(off_grid.sum())} of {len(members)} labelled cells in "
            f"{os.path.basename(labels_csv)} do not exist on the original grid given (the "
            f"registry), e.g. cell_id {bad[:5]}. The registry describes a different grid than "
            "the one that was clustered -- load the one saved with the run."
        )
    members = members.drop(columns="_merge")

    clusters = np.sort(mapping["cluster_label"].unique())
    col_of = {int(c): k for k, c in enumerate(clusters)}
    unmapped = sorted(set(members["cluster_label"].astype(int)) - set(col_of))
    if unmapped:
        raise ValueError(
            f"{len(unmapped)} cluster(s) of {os.path.basename(labels_csv)} have no cell in "
            f"{os.path.basename(distributed_mapping_file)} (e.g. {unmapped[:10]}), so their "
            "cells could not be filled. The label table and the mapping must come from the "
            "same create_clusters run."
        )

    with xr.open_dataset(cluster_file_path) as raw:
        ds = raw.load()
    lat_name, lon_name = _spatial_names(ds)
    where = os.path.basename(cluster_file_path)
    li = _axis_positions(ds[lat_name].values, mapping["grid_latitude"].values, "latitude", where)
    lj = _axis_positions(ds[lon_name].values, mapping["grid_longitude"].values, "longitude", where)
    slot_cluster = mapping["cluster_label"].map(col_of).to_numpy()

    if variable is not None:
        if variable not in ds.data_vars:
            raise ValueError(f"Variable {variable!r} not found in {where}: {list(ds.data_vars)}")
        names = [variable]
    else:
        names = [v for v in ds.data_vars if {lat_name, lon_name} <= set(ds[v].dims)]
        if not names:
            raise ValueError(f"{where} holds no variable on the {lat_name}/{lon_name} grid.")

    out_vars = {}
    coords: dict = {"latitude": orig_lats, "longitude": orig_lons}
    for name in names:
        da = ds[name].transpose(..., lat_name, lon_name)
        lead_dims = da.dims[:-2]
        vals = np.asarray(da.values, dtype="float64")
        slots = vals[..., li, lj]  # (*lead, n_slots)
        slots = np.where(np.abs(slots) >= 1e30, np.nan, slots)

        combined = np.full((*slots.shape[:-1], clusters.size), np.nan)
        for k in range(clusters.size):
            block = slots[..., slot_cluster == k]
            if block.shape[-1] == 0 or np.isnan(block).all():
                continue
            with np.errstate(invalid="ignore"), warnings.catch_warnings():
                warnings.simplefilter("ignore", RuntimeWarning)
                reducer = np.nanmean if aggregate == "mean" else np.nanmedian
                combined[..., k] = reducer(block, axis=-1)

        empty = [int(c) for k, c in enumerate(clusters) if np.isnan(combined[..., k]).all()]
        if empty:
            logger.warning(
                "%s: %d of %d clusters have no value at any chosen cell and stay NaN: %s",
                name,
                len(empty),
                clusters.size,
                empty[:10],
            )

        grid = np.full((*slots.shape[:-1], orig_lats.size, orig_lons.size), np.nan, "float32")
        grid[..., members["i"].to_numpy(int), members["j"].to_numpy(int)] = combined[
            ..., members["cluster_label"].map(col_of).to_numpy()
        ]
        # Collected over every variable: a static variable after a time-varying one must
        # not drop the time values the first one needs.
        coords.update({d: ds[d].values for d in lead_dims if d in ds.coords})
        attrs = _carried_attrs(ds[name])
        attrs["distributed_aggregate"] = aggregate
        out_vars[name] = ((*lead_dims, "latitude", "longitude"), grid, attrs)

    ds_out = xr.Dataset(out_vars, coords=coords)

    if output_path is None:
        stem = os.path.splitext(os.path.basename(cluster_file_path))[0]
        target_dir = output_dir or os.path.dirname(os.path.abspath(cluster_file_path))
        output_path = os.path.join(target_dir, f"{stem}_reconstructed_distributed.nc")
    os.makedirs(os.path.dirname(os.path.abspath(output_path)), exist_ok=True)
    ds_out.to_netcdf(
        output_path,
        encoding={v: {"zlib": True, "complevel": 4, "shuffle": True} for v in out_vars},
    )
    logger.info("Reconstructed distributed output (%s): %s", aggregate, output_path)
    return output_path
