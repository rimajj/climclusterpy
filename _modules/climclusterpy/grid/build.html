"""
Cluster NetCDF file generation module.

Converts hierarchical clustering results and timeseries data into
regular-grid NetCDF files suitable for land surface models.
"""

from __future__ import annotations

import logging
import os
from concurrent.futures import ProcessPoolExecutor

import numpy as np
import pandas as pd
import xarray as xr

from climclusterpy._mpi import is_mpi_root, mpi_rank
from climclusterpy.data.coords import COORD_PRECISION, standardize_coordinates

logger = logging.getLogger(__name__)


def _gather_cluster_centers(ds: xr.Dataset, labels_csv: str | None) -> pd.DataFrame:
    """Return a DataFrame with columns ['cluster_label','center_lat','center_lon'] in ds order.

    Preference order:
    1) Use center_lat/center_lon present in the dataset.
    2) If labels_csv provided and contains center_lat/center_lon, use those and order by ds label order.
    Otherwise raise ``KeyError``: the producer of ``ds`` must supply the centres.
    """
    if "cluster_label" not in ds:
        raise KeyError("Timeseries dataset must include 'cluster_label' as a data variable")
    # Get label order from dataset (coord or data var)
    labels = np.asarray(ds["cluster_label"].values).reshape(-1)

    # Case 1: present in dataset
    if "center_lat" in ds and "center_lon" in ds:
        return pd.DataFrame(
            {
                "cluster_label": labels.astype(int),
                "center_lat": np.asarray(ds["center_lat"].values).reshape(-1),
                "center_lon": np.asarray(ds["center_lon"].values).reshape(-1),
            }
        )

    # Case 1b: distributed-style NC — per-cell latitude/longitude, derive per-cluster mean centers.
    # The 'labels' array has one entry per cell (possibly repeated across npatch cells per cluster).
    # We group by cluster_label and average lat/lon to get a unique center per cluster.
    if "latitude" in ds and "longitude" in ds:
        lats = np.asarray(ds["latitude"].values).reshape(-1)
        lons = np.asarray(ds["longitude"].values).reshape(-1)
        df_cells = pd.DataFrame({"cluster_label": labels.astype(int), "lat": lats, "lon": lons})
        df_centers = (
            df_cells.groupby("cluster_label", sort=False)
            .agg(center_lat=("lat", "mean"), center_lon=("lon", "mean"))
            .reset_index()
        )
        # Return one row per unique label, in the order they first appear in 'labels'
        unique_labels, first_idx = np.unique(labels.astype(int), return_index=True)
        # Sort by first occurrence
        order = np.argsort(first_idx)
        df_centers = df_centers.set_index("cluster_label").loc[unique_labels[order]].reset_index()
        df_centers.columns = ["cluster_label", "center_lat", "center_lon"]
        return df_centers

    # Case 2: try labels CSV
    if labels_csv is not None and os.path.exists(labels_csv):
        df_lab = pd.read_csv(labels_csv)
        if {"cluster_label", "center_lat", "center_lon"}.issubset(df_lab.columns):
            # Ensure one row per cluster
            df_centers = df_lab.drop_duplicates("cluster_label")[
                ["cluster_label", "center_lat", "center_lon"]
            ].copy()
            # Reorder to dataset label order
            order_map = {int(lbl): i for i, lbl in enumerate(labels.astype(int))}
            df_centers = df_centers[df_centers["cluster_label"].astype(int).isin(order_map.keys())]
            df_centers["__ord"] = df_centers["cluster_label"].astype(int).map(order_map)
            df_centers = df_centers.sort_values("__ord").drop(columns="__ord")
            # If any missing, fall through
            if len(df_centers) == len(labels):
                return df_centers[["cluster_label", "center_lat", "center_lon"]].reset_index(
                    drop=True
                )

    # Case 3: synthetic centers (should not happen once producers are unified)
    raise KeyError(
        "center_lat/center_lon missing and labels CSV lacks centers; producer must include centers"
    )


def _circular_mode(residues: np.ndarray, bins: int = 80) -> float:
    """Find mode on [0, d) via histogram peak (returns bin center)."""
    max_val = residues.max() if len(residues) > 0 and residues.max() > 0 else 1.0
    hist, edges = np.histogram(residues, bins=bins, range=(0.0, max_val))
    idx = int(np.argmax(hist))
    return 0.5 * (edges[idx] + edges[idx + 1])


def _candidate_spacings(sorted_lats: np.ndarray, min_spacing: float) -> list:
    """Return a small set of sensible spacing candidates to evaluate."""
    uniq = np.sort(np.unique(sorted_lats))
    span = float(uniq[-1] - uniq[0]) if len(uniq) > 1 else 0.0
    diffs = np.diff(uniq)
    diffs = diffs[diffs > 1e-10]

    cand = set()
    # Always include min spacing
    cand.add(float(min_spacing))

    # Include common coarse spacings
    for v in [0.25, 0.5, 1.0, 2.0, 5.0]:
        if v >= min_spacing:
            cand.add(v)

    # Include data-driven candidates
    if len(diffs) > 0:
        cand.add(float(np.median(diffs)))
        # Mode of diffs via histogram
        n_bins = min(40, max(10, len(diffs) // 2))
        hist, bins = np.histogram(diffs, bins=n_bins)
        mode_idx = np.argmax(hist)
        mode_d = 0.5 * (bins[mode_idx] + bins[mode_idx + 1])
        if mode_d >= min_spacing:
            cand.add(float(mode_d))

    # Rational span splits (span / n) for small n
    if span > 0:
        for n in range(1, 9):
            d = span / n
            if d >= min_spacing:
                cand.add(float(d))

    # Sort and return
    return sorted([c for c in cand if c >= min_spacing])


def _evaluate_grid(sorted_lats: np.ndarray, d: float, ref: float = None) -> dict:
    """Compute best x0 (via residues mode/median) and diagnostics for given spacing d."""
    uniq = np.sort(np.unique(sorted_lats))
    if ref is None:
        ref = float(np.median(uniq))

    eps = 1e-10
    residues = np.mod(uniq - ref + eps, d)

    # Choose a robust residue: the histogram mode
    n_bins = min(80, max(20, len(residues) // 2))
    rmode = _circular_mode(residues, bins=n_bins)

    # Also compute IQR mask and possibly override with median of inliers
    Q1 = np.percentile(residues, 25)
    Q3 = np.percentile(residues, 75)
    IQR = Q3 - Q1
    mask = (residues >= Q1 - 1.5 * IQR) & (residues <= Q3 + 1.5 * IQR)
    if np.any(mask):
        robust_res = float(np.median(residues[mask]))
    else:
        robust_res = float(np.median(residues))

    # Use a blend: prefer mode if it is within reasonable distance to robust_res
    if abs(rmode - robust_res) <= 0.5 * d:
        chosen_res = rmode
    else:
        chosen_res = robust_res

    x0 = ref - chosen_res

    # Diagnostics on full set
    k = np.round((sorted_lats - x0) / d).astype(int)
    adjusted = x0 + k * d
    devs = np.abs(adjusted - sorted_lats)

    median_dev = float(np.median(devs))
    p90_dev = float(np.percentile(devs, 90))
    max_dev = float(np.max(devs))
    mean_dev = float(np.mean(devs))

    # Relative errors
    rel = devs / d if d != 0 else np.zeros_like(devs)

    # Coverage: fraction within 10% of d
    coverage = float(np.mean(rel <= 0.1))

    # Number of grid points spanned
    grid_points = int(np.max(k) - np.min(k) + 1) if len(k) > 0 else 1

    # Check monotonicity
    monotonic = bool(np.all(np.diff(k) >= -1e-6))

    return {
        "d": float(d),
        "x0": float(x0),
        "median_dev": median_dev,
        "p90_dev": p90_dev,
        "max_dev": max_dev,
        "mean_dev": mean_dev,
        "coverage": coverage,
        "grid_points": grid_points,
        "monotonic": monotonic,
    }

def calculate_grid_parameters(
    sorted_lats: np.ndarray,
    min_spacing: float = 0.25,
    beta: float = 0.01,
    tolerance: float | None = None,
    max_spacing: float = 5.0,
    spacing_steps: int = 100,
) -> dict:
    """
    Fast grid detection using candidate-based approach with tolerance-based coarsening.

    Algorithm features:
      - O(N) complexity: evaluates spacing candidates from linspace
      - Works on unique latitudes
      - Tests spacings uniformly distributed between min_spacing and max_spacing
      - Accepts grids where all points can shift ≤ tolerance to fit
      - Robust x0 via circular mode on residues with IQR outlier rejection
      - Scoring balances fit quality (median deviation) vs grid parsimony
      - Optional local refinement (±10%) around best candidate
      - Monotonicity validation

    ⚠ **On latitudes that already lie on a regular lattice, ``min_spacing`` is the answer,
    not a floor.** Every candidate spacing then fits to within ~0 deviation, so the parsimony
    term decides and it always prefers the finest candidate offered. Measured across 26 real
    cluster sets, this function returned the 0.1° floor in all 26 and never coarsened,
    which is why the grids built from cluster centres carry a latitude
    axis about five times longer than their cells need. Callers whose latitudes come off a
    regular source grid — which cluster centres do — should pass that grid's own cell size,
    or place the cells with :func:`~climclusterpy.build_reduced_grid`, which needs no search
    at all.

    Args:
        sorted_lats: Sorted array of latitude values
        min_spacing: Minimum allowed grid spacing in degrees (default 0.25°).
                     Also serves as tolerance if tolerance is None. See the note above: on
                     regularly-spaced input this value is what you get.
        beta: Parsimony penalty weight (default 0.01). Higher values favor coarser grids.
        tolerance: Maximum allowed shift for original latitudes (degrees).
                   If provided, only grids with max_dev ≤ tolerance are accepted.
                   If None, uses min_spacing as tolerance.
        max_spacing: Maximum grid spacing to test in degrees (default 5.0°).
        spacing_steps: Number of spacing candidates to test between min and max (default 100).

    Returns
    -------
        Dict with keys: d, x0, max_dev, mean_dev, median_dev, p90_dev,
                       coverage, adaptive_thresh, grid_points, monotonic
    """
    sorted_lats = np.asarray(sorted_lats, dtype=float).ravel()
    if sorted_lats.size == 0:
        raise ValueError("calculate_grid_parameters needs at least one latitude; got none.")
    if not np.all(np.isfinite(sorted_lats)):
        raise ValueError(
            f"calculate_grid_parameters got {int((~np.isfinite(sorted_lats)).sum())} missing "
            "(NaN) or infinite latitude(s); pass the cluster latitudes without gaps."
        )
    if not 0 < min_spacing <= max_spacing:
        raise ValueError(
            f"Need 0 < min_spacing <= max_spacing; got min_spacing={min_spacing}, "
            f"max_spacing={max_spacing} (degrees)."
        )
    sorted_lats = np.sort(sorted_lats)

    # Default tolerance to min_spacing if not provided
    if tolerance is None:
        tolerance = min_spacing

    # Edge cases
    if len(sorted_lats) <= 1:
        x0_val = float(sorted_lats[0]) if len(sorted_lats) else 0.0
        return {
            "d": float(min_spacing),
            "x0": x0_val,
            "max_dev": 0.0,
            "mean_dev": 0.0,
            "median_dev": 0.0,
            "p90_dev": 0.0,
            "coverage": 1.0,
            "adaptive_thresh": 0.0,
            "grid_points": len(sorted_lats),
            "monotonic": True,
        }

    uniq = np.sort(np.unique(sorted_lats))
    if len(uniq) <= 1:
        return {
            "d": float(min_spacing),
            "x0": float(uniq[0]),
            "max_dev": 0.0,
            "mean_dev": 0.0,
            "median_dev": 0.0,
            "p90_dev": 0.0,
            "coverage": 1.0,
            "adaptive_thresh": 0.0,
            "grid_points": len(uniq),
            "monotonic": True,
        }

    span = float(uniq[-1] - uniq[0])

    # STEP 1: Generate spacing candidates using linspace between min and max
    # Test spacings uniformly distributed for thorough coverage
    spacing_candidates = np.linspace(min_spacing, max_spacing, spacing_steps)

    # Filter candidates: only accept if max_dev ≤ tolerance
    # This dramatically reduces memory (e.g., 0.25° grid vs 1.0° grid = 4x fewer points)
    accepted_candidates = []
    for test_spacing in spacing_candidates:
        test_res = _evaluate_grid(sorted_lats, test_spacing)
        if test_res["max_dev"] <= tolerance:
            accepted_candidates.append(test_spacing)

    # If no candidates pass tolerance, fall back to min_spacing
    if len(accepted_candidates) == 0:
        logger.warning(
            f"  WARNING: No spacing candidates within tolerance={tolerance:.4f}°, using min_spacing={min_spacing:.3f}°"
        )
        accepted_candidates = [min_spacing]
    else:
        logger.info(
            f"  Grid search: {len(accepted_candidates)}/{len(spacing_candidates)} candidates within tolerance={tolerance:.4f}°"
        )
        logger.info(
            f"  Spacing range: {min(accepted_candidates):.3f}° to {max(accepted_candidates):.3f}°"
        )

    # STEP 2: Get additional standard candidate spacings (for backwards compatibility)
    candidates = _candidate_spacings(sorted_lats, min_spacing)
    if len(candidates) == 0:
        candidates = [min_spacing]

    # STEP 3: Combine all candidates (accepted + standard)
    all_candidates = sorted(set(accepted_candidates + candidates))

    # Evaluate all candidates and pick best by score
    best = None
    best_score = np.inf

    for d in all_candidates:
        res = _evaluate_grid(sorted_lats, d)
        # Score: small median_dev preferred; penalize many grid points (finer grids)
        # Normalize grid_points by expected count for this spacing
        expected_gp = (span / d) if span > 0 and d > 0 else res["grid_points"]
        norm_gp = res["grid_points"] / expected_gp if expected_gp > 0 else 1.0
        score = res["median_dev"] + beta * norm_gp * d
        res["score"] = score

        if score < best_score:
            best_score = score
            best = res

    # Optional local refinement around best d (±10%, 5 steps)
    if best is not None:
        d0 = best["d"]
        low = max(min_spacing, d0 * 0.9)
        high = d0 * 1.1
        for d in np.linspace(low, high, 5):
            res = _evaluate_grid(sorted_lats, d)
            expected_gp = (span / d) if span > 0 and d > 0 else res["grid_points"]
            norm_gp = res["grid_points"] / expected_gp if expected_gp > 0 else 1.0
            score = res["median_dev"] + beta * norm_gp * d
            res["score"] = score

            if score < best_score:
                best_score = score
                best = res

    # Add adaptive_thresh based on final grid's relative errors
    d_chosen = best["d"]
    x0_chosen = best["x0"]
    k = np.round((sorted_lats - x0_chosen) / d_chosen).astype(int)
    adjusted = x0_chosen + k * d_chosen
    devs = np.abs(adjusted - sorted_lats)
    rel = devs / d_chosen if d_chosen != 0 else np.zeros_like(devs)
    adaptive_thresh = float(np.percentile(rel, 90)) if len(rel) > 0 else 0.0
    best["adaptive_thresh"] = adaptive_thresh

    return best




def validate_even_spacing(adjusted_lats: np.ndarray, d: float) -> None:
    """Ensure all populated grid points lie on valid grid positions.

    Note: Grid can have empty points (gaps larger than d), but all populated
    points must be at valid grid positions (integer multiples of d from x0).
    """
    unique = np.unique(adjusted_lats)
    if len(unique) <= 1:
        return

    # Get first point
    x0 = unique[0]

    # For each populated point, check it aligns to a valid grid position
    for lat in unique:
        # How many d steps from x0?
        k = np.round((lat - x0) / d)
        # Check this point lies exactly on a grid position
        expected = x0 + k * d
        if not np.isclose(lat, expected, atol=1e-6):
            raise ValueError(
                f"Invalid grid position {lat:.6f}°. Nearest valid position would be {expected:.6f}°\n"
                f"Grid has spacing d={d:.6f}° starting from x0={x0:.6f}°"
            )

def create_regular_grid_from_dataset(
    ds: xr.Dataset,
    n_repeat: int,
    output_dir: str,
    labels_csv: str | None = None,
    min_grid_spacing: float = 0.1,
    max_grid_spacing: float = 5.0,
    grid_spacing_steps: int = 100,
) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray, float, float]:
    """Create grid with guaranteed even spacing and minimal deviation from input dataset.

    Places each cluster at its **centre** latitude — ``center_lat``, the latitude of
    whichever member sits nearest the cluster's median absolute latitude — on a lattice found
    by :func:`calculate_grid_parameters`. That is not the latitude of the cell whose climate
    the reduced cell carries: measured on 26 real cluster sets, the two are up to 107° of
    latitude apart. For the reduced-grid round trip use
    :func:`~climclusterpy.build_reduced_grid` instead, which places each cell at its own
    representative coordinates; this function remains the right call when the latitudes
    genuinely do not lie on a regular lattice and a spacing search is needed.

    Args:
        ds: Dataset with cluster centers
        n_repeat: Number of repeated cells per cluster
        output_dir: Directory for output files
        labels_csv: Optional CSV file with cluster labels
        min_grid_spacing: Minimum grid spacing in degrees, and the tolerance for accepting a
                         coarser one. Smaller = finer grid = more RAM. Default 0.1°. On
                         latitudes that already lie on a regular lattice the search returns
                         this value itself rather than coarsening, so it sets the row count
                         outright — see :func:`calculate_grid_parameters`.
        max_grid_spacing: Maximum grid spacing to test in degrees. Default: 5.0.
        grid_spacing_steps: Number of spacing candidates to test. Default: 100.

    Returns
    -------
        Tuple containing:
        - grid_lats: repeated adjusted latitudes (n_clusters * n_repeat)
        - grid_lons: all longitude values (n_clusters * n_repeat)
        - adjusted_lats: unique adjusted latitudes per cluster (n_clusters)
        - global_lons: all longitude values (n_clusters * n_repeat)
        - lat_spacing: the calculated latitude grid spacing
        - lat_offset: the calculated latitude grid offset (x0)
    """
    clusters = _gather_cluster_centers(ds, labels_csv)
    original_lats = clusters["center_lat"].values
    n_clusters = len(original_lats)
    sorted_lats = np.sort(original_lats)

    # Find optimal grid parameters using robust algorithm with tolerance-based coarsening
    # tolerance = min_grid_spacing: accept coarser grids if points fit within this tolerance
    logger.info(
        f"\nGrid Detection (min_spacing={min_grid_spacing}°, max_spacing={max_grid_spacing}°, steps={grid_spacing_steps}, tolerance={min_grid_spacing}°):"
    )
    grid_params = calculate_grid_parameters(
        sorted_lats,
        min_spacing=min_grid_spacing,
        tolerance=min_grid_spacing,
        max_spacing=max_grid_spacing,
        spacing_steps=grid_spacing_steps,
    )

    if grid_params["d"] == 0:
        raise ValueError("Invalid grid: spacing is zero (all latitudes identical)")

    # Round spacing and offset to COORD_PRECISION to ensure exact grid
    lat_spacing_rounded = np.round(grid_params["d"], COORD_PRECISION)
    lat_offset_rounded = np.round(grid_params["x0"], COORD_PRECISION)

    # Calculate final positions using rounded parameters
    k_values = np.round((sorted_lats - lat_offset_rounded) / lat_spacing_rounded)
    adjusted_sorted = lat_offset_rounded + k_values * lat_spacing_rounded
    # Round to same precision to ensure exact match with full grid later
    adjusted_sorted = np.round(adjusted_sorted, COORD_PRECISION).astype(np.float64)

    # Restore original order
    sort_idx = np.argsort(original_lats)
    adjusted_lats = adjusted_sorted[np.argsort(sort_idx)]

    # Validate even spacing using rounded spacing
    validate_even_spacing(adjusted_lats, lat_spacing_rounded)

    # Calculate and report deviations (using same order as grid_params was calculated with)
    deviations = np.abs(adjusted_lats - original_lats)

    # Comprehensive grid report
    unique_adjusted = np.unique(adjusted_sorted)
    n_populated = len(unique_adjusted)

    # Calculate total grid span and potential empty points
    if n_populated > 1:
        grid_span = (unique_adjusted[-1] - unique_adjusted[0]) / grid_params["d"]
        n_total_grid_points = int(np.round(grid_span)) + 1
        n_empty = n_total_grid_points - n_populated
    else:
        n_total_grid_points = 1
        n_empty = 0

    logger.info(f"\n{'=' * 60}")
    logger.info(f"Regular Grid Report ({n_clusters} clusters)")
    logger.info(f"{'=' * 60}")
    logger.info(f"Grid spacing (d):          {grid_params['d']:.6f}°")
    logger.info(f"Grid offset (x0):          {grid_params['x0']:.6f}°")
    logger.info("")
    logger.info("Latitude Adjustment Statistics:")
    # Use deviations calculated from adjusted_lats vs original_lats (actual final deviations)
    logger.info(f"  Max deviation:           {np.max(deviations):.6f}°")
    logger.info(f"  Mean deviation:          {np.mean(deviations):.6f}°")
    logger.info(f"  Median deviation:        {np.median(deviations):.6f}°")
    logger.info(f"  90th percentile:         {np.percentile(deviations, 90):.6f}°")
    logger.info("")
    logger.info("Grid Occupancy:")
    logger.info(f"  Populated grid points:   {n_populated}")
    logger.info(f"  Empty grid points:       {n_empty}")
    logger.info(f"  Total grid points:       {n_total_grid_points}")
    logger.info(f"  Occupancy rate:          {100 * n_populated / n_total_grid_points:.1f}%")
    logger.info("")
    logger.info("Grid Quality:")
    logger.info(f"  Coverage (adaptive):     {100 * grid_params['coverage']:.1f}%")
    logger.info(f"  Monotonic:               {'Yes' if grid_params['monotonic'] else 'No'}")
    logger.info(f"{'=' * 60}\n")

    # Save adjustment report - ensure directory exists first
    adjustment_report_path = os.path.join(output_dir, f"latitude_adjustments_{n_clusters}.csv")
    os.makedirs(os.path.dirname(adjustment_report_path), exist_ok=True)
    pd.DataFrame(
        {
            "cluster_label": clusters["cluster_label"],
            "original_lon": clusters["center_lon"],
            "original_lat": original_lats,
            "adjusted_lat": adjusted_lats,
            "deviation": deviations,
        }
    ).to_csv(adjustment_report_path, index=False)

    # Normalize longitudes from [0, 360] to [-180, 180] for CLM compatibility
    # CLM tools require longitudes in [-180, 180] range
    center_lons = clusters["center_lon"].values.copy()
    center_lons = np.where(center_lons > 180, center_lons - 360, center_lons)

    # Generate longitude grid
    min_lon = center_lons.min()
    max_lon = center_lons.max()
    # Special-case: if there's only one cluster (or all centers share same lon),
    # creating a linspace(min_lon, max_lon, n_clusters*n_repeat) would yield
    # identical values which later collapse to a single longitude dimension.
    # Create a tiny monotonic spread around the center to preserve n_repeat columns.
    if n_clusters == 1 or np.isclose(min_lon, max_lon):
        center_lon = float(min_lon)
        # Step chosen to remain distinct after rounding to COORD_PRECISION
        step = 10 ** (-COORD_PRECISION)
        half_span = step * (n_repeat - 1) / 2.0
        global_lons = np.linspace(center_lon - half_span, center_lon + half_span, n_repeat)
    else:
        global_lons = np.linspace(min_lon, max_lon, n_clusters * n_repeat)
    # Standardize longitudes for consistent dtype/precision
    global_lons = standardize_coordinates(global_lons, "global longitudes")

    return (
        np.repeat(adjusted_lats, n_repeat),
        global_lons,
        adjusted_lats,
        global_lons,
        lat_spacing_rounded,  # lat_spacing (already rounded)
        lat_offset_rounded,  # lat_offset (already rounded)
    )




def create_cluster_id_mapping(
    n_clusters: int,
    n_repeat: int,
    labels_csv: str,
    output_dir: str,
    cluster_labels_order: list[int] | None = None,
) -> tuple[pd.DataFrame, str]:
    """Create a CSV mapping original_cell_id -> new_cell_id with cluster_index and cluster_label.

    - new_cell_id is defined canonically as cluster_index * n_repeat (first artificial cell of that cluster)
    - cluster_index is zero-based order used when writing the NC files (lat row index)
    - cluster_label is the human-readable cluster label from labels CSV
    """
    os.makedirs(output_dir, exist_ok=True)
    labels_df = pd.read_csv(labels_csv)
    required = {"cell_id", "cluster_label"}
    if not required.issubset(labels_df.columns):
        raise ValueError(
            f"Labels CSV must contain columns {required}, got {labels_df.columns.tolist()}"
        )

    # Determine mapping from cluster_label -> cluster_index
    # Prefer the exact order used in the representative dataset if provided; otherwise fall back to sorted order
    if cluster_labels_order is not None and len(cluster_labels_order) == n_clusters:
        label_to_index = {int(lbl): idx for idx, lbl in enumerate(cluster_labels_order)}
    else:
        cluster_labels_sorted = sorted(labels_df["cluster_label"].unique())
        label_to_index = {int(lbl): idx for idx, lbl in enumerate(cluster_labels_sorted)}

    rows: list[dict] = []
    for _, row in labels_df.iterrows():
        lbl = int(row["cluster_label"])
        cluster_index = label_to_index[lbl]
        new_cell_id = cluster_index * n_repeat  # canonical representative artificial cell id
        rows.append(
            {
                "original_cell_id": int(row["cell_id"]),
                "new_cell_id": int(new_cell_id),
                "cluster_index": int(cluster_index),
                "cluster_label": int(lbl),
            }
        )

    mapping_df = pd.DataFrame(rows).drop_duplicates(
        subset=["original_cell_id"]
    )  # one row per original
    mapping_path = os.path.join(output_dir, f"cluster_id_mapping_{n_clusters}.csv")
    mapping_df.to_csv(mapping_path, index=False)
    # Keep a concise notice for the mapping file
    logger.info(f"ID mapping written: {os.path.basename(mapping_path)}")
    return mapping_df, mapping_path


#: The columns that say where each cluster sits; two position tables agree when these do.
_COMPACT_MAP_LAYOUT = ("cluster_label", "lat_index", "cluster_index_start", "cluster_index_end")


def _same_layout(a: pd.DataFrame, b: pd.DataFrame) -> bool:
    if not set(_COMPACT_MAP_LAYOUT) <= set(a.columns) or not set(_COMPACT_MAP_LAYOUT) <= set(
        b.columns
    ):
        return False
    key = list(_COMPACT_MAP_LAYOUT)
    a = a[key].sort_values(key).reset_index(drop=True).astype("int64")
    b = b[key].sort_values(key).reset_index(drop=True).astype("int64")
    return a.equals(b)


def _write_compact_map(new_map: pd.DataFrame, path: str, *, own: bool) -> None:
    """Write the position table of the grid just laid out, refusing to keep a stale one.

    The table is written once per variable, so on the second variable it already exists with
    the same layout -- that is fine. A table with a DIFFERENT layout is one left by another
    build: an earlier run into the same folder (replaced, because the folder is this grid's
    own), or another grid's table beside a shared ID mapping (an error, because keeping it
    projects every cluster to another cluster's cells and overwriting it breaks the other grid).
    """
    if os.path.exists(path):
        old = pd.read_csv(path)
        if not _same_layout(old, new_map):
            if not own:
                raise ValueError(
                    f"{path} already exists and describes a different reduced grid than the one "
                    "being written (its clusters sit in other rows or columns). It was probably "
                    "written by another build that shares this ID mapping. Pass compact_map_dir= "
                    "pointing at this grid's own output folder so each grid keeps its own table."
                )
            logger.info(f"  Replacing the position table left by an earlier run: {path}")
    os.makedirs(os.path.dirname(os.path.abspath(path)), exist_ok=True)
    new_map.to_csv(path, index=False)


def create_netcdf_from_dataset(
    variable: str,
    ds: xr.Dataset,
    grid_lats: np.ndarray,
    grid_lons: np.ndarray,
    adjusted_lats: np.ndarray,
    global_lons: np.ndarray,
    n_clusters: int,
    n_repeat: int,
    output_dir: str,
    aggregation_type: str,
    n_core: int,
    npatch: int,
    n_patch_result: int,
    timeseries_type: str,
    cluster_method: str | None = None,
    lat_spacing: float | None = None,
    lat_offset: float | None = None,
    id_mapping_file: str | None = None,
    lon_spacing: float | None = None,
    compact_map_dir: str | None = None,
) -> None:
    """Create memory-efficient NetCDF with regular lat/lon grid (CLM-compatible).

    Vectorized replacement for the original loop-based implementation.
    The output file format, filename, and grid structure are identical to the
    original — only the internal fill strategy has changed (numpy fancy indexing
    instead of Python for-loops), giving a ~500x speedup for large cluster counts.

    ``compact_map_dir``, when given, is where ``cluster_compact_map_{K}.csv`` goes — the one
    table the projection back reads. Left at None it goes beside ``id_mapping_file``, which is
    the historical location and correct only when the two were written by the same build.

    ``lon_spacing``, when given, lays the columns out that many degrees apart starting at
    ``global_lons.min()`` instead of spreading them over the range of the cluster centre
    longitudes. The step it produces is what ``cdf2clm`` writes into the header and the
    vegetation model reads back as ``config->resolution``, so passing the source grid's own
    cell size is what makes the declared cell size correct. Left at
    None the layout is unchanged, including the 0.1° stand-in column a single-column grid is
    padded with.
    """
    from collections import defaultdict

    # ── 1. Build the same lat/lon grid as the original ────────────
    unique_populated_lats = np.unique(adjusted_lats)
    lat_to_clusters: dict = defaultdict(list)
    for i in range(n_clusters):
        lat_to_clusters[adjusted_lats[i]].append(i)

    if lat_spacing is not None and lat_offset is not None:
        scale = 10**COORD_PRECISION
        lat_offset_int = round(lat_offset * scale)
        lat_spacing_int = round(lat_spacing * scale)
        k_min = round((unique_populated_lats.min() - lat_offset) / lat_spacing)
        k_max = round((unique_populated_lats.max() - lat_offset) / lat_spacing)
        unique_lats = np.array(
            [(lat_offset_int + k * lat_spacing_int) / scale for k in range(k_min, k_max + 1)]
        )
    else:
        unique_lats = unique_populated_lats.copy()
        logger.warning(
            "  WARNING: No lat_spacing provided - grid may not be regular (CLM tools may fail)"
        )

    max_clusters_per_lat = max(len(v) for v in lat_to_clusters.values())
    lon_dim = max_clusters_per_lat * n_repeat
    lat_dim = len(unique_lats)

    # Ensure at least 2D grid (CLM requirement)
    expanded_lat = expanded_lon = False
    if lat_dim == 1:
        unique_lats = np.array([unique_lats[0], unique_lats[0] + (lat_spacing or 1.0)])
        lat_dim = 2
        expanded_lat = True
        logger.info(f"  Note: Grid had single latitude - added NaN row at {unique_lats[1]:.5f}°")

    if lon_dim == 1:
        pad_step = float(lon_spacing) if lon_spacing else 0.1
        unique_lons = np.array([global_lons.min(), global_lons.min() + pad_step])
        lon_dim = 2
        expanded_lon = True
        logger.info(
            f"  Note: Grid had single longitude - added NaN column at {unique_lons[1]:.5f}°"
        )
    elif lon_spacing is not None:
        unique_lons = global_lons.min() + np.arange(lon_dim) * float(lon_spacing)
    else:
        unique_lons = np.linspace(global_lons.min(), global_lons.max(), lon_dim)

    unique_lats = standardize_coordinates(unique_lats, "output latitudes")
    unique_lons = standardize_coordinates(unique_lons, "output longitudes")

    logger.info(
        f"  Regular grid: {lat_dim} lat × {lon_dim} lon "
        f"(max {max_clusters_per_lat} clusters/lat × {n_repeat} repeat)"
    )
    if expanded_lat or expanded_lon:
        logger.info(
            f"  Grid expanded from 1D to 2D (added NaN "
            f"{'row' if expanded_lat else ''}{',' if expanded_lat and expanded_lon else ''}"
            f"{'column' if expanded_lon else ''})"
        )
    logger.info(
        f"  Memory reduction vs full sparse: "
        f"~{100 * (1 - lon_dim / max(len(np.unique(global_lons)), 1)):.1f}%"
    )

    # ── 2. Build index arrays vectorized (no Python loop over clusters) ──
    lat_to_idx = {lat: i for i, lat in enumerate(unique_lats)}

    lat_indices = np.empty(n_clusters, dtype=np.int32)
    lon_start = np.empty(n_clusters, dtype=np.int32)

    for cluster_idx in range(n_clusters):
        clat = adjusted_lats[cluster_idx]
        lat_indices[cluster_idx] = lat_to_idx.get(clat, int(np.argmin(np.abs(unique_lats - clat))))
        position_in_lat = lat_to_clusters[clat].index(cluster_idx)
        lon_start[cluster_idx] = position_in_lat * n_repeat

    # Expand to one entry per (cluster × repeat) slot
    repeat_offsets = np.arange(n_repeat, dtype=np.int32)
    lat_idx_exp = np.repeat(lat_indices, n_repeat)  # (n_clusters*n_repeat,)
    lon_idx_exp = (lon_start[:, None] + repeat_offsets[None, :]).ravel()  # (n_clusters*n_repeat,)

    # ── 3. Fill output array with a single fancy-index assignment ────
    is_static = variable in ("soil_depth", "soil_code")

    if is_static:
        data = np.full((lat_dim, lon_dim), -1e32, dtype=np.float32)
        vals = ds[variable].values  # shape (n_clusters,)
        if vals.ndim == 0:
            vals = np.full(n_clusters, float(vals))
        vals_exp = np.repeat(vals.astype(np.float32), n_repeat)
        data[lat_idx_exp, lon_idx_exp] = vals_exp

    else:
        time_values = ds["time"].values
        n_time = len(time_values)
        data = np.full((n_time, lat_dim, lon_dim), -1e32, dtype=np.float32)

        da = ds[variable]
        if "cluster" in da.dims and "time" in da.dims:
            ts_all = da.transpose("time", "cluster").values.astype(
                np.float32
            )  # (n_time, n_clusters)
        elif "cluster" in da.dims:
            scalar_vals = da.values.astype(np.float32)
            ts_all = np.tile(scalar_vals, (n_time, 1))
        else:
            raise ValueError(
                f"Variable '{variable}' must have dim 'cluster' (and optionally 'time')"
            )

        # Expand cluster axis: each cluster column repeated n_repeat times
        ts_exp = np.repeat(ts_all, n_repeat, axis=1)  # (n_time, n_clusters*n_repeat)

        # Single vectorized write — no Python loop
        data[:, lat_idx_exp, lon_idx_exp] = ts_exp

    # ── 4. soil_code integer conversion (identical to original) ──────
    if variable == "soil_code":
        data_int = np.full(data.shape, -9999, dtype=np.int32)
        valid_mask = data != -1e32
        data_int[valid_mask] = data[valid_mask].astype(np.int32)
        data = data_int
        fill_value = -9999
    else:
        fill_value = -1e32

    # ── 5. Write compact grid mapping CSV (identical logic to original) ──
    # WHERE it goes decides whether the projection back is correct. The map is a property of
    # the grid THIS call just laid out — which row each cluster sits in — so it belongs with
    # that grid's own artifacts. Defaulting it to the ID mapping's directory is only safe
    # while the two are written together: a caller who REUSES an ID mapping from a shared
    # directory (`build_reduced_grid` with `id_mapping_file=`, which is the whole point of
    # reusing one) looked for the map there, found a map left by a different build of a
    # different grid, kept it, and wrote none of its own. Measured on four test grids at
    # K = 67: of the inherited map's 67 rows, 63
    # named a row outside the grid (so the projection raises) and 3 named a real row holding
    # ANOTHER cluster, which it applies without complaint.
    latrow_map_dir = compact_map_dir
    if latrow_map_dir is None and id_mapping_file is not None:
        latrow_map_dir = os.path.dirname(id_mapping_file)
    if latrow_map_dir is not None:
        labels_arr = ds["cluster_label"].values if "cluster_label" in ds else np.arange(n_clusters)
        lat_to_grid_idx = {}
        for cluster_lat in np.unique(adjusted_lats):
            matches = np.where(np.abs(unique_lats - cluster_lat) < 1e-6)[0]
            lat_to_grid_idx[cluster_lat] = (
                int(matches[0])
                if len(matches) > 0
                else int(np.argmin(np.abs(unique_lats - cluster_lat)))
            )

        lat_to_cluster_list: dict = defaultdict(list)
        for cluster_idx in range(n_clusters):
            lat_to_cluster_list[adjusted_lats[cluster_idx]].append(cluster_idx)

        rows = []
        for cluster_idx in range(n_clusters):
            clat = adjusted_lats[cluster_idx]
            lat_idx_val = lat_to_grid_idx[clat]
            position_in_lat = lat_to_cluster_list[clat].index(cluster_idx)
            cluster_start_idx = position_in_lat * n_repeat
            cluster_end_idx = cluster_start_idx + n_repeat - 1
            rows.append(
                {
                    "cluster_label": int(labels_arr[cluster_idx]),
                    "lat_index": int(lat_idx_val),
                    "cluster_index_start": int(cluster_start_idx),
                    "cluster_index_end": int(cluster_end_idx),
                    "n_repeat": int(n_repeat),
                    # The coordinates of the same cells, so the projection back can find them
                    # in model output by position on the map rather than by array index, and
                    # can tell output written on some other grid from output written on this one.
                    "grid_latitude": float(unique_lats[lat_idx_val]),
                    "grid_longitude_start": float(unique_lons[cluster_start_idx]),
                    "grid_longitude_end": float(unique_lons[cluster_end_idx]),
                }
            )
        new_map = pd.DataFrame(rows).sort_values(["lat_index", "cluster_index_start"])
        latrow_map_path = os.path.join(latrow_map_dir, f"cluster_compact_map_{n_clusters}.csv")
        # The folder is this grid's own when the caller named it, or when it is where the grid's
        # NetCDFs go; otherwise it is the folder of an ID mapping that may be shared.
        own = compact_map_dir is not None or (
            os.path.abspath(latrow_map_dir) == os.path.abspath(output_dir)
        )
        _write_compact_map(new_map, latrow_map_path, own=own)

    # ── 6. Build output xarray Dataset (identical structure to original) ──
    time_values = ds["time"].values if "time" in ds else None
    dims = ("latitude", "longitude") if is_static else ("time", "latitude", "longitude")

    ds_out = xr.Dataset(
        {variable: (dims, data)},
        coords={"latitude": unique_lats, "longitude": unique_lons},
    )
    ds_out.attrs.update(
        {
            "grid_format": "regular_minimal",
            "description": "Regular lat/lon grid with minimal lon dimension based on max cluster occupancy",
            "n_clusters": n_clusters,
            "n_repeat": n_repeat,
            "max_clusters_per_lat": max_clusters_per_lat,
        }
    )

    if not is_static and time_values is not None:
        ds_out["time"] = time_values
        ds_out.time.encoding.update({"units": "days since 2015-01-01", "calendar": "noleap"})

    ds_out[variable].encoding["_FillValue"] = fill_value
    ds_out[variable].attrs["missing_value"] = fill_value

    # Attach cluster_label grid (identical logic to original)
    labels_arr = ds["cluster_label"].values if "cluster_label" in ds else np.arange(n_clusters)
    cl_grid = np.full((lat_dim, lon_dim), -1, dtype=np.int32)
    cl_grid[lat_idx_exp, lon_idx_exp] = np.repeat(labels_arr.astype(np.int32), n_repeat)
    ds_out["cluster_label"] = (["latitude", "longitude"], cl_grid)
    ds_out["cluster_label"].attrs.update(
        {
            "long_name": "cluster label",
            "description": "Cluster identifier for each grid cell",
        }
    )

    # ── 7. Write to disk (identical format to original) ───────────
    if not is_mpi_root():
        logger.info(f"  MPI rank {mpi_rank()}: Skipping write (handled by rank 0)")
        return

    method_suffix = f"_{cluster_method}" if cluster_method else ""
    output_filename = (
        f"{variable}_{n_clusters}_clusters_clustering_ncells_{n_core}"
        f"_{timeseries_type}{method_suffix}.nc"
    )
    output_path = os.path.join(output_dir, output_filename)
    os.makedirs(output_dir, exist_ok=True)

    try:
        ds_out.to_netcdf(
            output_path,
            mode="w",
            format="NETCDF4_CLASSIC",
            engine="netcdf4",
            compute=True,
        )
        logger.info(f"  Created: {output_path}")
        import subprocess

        # A best-effort flush for shared file systems: the file is already complete, so a
        # missing ``sync`` program or a slow one is not an error.
        try:
            subprocess.run(["sync"], timeout=10, check=False)
        except (OSError, subprocess.TimeoutExpired):
            pass
    except Exception as e:
        logger.error(f"  ERROR writing {output_path}: {e}")
        raise
    finally:
        del ds_out, data
        import gc

        gc.collect()


def create_cluster_nc_files(
    timeseries_dataset: xr.Dataset,
    output_dir: str,
    n_repeat: int,
    npatch: int,
    aggregation_type: str,
    timeseries_type: str = "representative",
    parallel: bool = True,
    labels_csv: str | None = None,
    precomputed_grid: tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray] | None = None,
    id_mapping_file: str | None = None,
    workers: int | None = None,
    cluster_method: str | None = None,
    min_grid_spacing: float = 0.1,
    max_grid_spacing: float = 5.0,
    grid_spacing_steps: int = 100,
    overwrite: bool = True,
    artifacts_dir: str | None = None,
    lon_spacing: float | None = None,
    compact_map_dir: str | None = None,
) -> dict[str, str]:
    """
    Create cluster NetCDF files from timeseries dataset.

    Uses robust grid parameter detection that adaptively finds optimal spacing
    without requiring tolerance thresholds.

    Args:
        timeseries_dataset: Dataset with cluster timeseries (from mean or representative extraction)
        output_dir: Directory to save output files
        n_repeat: Number of cells per cluster
        npatch: Number of patches per cluster
        aggregation_type: Temporal aggregation (monthly, seasonal, yearly)
        timeseries_type: Type of timeseries (mean, representative)
        parallel: Whether to process variables in parallel
        labels_csv: Optional CSV file with cluster labels
        precomputed_grid: Optional precomputed grid tuple. Pass a
                   :class:`~climclusterpy.grid.native.NativePlacement` here (together with
                   ``lon_spacing``) to place each cell at its own representative
                   coordinates instead of at the cluster centre — or call
                   :func:`~climclusterpy.build_reduced_grid`, which does both.
        id_mapping_file: Optional ID mapping file
        workers: Optional number of workers for parallel processing
        cluster_method: Optional cluster method name for output filename
        min_grid_spacing: Minimum grid spacing in degrees, and the tolerance for accepting a
                   coarser one. Smaller = finer grid = more RAM. Default 0.1°. NOTE: on
                   latitudes that already lie on a regular lattice — which cluster centres
                   taken from a regular source grid always do — the search returns this value
                   itself rather than treating it as a floor, so it sets the row count
                   outright (it did so on all 26 real cluster sets it was measured on). See
                   :func:`calculate_grid_parameters`.
        max_grid_spacing: Maximum grid spacing to test in degrees. Default: 5.0°.
        grid_spacing_steps: Number of spacing candidates to test. Default: 100.
        overwrite: If False, skip variables whose output NC already exists and is non-empty.
                   Default: True (always overwrite).
        artifacts_dir: Directory for CSV artifacts (cluster_id_mapping, cluster_compact_map,
                   latitude_adjustments). When None, defaults to ``output_dir`` so existing
                   callers are unaffected. Pass a ``__grid_artifacts__`` sub-directory here
                   to keep the CSVs out of the directory the NetCDFs go in.
        compact_map_dir: Where ``cluster_compact_map_{K}.csv`` goes. When None it goes beside
                   ``id_mapping_file`` — the historical location, and correct only when this
                   call also wrote that mapping. A caller that REUSES an ID mapping from a
                   shared directory must pass this: if an earlier build of a different grid left
                   a table there, the build stops with an error rather than keep it.
        lon_spacing: Optional longitude step in degrees. When given, columns are laid out
                   that far apart starting at the westernmost longitude, which makes the cell
                   size the model reads back out of the header the real one; when None the
                   columns are spread over the range of the centre longitudes, as before.

    Returns
    -------
        Dict mapping variable names to output file paths
    """
    ds = timeseries_dataset
    if "cluster_label" not in ds:
        raise KeyError("Timeseries dataset must include 'cluster_label' variable")
    n_clusters = len(ds["cluster_label"].values)
    n_core = int(n_clusters * n_repeat)
    n_patch_result = int(npatch / n_repeat)

    # Ensure output directory exists before writing any files
    os.makedirs(output_dir, exist_ok=True)

    # Artifacts directory: CSV mappings live here; defaults to output_dir for backward compat.
    _artifacts_dir = artifacts_dir if artifacts_dir is not None else output_dir
    os.makedirs(_artifacts_dir, exist_ok=True)

    # Create grid coordinates
    if precomputed_grid is not None:
        # Unpack grid tuple - may be 4-tuple (old) or 6-tuple (new with spacing/offset)
        if len(precomputed_grid) == 6:
            grid_lats, grid_lons, adjusted_lats, unique_lons, lat_spacing, lat_offset = (
                precomputed_grid
            )
        elif len(precomputed_grid) == 4:
            grid_lats, grid_lons, adjusted_lats, unique_lons = precomputed_grid
            lat_spacing, lat_offset = None, None
        else:
            raise ValueError(f"Unexpected precomputed_grid tuple length: {len(precomputed_grid)}")
    else:
        grid_lats, grid_lons, adjusted_lats, unique_lons, lat_spacing, lat_offset = (
            create_regular_grid_from_dataset(
                ds,
                n_repeat,
                output_dir,
                labels_csv,
                min_grid_spacing=min_grid_spacing,
                max_grid_spacing=max_grid_spacing,
                grid_spacing_steps=grid_spacing_steps,
            )
        )

    # Ensure latitude adjustments report exists in _artifacts_dir for reconstruction ordering
    adj_report = os.path.join(_artifacts_dir, f"latitude_adjustments_{n_clusters}.csv")
    if not os.path.exists(adj_report):
        centers = _gather_cluster_centers(ds, labels_csv)
        original_lats = centers["center_lat"].values
        deviations = np.abs(adjusted_lats - original_lats)
        pd.DataFrame(
            {
                "cluster_label": centers["cluster_label"],
                "original_lon": centers["center_lon"],
                "original_lat": original_lats,
                "adjusted_lat": adjusted_lats,
                "deviation": deviations,
            }
        ).to_csv(adj_report, index=False)

    # Create original<->artificial ID mapping (no coordinates)
    mapping_path = id_mapping_file
    if mapping_path is None and labels_csv is not None:
        # Ensure cluster_index ordering matches the dataset's cluster label order in ds
        rep_order = ds["cluster_label"].values.tolist() if "cluster_label" in ds else None
        _df, mapping_path = create_cluster_id_mapping(
            n_clusters=n_clusters,
            n_repeat=n_repeat,
            labels_csv=labels_csv,
            output_dir=_artifacts_dir,
            cluster_labels_order=rep_order,
        )

    # Get variables to process (exclude metadata)
    exclude_vars = {"center_lat", "center_lon", "time", "latitude", "longitude", "cluster_label"}
    variables = [v for v in ds.data_vars if v not in exclude_vars]
    # Debug: list variables once
    logger.info(f"Variables to export: {variables}")

    output_files = {}

    # IMPORTANT: Disable parallel processing for NetCDF writing to avoid HDF5 threading issues
    # The HDF5 library used by netCDF4 is not thread-safe when multiple processes
    # write files simultaneously, leading to corrupted files (48-byte empty files)
    # See: https://github.com/Unidata/netcdf-c/issues/1388
    use_parallel = False  # Force sequential for reliability

    if use_parallel and parallel and len(variables) > 1:
        # Process variables in parallel (DISABLED due to HDF5 threading issues)
        logger.info("Processing variables in parallel...")
        max_workers = None if (workers is None or workers < 1) else workers
        with ProcessPoolExecutor(max_workers=max_workers) as executor:
            futures = []
            for var in variables:
                futures.append(
                    executor.submit(
                        create_netcdf_from_dataset,
                        var,
                        ds,
                        grid_lats,
                        grid_lons,
                        adjusted_lats,
                        unique_lons,
                        n_clusters,
                        n_repeat,
                        output_dir,
                        aggregation_type,
                        n_core,
                        npatch,
                        n_patch_result,
                        timeseries_type,
                        cluster_method,
                        lat_spacing,
                        lat_offset,
                        # `mapping_path`, not `id_mapping_file`: the compact map is written
                        # beside the ID mapping, and it is the ONLY thing the projection back
                        # reads. Passing the raw argument meant a caller who supplied
                        # `labels_csv` and let this function write the mapping got no compact
                        # map at all, and found out one step later, at reconstruction.
                        mapping_path,
                        lon_spacing,
                        compact_map_dir,
                    )
                )

            for var, future in zip(variables, futures):
                try:
                    future.result()
                    method_suffix = f"_{cluster_method}" if cluster_method else ""
                    output_filename = (
                        f"{var}_{n_clusters}_clusters_{aggregation_type}_ncells_{n_core}"
                        f"_{timeseries_type}{method_suffix}.nc"
                    )
                    output_files[var] = os.path.join(output_dir, output_filename)
                except Exception as e:
                    # A missing variable file would leave the model with incomplete inputs.
                    raise RuntimeError(
                        f"Writing variable '{var}' to the reduced grid failed: {e!s}"
                    ) from e
    else:
        # Process variables sequentially
        # Memory optimisation: build a minimal dataset per variable so only one
        # climate array is ever referenced at a time (important when the input
        # ds is a lazily-opened 20 GB timeseries NC with 7 variables).
        metadata_vars = ["cluster_label", "time"]
        for var in variables:
            # Per-variable resume: skip if output NC already exists and is non-empty
            method_suffix = f"_{cluster_method}" if cluster_method else ""
            output_filename = (
                f"{var}_{n_clusters}_clusters_clustering_ncells_{n_core}"
                f"_{timeseries_type}{method_suffix}.nc"
            )
            expected_path = os.path.join(output_dir, output_filename)
            if (
                not overwrite
                and os.path.exists(expected_path)
                and os.path.getsize(expected_path) > 0
            ):
                logger.info(
                    f"  [RESUME] {output_filename} already exists ({os.path.getsize(expected_path) / 1e6:.1f} MB), skipping"
                )
                output_files[var] = expected_path
                continue

            try:
                # Build a lightweight dataset slice containing only this variable
                # plus required metadata (cluster_label, time).  This prevents
                # xarray / netCDF4 from eagerly caching other variables' data
                # and keeps peak RSS ≈ one output grid array (~100 GB) instead
                # of one grid + all input variables.
                keep = [v for v in metadata_vars if v in ds] + [var]
                ds_slim = ds[keep]

                create_netcdf_from_dataset(
                    var,
                    ds_slim,
                    grid_lats,
                    grid_lons,
                    adjusted_lats,
                    unique_lons,
                    n_clusters,
                    n_repeat,
                    output_dir,
                    aggregation_type,
                    n_core,
                    npatch,
                    n_patch_result,
                    timeseries_type,
                    cluster_method,
                    lat_spacing,
                    lat_offset,
                    mapping_path,  # see the note in the parallel branch above
                    lon_spacing,
                    compact_map_dir,
                )
                output_files[var] = expected_path
            except Exception as e:
                # A missing variable file would leave the model with incomplete inputs.
                raise RuntimeError(
                    f"Writing variable '{var}' to the reduced grid failed: {e!s}"
                ) from e
            finally:
                # Ensure the slim slice is freed before the next variable
                try:
                    del ds_slim
                except NameError:
                    pass
                import gc

                gc.collect()

    # If we created a mapping file, include it under a special key for consumers
    if mapping_path:
        output_files["__id_mapping__"] = mapping_path
    return output_files
